(function(){
  var find  = document.getElementById('labFind');
  var cats  = document.getElementById('labCats');
  var empty = document.getElementById('labEmpty');
  var cards = document.querySelectorAll('a.card');
  var soons = document.querySelectorAll('.soon');
  var pick  = '';

  function run(){
    var word = (find.value || '').trim().toLowerCase();
    var seen = 0;
    var i, c, txt, ok;
    for(i = 0; i < cards.length; i++){
      c   = cards[i];
      txt = (c.textContent || '').toLowerCase();
      ok  = true;
      if(pick){ if(c.getAttribute('data-cat') !== pick){ ok = false; } }
      if(ok){ if(word){ if(txt.indexOf(word) < 0){ ok = false; } } }
      if(ok){ c.classList.remove('off'); seen = seen + 1; }
      else { c.classList.add('off'); }
    }
    /* 묶음 안이 다 숨으면 묶음 이름도 숨깁니다 */
    for(i = 0; i < soons.length; i++){
      var live = soons[i].querySelectorAll('a.card:not(.off)').length;
      var has  = soons[i].querySelectorAll('a.card').length;
      if(has === 0){
        if(word){ soons[i].classList.add('off'); }
        else if(pick){ soons[i].classList.add('off'); }
        else { soons[i].classList.remove('off'); }
      }
      else if(live === 0){ soons[i].classList.add('off'); }
      else { soons[i].classList.remove('off'); }
    }
    if(seen === 0){ empty.classList.add('on'); }
    else { empty.classList.remove('on'); }
  }

  find.addEventListener('input', run);
  cats.addEventListener('click', function(e){
    var b = e.target.closest('.cat');
    if(!b){ return; }
    pick = b.getAttribute('data-cat') || '';
    var all = cats.querySelectorAll('.cat');
    for(var i = 0; i < all.length; i++){ all[i].classList.remove('on'); }
    b.classList.add('on');
    run();
  });
})();
