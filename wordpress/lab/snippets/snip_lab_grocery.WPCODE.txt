<script>
const DATA = JSON.parse(document.getElementById('quiz-data').textContent);
const Q = DATA.questions;
const RESULTS = DATA.results;

/* 카카오 개발자 > 앱 > 플랫폼 키 > JavaScript 키
   해당 키의 [JavaScript SDK 도메인]에 이 페이지 도메인을 등록해야 작동합니다. */
const KAKAO_JS_KEY = '7806d048aad9905e98fb7dd4217236d2';
const OG_IMAGE = '';   /* 1200x630 그림을 올리면 그 주소를 여기 넣으세요 */
function initKakao(){
  if(!KAKAO_JS_KEY || !window.Kakao) return false;
  if(!window.Kakao.isInitialized()) window.Kakao.init(KAKAO_JS_KEY);
  return true;
}

let state = { step: -1, answers: [] };

const brandTitle = document.getElementById('brandTitle');
const card = document.getElementById('card');
const routeBar = document.getElementById('routeBar');
const routeFill = document.getElementById('routeFill');
const routeDots = document.getElementById('routeDots');
const routeCart = document.getElementById('routeCart');

function buildRouteDots(){
  routeDots.innerHTML = '';
  for(let i=0;i<Q.length;i++) routeDots.appendChild(document.createElement('span'));
}
buildRouteDots();

function updateRoute(step){
  const total = Q.length;
  const pct = total <= 1 ? 0 : (step/(total-1))*100;
  routeFill.style.width = 'calc(' + pct + '% )';
  routeCart.style.left = pct + '%';
  const dots = routeDots.children;
  for(let i=0;i<dots.length;i++){
    dots[i].classList.remove('done','now');
    if(i < step) dots[i].classList.add('done');
    else if(i === step) dots[i].classList.add('now');
  }
}

function renderIntro(){
  brandTitle.textContent = '';
  routeBar.style.display = 'none';
  card.innerHTML = `
    <div class="intro">
      <div class="hero">🛒</div>
      <h2>${DATA.introTitle}</h2>
      <p>${DATA.introSub}</p>
      <button class="btn" id="startBtn">${DATA.startLabel}</button>
    </div>
  `;
  document.getElementById('startBtn').onclick = () => {
    state = { step: 0, answers: [] };
    renderQuestion();
  };
}

function renderQuestion(){
  const q = Q[state.step];
  brandTitle.textContent = DATA.brand;
  routeBar.style.display = 'block';
  updateRoute(state.step);

  const optsHtml = q.options.map((opt, i) => `
    <button class="opt" data-trait="${opt.trait}" data-idx="${i}">
      <span class="tile">${opt.svg || opt.icon}</span>
      <p>${opt.text}</p>
    </button>
  `).join('');

  card.innerHTML = `
    <div class="qhead">
      <button class="backbtn" id="backBtn">← 이전</button>
      <div class="stamp">
        <span class="day">${q.day}</span>
        <span class="phase">${q.phase}</span>
      </div>
    </div>
    <div class="qtext">${q.q}</div>
    <div class="options">${optsHtml}</div>
  `;

  document.getElementById('backBtn').onclick = () => {
    if(card.dataset.locked === '1') return;
    if(state.step === 0){
      state = { step: -1, answers: [] };
      renderIntro();
    } else {
      state.answers.pop();
      state.step -= 1;
      renderQuestion();
    }
  };

  Array.from(card.querySelectorAll('.opt')).forEach(btn => {
    btn.onclick = () => {
      if(card.dataset.locked === '1') return;
      card.dataset.locked = '1';
      btn.classList.add('picked');
      state.answers.push({ axis: q.axis, trait: btn.dataset.trait });
      setTimeout(() => {
        card.dataset.locked = '0';
        if(state.step < Q.length - 1){
          state.step += 1;
          renderQuestion();
        } else {
          renderResult();
        }
      }, 320);
    };
  });
}

function computeType(){
  const c = { plan:0, impulse:0, price:0, taste:0 };
  state.answers.forEach(a => { c[a.trait] = (c[a.trait]||0) + 1; });
  const axis1 = c.plan  >= c.impulse ? 'plan'  : 'impulse';
  const axis2 = c.price >= c.taste   ? 'price' : 'taste';
  return axis1 + '_' + axis2;
}

function renderResult(forcedType){
  routeBar.style.display = 'none';
  brandTitle.textContent = '나의 장보기 스타일';
  const type = forcedType || computeType();
  const r = RESULTS[type];
  const opp = RESULTS[r.opp];

  card.innerHTML = `
    <div class="result">
      <div class="ticket">
        <div class="remoji">${r.emoji}</div>
        <p class="nick">${r.nick}</p>
        <p class="tag">${r.tag}</p>
        <p class="punch">${r.punch}</p>
        <div class="cart-list">
          ${r.cart.map(item => `<div class="row"><span>${item[0]}</span><span>${item[1]}</span></div>`).join('')}
        </div>
      </div>
      <p class="desc">${r.desc}</p>
      <div class="opposite">
        <div class="label">나와 정반대 유형</div>
        <div class="name">${opp.emoji} ${opp.nick}</div>
        <div class="note">${r.oppNote}</div>
      </div>
      <p class="share-title">친구는 어떤 타입일까?</p>
      <div class="share-row">
        <button class="share-item" id="shKakao"><span class="ic kakao">톡</span><span class="lb">카카오톡</span></button>
        <button class="share-item" id="shX"><span class="ic x">𝕏</span><span class="lb">X</span></button>
        <button class="share-item" id="shLink"><span class="ic link">🔗</span><span class="lb" id="lbLink">링크 복사</span></button>
        <button class="share-item" id="shMore" style="display:none;"><span class="ic more">···</span><span class="lb">다른 앱</span></button>
      </div>
      <button class="btn ghost" id="retryBtn">다시 장보러 가기</button>
      <div style="margin-top:12px;">
        <button class="backbtn" id="backFromResult" style="margin:0 auto;">← 마지막 질문 다시 보기</button>
      </div>
    </div>
  `;

  document.getElementById('retryBtn').onclick = () => {
    state = { step: -1, answers: [] };
    renderIntro();
  };

  const backRes = document.getElementById('backFromResult');
  if(state.answers.length === Q.length){
    backRes.onclick = () => {
      state.answers.pop();
      state.step = Q.length - 1;
      renderQuestion();
    };
  } else {
    backRes.style.display = 'none';
  }

  /* ── 공유 ─────────────────────────────── */
  const shareUrl  = location.origin + location.pathname + '?r=' + type;
  /* ── 공유 글 ─ 카카오는 「제목」만 큰 글씨로 보여줍니다.
     그래서 검사 이름을 제목 안에 넣습니다. ── */
  const shareTitle = '장보기 스타일' + ' · ' + (r.name || r.nick);
  const shareLine  = (r.name ? r.nick : (r.tag || r.punch || ''));
  const shareDesc  = shareLine + '\n스텔라 랩 — 나는 어느 쪽일까';
  const shareText  = shareTitle + '\n' + shareLine;
  document.getElementById('shKakao').onclick = () => {
    if(!initKakao()){ alert('카카오 공유는 등록된 도메인에서만 작동해요.'); return; }
    const _lnk = { mobileWebUrl: shareUrl, webUrl: shareUrl };
    const _btn = [{ title: '나도 해보기', link: _lnk }];
    const _img = (r.img || OG_IMAGE || '');
    if (_img) {
      window.Kakao.Share.sendDefault({
        objectType: 'feed',
        content: { title: shareTitle, description: shareDesc,
                   imageUrl: _img, link: _lnk },
        buttons: _btn
      });
    } else {
      /* 그림이 없을 때는 글 본을 씁니다. 깨진 그림 자리가 안 생기고
         글씨가 더 크게 나옵니다. */
      window.Kakao.Share.sendDefault({
        objectType: 'text',
        text: shareTitle + '\n' + shareDesc,
        link: _lnk,
        buttonTitle: '나도 해보기'
      });
    }
  };

  document.getElementById('shX').onclick = () => {
    window.open('https://twitter.com/intent/tweet?text=' + encodeURIComponent(shareText) +
      '&url=' + encodeURIComponent(shareUrl), '_blank', 'noopener,width=560,height=520');
  };

  const lbLink = document.getElementById('lbLink');
  document.getElementById('shLink').onclick = () => {
    const done = (m) => { lbLink.textContent = m; setTimeout(()=>lbLink.textContent='링크 복사', 1600); };
    if(navigator.clipboard){
      navigator.clipboard.writeText(shareText + '\n' + shareUrl).then(()=>done('복사됨')).catch(()=>done('복사 실패'));
    } else { done('복사 실패'); }
  };

  /* 인스타그램은 외부 웹에서 바로 올릴 방법이 없어 기본 공유 시트로 대체 */
  if(navigator.share){
    const more = document.getElementById('shMore');
    more.style.display = 'flex';
    more.onclick = () => navigator.share({ text: shareText, url: shareUrl }).catch(()=>{});
  }
}

/* 공유 링크(?r=유형)로 들어오면 해당 결과를 먼저 보여줍니다 */
const shared = new URLSearchParams(location.search).get('r');
if(shared){ if(RESULTS[shared]){ renderResult(shared); } else { renderIntro(); } }
else renderIntro();
</script>
</div>

	exit;
