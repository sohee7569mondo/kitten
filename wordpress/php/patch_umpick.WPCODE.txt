<!-- stella 유앤미 · 이름 골라주기 · 판 2026-09-16 -->
<script>
(function(){
  if(window.StellaUmPick){ return; }
  window.StellaUmPick = 1;

  var KEY = 'stella_celeb_go';
  var want = '';
  try{ want = localStorage.getItem(KEY); }catch(e){ want = ''; }
  if(!want){ return; }          /* ★ 담아둔 이름이 없으면 아무것도 안 합니다 */

  function tidy(v){
    var s = String(v === null ? '' : v), out = '', i, ch;
    var SP = ' ' + String.fromCharCode(9) + String.fromCharCode(10) + String.fromCharCode(13);
    for(i = 0; i < s.length; i++){
      ch = s.charAt(i);
      if(SP.indexOf(ch) >= 0){ ch = ' '; }
      out += ch;
    }
    while(out.indexOf('  ') >= 0){ out = out.split('  ').join(' '); }
    return out.trim();
  }
  function squash(v){
    var s = tidy(v), out = '', i;
    for(i = 0; i < s.length; i++){ if(s.charAt(i) !== ' '){ out += s.charAt(i); } }
    return out;
  }

  /* 연예인 목록이 떠 있으면 그 사람을 누릅니다 */
  function pickFromSheet(){
    var sheet = document.getElementById('um-celeb');
    if(!sheet){ return 0; }
    var q = squash(want);
    var bs = sheet.querySelectorAll('button, li, div[data-n], a'), i, el, t;
    for(i = 0; i < bs.length; i++){
      el = bs[i];
      t = el.getAttribute('data-n');
      if(!t){ t = tidy(el.textContent); }
      if(squash(t) !== q){ continue; }
      try{ el.click(); }catch(e){}
      return 1;
    }
    return 0;
  }

  /* 아직이면 「연예인이랑 볼래요」 단추를 눌러 목록을 엽니다 */
  function openList(){
    try{
      var b = document.querySelector('#celeb');
      if(!b){ return; }
      var st = window.getComputedStyle(b);
      if(st){ if(st.display === 'none'){ return; } }
      b.click();
    }catch(e){}
  }

  function tick(){
    if(!want){ return; }
    if(pickFromSheet()){
      want = '';
      try{ localStorage.removeItem(KEY); }catch(e){}
      return;
    }
    openList();
  }

  function start(){
    tick();
    var k = 0;
    var t = setInterval(function(){
      k++;
      tick();
      if(!want){ clearInterval(t); return; }   /* 다 했으면 그만 봅니다 */
      if(k > 120){ clearInterval(t); }         /* 1분쯤 보다가 멈춥니다 */
    }, 500);
  }
  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', start);
  } else { start(); }
})();
</script>
