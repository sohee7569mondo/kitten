/* ════════════════════════════════════════════════════════════
   마루 1부 「사주로 당신을 그리면」 — 직장운 칸 더하기
   ★ 2026-09-11 · 소희 님께

   ── 지금 어떻게 되어 있나 ────────────────────────────
   책 맨 앞 window.__stellaCareerLens 의 LENS_TOPIC 표에
   주제별 글이 다섯 벌 들어 있습니다.

       취업운 · 이직운 · 퇴사운 · 사업운 · 재물운

   직장운은 일부러 안 넣으셨습니다. patch160_lens_career 머리말에
   「직장운은 지금 글을 그대로 씁니다」라고 적혀 있습니다.
   마루의 본래 글이 이미 직장 이야기였으니까요.

   ── 무엇이 바뀌나 ────────────────────────────────────
   오늘 확정하신 **직장운세 ① 「나는 어떤 방식으로 일하는가」**를
   그 자리에 넣습니다. 십성 갈래 다섯 벌이라 표 모양이 꼭 맞습니다.

   표 리터럴은 한 글자도 안 건드립니다.
   ALIAS 줄 앞에 한 덩어리를 더할 뿐입니다.

       LENS_TOPIC['직장운'] = { 비겁 … 인성 };

   그래서 다른 다섯 주제는 지금 그대로입니다.

   ★ 이 안에서 두 글자짜리 논리기호(앰퍼샌드 둘)를 쓰지 않습니다.

   해보기   https://stellasaju.com/?stella_job=preview
   붙이기   https://stellasaju.com/?stella_job=apply
   되돌리기 https://stellasaju.com/?stella_job=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_job'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	$mode = sanitize_text_field( wp_unslash( $_GET['stella_job'] ) );
	$META = '_stella_job_bak_0911';
	$ID   = 160;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<!DOCTYPE html><html lang="ko" translate="no"><head><meta charset="utf-8">'
	   . '<meta name="google" content="notranslate">'
	   . '<meta name="viewport" content="width=device-width,initial-scale=1">'
	   . '<title>JOB</title><style>body{background:#12101B;color:#F2ECE0;margin:0;'
	   . 'padding:18px;font:13px/1.7 ui-monospace,Menlo,monospace;max-width:900px}'
	   . 'h1{color:#D4AF6A;font-size:15px;margin:0 0 14px}'
	   . '.ok{color:#9AD1A5}.no{color:#E5A0A0}.dim{color:#8B849C}'
	   . 'pre{white-space:pre-wrap;word-break:break-all;background:#000;padding:10px;'
	   . 'border-left:3px solid #D4AF6A}</style></head>'
	   . '<body class="notranslate" translate="no">';

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $ID, $META, true );
		if ( '' === $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $ID ) );
		clean_post_cache( $ID );
		delete_post_meta( $ID, $META );
		echo '<p class="ok">되돌렸습니다.</p>';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $ID ) );
	if ( '' === $c ) { echo '<p class="no">책을 못 읽었습니다.</p>'; exit; }

	/* ── 넣을 글 (nowdoc) ─────────────────────────────── */
	$ADD = <<<'JSTAG'
LENS_TOPIC['직장운'] = {
      '비겁':'맡은 일은 내가 끝까지 들고 가는 자리입니다. 나눠주면 편해지는 게 아니라 신경이 더 쓰여요. 설명하는 시간에 내가 하는 게 빠릅니다. 그래서 혼자 책임지고 끝내는 일에서 힘이 납니다. 다만 이 방식은 일이 커지면 막힙니다. 내 손에 다 들어오는 크기까지만 자라기 때문이에요. 한마디로 <em>혼자 끝내는 힘이 밑천인데, 넘기는 법을 배워야 다음 자리가 열리는 사람</em>입니다. 넘길 수 있는 일 하나를 정해두세요.',
      '식상':'하던 대로만 하는 일은 오래 못 견디는 자리입니다. 같은 일도 이렇게 하면 낫지 않나가 먼저 보이고, 그 눈으로 일하세요. 그래서 새로운 것을 만들고 내놓는 자리에서 힘이 납니다. 다만 이 방식은 마무리에서 샙니다. 다음 것이 벌써 눈에 들어와서 앞의 것을 끝까지 붙들지 못해요. 한마디로 <em>만드는 힘은 넉넉한데 끝내는 기록이 필요한 사람</em>입니다. 끝내는 날을 먼저 정해두세요. 끝낸 기록이 쌓여야 사람들이 다시 맡깁니다.',
      '재성':'들인 것과 나오는 것이 맞아야 움직이는 자리입니다. 일 자체보다 그 일이 값이 되는지를 먼저 보세요. 의미만 있고 셈이 안 맞으면 오래 못 합니다. 그래서 숫자와 결과가 보이는 자리에서 힘이 납니다. 다만 이 방식은 당장 값이 안 되는 일을 놓칩니다. 배우는 일과 사람을 얻는 일은 셈이 늦게 돌아와요. 한마디로 <em>셈이 빠른 만큼 늦게 오는 값을 일부러 챙겨야 하는 사람</em>입니다. 이 년 뒤에 값이 될 일 하나를 지금 잡아두세요.',
      '관성':'정해진 것을 정해진 대로 해내는 자리입니다. 해야 할 일이 분명하고 책임질 자리가 있으면 그 성실함이 그대로 값이 됩니다. 그래서 틀이 있는 자리에서 힘이 납니다. 다만 이 방식은 아무도 정해주지 않은 자리에서 멈춥니다. 능력이 모자라서가 아니라 방향을 잡느라 힘을 쓰기 때문이에요. 한마디로 <em>맡은 것은 끝까지 지키는데, 범위가 흐리면 힘이 두 배로 드는 사람</em>입니다. 일을 받을 때 여기까지 언제까지를 먼저 물으세요. 묻는 것이 따지는 게 아니라 당신이 일하는 법입니다.',
      '인성':'준비가 되어야 손이 나가는 자리입니다. 모르는 채로 뛰어드는 것을 잘 못 하시는 대신 한 번 익히면 오래 갑니다. 그래서 배우고 쌓이는 자리에서 힘이 납니다. 다만 이 방식은 시작이 늦습니다. 그리고 준비가 끝나는 날은 오지 않아요. 한마디로 <em>배운 것이 밑천인데, 시작하는 날을 스스로 정해야 하는 사람</em>입니다. 칠 할쯤 알았다 싶으면 시작하세요. 나머지는 하면서 배우면 됩니다.'
    };

    
JSTAG;

	/* ── 바꿀 자리 — 들여쓰기·줄바꿈에 기대지 않습니다 ── */
	$CAND = array(
		"var ALIAS = { '직업운':'직장운' };",
		"var ALIAS={ '직업운':'직장운' };",
		"var ALIAS = {'직업운':'직장운'};",
	);

	$rows = array(); $OLD = '';
	foreach ( $CAND as $x ) {
		$n = substr_count( $c, $x );
		$rows[] = array( $x, $n );
	}
	foreach ( $rows as $r ) { if ( 1 === $r[1] ) { $OLD = $r[0]; break; } }

	echo '<h1>직장운 칸 더하기 — '
	   . ( 'apply' === $mode ? '바꿉니다' : '미리보기 (아직 안 바꿈)' ) . '</h1>';
	echo '<p class="dim">쪽 ' . (int) $ID . ' · '
	   . number_format( strlen( $c ) ) . '글자</p>';

	$d = "찾아본 토막\n";
	foreach ( $rows as $r ) { $d .= sprintf( "  %-3d번  %s\n", $r[1], $r[0] ); }
	$d .= "\n지금 표에 있는 주제\n";
	foreach ( array( '취업운', '직장운', '이직운', '퇴사운', '사업운', '재물운' ) as $t ) {
		$has = ( false !== strpos( $c, "'" . $t . "': {" ) )
		    || ( false !== strpos( $c, "LENS_TOPIC['" . $t . "']" ) );
		$d .= sprintf( "  %-6s %s\n", $t, $has ? '있음' : '없음' );
	}
	echo '<pre>' . esc_html( $d ) . '</pre>';

	if ( false !== strpos( $c, "LENS_TOPIC['직장운']" ) ) {
		echo '<p class="ok">이미 들어가 있습니다.</p>';
		exit;
	}
	if ( '' === $OLD ) {
		echo '<p class="no">딱 한 번 나오는 토막을 못 찾아 아무것도 안 바꿨습니다.</p>';
		echo '<p class="dim">위 표를 보내주시면 다음 판을 만들겠습니다.</p>';
		exit;
	}

	$amp = substr_count( $ADD, '&' );
	echo '<p class="dim">넣는 글 안의 앰퍼샌드 : ' . (int) $amp
	   . ( $amp ? ' <span class="no">(0 이어야 합니다)</span>'
	            : ' <span class="ok">(좋습니다)</span>' ) . '</p>';
	if ( $amp ) { echo '<p class="no">앰퍼샌드가 들어 있어 멈췄습니다.</p>'; exit; }

	$new = str_replace( $OLD, $ADD . $OLD, $c );

	echo '<p class="dim">' . number_format( strlen( $c ) ) . ' → '
	   . number_format( strlen( $new ) ) . ' 글자</p>';

	if ( 'apply' !== $mode ) {
		echo '<p class="ok">자리를 찾았습니다.</p>';
		echo '<p class="dim">이대로 괜찮으시면 /?stella_job=apply 를 여세요.</p>';
		echo '<p class="dim">직장운으로 책을 여시면 1부 첫 장이 바뀝니다. '
		   . '다른 주제는 지금 그대로입니다.</p>';
		exit;
	}

	if ( '' === (string) get_post_meta( $ID, $META, true ) ) {
		update_post_meta( $ID, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $ID ) );
	clean_post_cache( $ID );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<p class="ok">바꿨습니다. 직장운으로 책을 열어 1부 첫 장을 보세요.</p>';
	echo '<p class="dim">되돌리려면 /?stella_job=undo</p>';
	exit;
} );
