/* ════════════════════════════════════════════════════════════
   책 속 지도 그리기 (MAP-1)   ★ 2026-09-11 · 소희 님께

   결혼운세 · 재물운세 새 원고를 실제 책에 넣기 전에,
   **책이 실제로 어떻게 생겼는지** 먼저 읽습니다.

   오늘 제가 짐작으로 세 번 헛걸음했으므로
   이번에는 짐작하지 않고 도구로 읽습니다.

   아무것도 바꾸지 않습니다. 읽기만 합니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   열기   https://stellasaju.com/?stella_map911=1
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_map911'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	echo "MAP-1  책 속 지도  ★ 2026-09-11 판\n";
	echo str_repeat( '=', 62 ) . "\n\n";

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo "책 쪽이 없습니다.\n"; exit; }
	$id = (int) $page->ID;

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	echo "쪽 번호      " . $id . "\n";
	echo "글자 수      " . number_format( strlen( $c ) ) . "\n";
	echo "script 열림  " . substr_count( $c, '<script' ) . "\n";
	echo "script 닫힘  " . substr_count( $c, '</scr' . 'ipt>' ) . "\n\n";

	/* 한 조각 보여주기 */
	$peek = function ( $s, $at, $before, $after ) {
		$a = $at - $before; if ( $a < 0 ) { $a = 0; }
		$t = substr( $s, $a, $before + $after );
		$t = str_replace( array( "\r", "\n", "\t" ), array( '', ' ⏎ ', ' ' ), $t );
		$t = preg_replace( '/ {2,}/', ' ', $t );
		return trim( $t );
	};

	/* 어느 줄인지 */
	$lineno = function ( $s, $at ) {
		return substr_count( substr( $s, 0, $at ), "\n" ) + 1;
	};

	/* ── 1 · 주제가 갈라지는 자리 ──────────────────── */
	echo str_repeat( '-', 62 ) . "\n";
	echo "1 · 주제(topic)가 갈라지는 자리\n";
	echo str_repeat( '-', 62 ) . "\n";

	$n = 0;
	if ( preg_match_all( "/topic\s*===?\s*\\\\?'([^'\\\\]{1,30})\\\\?'/", $c, $m, PREG_OFFSET_CAPTURE ) ) {
		$seen = array();
		foreach ( $m[1] as $k => $hit ) {
			$name = $hit[0];
			if ( isset( $seen[ $name ] ) ) { $seen[ $name ]++; continue; }
			$seen[ $name ] = 1;
			$n++;
			echo sprintf( "  %-10s  줄 %-7d  %s\n", $name,
				$lineno( $c, $m[0][ $k ][1] ), $peek( $c, $m[0][ $k ][1], 0, 70 ) );
		}
		echo "\n  주제 이름과 나온 횟수\n";
		foreach ( $seen as $name => $cnt ) {
			echo sprintf( "    %-12s %d번\n", $name, $cnt );
		}
	}
	if ( ! $n ) { echo "  못 찾았습니다.\n"; }
	echo "\n";

	/* ── 2 · 장 제목 ───────────────────────────────── */
	echo str_repeat( '-', 62 ) . "\n";
	echo "2 · 장 제목처럼 보이는 것\n";
	echo str_repeat( '-', 62 ) . "\n";

	$hit2 = 0;
	/* 09장 · 08장 같은 두 자리 숫자 + 장 */
	if ( preg_match_all( '/[0-9]{2}\s*장/', $c, $m2, PREG_OFFSET_CAPTURE ) ) {
		$seen2 = array();
		foreach ( $m2[0] as $hit ) {
			$key = trim( $hit[0] );
			if ( isset( $seen2[ $key ] ) ) { continue; }
			$seen2[ $key ] = 1;
			$hit2++;
			echo sprintf( "  %-8s 줄 %-7d %s\n", $key, $lineno( $c, $hit[1] ),
				$peek( $c, $hit[1], 40, 90 ) );
			if ( $hit2 > 24 ) { break; }
		}
	}
	if ( ! $hit2 ) { echo "  두 자리 숫자 + 장 꼴은 없습니다.\n"; }
	echo "\n";

	/* ── 3 · 결혼 · 재물이 나오는 자리 ─────────────── */
	echo str_repeat( '-', 62 ) . "\n";
	echo "3 · 결혼 · 재물 · 배우자가 나오는 자리 (앞 40 뒤 110)\n";
	echo str_repeat( '-', 62 ) . "\n";

	foreach ( array( '결혼운', '재물운', '배우자', '궁합' ) as $w ) {
		echo "\n  [" . $w . "]\n";
		$from = 0; $cnt = 0;
		while ( true ) {
			$at = strpos( $c, $w, $from );
			if ( false === $at ) { break; }
			$from = $at + 1;
			$cnt++;
			if ( $cnt > 12 ) { echo "    … 더 있습니다\n"; break; }
			echo sprintf( "    줄 %-7d %s\n", $lineno( $c, $at ),
				$peek( $c, $at, 40, 110 ) );
		}
		if ( ! $cnt ) { echo "    없습니다.\n"; }
	}
	echo "\n";

	/* ── 4 · 붙일 자리로 쓸 만한 앵커 ──────────────── */
	echo str_repeat( '-', 62 ) . "\n";
	echo "4 · 지난 패치들이 앵커로 쓰던 줄이 아직 있는지\n";
	echo str_repeat( '-', 62 ) . "\n";

	$ANCHORS = array(
		'재물운 다섯갈래'  => 'var topSip = it.top.map(function(t){ return t.sip; });',
		'다섯갈래 들어감'  => '__isMoney',
		'주제 이름'       => 'var topic',
		'이름'           => 'var name',
		'갈래 개수'       => 'it.groups',
		'십성 목록'       => 'it.top',
		'일지'           => 'it.day',
		'월지'           => 'it.month',
		'시지'           => 'it.hour',
		'세운'           => 'it.year',
		'대운'           => 'it.dae',
		'장 만들기'       => 'function chap',
		'장 넣기'        => 'push(chap',
	);
	foreach ( $ANCHORS as $label => $needle ) {
		$cnt = substr_count( $c, $needle );
		echo sprintf( "  %-14s %-3d번  %s\n", $label, $cnt,
			$cnt ? $peek( $c, strpos( $c, $needle ), 0, 80 ) : '—' );
	}
	echo "\n";

	/* ── 5 · 책이 여러 쪽에 나뉘어 있는지 ──────────── */
	echo str_repeat( '-', 62 ) . "\n";
	echo "5 · 책과 이어진 다른 쪽\n";
	echo str_repeat( '-', 62 ) . "\n";

	$rows = $wpdb->get_results(
		"SELECT ID, post_name, post_type, post_status, LENGTH(post_content) AS len
		   FROM {$wpdb->posts}
		  WHERE post_status IN ('publish','draft','private')
		    AND post_type IN ('page','post','wp_template','wp_template_part')
		    AND LENGTH(post_content) > 20000
		  ORDER BY len DESC LIMIT 20" );
	foreach ( (array) $rows as $r ) {
		echo sprintf( "  %-6d %-28s %-18s %10s\n", $r->ID,
			substr( (string) $r->post_name, 0, 28 ), $r->post_type,
			number_format( (int) $r->len ) );
	}
	echo "\n";

	/* ── 6 · 켜져 있는 WPCode 스니펫 중 책을 고치는 것 ── */
	echo str_repeat( '-', 62 ) . "\n";
	echo "6 · 켜져 있는 스니펫 중 reading-book 을 건드리는 것\n";
	echo str_repeat( '-', 62 ) . "\n";

	$snips = $wpdb->get_results(
		"SELECT ID, post_title, post_content FROM {$wpdb->posts}
		  WHERE post_type = 'wpcode' AND post_status = 'publish'
		  ORDER BY ID ASC" );
	$k = 0;
	foreach ( (array) $snips as $s ) {
		$code = (string) $s->post_content;
		if ( '' === $code ) { continue; }
		if ( false === strpos( $code, 'reading-book' ) ) { continue; }
		$k++;
		echo sprintf( "  %-6d %s\n", $s->ID, substr( (string) $s->post_title, 0, 54 ) );
	}
	if ( ! $k ) { echo "  없습니다.\n"; }

	echo "\n" . str_repeat( '=', 62 ) . "\n";
	echo "끝. 아무것도 바꾸지 않았습니다.\n";
	exit;
} );
