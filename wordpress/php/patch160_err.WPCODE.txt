   ── 이 조각이 하는 일 ─────────────────────────────────

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active

add_action( 'wp_head', function () {

	if ( ! isset( $_GET['stella_err'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	$js = <<<'STELLAERR'
<script>
(function(){
  var box=null, n=0;
  function show(m){
    n=n+1;
    if(!box){
      box=document.createElement('div');
      box.setAttribute('id','stellaErrBox');
      box.style.cssText='position:fixed;left:0;right:0;top:0;z-index:2147483647;'+
        'background:#6E1020;color:#FFECEC;font:12px/1.65 ui-monospace,Menlo,monospace;'+
        'padding:12px 14px;max-height:70vh;overflow:auto;white-space:pre-wrap;'+
        'border-bottom:3px solid #D4AF6A';
      var host=document.body;
      if(!host){ host=document.documentElement; }
      host.appendChild(box);
    }
    box.appendChild(document.createTextNode(n+'. '+m+'\n'));
  }

  window.onerror=function(msg,src,line,col){
    show('넘어짐  '+line+'줄 '+col+'칸\n    '+msg);
    return false;
  };

  /* ★ 화면에 들어 있는 <script> 를 하나씩 직접 문법 검사합니다.
     네트워크를 안 씁니다. 막힐 데가 없습니다. */
  function scan(){
    var ss=document.getElementsByTagName('script');
    var i, t, k, bad=0;
    for(i=0;i<ss.length;i++){
      if(ss[i].src){ continue; }
      t=ss[i].textContent;
      if(!t){ continue; }
      if(t.length<40){ continue; }
      try{ new Function(t); continue; }catch(err){ }
      bad=bad+1;
      show('★★ ' + bad + '번째 다친 덩어리 — 화면의 ' + i + '번째 script, ' + t.length + '자');

      var e2=null;
      try{ new Function(t); }catch(err){ e2=err; }
      show('    ' + String(e2));

      /* ★ 이분법으로 다친 자리를 정확히 짚습니다.
         앞에서부터 조금씩 잘라 넣어보다가, 같은 오류가
         처음 나는 자리가 곧 다친 자리입니다. */
      var want='regular expression';
      var hurts=function(k){
        try{ new Function(t.slice(0,k)); return 0; }
        catch(e3){
          if(String(e3.message||e3).indexOf(want)>=0){ return 1; }
          return 0;
        }
      };
      if(hurts(t.length)===1){
        var lo=0, hi=t.length, mid;
        while(lo<hi){
          mid=Math.floor((lo+hi)/2);
          if(hurts(mid)===1){ hi=mid; } else { lo=mid+1; }
        }
        var ln2=t.slice(0,lo).split('\n').length;
        var from=lo-200;
        if(from<0){ from=0; }
        show('    ★ 다친 자리 ' + lo + '번째 글자 · ' + ln2 + '줄');
        show('    ┌ 앞 200자');
        show('    ' + t.slice(from,lo).replace(/\n/g,' '));
        show('    └ 뒤 60자');
        show('    ' + t.slice(lo,lo+60).replace(/\n/g,' '));
      } else {
        show('    이분법으로 못 짚었습니다');
      }

      show('    x26 꼴 ' + t.split('x26').length + '군데');
      show('    ┌ 머리 200자');
      show('    ' + t.slice(0,200).replace(/\n/g,' '));
      if(bad>=3){ return bad; }
    }
    return bad;
  }

  function look(){
    var b=document.getElementById('ssb');
    if(!b){ show('ssb 칸이 없습니다'); }
    else {
      show('쪽 개수      '+b.getElementsByClassName('page').length);
    }
    show('계산기       '+(window.StellaSaju ? '있음' : '★ 없음'));
    show('풀이말       '+(window.StellaRead ? '있음' : '★ 없음'));
    var o=null;
    try{ o=localStorage.getItem('stella_demo'); }catch(err){ o='읽기 막힘'; }
    if(!o){ show('주문 자료    ★ 없음'); }
    else { show('주문 자료    있음 '+String(o).length+'자'); }
    var bad=scan();
    if(bad===0){ show('★ 다친 덩어리를 못 찾았습니다'); }
  }

  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', function(){ setTimeout(look, 2500); });
  } else { setTimeout(look, 2500); }
})();
</script>
STELLAERR;

	echo $js;
}, 0 );
