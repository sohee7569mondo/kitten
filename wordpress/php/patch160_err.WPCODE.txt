/* ══════════════════════════════════════════════════════════
   자바스크립트가 넘어지는 자리를 화면에 찍습니다 (읽기만)
   ★ 2026-09-12 · 소희 님께

   ── 여기까지 알아낸 것 ────────────────────────────────
   · 책 쪽 글은 멀쩡합니다 (1,446,093자, script 39개)
   · 워드프레스가 손댄 것은 <img> 에 decoding="async" 를
     한 번 더 끼운 것 하나뿐입니다. 같은 표가 두 번 들어간
     것이라 브라우저가 그냥 무시합니다. 상처가 아닙니다.
   · 되돌리기는 실행조차 안 됐습니다
   → 그러니 자바스크립트가 돌다가 어디선가 넘어지는 것입니다.

   ── 이 조각이 하는 일 ─────────────────────────────────
   책 쪽 맨 위에 빨간 칸을 하나 띄우고, 자바스크립트가
   넘어질 때마다 그 말을 그대로 받아 적습니다.
   그리고 두 걸음 반 뒤에 이것도 적습니다 —
       쪽이 몇 개 그려졌나
       계산기(StellaSaju)가 실려 있나
       풀이말(StellaRead)이 실려 있나
       주문 자료가 있나
   ★ 그리고 넘어진 줄을 원문에서 그대로 읽어와 보여줍니다.

   ★ 주소 끝에 ?stella_err=1 을 붙였을 때만 뜹니다.
     손님 화면에는 절대 안 나옵니다.
   ★ 아무것도 안 바꿉니다.

   보기  https://stellasaju.com/reading-book/?stella_err=1

   ★ 이번에는 주소창에 그대로 치셔도 됩니다.
     빈 책이어도 상관없습니다 — 넘어지는 자리를 보는 것이니까요.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {

	if ( ! isset( $_GET['stella_err'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	$js = <<<'STELLAERR'
<script>
(function(){
  var box=null, n=0;
  function show(m){
    n=n+1;
    if(!box){
      box=document.createElement('div');
      box.setAttribute('id','stellaErrBox');
      box.style.cssText='position:fixed;left:0;right:0;top:0;z-index:2147483647;'+
        'background:#6E1020;color:#FFECEC;font:12px/1.65 ui-monospace,Menlo,monospace;'+
        'padding:12px 14px;max-height:70vh;overflow:auto;white-space:pre-wrap;'+
        'border-bottom:3px solid #D4AF6A';
      var host=document.body;
      if(!host){ host=document.documentElement; }
      host.appendChild(box);
    }
    box.appendChild(document.createTextNode(n+'. '+m+'\n'));
  }

  window.onerror=function(msg,src,line,col){
    show('넘어짐  '+line+'줄 '+col+'칸\n    '+msg);
    return false;
  };

  /* ★ 화면에 들어 있는 <script> 를 하나씩 직접 문법 검사합니다.
     네트워크를 안 씁니다. 막힐 데가 없습니다. */
  function scan(){
    var ss=document.getElementsByTagName('script');
    var i, t, k, bad=0;
    for(i=0;i<ss.length;i++){
      if(ss[i].src){ continue; }
      t=ss[i].textContent;
      if(!t){ continue; }
      if(t.length<40){ continue; }
      try{ new Function(t); continue; }catch(err){ }
      bad=bad+1;
      show('★★ ' + bad + '번째 다친 덩어리 — 화면의 ' + i + '번째 script, ' + t.length + '자');

      var e2=null;
      try{ new Function(t); }catch(err){ e2=err; }
      show('    ' + String(e2));

      /* 어느 줄인지 브라우저가 알려줍니다 */
      var ln=0, cl=0;
      var st=String(e2 ? (e2.stack||'') : '');
      var m=st.match(/<anonymous>:(\d+):(\d+)/);
      if(m){ ln=parseInt(m[1],10)-2; cl=parseInt(m[2],10); }
      var L=t.split('\n');
      if(ln>0){
        show('    다친 줄 ' + ln + ' / 모두 ' + L.length + '줄');
        for(k=ln-1;k<=ln+1;k++){
          if(k<1){ continue; }
          if(k>L.length){ break; }
          var row=L[k-1];
          if(row.length>260){ row=row.slice(0,260)+'…'; }
          show('    ' + k + '| ' + row);
        }
      } else {
        show('    줄 번호를 못 받았습니다 / 모두 ' + L.length + '줄');
      }

      show('    ┌ 머리 200자');
      show('    ' + t.slice(0,200).replace(/\n/g,' '));
      show('    └ 꼬리 120자');
      show('    ' + t.slice(-120).replace(/\n/g,' '));
      if(bad>=3){ return bad; }
    }
    return bad;
  }

  function look(){
    var b=document.getElementById('ssb');
    if(!b){ show('ssb 칸이 없습니다'); }
    else {
      show('쪽 개수      '+b.getElementsByClassName('page').length);
    }
    show('계산기       '+(window.StellaSaju ? '있음' : '★ 없음'));
    show('풀이말       '+(window.StellaRead ? '있음' : '★ 없음'));
    var o=null;
    try{ o=localStorage.getItem('stella_demo'); }catch(err){ o='읽기 막힘'; }
    if(!o){ show('주문 자료    ★ 없음'); }
    else { show('주문 자료    있음 '+String(o).length+'자'); }
    var bad=scan();
    if(bad===0){ show('★ 다친 덩어리를 못 찾았습니다'); }
  }

  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', function(){ setTimeout(look, 2500); });
  } else { setTimeout(look, 2500); }
})();
</script>
STELLAERR;

	echo $js;
}, 0 );
