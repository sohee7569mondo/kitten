/* ════════════════════════════════════════════════════════════
   책 · 오행 오각형 그래프   ★ 2026-09-09 · 소희 님께

   「기운의 저울」 장의 막대 다섯 줄을 오각형 그래프로 바꿉니다.
   목 → 화 → 토 → 금 → 수, 상생 차례로 시계방향입니다.
   꼭짓점마다 이름과 개수를 답니다.
   가운데에 가까울수록 그 기운이 적습니다.

   숫자는 새로 만들지 않습니다. 이미 책이 세어 둔 chart.five 를
   그대로 그립니다. 색도 이미 책에 있는 오행 다섯 빛깔을 씁니다.

   그림 그리다 문제가 생기면 빈 글을 돌려주고, 그러면 원래 막대가
   그대로 나옵니다. 책이 깨지지 않습니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   미리보기   https://stellasaju.com/?stella_pent=1
   넣기       https://stellasaju.com/?stella_pent=apply
   되돌리기   https://stellasaju.com/?stella_pent=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_pent'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_pent'] ) );
	$META = '_stella_pent_bak_0909';
	$SLUG = 'reading-book';
	$MARK = 'STELLA-PENT-v1';

	/* 이 줄 바로 뒤에 그림을 끼웁니다 */
	$ANCHOR = "'<p class=\"dek\">여덟 글자를 오행으로 나눠 세었습니다</p>'+";

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="google" content="notranslate">'
	   . '<style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.75;'
	   . 'padding:28px 18px;max-width:900px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.2rem;margin:0 0 10px}'
	   . 'h2{color:#9AD1A5;font-size:1rem;margin:26px 0 6px;'
	   . 'border-top:1px solid rgba(255,255,255,.12);padding-top:14px}'
	   . '.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:12px/1.7 monospace;border-left:3px solid #D4AF6A}'
	   . '</style>';

	$p = get_page_by_path( $SLUG );
	if ( ! $p ) { echo '<p class="no">책 쪽이 없습니다.</p>'; exit; }

	/* ── 되돌리기 ─────────────────────────────────────── */
	if ( 'undo' === $mode ) {
		echo '<h1>되돌립니다</h1>';
		$bak = get_post_meta( $p->ID, $META, true );
		if ( '' === $bak ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ),
			array( 'ID' => $p->ID ) );
		clean_post_cache( $p->ID );
		delete_post_meta( $p->ID, $META );
		if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		echo '<p class="ok">막대 다섯 줄로 돌아갔습니다.</p>';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );

	$go = ( 'apply' === $mode );
	echo '<h1>오행 오각형 — '
	   . ( $go ? '넣습니다' : '미리보기 (아직 안 넣음)' ) . '</h1>';
	echo '<p class="dim">쪽 번호 ' . (int) $p->ID
	   . ' &middot; 크기 ' . number_format( strlen( $c ) ) . ' 바이트</p>';

	if ( false !== strpos( $c, $MARK ) ) {
		echo '<p class="ok">이미 들어 있습니다. (' . esc_html( $MARK ) . ')</p>';
		echo '<p class="dim">되돌리시려면 /?stella_pent=undo 를 여세요.</p>';
		exit;
	}

	$n = substr_count( $c, $ANCHOR );
	echo '<h2>넣을 자리</h2><pre>' . esc_html( $ANCHOR ) . "\n\n"
	   . '찾은 개수  ' . (int) $n . '</pre>';

	if ( 1 !== $n ) {
		echo '<p class="no">자리가 하나가 아닙니다. 넣지 않았습니다.</p>';
		exit;
	}

	$PENT = <<<'STELLA_PENT_JS'
(function(){
  try{
    var five = chart.five;
    var NM = { '목':'나무', '화':'불', '토':'흙', '금':'쇠', '수':'물' };
    var ORDER = ['목','화','토','금','수'];   /* 상생 차례로 시계방향 */
    var by = {}, i, j;
    for(i=0; i<five.length; i++){ by[five[i].element] = five[i]; }

    var max = 0;
    for(i=0; i<ORDER.length; i++){
      var f0 = by[ORDER[i]];
      if(!f0){ continue; }
      if(f0.count > max){ max = f0.count; }
    }
    if(max < 1){ return ''; }

    var CX = 170, CY = 130, R = 88;
    function pt(k, m){
      var a = (-90 + k * 72) * Math.PI / 180;
      return [ CX + Math.cos(a) * R * m, CY + Math.sin(a) * R * m ];
    }
    function poly(m){
      var s = [], p;
      for(j=0; j<5; j++){ p = pt(j, m); s.push(p[0].toFixed(1) + ',' + p[1].toFixed(1)); }
      return s.join(' ');
    }

    /* 거미줄 — 네 겹의 오각형과 다섯 갈래 */
    var web = '';
    var rings = [0.25, 0.5, 0.75, 1];
    for(i=0; i<rings.length; i++){
      var op = '.16';
      if(rings[i] === 1){ op = '.34'; }
      web += '<polygon points="' + poly(rings[i]) + '" fill="none" '
           + 'stroke="currentColor" stroke-width="1" opacity="' + op + '"/>';
    }
    for(i=0; i<5; i++){
      var e1 = pt(i, 1);
      web += '<line x1="' + CX + '" y1="' + CY + '" x2="' + e1[0].toFixed(1)
           + '" y2="' + e1[1].toFixed(1) + '" stroke="currentColor" '
           + 'stroke-width="1" opacity=".16"/>';
    }

    /* 내 모양 */
    var pts = [], dots = '', labs = '';
    for(i=0; i<5; i++){
      var el = ORDER[i];
      var f1 = by[el];
      var n = 0;
      if(f1){ n = f1.count; }
      var p1 = pt(i, n / max);
      pts.push(p1[0].toFixed(1) + ',' + p1[1].toFixed(1));
      dots += '<circle data-el="' + el + '" cx="' + p1[0].toFixed(1)
            + '" cy="' + p1[1].toFixed(1) + '" r="4.5" fill="currentColor"/>';

      var lp = pt(i, 1.24);
      var an = 'middle';
      if(i === 1){ an = 'start'; }
      if(i === 2){ an = 'start'; }
      if(i === 3){ an = 'end'; }
      if(i === 4){ an = 'end'; }
      labs += '<text data-el="' + el + '" class="pent-l" x="' + lp[0].toFixed(1)
            + '" y="' + lp[1].toFixed(1) + '" text-anchor="' + an + '" '
            + 'dominant-baseline="middle" fill="currentColor">'
            + el + ' ' + NM[el]
            + '<tspan class="pent-n" dx="6">' + n + '</tspan></text>';
    }

    var css = '<style>'
      + '#ssb .pent{ margin:24px 0 2px; }'
      + '#ssb .pent svg{ display:block; width:100%; max-width:440px; height:auto;'
      + ' margin:0 auto; color:var(--dim); overflow:visible; }'
      + '#ssb .pent-area{ fill:rgba(138,106,36,.15); stroke:var(--gold);'
      + ' stroke-width:2; stroke-linejoin:round; }'
      + '#ssb .pent-l{ font-family:var(--serif); font-size:13px; font-weight:700; }'
      + '#ssb .pent-n{ font-family:var(--mono); font-size:16px; font-weight:700; }'
      + '#ssb .pent-cap{ text-align:center; font-size:.8rem; color:var(--dim);'
      + ' margin:8px 0 0; }'
      + '#ssb .scale{ display:none; }'   /* 오각형이 막대를 대신합니다 */
      + '@media(max-width:640px){ #ssb .pent-l{ font-size:12px; }'
      + ' #ssb .pent-n{ font-size:14px; } }'
      + '</style>';

    return css
      + '<div class="pent"><!-- STELLA-PENT-v1 -->'
      + '<svg viewBox="0 0 340 240" role="img" aria-label="오행 오각형">'
      + '<g>' + web + '</g>'
      + '<polygon class="pent-area" points="' + pts.join(' ') + '"/>'
      + dots + labs
      + '</svg>'
      + '<p class="pent-cap">가운데에 가까울수록 그 기운이 적습니다</p>'
      + '</div>';
  }catch(err){
    return '';
  }
})()
STELLA_PENT_JS;

	$add = "\n    " . $PENT . "+";

	echo '<h2>끼워 넣을 그림 (자바스크립트)</h2>';
	echo '<p class="dim">' . number_format( strlen( $add ) ) . ' 바이트</p>';
	echo '<pre>' . esc_html( substr( $add, 0, 900 ) ) . "\n…</pre>";

	/* 홑 앰퍼샌드가 있으면 워드프레스가 스크립트를 망가뜨립니다 */
	if ( false !== strpos( $add, '&' ) ) {
		echo '<p class="no">앰퍼샌드가 들어 있습니다. 넣지 않았습니다.</p>';
		exit;
	}

	if ( ! $go ) {
		echo '<h2>끝</h2><p class="dim">이대로 괜찮으시면 '
		   . '/?stella_pent=apply 를 여세요.</p>';
		exit;
	}

	$new = str_replace( $ANCHOR, $ANCHOR . $add, $c );

	if ( '' === get_post_meta( $p->ID, $META, true ) ) {
		update_post_meta( $p->ID, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ),
		array( 'ID' => $p->ID ) );
	clean_post_cache( $p->ID );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<h2>끝</h2><p class="ok">넣었습니다. '
	   . number_format( strlen( $c ) ) . ' -> '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p class="dim">책을 열어 「기운의 저울」 장을 보세요. '
	   . '되돌리시려면 /?stella_pent=undo 를 여세요.</p>';
	exit;
} );
