/* ════════════════════════════════════════════════════════════
   스타 사주 · 맨 위 여백을 **재지 말고 못 박습니다**
   판 2026-09-16 · 6/6

   소희 님 : 「이것만 지금 몇시간인데 그냥 페이지로 만들면
              괜찮지 않아?」

   ★ 소희 님 말씀이 맞습니다. 여태 여백이 안 잡힌 까닭은
     **자바스크립트로 머리글 높이를 재서 밀고 있었기** 때문입니다.
     재는 방식은 —
         · 재는 순간에 따라 값이 달라지고
         · 스크롤하면 또 달라지고
         · 한 번 그린 뒤에 미니 화면이 한 번 덜컥 움직입니다

     안 잽니다. **CSS 여백 한 줄로 못 박습니다.** 계산이 없으니
     달라질 일도 없고, 밀리는 것도 없고, 늘어나지도 않습니다.

   무엇을 고치나 — 셋

     ① 덮개(.sheet)의 위 여백    26px → 100px
     ② 좁은 화면(600 이하)        84px
     ③ 머리글을 재서 미는 자바스크립트 한 줄을 **끕니다**
        (그 줄이 있는 한, CSS 로 아무리 못 박아도 덮어씁니다)

   ── 쓰는 법 ─────────────────────────────────────────
   ① 고치기    https://stellasaju.com/?stella_star6=1
              ★ 미리보기 없이 **바로 고칩니다.** 열기만 하면 됩니다.
                넣기 전 쪽을 통째로 백업합니다.
   ② 값 바꾸기 https://stellasaju.com/?stella_star6=1&px=120
              여백이 모자라거나 남으면 숫자만 바꿔 다시 여세요.
              (좁은 화면은 그 값에서 16 을 뺀 값을 씁니다)
   ③ 되돌리기  https://stellasaju.com/?stella_star6=1&undo=1

   ★ 이 조각은 두 번 눌러도 안전합니다 — 이미 고쳐져 있으면
     값만 갈아끼웁니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_star6'] ) ) { return; }

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:15px/1.8 system-ui;'
		. 'background:#141018;color:#eee;padding:24px;max-width:900px">';

	if ( ! current_user_can( 'manage_options' ) ) {
		echo '<h1 style="color:#ff7b7b">관리자만 볼 수 있습니다.</h1>';
		exit;
	}

	$pid  = 406;
	$bak  = 'stella_bak_406_p6';
	$page = get_post( $pid );
	if ( ! $page ) { echo '<h1 style="color:#ff7b7b">쪽 406 을 못 찾았습니다.</h1>'; exit; }
	$body = $page->post_content;
	$BS   = chr( 92 );
	$b4   = substr_count( $body, $BS );

	/* ── 되돌리기 ─────────────────────────────────────── */
	if ( isset( $_GET['undo'] ) ) {
		$old = get_post_meta( $pid, $bak, true );
		if ( ! $old ) { echo '<h1 style="color:#ff7b7b">되돌릴 것이 없습니다.</h1>'; exit; }
		if ( ! isset( $_GET['go'] ) ) {
			printf( '<h1 style="color:#ffd76a">되돌리기</h1><p>지금 %s자 → %s자</p>',
				number_format( strlen( $body ) ), number_format( strlen( $old ) ) );
			echo '<p style="font-size:18px"><a style="color:#ffd76a" '
				. 'href="?stella_star6=1&amp;undo=1&amp;go=1">→ 이대로 되돌리기</a></p>';
			exit;
		}
		wp_update_post( array( 'ID' => $pid, 'post_content' => wp_slash( $old ) ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $bak );
		echo '<h2 style="color:#7bff9b">되돌렸습니다.</h2>';
		exit;
	}

	/* 여백 값 — 주소로 바꿀 수 있습니다 */
	$PX = isset( $_GET['px'] ) ? (int) $_GET['px'] : 100;
	if ( $PX < 30 ) { $PX = 30; }
	if ( $PX > 260 ) { $PX = 260; }
	$PS = $PX - 16;

	echo '<h1 style="color:#ffd76a">맨 위 여백을 못 박습니다</h1>';
	printf( '<p>쪽 406 · 지금 <b>%s자</b> · 역빗금 <b>%d개</b></p>',
		number_format( strlen( $body ) ), $b4 );
	printf( '<p>넓은 화면 <b>%dpx</b> · 좁은 화면 <b>%dpx</b></p>', $PX, $PS );

	$new  = $body;
	$done = array();
	$fail = array();

	/* ── ① 덮개의 위 여백 ─────────────────────────────── */
	$pat1 = '#padding:\s*([0-9]+)px\s+16px\s+60px;\}#u';
	$n1   = preg_match_all( $pat1, $new, $m1 );
	if ( 1 === $n1 ) {
		$was  = (int) $m1[1][0];
		$new  = preg_replace( $pat1, 'padding:' . $PX . 'px 16px 60px;}', $new, 1 );
		$done[] = '① 덮개 위 여백  ' . $was . 'px → ' . $PX . 'px';
	} else {
		$fail[] = '① 덮개 위 여백 — ' . $n1 . '군데 (하나여야 합니다)';
	}

	/* ── ② 좁은 화면 ─────────────────────────────────── */
	$A2 = '@media(max-width:600px){ #ssv .card{padding:30px 20px 36px;} }';
	$B2 = $A2 . chr( 10 )
		. '@media(max-width:600px){ #ssv .sheet{padding-top:' . $PS . 'px;} }';
	$old2 = '#@media\(max-width:600px\)\{\s*\#ssv\s+\.sheet\{padding-top:[0-9]+px;\}\s*\}#u';
	if ( preg_match( $old2, $new ) ) {
		/* 이미 넣어둔 것이 있으면 값만 갈아끼웁니다 */
		$new = preg_replace( $old2,
			'@media(max-width:600px){ #ssv .sheet{padding-top:' . $PS . 'px;} }', $new, 1 );
		$done[] = '② 좁은 화면  값만 갈아끼움 → ' . $PS . 'px';
	} elseif ( 1 === substr_count( $new, $A2 ) ) {
		$new    = str_replace( $A2, $B2, $new );
		$done[] = '② 좁은 화면  새로 넣음 → ' . $PS . 'px';
	} else {
		$fail[] = '② 좁은 화면 — 붙일 자리를 못 찾았습니다';
	}

	/* ── ③ 머리글을 재서 미는 줄을 끕니다 ─────────────── */
	$pat3 = '#[^' . chr( 10 ) . ']*cd\.style\.paddingTop[^' . chr( 10 ) . ']*#u';
	$n3   = preg_match_all( $pat3, $new );
	if ( $n3 > 0 ) {
		$new = preg_replace( $pat3,
			'        /* 여백은 CSS 가 정합니다 — 재지 않습니다 */', $new );
		$done[] = '③ 재서 미는 줄  ' . $n3 . '군데를 껐습니다';
	} else {
		if ( false !== strpos( $new, '여백은 CSS 가 정합니다' ) ) {
			$done[] = '③ 재서 미는 줄  이미 꺼져 있습니다';
		} else {
			$fail[] = '③ 재서 미는 줄 — 못 찾았습니다';
		}
	}

	echo '<table cellpadding="7" style="border-collapse:collapse;font-size:14px">';
	foreach ( $done as $d ) {
		printf( '<tr><td style="color:#7bff9b">됨</td><td>%s</td></tr>', esc_html( $d ) );
	}
	foreach ( $fail as $d ) {
		printf( '<tr><td style="color:#ff7b7b">못함</td><td>%s</td></tr>', esc_html( $d ) );
	}
	echo '</table>';

	if ( count( $fail ) ) {
		if ( ! count( $done ) ) {
			echo '<h2 style="color:#ff7b7b">하나도 못 고쳤습니다. 이 화면을 저에게 보여주세요.</h2>';
			exit;
		}
	}

	if ( $new === $body ) {
		echo '<h2 style="color:#7bff9b">이미 이 값으로 되어 있습니다. 안 건드렸습니다.</h2>';
		printf( '<p style="font-size:18px"><a style="color:#ffd76a" href="%s">→ 스타 사주 열어보기</a></p>',
			esc_url( get_permalink( $pid ) ) );
		exit;
	}

	/* 백업은 늘 지금 쪽으로 갱신합니다 */
	update_post_meta( $pid, $bak, $body );
	/* ★ wp_slash — 안 감싸면 역빗금이 벗겨집니다 */
	wp_update_post( array( 'ID' => $pid, 'post_content' => wp_slash( $new ) ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	clean_post_cache( $pid );
	$chk = get_post( $pid );
	$af  = substr_count( $chk->post_content, $BS );

	echo '<h2 style="color:#7bff9b">고쳤습니다.</h2>';
	printf( '<p>다시 읽어보니 <b>%s자</b> · 역빗금 <b style="color:%s">%d개</b> '
		. '(넣기 전과 같아야 합니다 — %d개)</p>',
		number_format( strlen( $chk->post_content ) ),
		( $af === $b4 ) ? '#7bff9b' : '#ff7b7b', $af, $b4 );
	if ( $af !== $b4 ) {
		echo '<p style="color:#ff7b7b">역빗금 수가 달라졌습니다. 이 숫자를 저에게 알려주세요.</p>';
	}
	printf( '<p style="font-size:18px"><a style="color:#ffd76a" href="%s">→ 스타 사주 열어보기</a></p>',
		esc_url( get_permalink( $pid ) ) );
	echo '<p style="font-size:13px;color:#999">여백이 모자라거나 남으면 '
		. '?stella_star6=1&amp;px=120 처럼 숫자만 바꿔 다시 여세요.</p>';
	exit;

}, 1 );
