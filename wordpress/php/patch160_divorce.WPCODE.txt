add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_divorce'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_divorce'] ) );
	$META = '_stella_divorce_bak';
	$MARK = 'STELLA_DIVORCE_1';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "바이트\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

	echo "이혼운을 새 주제로 붙입니다   판 DIVORCE-1\n";
	echo "지금 책 : " . number_format( strlen( $c ) ) . "바이트\n\n";

	if ( false !== strpos( $c, $MARK ) ) {
		echo "★ 이미 넣으셨습니다. 다시 넣지 않습니다.\n";
		echo "   고치고 싶으시면 먼저 ?stella_divorce=undo 를 하세요.\n";
		exit;
	}

	$jobs = array();

	/* ─ ① 문의 주제 목록에 이혼운을 더합니다 ───────────────────── */
	$jobs[] = array(
		'name' => '① 문의 주제 목록 (door-love)',
		'kind' => 'list',
	);

	/* ─ ②~⑨ 표 여덟 곳. 선언 바로 뒤에 새 칸을 끼워 넣습니다 ──── */
	$tables = array(
		array( '② 저울 다섯 개',   'AXIS_WARM_TOPIC',  'WARM' ),
		array( '③ 저울 제목',      'AXIS_TITLE_TOPIC', 'TITLE' ),
		array( '④ 저울 쪽 이름',   'AXIS_SIDE_TOPIC',  'SIDE' ),
		array( '⑤ 조합 풀이',      'COMBO_TOPIC',      'COMBO' ),
		array( '⑥ 편지',           'CLOSE_TOPIC',      'CLOSE' ),
		array( '⑦ 배웅 한 줄',     'LAST_TOPIC',       'LAST' ),
		array( '⑧ 1부 렌즈',       'LENS_TOPIC',       'LENS' ),
		array( '⑨ 맺음말 세 가지', 'KEEPTOPIC',        'KEEP' ),
	);
	foreach ( $tables as $t ) {
		$jobs[] = array( 'name' => $t[0], 'kind' => 'table', 'var' => $t[1], 'piece' => $t[2] );
	}

	$found = 0; $skip = 0; $out = $c;

	foreach ( $jobs as $j ) {

		if ( 'list' === $j['kind'] ) {
			/* 'door-love':  ['결혼운',…,'궁합']  안에 이혼운을 끼웁니다.
			   이별운 바로 뒤가 제 자리입니다. */
			$hit = preg_match( "/('door-love'\s*:\s*\[[^\]]*?)'이별운'/u", $out, $m, PREG_OFFSET_CAPTURE );
			if ( 1 !== $hit ) {
				echo $j['name'] . " — 자리를 못 찾았습니다. 건너뜁니다.\n";
				$skip++;
				continue;
			}
			$at  = (int) $m[0][1];
			$len = strlen( $m[0][0] );
			$new = $m[1][0] . "'이별운','이혼운'";
			if ( 'preview' === $mode ) {
				echo $j['name'] . " — 찾았습니다\n";
				echo "    전 : " . substr( $out, $at, $len ) . "\n";
				echo "    후 : " . $new . "\n\n";
			} else {
				$out = substr_replace( $out, $new, $at, $len );
			}
			$found++;
			continue;
		}

		/* 표 여덟 곳 */
		$hit = preg_match( '/var\s+' . preg_quote( $j['var'], '/' ) . '\s*=\s*\{/u',
			$out, $m, PREG_OFFSET_CAPTURE );
		if ( 1 !== $hit ) {
			echo $j['name'] . " (" . $j['var'] . ") — 이 표가 책에 없습니다. 건너뜁니다.\n";
			$skip++;
			continue;
		}
		$at  = (int) $m[0][1];
		$len = strlen( $m[0][0] );
		$piece = stella_divorce_piece( $j['piece'] );
		if ( 'preview' === $mode ) {
			echo $j['name'] . " (" . $j['var'] . ") — 찾았습니다 · 넣을 글 "
				. number_format( strlen( $piece ) ) . "바이트\n";
			echo "    자리 : " . substr( $out, $at, $len ) . "\n";
			echo "    넣을 앞머리 : " . stella_divorce_peek( $piece ) . "\n\n";
		} else {
			$out = substr_replace( $out, $m[0][0] . $piece, $at, $len );
		}
		$found++;
	}

	echo "\n찾음 " . $found . "군데 · 건너뜀 " . $skip . "군데\n";

	if ( 'preview' === $mode ) {
		echo "\n아직 아무것도 안 바꿨습니다.\n";
		echo "정말 넣으시려면 ?stella_divorce=apply 로 오세요.\n";
		exit;
	}

	if ( 0 === $found ) { echo "바꿀 자리가 없어 그만둡니다.\n"; exit; }

	$out = $out . "\n<!-- " . $MARK . " -->\n";

	update_post_meta( $id, $META, $c );
	$wpdb->update( $wpdb->posts, array( 'post_content' => $out ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "\n넣었습니다.\n";
	echo "전 : " . number_format( strlen( $c ) ) . "바이트\n";
	echo "후 : " . number_format( strlen( $out ) ) . "바이트\n";
	echo "늘어난 만큼 : " . number_format( strlen( $out ) - strlen( $c ) ) . "바이트\n";
	echo "\n되돌리기 : ?stella_divorce=undo\n";
	exit;
}, 1 );

/* 넣을 글의 앞머리만 한 줄로 보여줍니다 */
function stella_divorce_peek( $s ) {
	$s = preg_replace( '/\s+/u', ' ', $s );
	$s = trim( $s );
	if ( mb_strlen( $s ) > 70 ) { return mb_substr( $s, 0, 70 ) . ' …'; }
	return $s;
}
