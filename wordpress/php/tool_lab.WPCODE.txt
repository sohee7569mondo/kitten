/* ════════════════════════════════════════════════════════════
   스텔라 랩 · 어느 카드가 어디로 가나 — 보기만 하는 도구
   판 2026-09-17

   소희 님 : 「스텔라 랩에서 일본여행으로 테스트 하는 MBTI 는
              누르면 옛날 스텔라랩홈으로 가」

   랩의 카드 하나하나가 **따로 있는 조각**이 맡습니다.
   그 조각이 꺼져 있으면 랩 본체(804)가 받아 **옛 홈**을 보여줍니다.
   어느 카드가 그런지 눈으로 찾을 필요 없습니다.

       https://stellasaju.com/?stella_lab=1

   하는 일 —
       ① /lab/ 을 실제로 한 번 열어 카드 링크를 다 뽑습니다
       ② 켜진 조각 가운데 그 주소를 맡는 것이 있는지 봅니다
       ③ 없는 카드에 ★ 를 답니다 — 그것이 옛 홈으로 갑니다

   ★ 보안 — 조각 **이름과 번호만** 찍습니다. 안에 든 글은 한 자도
     안 찍습니다. 열쇠가 든 조각이 있어 그렇습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_lab'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">'
		. '<body style="font:15px/1.7 system-ui;background:#141018;color:#eee;'
		. 'padding:22px;max-width:920px;margin:0 auto">';

	echo '<h1 style="color:#ffd76a;font-size:20px">스텔라 랩 · 카드가 어디로 가나</h1>';

	/* ── ① /lab/ 을 실제로 열어 카드 링크를 뽑습니다 ── */
	$home = home_url( '/lab/' );
	$r    = wp_remote_get( $home, array( 'timeout' => 25, 'sslverify' => false ) );
	if ( is_wp_error( $r ) ) {
		printf( '<p style="color:#ff7b7b">/lab/ 을 못 열었습니다 — %s</p>',
			esc_html( $r->get_error_message() ) );
		exit;
	}
	$html = wp_remote_retrieve_body( $r );
	printf( '<p style="font-size:13px;color:#999">%s 을 열어 보았습니다 · %s자</p>',
		esc_html( $home ), number_format( strlen( $html ) ) );

	/* 목록이 새 것인지 옛 것인지 */
	$isNew = ( false !== strpos( $html, 'labFind' ) );
	printf( '<p>목록 화면 : <b style="color:%s">%s</b></p>',
		$isNew ? '#7bff9b' : '#ffb76a',
		$isNew ? '새 목록 (찾기칸이 있습니다)' : '옛 목록 (랩 본체가 그립니다)' );

	/* 카드 링크 뽑기 */
	$cards = array();
	if ( preg_match_all( '#href=[\'"]([^\'"]*?/lab/([a-z0-9-]+)/?)[\'"]#i', $html, $mm, PREG_SET_ORDER ) ) {
		foreach ( $mm as $one ) {
			$slug = strtolower( $one[2] );
			if ( '' === $slug ) { continue; }
			$cards[ $slug ] = $one[1];
		}
	}
	/* 카드 딱지(이름)도 같이 — 링크 뒤에 오는 첫 제목을 씁니다 */
	$names = array();
	if ( preg_match_all( '#/lab/([a-z0-9-]+)/?[\'"][^>]*>(.{0,400}?)</a>#is', $html, $nn, PREG_SET_ORDER ) ) {
		foreach ( $nn as $one ) {
			$t = wp_strip_all_tags( $one[2] );
			$t = trim( preg_replace( '/\s+/u', ' ', $t ) );
			if ( '' === $t ) { continue; }
			$names[ strtolower( $one[1] ) ] = mb_substr( $t, 0, 40 );
		}
	}

	if ( ! $cards ) {
		echo '<p style="color:#ff7b7b">카드 링크를 하나도 못 찾았습니다.</p>';
		exit;
	}
	printf( '<p>카드 <b>%d</b>장을 찾았습니다.</p>', count( $cards ) );

	/* ── ② 켜진 조각 가운데 그 주소를 맡는 것 ── */
	global $wpdb;
	$rows = $wpdb->get_results(
		"SELECT ID, post_title, post_status, post_content FROM {$wpdb->posts}
		  WHERE post_type = 'wpcode'" );
	if ( ! $rows ) { $rows = array(); }

	$owner = array();   /* slug => array( 번호, 이름, 켜짐 ) */
	foreach ( $rows as $sn ) {
		$code = (string) $sn->post_content;
		if ( false === strpos( $code, '/lab/' ) ) { continue; }
		foreach ( $cards as $slug => $href ) {
			/* 그 주소를 **맡는** 조각인지 — 따옴표로 감싼 '/lab/xxx' 가 있어야 합니다.
			   카드 링크로만 적힌 목록 조각은 걸리지 않게 합니다. */
			$need = "'/lab/" . $slug . "'";
			if ( false === strpos( $code, $need ) ) { continue; }
			$on = ( 'publish' === $sn->post_status );
			if ( isset( $owner[ $slug ] ) ) { if ( $owner[ $slug ][2] ) { continue; } }
			$owner[ $slug ] = array( $sn->ID, $sn->post_title, $on );
		}
	}

	/* ── ③ 표 ── */
	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%;font-size:13.5px;margin-top:12px">';
	echo '<tr style="color:#9fd;text-align:left">'
		. '<th>카드</th><th>주소</th><th>맡는 조각</th><th></th></tr>';

	$bad = 0;
	$off = 0;
	ksort( $cards );
	foreach ( $cards as $slug => $href ) {
		$nm  = isset( $names[ $slug ] ) ? $names[ $slug ] : $slug;
		$has = isset( $owner[ $slug ] );
		$on  = $has ? $owner[ $slug ][2] : false;

		if ( ! $has ) { $bad++; }
		elseif ( ! $on ) { $off++; }

		$col = '#7bff9b';
		if ( ! $has ) { $col = '#ff7b7b'; }
		elseif ( ! $on ) { $col = '#ffb76a'; }

		$who = '<span style="color:#ff7b7b">맡는 조각이 없습니다</span>';
		if ( $has ) {
			$who = sprintf( '<span style="color:%s">%d · %s · %s</span>',
				$on ? '#7bff9b' : '#ffb76a',
				$owner[ $slug ][0], esc_html( mb_substr( $owner[ $slug ][1], 0, 26 ) ),
				$on ? '켜짐' : '꺼짐' );
		}

		$fix = '';
		if ( ! $has ) { $fix = '<span style="color:#ff7b7b">→ 옛 홈으로 갑니다</span>'; }
		elseif ( ! $on ) {
			$fix = sprintf( '<a style="color:#ffd76a" href="%s">→ 켜러 가기</a>',
				esc_url( admin_url( 'admin.php?page=wpcode-snippet-manager'
					. chr( 38 ) . 'snippet_id=' . $owner[ $slug ][0] ) ) );
		}

		printf( '<tr style="border-top:1px solid #2a2438">'
			. '<td style="color:%s"><b>%s</b></td>'
			. '<td><a style="color:#8ab" href="%s">/lab/%s/</a></td>'
			. '<td>%s</td><td>%s</td></tr>',
			$col, esc_html( $nm ), esc_url( $href ), esc_html( $slug ), $who, $fix );
	}
	echo '</table>';

	echo '<hr style="border:0;border-top:1px solid #333;margin:20px 0">';
	printf( '<p>카드 <b>%d</b>장 가운데 — '
		. '<b style="color:#7bff9b">%d</b>장은 제 조각이 맡고 있고, '
		. '<b style="color:#ffb76a">%d</b>장은 조각이 <b>꺼져</b> 있고, '
		. '<b style="color:#ff7b7b">%d</b>장은 <b>맡는 조각이 없습니다</b>.</p>',
		count( $cards ), count( $cards ) - $bad - $off, $off, $bad );

	if ( $bad ) {
		echo '<p style="color:#ffb76a;font-size:13.5px">★ 맡는 조각이 없는 카드는 랩 본체(804)가 '
			. '받아 <b>옛 홈</b>을 보여줍니다. 그 검사를 아직 안 만들었거나, '
			. '조각을 아직 안 붙이신 것입니다. 어느 쪽인지는 제가 저장소를 보고 '
			. '알려드릴 수 있으니 이 화면을 저에게 보여주세요.</p>';
	}
	echo '<p style="font-size:12.5px;color:#777">★ 조각 이름과 번호만 찍었습니다. '
		. '안에 든 글은 한 자도 안 찍습니다.</p>';
	exit;

}, 1 );
