/* ════════════════════════════════════════════════════════════
   랩 여덟 장이 누구 손에 있나 — 보기만 하는 도구
   ★ 2026-09-09 · 소희 님께

   「도쿄를 눌렀는데 옛날 쪽이 나온다」 같은 일이 왜 생기는지
   봅니다. 아무것도 안 바꿉니다.

   보기   https://stellasaju.com/?stella_lab=1

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_lab'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/plain; charset=utf-8' );

	$slugs = array( 'night-snack', 'cafe', 'camping', 'grocery',
	                'money', 'breakup', 'tokyo', 'family' );

	echo "═══ 랩 여덟 장 ═══\n\n";

	/* ── WPCode 스니펫을 모두 훑습니다 ─────────────── */
	$snips = get_posts( array(
		'post_type'      => 'wpcode',
		'post_status'    => 'any',
		'posts_per_page' => 200,
	) );

	echo '── WPCode 스니펫 ' . count( $snips ) . "개 ──\n\n";

	$owner = array();   /* 슬러그 => 스니펫 이름들 */
	foreach ( $slugs as $s ) { $owner[ $s ] = array(); }

	foreach ( $snips as $sn ) {
		$code = (string) $sn->post_content;
		$hits = array();
		foreach ( $slugs as $s ) {
			if ( false !== strpos( $code, '/lab/' . $s ) ) { $hits[] = $s; }
		}
		if ( ! $hits ) { continue; }

		/* 켜져 있는지 — WPCode 가 쓰는 이름이 판마다 달라서
		   'active' 가 들어간 값을 모두 보여줍니다. */
		$on = array();
		foreach ( (array) get_post_meta( $sn->ID ) as $k => $v ) {
			if ( false === strpos( $k, 'active' ) ) { if ( false === strpos( $k, 'location' ) ) { continue; } }
			$val = is_array( $v ) ? reset( $v ) : $v;
			if ( is_array( $val ) ) { $val = wp_json_encode( $val ); }
			$on[] = $k . '=' . (string) $val;
		}

		echo '  · ' . $sn->post_title . "\n";
		echo '      상태  ' . $sn->post_status . '   ' . implode( '  ', $on ) . "\n";
		echo '      맡은 곳  ' . implode( ', ', $hits ) . "\n\n";

		foreach ( $hits as $s ) { $owner[ $s ][] = $sn->post_title . ' (' . $sn->post_status . ')'; }
	}

	/* ── 장마다 누가 있나 ───────────────────────────── */
	echo "── 장마다 ──\n";
	foreach ( $slugs as $s ) {
		$who = $owner[ $s ];
		if ( ! $who ) {
			echo sprintf( "  %-12s ★ 맡은 스니펫이 없습니다 — 옛날 쪽이 나옵니다\n", $s );
		} else {
			echo sprintf( "  %-12s %s\n", $s, implode( ' / ', $who ) );
		}
	}
	echo "\n";

	/* ── 옛 랩 본체가 아직 /lab/ 을 붙들고 있나 ─────── */
	echo "── 옛 랩 쪽 ──\n";
	foreach ( array( 'lab', 'lab/tokyo', 'lab/breakup' ) as $path ) {
		$p = get_page_by_path( $path );
		echo sprintf( "  /%s%s\n", $path,
			$p ? '  → 쪽 ' . $p->ID . ' (' . $p->post_title . ', ' . $p->post_status . ')' : '  (쪽 없음)' );
	}
	echo "\n";

	echo "── 읽는 법 ──\n";
	echo "  「맡은 스니펫이 없습니다」면 그 장은 아직 옛날 판입니다.\n";
	echo "  스니펫이 있는데도 옛날 판이 나오면 ─ 상태가 draft 이거나\n";
	echo "  위치(Location)가 「어디서나 실행」이 아닙니다.\n";

	echo "\n═══ 끝 ═══\n";
	exit;
}, 1 );
