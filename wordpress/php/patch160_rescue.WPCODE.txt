/* ══════════════════════════════════════════════════════════
   구조 — 책 쪽을 되살립니다
   ★ 2026-09-12 · 소희 님께

   ── 무슨 일이 있었나 ──────────────────────────────────
   여백 조각의 「되돌리기」를 누르셨더니 책이 안 보입니다.
   제 잘못입니다. 되돌림 꾸러미를 담는 방법이 틀렸습니다.

   워드프레스의 update_post_meta 는 값을 담을 때
   역슬래시(\)를 한 겹 벗겨냅니다. 책 본문의 자바스크립트에는

       /\s+/      틈을 찾는 자리
       '<\/p>'    닫는 표를 글자로 쓴 자리
       \'         따옴표를 글자로 쓴 자리

   같은 역슬래시가 수백 군데 있습니다. 그게 꾸러미에 담기는
   순간 전부 사라졌습니다. 그리고 되돌리기가 그 판을 책 쪽에
   덮었습니다. 그래서 자바스크립트가 문법부터 깨졌습니다.

   ★ 글자는 한 자도 안 없어졌습니다. 역슬래시만 빠졌습니다.

   ── 이 조각이 하는 일 ─────────────────────────────────
   ① 지금 상태를 잽니다 — 글자수 · 역슬래시 개수 · 붙어 있는 조각들
   ② 되살릴 수 있는 자리를 전부 찾아 늘어놓습니다
        · 판올림 기록(revision)  ← 워드프레스가 스스로 남긴 것
        · 되돌림 꾸러미(백업)     ← 조각들이 남긴 것
      각각 역슬래시가 몇 개인지 같이 적습니다.
      역슬래시가 남아 있는 것이 성한 판입니다.
   ③ 고른 자리에서 책 쪽을 되살립니다

   ★ 되살리기 전에 지금 판을 따로 담아둡니다. 잘못 골라도
     ?stella_rescue=back 으로 이 자리로 돌아옵니다.
     이번에는 update_option 으로 담습니다 — 역슬래시를
     벗겨내지 않는 자리입니다.

   ── 먼저 이것 ─────────────────────────────────────────
   재보기  https://stellasaju.com/?stella_rescue=look

   그 결과를 그대로 저에게 보내주세요. 어디서 되살릴지
   제가 골라 알려드립니다.

   ── 되살리기 (제가 알려드린 뒤에) ──────────────────────
   판올림에서   https://stellasaju.com/?stella_rescue=rev-번호
   꾸러미에서   https://stellasaju.com/?stella_rescue=bak-이름
   지금으로     https://stellasaju.com/?stella_rescue=back

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ★ ★ 그리고 오늘부터 지켜야 할 것 ────────────────────
   다른 조각의 「되돌리기(undo)」도 모두 같은 방법으로 담았습니다.
   그래서 어느 조각이든 되돌리기를 누르면 똑같이 깨집니다.
   되돌리고 싶으실 때는 누르지 마시고 저에게 말씀해 주세요.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_rescue'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_rescue'] ) );
	$HOLD = 'stella_rescue_hold';
	$BS   = '\\';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	$now = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	$MARKS = array(
		'BUILD: TAIL-', 'BUILD: SPACE-', 'BUILD: DIV', 'BUILD: VM-',
		'BUILD: M9-', 'BUILD: ARC', 'SUM-3', 'BUILD: LV3',
		'BUILD: MOON-1', 'BUILD: CH1', 'BUILD: K1',
	);

	$sizeup = function ( $s ) use ( $BS ) {
		$s = (string) $s;
		return number_format( strlen( $s ) ) . '자 · 역슬래시 '
			. number_format( substr_count( $s, $BS ) );
	};

	/* ── 하나 · 지금 상태를 잽니다 ─────────────────────── */
	if ( 'look' === $mode ) {

		echo "═══ 지금 책 쪽 ═══════════════════════════\n";
		echo '  쪽 번호    : ' . $id . "\n";
		echo '  크기       : ' . $sizeup( $now ) . "\n";
		echo '  ssb 칸     : ' . substr_count( $now, 'id="ssb"' ) . "개\n";
		echo '  script 칸  : ' . substr_count( $now, '<script' ) . "개\n";
		echo '  /\\s  꼴    : ' . substr_count( $now, $BS . 's' ) . "개\n";
		echo '  <\\/  꼴    : ' . substr_count( $now, '<' . $BS . '/' ) . "개\n\n";

		echo "  붙어 있는 조각\n";
		foreach ( $MARKS as $m ) {
			$n = substr_count( $now, $m );
			if ( 0 === $n ) { continue; }
			echo '    ' . $m . '  ' . $n . "개\n";
		}
		echo "\n";

		echo "═══ 판올림 기록(revision) ═════════════════\n";
		$revs = wp_get_post_revisions( $id, array( 'numberposts' => 20 ) );
		if ( empty( $revs ) ) {
			echo "  하나도 없습니다.\n";
		} else {
			foreach ( $revs as $r ) {
				$rc = (string) $wpdb->get_var( $wpdb->prepare(
					"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", (int) $r->ID ) );
				echo '  rev-' . (int) $r->ID . '  ' . $r->post_modified
					. '  ' . $sizeup( $rc ) . "\n";
			}
		}
		echo "\n";

		echo "═══ 되돌림 꾸러미(백업) ═══════════════════\n";
		$rows = $wpdb->get_results( $wpdb->prepare(
			"SELECT meta_key, meta_value FROM {$wpdb->postmeta} WHERE post_id = %d", $id ) );
		$seen = 0;
		if ( $rows ) {
			foreach ( $rows as $m ) {
				$v = (string) $m->meta_value;
				if ( strlen( $v ) < 50000 ) { continue; }
				$seen = $seen + 1;
				echo '  bak-' . $m->meta_key . "\n";
				echo '      ' . $sizeup( $v ) . "\n";
			}
		}
		if ( 0 === $seen ) { echo "  큰 꾸러미가 없습니다.\n"; }
		echo "\n";

		$held = (string) get_option( $HOLD, '' );
		if ( '' !== $held ) {
			echo "═══ 따로 담아둔 판 ═══════════════════════\n";
			echo '  ' . $sizeup( $held ) . "\n";
			echo "  ?stella_rescue=back 으로 이 판으로 돌아옵니다.\n\n";
		}

		echo "★ 역슬래시가 넉넉히 남아 있는 줄이 성한 판입니다.\n";
		echo "  이 화면을 그대로 저에게 보내주세요.\n";
		exit;
	}

	/* ── 둘 · 따로 담아둔 판으로 돌아갑니다 ─────────────── */
	if ( 'back' === $mode ) {
		$held = (string) get_option( $HOLD, '' );
		if ( '' === $held ) { echo '따로 담아둔 판이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $held ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		echo "담아둔 판으로 돌아왔습니다.\n";
		echo '  ' . $sizeup( $held ) . "\n";
		exit;
	}

	/* ── 셋 · 판올림 기록에서 되살립니다 ────────────────── */
	if ( 0 === strpos( $mode, 'rev-' ) ) {
		$rid = (int) substr( $mode, 4 );
		if ( $rid < 1 ) { echo '번호를 못 읽었습니다.'; exit; }
		$row = $wpdb->get_row( $wpdb->prepare(
			"SELECT post_parent, post_content FROM {$wpdb->posts} WHERE ID = %d", $rid ) );
		if ( ! $row ) { echo '그 번호가 없습니다.'; exit; }
		if ( (int) $row->post_parent !== $id ) { echo '책 쪽의 기록이 아닙니다.'; exit; }

		update_option( $HOLD, $now, false );
		$wpdb->update( $wpdb->posts,
			array( 'post_content' => (string) $row->post_content ), array( 'ID' => $id ) );
		clean_post_cache( $id );

		echo "되살렸습니다 — rev-" . $rid . "\n";
		echo '  ' . $sizeup( $row->post_content ) . "\n\n";
		echo "책을 열어보세요. 아니면 ?stella_rescue=back\n";
		exit;
	}

	/* ── 넷 · 꾸러미에서 되살립니다 ─────────────────────── */
	if ( 0 === strpos( $mode, 'bak-' ) ) {
		$key = substr( $mode, 4 );
		if ( '' === $key ) { echo '이름을 못 읽었습니다.'; exit; }
		$v = (string) get_post_meta( $id, $key, true );
		if ( '' === $v ) { echo '그 이름의 꾸러미가 비어 있습니다.'; exit; }

		update_option( $HOLD, $now, false );
		$wpdb->update( $wpdb->posts, array( 'post_content' => $v ), array( 'ID' => $id ) );
		clean_post_cache( $id );

		echo "되살렸습니다 — " . $key . "\n";
		echo '  ' . $sizeup( $v ) . "\n\n";
		echo "책을 열어보세요. 아니면 ?stella_rescue=back\n";
		exit;
	}

	echo "?stella_rescue=look 부터 보세요.";
	exit;
}, 1 );
