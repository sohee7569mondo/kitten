/* ════════════════════════════════════════════════════════════
   스타 사주 — 사람 표를 밖으로 내놓는 한 줄
   ★ 2026-09-15 · 소희 님께

   왜 필요한가
     사람 표(PEOPLE)가 쪽 안쪽 울타리에 갇혀 있어 조각이 손을 못 댑니다.
     한 줄만 내놓으면 그 뒤로는 조각이 다 합니다 —
       · 같은 사람이 두 줄 들어간 것 합치기 (가수 + 배우)
       · 동명이인이면 생년월일을 보여주고 고르게 하기
       · 관리자에게 표 상태를 세어 보여주기

   넣는 것은 딱 이 한 줄입니다
       window.StellaCeleb=PEOPLE;

   어디에 넣나
     findPerson 함수 바로 앞. 그때 PEOPLE 은 이미 만들어져 있습니다.

   ── 쓰는 법 (차례대로) ──
     ?stella_celeb=hunt      표가 어디 있는지 낱말로 훑습니다 (아무것도 안 바꿈)
     ?stella_celeb=peek      쪽 하나를 뜯어봅니다 (기본 193 · ?id= 로 바꿉니다)
     ?stella_celeb=stars     쪽에 든 사람을 세어 봅니다 — 겹친 이름 · 동명이인
     ?stella_celeb=find      찾아낸 글을 확인합니다 (아무것도 안 바꿈)
     ?stella_celeb=preview   무엇이 어떻게 바뀌는지 그대로 보여줍니다
     ?stella_celeb=apply     실제로 넣습니다 (넣기 전 쪽 전체를 백업)
     ?stella_celeb=undo      백업으로 되돌립니다

   ★ 관리자로 로그인하신 뒤에 여세요.
   ★ 편집기로 저장하지 마세요 — 워드프레스가 역빗금을 벗겨냅니다.
     고칠 것은 늘 조각으로.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_celeb'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_celeb'] ) );
	$mark = 'window.StellaCeleb=PEOPLE;';
	$re   = '#(function\s+findPerson\s*\()#u';

	echo "═══ 스타 사주 · 사람 표 내놓기 ═══\n\n";

	/* ── 어디에 있든 찾습니다 ──────────────────────────────
	   2026-09-15 · 첫 판은 page 와 post 만 뒤져서 못 찾았습니다.
	   500명짜리 표는 쪽이 아니라 WPCode 스니펫에 들어 있을 수
	   있습니다. 그래서 글종류를 가리지 않고 훑습니다. */
	global $wpdb;
	$types = $wpdb->get_col(
		"SELECT DISTINCT post_type FROM {$wpdb->posts}" );

	$WORDS = array( 'function findPerson', 'var PEOPLE', 'cRoster',
	                'StellaCeleb', 'PEOPLE', 'findPerson' );

	if ( 'hunt' === $mode ) {
		echo "── 어떤 글종류가 있나 ──\n";
		foreach ( $types as $t ) {
			$n = (int) $wpdb->get_var( $wpdb->prepare(
				"SELECT COUNT(*) FROM {$wpdb->posts} WHERE post_type = %s", $t ) );
			printf( "  %-22s %d개\n", $t, $n );
		}
		echo "\n── 낱말이 어디에 있나 ──\n";
		foreach ( $WORDS as $w ) {
			$rows = $wpdb->get_results( $wpdb->prepare(
				"SELECT ID, post_type, post_title, post_name,
				        CHAR_LENGTH(post_content) AS n
				   FROM {$wpdb->posts}
				  WHERE post_content LIKE %s
			   ORDER BY n DESC LIMIT 12",
				'%' . $wpdb->esc_like( $w ) . '%' ) );
			printf( "\n  「%s」 — %d곳\n", $w, count( $rows ) );
			foreach ( $rows as $r ) {
				printf( "     %-9s %5d  %-26s %s  %d자\n",
					$r->post_type, $r->ID,
					mb_substr( (string) $r->post_title, 0, 24 ),
					$r->post_name, $r->n );
			}
		}
		echo "\n찾으신 자리를 알려주시면 그 자리에 맞춰 넣겠습니다.\n";
		exit;
	}

	/* ── 쪽 하나를 뜯어봅니다 ──────────────────────────────
	   2026-09-15 · hunt 로 알아낸 것 —
	     살아 있는 스타 사주는 page 193 「인물 사주」(celeb) 25,472자.
	     제 사본(68,741자)과 딴 물건이고, findPerson 도 PEOPLE 도
	     없습니다. 500명은 그 쪽에 안 들어갑니다.
	   그래서 쪽이 표를 어디서 가져오는지 찾습니다. */
	if ( 'peek' === $mode ) {
		$pid = isset( $_GET['id'] ) ? (int) $_GET['id'] : 193;
		$post = get_post( $pid );
		if ( ! $post ) { echo "★ 쪽 {$pid} 이 없습니다\n"; exit; }
		$c = $post->post_content;
		printf( "쪽 %d 「%s」 · 슬러그 %s · %d자\n\n",
			$pid, $post->post_title, $post->post_name, strlen( $c ) );

		echo "── 큰 표 같은 변수 선언 ──\n";
		if ( preg_match_all( '#(var|const|let)\s+([A-Za-z_$][\w$]*)\s*=\s*[\[{]#u',
		                     $c, $mm, PREG_SET_ORDER ) ) {
			foreach ( $mm as $m ) { printf( "   %s %s\n", $m[1], $m[2] ); }
		} else { echo "   없습니다\n"; }

		echo "\n── 밖에서 가져오는 자리 ──\n";
		$hit = 0;
		foreach ( array( 'fetch(', 'wp-json', '.json', 'XMLHttpRequest',
		                 'ajax', 'admin-ajax', 'window.Stella', 'apply_filters',
		                 'shortcode', 'wp_localize' ) as $w ) {
			$n = substr_count( $c, $w );
			if ( ! $n ) { continue; }
			$hit++;
			printf( "   %-16s %d번\n", $w, $n );
			$at = strpos( $c, $w );
			echo "      " . str_replace( "\n", " ",
				substr( $c, max( 0, $at - 60 ), 170 ) ) . "\n";
		}
		if ( ! $hit ) { echo "   없습니다 — 표가 쪽 안에 통째로 있을 수 있습니다\n"; }

		echo "\n── window 에 내놓는 이름 ──\n";
		if ( preg_match_all( '#window\.([A-Za-z_$][\w$]*)#u', $c, $mm ) ) {
			$u = array_unique( $mm[1] );
			foreach ( $u as $w ) { echo "   window.$w\n"; }
		} else { echo "   없습니다\n"; }

		echo "\n── 이름이 celeb · star · 스타 · 인물 인 스니펫 ──\n";
		global $wpdb;
		$rows = $wpdb->get_results(
			"SELECT ID, post_title, post_name, CHAR_LENGTH(post_content) AS n
			   FROM {$wpdb->posts}
			  WHERE post_type = 'wpcode'
			    AND ( post_title LIKE '%celeb%' OR post_title LIKE '%star%'
			       OR post_title LIKE '%스타%' OR post_title LIKE '%인물%'
			       OR post_name  LIKE '%celeb%' OR post_name  LIKE '%star%' )
		   ORDER BY n DESC LIMIT 20" );
		if ( $rows ) {
			foreach ( $rows as $r ) {
				printf( "   %5d  %-30s %s  %d자\n", $r->ID,
					mb_substr( (string) $r->post_title, 0, 28 ), $r->post_name, $r->n );
			}
		} else { echo "   없습니다\n"; }

		echo "\n── 큰 스니펫 열 개 (표가 든 것을 찾습니다) ──\n";
		$rows = $wpdb->get_results(
			"SELECT ID, post_title, CHAR_LENGTH(post_content) AS n
			   FROM {$wpdb->posts} WHERE post_type = 'wpcode'
		   ORDER BY n DESC LIMIT 10" );
		foreach ( $rows as $r ) {
			printf( "   %5d  %-34s %d자\n", $r->ID,
				mb_substr( (string) $r->post_title, 0, 32 ), $r->n );
		}
		exit;
	}

	/* ── 쪽에 든 사람 자료를 세어 봅니다 ───────────────────
	   2026-09-15 · peek 으로 알아낸 것 —
	     사람 자료는 쪽 193 안에 {"n":"이름", …} 꼴 JSON 으로 들어
	     있습니다. patch_star1/2 가 필터로 모아 쪽 글에 직접 합칩니다.
	     ?stella_starmerge=1 이 홈으로 튕기면 그 조각이 꺼진 것이므로,
	     여기서 직접 세어 봅니다. */
	if ( 'stars' === $mode ) {
		$pid  = isset( $_GET['id'] ) ? (int) $_GET['id'] : 193;
		$post = get_post( $pid );
		if ( ! $post ) { echo "★ 쪽 {$pid} 이 없습니다\n"; exit; }
		$c = $post->post_content;
		printf( "쪽 %d 「%s」 · %d자\n\n", $pid, $post->post_title, strlen( $c ) );

		/* JSON 을 하나씩 떠냅니다 (patch_star2 와 같은 방식) */
		$rows = array(); $from = 0;
		while ( true ) {
			$at = strpos( $c, '{"n":"', $from );
			if ( false === $at ) { break; }
			$depth = 0; $end = 0; $len = strlen( $c ); $inStr = false;
			for ( $i = $at; $i < $len; $i++ ) {
				$ch = $c[ $i ];
				if ( '"' === $ch ) {
					if ( $i > 0 ) { if ( '\\' !== $c[ $i - 1 ] ) { $inStr = ! $inStr; } }
					continue;
				}
				if ( $inStr ) { continue; }
				if ( '{' === $ch ) { $depth++; }
				if ( '}' === $ch ) { $depth--; if ( 0 === $depth ) { $end = $i; break; } }
			}
			if ( $end <= $at ) { break; }
			$one = json_decode( substr( $c, $at, $end - $at + 1 ), true );
			if ( is_array( $one ) ) { $rows[] = $one; }
			$from = $end + 1;
		}

		printf( "모두 %d명\n\n", count( $rows ) );
		if ( ! $rows ) {
			echo "★ 사람 자료를 못 찾았습니다. 자료 모양이 바뀌었을 수 있습니다.\n";
			exit;
		}

		/* 직업 분포 */
		$job = array();
		foreach ( $rows as $r ) {
			$j = isset( $r['j'] ) ? $r['j'] : '(없음)';
			$job[ $j ] = isset( $job[ $j ] ) ? $job[ $j ] + 1 : 1;
		}
		arsort( $job );
		echo "── 직업 ──\n";
		foreach ( $job as $k => $v ) { printf( "   %-12s %d명\n", $k, $v ); }

		/* 이름이 겹치는 것 — 괄호를 뗀 기본 이름으로 묶습니다 */
		$byName = array();
		foreach ( $rows as $r ) {
			$n = isset( $r['n'] ) ? $r['n'] : '';
			$base = $n;
			$br = strpos( $base, '(' );
			if ( false !== $br ) { $base = substr( $base, 0, $br ); }
			$base = trim( $base );
			if ( '' === $base ) { continue; }
			if ( ! isset( $byName[ $base ] ) ) { $byName[ $base ] = array(); }
			$byName[ $base ][] = $r;
		}

		$same = array(); $diff = array();
		foreach ( $byName as $base => $list ) {
			if ( count( $list ) < 2 ) { continue; }
			$births = array();
			foreach ( $list as $r ) {
				$b = isset( $r['b'] ) ? $r['b'] : '';
				$births[ $b ] = 1;
			}
			if ( count( $births ) === 1 ) { $same[ $base ] = $list; }
			else { $diff[ $base ] = $list; }
		}

		printf( "\n── 같은 이름이 여럿 ── (%d 쌍)\n", count( $byName ) ? count( $same ) + count( $diff ) : 0 );

		printf( "\n★ 같은 사람이 두 줄 (생일이 같음) — %d명\n", count( $same ) );
		foreach ( $same as $base => $list ) {
			printf( "   %s · %s\n", $base, isset( $list[0]['b'] ) ? $list[0]['b'] : '?' );
			foreach ( $list as $r ) {
				printf( "      %-18s %s\n",
					isset( $r['n'] ) ? $r['n'] : '?',
					isset( $r['j'] ) ? $r['j'] : '?' );
			}
		}
		if ( ! $same ) { echo "   없습니다 — 중복 등록은 없습니다\n"; }

		printf( "\n★ 진짜 동명이인 (생일이 다름) — %d명\n", count( $diff ) );
		foreach ( $diff as $base => $list ) {
			printf( "   %s\n", $base );
			foreach ( $list as $r ) {
				printf( "      %-18s %-8s %s\n",
					isset( $r['n'] ) ? $r['n'] : '?',
					isset( $r['j'] ) ? $r['j'] : '?',
					isset( $r['b'] ) ? $r['b'] : '?' );
			}
		}
		if ( ! $diff ) { echo "   없습니다\n"; }

		/* 괄호가 붙은 이름 */
		$par = array();
		foreach ( $rows as $r ) {
			$n = isset( $r['n'] ) ? $r['n'] : '';
			if ( strpos( $n, '(' ) !== false ) { $par[] = $n; }
		}
		printf( "\n── 괄호가 붙은 이름 %d명 ──\n", count( $par ) );
		$k = 0;
		foreach ( $par as $n ) {
			$k++;
			printf( "   %s%s", $n, ( $k % 4 === 0 ) ? "\n" : "  " );
		}
		if ( $k % 4 !== 0 ) { echo "\n"; }
		exit;
	}

	/* ── 표가 있는 글을 찾습니다 (글종류를 안 가립니다) ───── */
	$found = array();
	$ids = $wpdb->get_col( $wpdb->prepare(
		"SELECT ID FROM {$wpdb->posts} WHERE post_content LIKE %s",
		'%' . $wpdb->esc_like( 'findPerson' ) . '%' ) );
	foreach ( $ids as $pid ) {
		$c = get_post_field( 'post_content', $pid );
		if ( ! preg_match( $re, $c ) ) { continue; }
		$found[] = (int) $pid;
	}

	if ( ! $found ) {
		echo "★ 표를 품은 글을 못 찾았습니다.\n\n";
		echo "  ?stella_celeb=hunt 로 여시면 어디에 무엇이 있는지\n";
		echo "  낱말별로 훑어 표로 찍어 드립니다. 그것을 보여주세요.\n";
		exit;
	}

	foreach ( $found as $pid ) {
		$post = get_post( $pid );
		$c    = $post->post_content;
		$has  = ( strpos( $c, $mark ) !== false );
		printf( "쪽 %d 「%s」 · %s\n", $pid, $post->post_title,
			$has ? '이미 넣어져 있습니다' : '아직 안 넣었습니다' );
		printf( "   슬러그 %s · 글자 %d\n", $post->post_name, strlen( $c ) );

		if ( 'find' === $mode ) { continue; }

		/* 그 자리 글을 그대로 찍어 드립니다 */
		if ( preg_match( $re, $c, $m, PREG_OFFSET_CAPTURE ) ) {
			$at = $m[1][1];
			$from = max( 0, $at - 120 );
			echo "\n   ── 넣을 자리 앞뒤 ──\n";
			echo "   " . str_replace( "\n", "\n   ",
				substr( $c, $from, 240 ) ) . "\n";
		}

		if ( 'preview' === $mode ) {
			if ( $has ) { echo "\n   → 이미 있으니 apply 해도 안 바뀝니다.\n\n"; continue; }
			echo "\n   ── 넣고 나면 ──\n";
			echo "   " . $mark . "\n";
			echo "   function findPerson(...\n";
			echo "\n   ?stella_celeb=apply 로 실제로 넣습니다.\n\n";
			continue;
		}

		if ( 'apply' === $mode ) {
			if ( $has ) { echo "   → 이미 있습니다. 안 건드립니다.\n\n"; continue; }
			update_post_meta( $pid, '_stella_celeb_backup', $c );
			$new = preg_replace( $re, $mark . "\n  $1", $c, 1 );
			if ( $new === null ) { echo "   ★ 바꾸다 탈이 났습니다\n\n"; continue; }
			if ( $new === $c )   { echo "   ★ 한 군데도 안 바뀌었습니다\n\n"; continue; }
			wp_update_post( array( 'ID' => $pid, 'post_content' => $new ) );
			printf( "   → 넣었습니다 (글자 %d → %d)\n", strlen( $c ), strlen( $new ) );
			echo "   되돌리려면 ?stella_celeb=undo\n\n";
			continue;
		}

		if ( 'undo' === $mode ) {
			$bak = get_post_meta( $pid, '_stella_celeb_backup', true );
			if ( ! $bak ) { echo "   ★ 백업이 없습니다\n\n"; continue; }
			printf( "   지금 %d 자 → 백업 %d 자\n", strlen( $c ), strlen( $bak ) );
			if ( abs( strlen( $c ) - strlen( $bak ) ) > 2000 ) {
				echo "   ★ 차이가 너무 큽니다. 그 사이에 다른 것이 바뀐 듯합니다.\n";
				echo "     정말 되돌리시려면 ?stella_celeb=undo2 로 여세요.\n\n";
				continue;
			}
			wp_update_post( array( 'ID' => $pid, 'post_content' => $bak ) );
			echo "   → 되돌렸습니다\n\n";
			continue;
		}

		if ( 'undo2' === $mode ) {
			$bak = get_post_meta( $pid, '_stella_celeb_backup', true );
			if ( ! $bak ) { echo "   ★ 백업이 없습니다\n\n"; continue; }
			wp_update_post( array( 'ID' => $pid, 'post_content' => $bak ) );
			echo "   → 크기 차이를 무릅쓰고 되돌렸습니다\n\n";
			continue;
		}
	}

	echo "\n쓸 수 있는 주소 : hunt · peek · stars · find · preview · apply · undo\n";
	exit;
} );
