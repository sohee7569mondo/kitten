		if ( substr_count( $out, $s[3] ) > 0 ) {
			continue;
		}
		$ins  = $s[2];
		$mode = $s[4];
		$out  = preg_replace_callback(
			$s[1],
			function ( $mm ) use ( $ins, $mode ) {
				if ( 'after' === $mode ) {
					return $mm[0] . $ins;
				}
				if ( 'call1' === $mode ) {
					return str_replace( 'easyTxt+', 'easyTxt+__hWhen+', $mm[0] );
				}
				return $mm[0] . '+__hBody';
			},
			$out,
			1
		);
	}

	$wpdb->update( $wpdb->posts, array( 'post_content' => $out ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) {
		wp_cache_flush();
	}

	printf( '<h2 style="color:#7bff9b">넣었습니다 — %d → %d 바이트</h2>', strlen( $now ), strlen( $out ) );
	echo '<p><a style="color:#ffd76a" href="/door-health/">→ 문에서 확인</a></p>';
	echo '<p style="color:#888">되돌리려면 ?stella_health=1&undo=1</p>';
	exit;
