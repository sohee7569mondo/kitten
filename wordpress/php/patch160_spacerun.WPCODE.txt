add_action( 'wp_head', function () {

	if ( is_admin() ) { return; }
	if ( ! is_singular() ) { return; }
	$p = get_post();
	if ( ! $p ) { return; }
	if ( 'reading-book' !== $p->post_name ) { return; }

	echo "\n<style>\n";
	echo stella_space_run_css();
	echo "\n</style>\n";
}, 99 );
