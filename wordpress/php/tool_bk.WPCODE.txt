/* ── ② 어떤 쪽의 겉모습 덩어리들을 훑어봅니다 ── */
add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_styles'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$pid = (int) $_GET['stella_styles'];
	if ( $pid < 1 ) { $pid = 172; }

	$safe = function ( $s ) {
		if ( function_exists( 'mb_convert_encoding' ) ) {
			$s = mb_convert_encoding( $s, 'UTF-8', 'UTF-8' );
		}
		return esc_html( $s );
	};

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:14px/1.7 -apple-system,system-ui,sans-serif;'
		. 'background:#111;color:#eee;padding:24px;max-width:1000px">';
	printf( '<h1 style="color:#ffd76a">%d 쪽의 겉모습 덩어리</h1>', $pid );
	echo '<p style="color:#ff9b9b">브라우저 번역을 꺼주세요.</p>';

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT post_title, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<p style="color:#ff7b7b">못 읽었습니다.</p></body>'; exit; }

	$c = $row->post_content;
	printf( '<p>%s · %s 글자</p>', esc_html( $row->post_title ), number_format( strlen( $c ) ) );

	$off = 0;
	$i   = 0;
	while ( ( $s1 = strpos( $c, '<style', $off ) ) !== false ) {
		$s2 = strpos( $c, '>', $s1 );
		$s3 = strpos( $c, '</style>', $s2 );
		if ( false === $s2 ) { break; }
		if ( false === $s3 ) { break; }
		$i++;
		$body = substr( $c, $s2 + 1, $s3 - $s2 - 1 );
		$head = substr( ltrim( $body ), 0, 260 );
		printf(
			'<p style="margin:16px 0 4px;color:#ffd76a">덩어리 %d · %s 글자 · 시작 %d</p>'
			. '<pre style="white-space:pre-wrap;background:#000;padding:10px;'
			. 'border-left:3px solid #ffd76a;font:12px/1.5 monospace">%s</pre>',
			$i, number_format( strlen( $body ) ), $s1, $safe( $head )
		);
		$off = $s3 + 8;
	}
	printf( '<p style="color:#9fd">모두 <b>%d</b> 덩어리.</p>', $i );

	echo '<h2 style="color:#9fd">찾는 말이 있나</h2><table cellpadding="7" style="border-collapse:collapse">';
	$words = array(
		'2026-09-08 · 문을 흰 바탕으로',
		'2026-09-08 · 남은 쪽을 흰 바탕으로',
		'흰 바탕으로',
		'--void:#F1EDE3',
		'밤하늘 별은 흰 바탕에서',
	);
	foreach ( $words as $w ) {
		$n = substr_count( $c, $w );
		printf(
			'<tr style="border-top:1px solid #333"><td><code>%s</code></td>'
			. '<td style="color:%s"><b>%d</b></td></tr>',
			esc_html( $w ), $n ? '#7bff9b' : '#888', $n
		);
	}
	echo '</table>';
	echo '<h2 style="color:#7bff9b">끝났습니다. 아무것도 바뀌지 않았습니다.</h2></body>';
	exit;
}, 0 );
