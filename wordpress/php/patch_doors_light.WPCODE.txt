/* ════════════════════════════════════════════════════════════
   문 여섯 쪽을 흰 바탕으로

   소희 님 —
     「전체적으로 기본 배경색은 흰색으로 하는거 먼저 고치자」

   ── 어떻게 바꾸나 ──────────────────────────────────
   문 쪽들은 색을 색표(#ssg 안의 --void, --star …) 하나에 모아
   쓰고 있습니다. 그 표를 지우지 않고 뒤에서 덮어씁니다.
   그래서 되돌리기가 쉽고, 사이트 글은 한 글자도 안 건드립니다.

   표 밖에 박혀 있던 색도 같이 고칩니다 — 흰 바탕에서 안 보이게
   되는 것들입니다.
     맞음 연초록 → 진초록 · 틀림 연분홍 → 진빨강
     견본 라벨 · 넣는 칸 바탕 · 목록 펼침
     그림 위 어두운 그라데이션 · 밤하늘 별점

   ── 문마다 다른 빛깔은 살립니다 ────────────────────
   다만 한 단 진하게 잡습니다. 흰 바탕에서 파스텔은 글자로 안 읽혀요.
     연애  #D2909E → #A44E62      건강  #93BBA3 → #3F7A59
     일    #C9A66B → #8C6A2E      흐름  #A98CD8 → #6B4FB0
     별    #86B4D8 → #33688F      타로  #CB7F5E → #A0522D

   ★ 여섯 쪽을 한 번에 봅니다. 못 찾은 쪽은 건너뛰고 나머지만
     바꿉니다. 무엇을 건너뛰었는지 화면에 적습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_light=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_light=apply
   ③ 되돌리기    https://stellasaju.com/?stella_light=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_light'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_light_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_light'] ) );
	$MARK = '2026-09-08 · 문을 흰 바탕으로';

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	/* 문 여섯과 그 문의 빛깔 (흰 바탕용으로 한 단 진하게) */
	$doors = array(
		'door-love'    => array( '연애 · 결혼',  '#A44E62' ),
		'door-career'  => array( '일 · 돈',      '#8C6A2E' ),
		'door-health'  => array( '건강 · 가족',  '#3F7A59' ),
		'door-fortune' => array( '흐름 · 시기',  '#6B4FB0' ),
		'door-astro'   => array( '점성술',       '#33688F' ),
		'door-tarot'   => array( '스텔라타로',   '#A0522D' ),
	);

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$n = 0;
		foreach ( $doors as $slug => $d ) {
			$p = get_page_by_path( $slug );
			if ( ! $p ) { continue; }
			$old = get_post_meta( $p->ID, $META, true );
			if ( ! $old ) { echo '<p>' . esc_html( $slug ) . ' — 되돌릴 것이 없습니다.</p>'; continue; }
			$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $p->ID ) );
			clean_post_cache( $p->ID );
			delete_post_meta( $p->ID, $META );
			echo '<p class="ok">' . esc_html( $slug ) . ' — '
			   . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
			$n++;
		}
		echo '<p>' . intval( $n ) . '쪽을 되돌렸습니다.</p>';
		exit;
	}

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$done = 0; $skip = 0;

	foreach ( $doors as $slug => $d ) {
		$label  = $d[0];
		$accent = $d[1];

		echo '<h2>' . esc_html( $slug ) . ' &middot; ' . esc_html( $label ) . '</h2>';

		$p = get_page_by_path( $slug );
		if ( ! $p ) { echo '<p class="no">쪽을 못 찾았습니다. 건너뜁니다.</p>'; $skip++; continue; }

		$row = $wpdb->get_row( $wpdb->prepare(
			"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
		if ( ! $row ) { echo '<p class="no">내용을 못 읽었습니다. 건너뜁니다.</p>'; $skip++; continue; }
		$c = $row->post_content;

		if ( false !== strpos( $c, $MARK ) ) {
			echo '<p class="no">이미 흰 바탕입니다. 건너뜁니다.</p>'; $skip++; continue;
		}

		/* 2026-09-08 · 자리를 <div id="ssg"> 하나로 잡습니다.
		   전에는 </style> 바로 뒤라고 봤는데, 타로 문은 그 사이에
		   소개 블록이 끼어 있어서 못 찾았습니다. 이 자리는 어느 문에나
		   딱 한 번 나오고, 바로 앞에 넣으면 앞의 색표를 덮습니다. */
		$find = '/<div id="ssg"/u';
		$n = preg_match_all( $find, $c );
		if ( 1 !== $n ) {
			echo '<p class="no">붙일 자리를 ' . intval( $n ) . '군데 찾았습니다. '
			   . '한 군데여야 합니다. 건너뜁니다.</p>';
			$skip++; continue;
		}

		$css = "<style>\n/* ══ " . $MARK . " ═══════════════════════════════\n"
		     . "   앞의 색표를 지우지 않고 뒤에서 덮어씁니다. 되돌리기가 쉽습니다.\n"
		     . "   홈과 같은 종이빛을 씁니다. */\n"
		     . "#ssg{\n"
		     . "  --void:#FFFDF9; --void-deep:#F5F1E8;\n"
		     . "  --nebula:#FFFFFF; --nebula-soft:#F5F1E8;\n"
		     . "  --purple-glow:rgba(107,79,176,.16);\n"
		     . "  --gold:#A9791F; --gold-soft:#B98A47; --gold-lite:#8A6520;\n"
		     . "  --star:#221C33;\n"
		     . "  --ink-soft:#4E4763; --ink-dim:#8B849C;\n"
		     . "  --line:#E7E1D6;\n"
		     . "  --g-accent:" . $accent . " !important;\n"
		     . "  color:var(--star);\n"
		     . "}\n"
		     . "/* 밤하늘 별은 흰 바탕에서 안 보입니다. 아주 옅은 결만 남깁니다. */\n"
		     . "#ssg .g-stars{\n"
		     . "  background:\n"
		     . "    radial-gradient(58vw 44vw at 12% 2%, rgba(107,79,176,.05), transparent 62%),\n"
		     . "    radial-gradient(52vw 40vw at 92% 24%, rgba(169,121,31,.05), transparent 60%),\n"
		     . "    var(--void) !important;}\n"
		     . "/* 카드 · 상자 — 흰 종이에 얇은 테 */\n"
		     . "#ssg .g-card, #ssg .step, #ssg .saved{\n"
		     . "  background:var(--nebula) !important; border:1px solid var(--line) !important;\n"
		     . "  box-shadow:0 10px 30px -18px rgba(34,28,51,.28) !important;}\n"
		     . "#ssg .bill{ background:var(--nebula) !important; }\n"
		     . "/* 넣는 칸 */\n"
		     . "#ssg .f input, #ssg .f select, #ssg .oneline input, #ssg .memo,\n"
		     . "#ssg .topic span{ background:var(--void) !important; color:var(--star) !important;\n"
		     . "  border-color:#D8D0BF !important; }\n"
		     . "#ssg select option{ background:#FFFFFF !important; color:#221C33 !important; }\n"
		     . "#ssg ::placeholder{ color:#B7B0C4 !important; }\n"
		     . "#ssg .topic input:checked + span{\n"
		     . "  background:#F7F2EA !important; border-color:var(--g-accent) !important;\n"
		     . "  color:var(--star) !important; }\n"
		     . "/* 그림 위에 얹던 어두운 그라데이션 */\n"
		     . "#ssg .g-portrait::after{\n"
		     . "  background:linear-gradient(to top, rgba(255,255,255,.96) 2%,\n"
		     . "             rgba(255,255,255,.3) 55%, transparent 80%) !important;}\n"
		     . "/* 흰 바탕에서 안 보이던 글자들 */\n"
		     . "#ssg .msg.ok{ color:#2F7D4A !important; }\n"
		     . "#ssg .msg.err, #ssg .lack{ color:#C4453A !important; }\n"
		     . "#ssg .sample{ background:#FFFFFF !important; border-color:#D8D0BF !important; }\n"
		     . "#ssg .sample:hover{ border-color:var(--g-accent) !important; }\n"
		     . "#ssg .sample-k{ color:var(--g-accent) !important; }\n"
		     . "#ssg .step-no{ color:#FFFFFF !important; }\n"
		     . "#ssg .cal-switch button.on{ color:#FFFFFF !important; }\n"
		     . "#ssg .go{ color:#FFFDF9 !important;\n"
		     . "  background:linear-gradient(120deg,#3A2E77,#241C4E) !important;\n"
		     . "  box-shadow:0 16px 34px -16px rgba(36,28,78,.7) !important; }\n"
		     . "#ssg .signup-note{ background:#F3EFFA !important; border-color:#D9CFF2 !important; }\n"
		     . "#ssg .pay-note, #ssg .gunghap-box, #ssg .g-greet, #ssg .g-method{\n"
		     . "  background:#FAF6EE !important; border-color:#E7E1D6 !important; }\n"
		     . "#ssg .pay-note b{ color:var(--gold) !important; }\n"
		     . "</style>\n<div id=\"ssg\"";

		if ( ! $apply ) {
			echo '<p class="ok">한 군데 찾았습니다 — 빛깔 <b>' . esc_html( $accent )
			   . '</b> 로 흰 바탕을 얹습니다.</p>';
			$done++;
			continue;
		}

		$new = preg_replace( $find, str_replace( '\\', '\\\\', $css ), $c, 1 );
		if ( null === $new ) { echo '<p class="no">바꾸다가 멈췄습니다. 건너뜁니다.</p>'; $skip++; continue; }

		if ( ! get_post_meta( $p->ID, $META, true ) ) {
			update_post_meta( $p->ID, $META, $c );
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $p->ID ) );
		clean_post_cache( $p->ID );

		echo '<p class="ok">' . number_format( strlen( $c ) ) . ' → '
		   . number_format( strlen( $new ) ) . ' 바이트</p>';
		$done++;
	}

	echo '<h2>정리</h2>';
	echo '<p>' . intval( $done ) . '쪽 ' . ( $apply ? '바꿨습니다' : '바꿀 수 있습니다' )
	   . ' &middot; ' . intval( $skip ) . '쪽 건너뜀</p>';
	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_light=apply</code> 를 여세요.</p>';
	} else {
		echo '<p><a href="/door-love/">연애 문 열어보기</a></p>';
		echo '<p class="ok">되돌리시려면 <code>/?stella_light=undo</code> 를 여세요.</p>';
	}
	exit;
}, 0 );
