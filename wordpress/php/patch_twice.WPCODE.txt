/* ════════════════════════════════════════════════════════════
   여는 글에 「두 번 읽으시면 좋습니다」 한 줄   ★ 2026-09-08

   소희 님이 주신 글 그대로 —

     제가 전해드리고 싶은 말이 많아서 양이 많습니다.
     지금 읽으시고 나중에 다시 한번 읽으시면 더욱 좋습니다.

   「아래글은 좀 작게」 하셔서 책의 작은 글씨(.dek)로 넣습니다.
   여는 글 제목 바로 아래가 아니라 여는 글 맨 끝에 둡니다 —
   무슨 책인지 먼저 말하고, 읽는 법을 마지막에 일러드리는 차례가
   자연스럽습니다.

   ── 어디에 ─────────────────────────────────────
     /reading-book/    손님이 받으시는 책      (여는 글 끝)
     /reading-sample/  덕선 · 일과 돈          (여는 글 끝)
     /reading-wealth/  선우 · 재물운           (여는 글 끝)

   ★ 견본 책 넷(정환 · 동룡 · 보라 · 택)은 여는 글 짜임이 달라서
     먼저 보기에서 「못 찾음」으로 나올 수 있습니다. 그러면 그
     쪽만 따로 잡겠습니다.

   ── 다시 넣어도 됩니다 ─────────────────────────
   이미 들어간 쪽은 0군데로 나오고 건너뜁니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_twice=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_twice=apply
   ③ 되돌리기    https://stellasaju.com/?stella_twice=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_twice'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_twice_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_twice'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:880px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}.dim{color:#8B849C;font-size:.86rem}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$LINE = '제가 전해드리고 싶은 말이 많아서 양이 많습니다. '
	      . '지금 읽으시고 나중에 다시 한번 읽으시면 더욱 좋습니다.';

	$pages = array(
		'reading-book'    => '손님이 받으시는 책',
		'reading-sample'  => '덕선 · 일과 돈',
		'reading-wealth'  => '선우 · 재물운',
		'reading-love'    => '정환 · 연애운',
		'reading-health'  => '동룡 · 건강운',
		'reading-fortune' => '보라 · 시기운',
		'reading-astro'   => '택 · 점성술',
	);

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$back = 0;
		foreach ( $pages as $slug => $nm ) {
			$p = get_page_by_path( $slug );
			if ( ! $p ) { continue; }
			$old = get_post_meta( $p->ID, $META, true );
			if ( ! $old ) { continue; }
			$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $p->ID ) );
			clean_post_cache( $p->ID );
			delete_post_meta( $p->ID, $META );
			echo '<p class="ok">' . esc_html( $nm ) . ' — 되돌렸습니다.</p>';
			$back++;
		}
		echo '<h2>정리</h2><p>' . intval( $back ) . '쪽 되돌렸습니다.</p>';
		exit;
	}

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';
	echo '<p class="dim">넣는 글 : ' . esc_html( $LINE ) . '</p>';

	$done = 0; $skip = 0;

	foreach ( $pages as $slug => $nm ) {

		echo '<h2>' . esc_html( $slug ) . ' &middot; ' . esc_html( $nm ) . '</h2>';

		$p = get_page_by_path( $slug );
		if ( ! $p ) { echo '<p class="no">쪽을 못 찾았습니다. 건너뜁니다.</p>'; $skip++; continue; }

		$row = $wpdb->get_row( $wpdb->prepare(
			"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
		if ( ! $row ) { echo '<p class="no">내용을 못 읽었습니다. 건너뜁니다.</p>'; $skip++; continue; }
		$c = $row->post_content;

		if ( false !== strpos( $c, $LINE ) ) {
			echo '<p class="dim">이미 들어가 있습니다. 건너뜁니다.</p>'; $skip++; continue;
		}

		/* ① 살아 있는 책 — 여는 글 마지막 문단 뒤 (자바스크립트로 짜는 쪽) */
		$find1 = '/어느 쪽이 더 당신다운지는, 마지막에 당신이 직접 판단해 주세요\.<\/p>\'\);/u';
		$to1   = "어느 쪽이 더 당신다운지는, 마지막에 당신이 직접 판단해 주세요.</p>'+\n"
		       . "    /* 2026-09-08 · 소희 님 — 글이 많으니 두 번 읽으시라고 */\n"
		       . "    '<p class=\"dek\">" . $LINE . "</p>');";

		/* ② 붙박이 견본 책 — 여는 글의 마지막 </p> 뒤, 쪽번호(folio) 앞 */
		$find2 = '/(<p class="eyebrow">여는 글<\/p>.*?)(\s*<div class="folio">)/su';
		$to2   = '$1' . "\n    <p class=\"dek\">" . $LINE . "</p>" . '$2';

		$n1 = preg_match_all( $find1, $c );
		$n2 = preg_match_all( $find2, $c );

		if ( 1 === $n1 ) {
			$which = '살아 있는 책 · 여는 글 끝';
			$find  = $find1; $to = $to1;
		} elseif ( 1 === $n2 ) {
			$which = '붙박이 견본 · 여는 글 끝';
			$find  = $find2; $to = $to2;
		} else {
			echo '<p class="no">붙일 자리를 못 찾았습니다 (살아 있는 책 ' . intval( $n1 )
			   . ' · 붙박이 ' . intval( $n2 ) . '). 건너뜁니다.</p>';
			$skip++; continue;
		}

		if ( ! $apply ) {
			echo '<p class="ok">자리를 찾았습니다 — ' . esc_html( $which ) . '</p>';
			$done++; continue;
		}

		$new = preg_replace( $find, str_replace( '\\', '\\\\', $to ), $c, 1 );
		if ( null === $new ) {
			echo '<p class="no">바꾸다가 멈췄습니다. 이 쪽은 안 바꿨습니다.</p>';
			$skip++; continue;
		}

		if ( ! get_post_meta( $p->ID, $META, true ) ) {
			update_post_meta( $p->ID, $META, $c );
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $p->ID ) );
		clean_post_cache( $p->ID );

		echo '<p class="ok">' . esc_html( $which ) . ' &middot; '
		   . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $new ) ) . ' 바이트</p>';
		$done++;
	}

	echo '<h2>정리</h2>';
	echo '<p>' . intval( $done ) . '쪽 ' . ( $apply ? '바꿨습니다' : '바꿀 수 있습니다' )
	   . ' &middot; ' . intval( $skip ) . '쪽 건너뜀</p>';
	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_twice=apply</code> 를 여세요.</p>';
	} else {
		echo '<p class="ok">되돌리시려면 <code>/?stella_twice=undo</code> 를 여세요.</p>';
	}
	exit;
}, 0 );
