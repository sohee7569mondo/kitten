	$b .= "    /* 주제를 고르지 않은 채로는 미리보기를 안 건드립니다.\n";
	$b .= "       원래 견본이 「덕선의 직업운」인데 이름만 바꾸면 거짓말이 됩니다. */\n";
	$b .= "    if(want){ dress(current()); }\n";
	$b .= "    if(fixed){ renumber(); }\n";
	$b .= "    unsay();\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  function start(){\n";
	$b .= "    run();\n";
	$b .= "    document.addEventListener('change', function(ev){\n";
	$b .= "      var t = ev.target;\n";
	$b .= "      if(!t){ return; }\n";
	$b .= "      if(t.name !== 'ssg-topic'){ return; }\n";
	$b .= "      dress(t.value);\n";
	$b .= "    });\n";
	$b .= "    var k = 0;\n";
	$b .= "    var tick = setInterval(function(){\n";
	$b .= "      k = k + 1;\n";
	$b .= "      run();\n";
	$b .= "      if(k > 24){ clearInterval(tick); }\n";
	$b .= "    }, 400);\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  if(document.readyState === 'loading'){\n";
	$b .= "    document.addEventListener('DOMContentLoaded', start);\n";
	$b .= "  } else { start(); }\n";
	$b .= "})();\n";
	$b .= "</script>\n" . $e . "\n";

	if ( false !== strpos( $b, '&' ) ) {
		echo '<h1 class="no">앰퍼샌드가 섞여 있습니다. 넣지 않았습니다.</h1>';
		exit;
	}

	echo '<h1>' . ( 'apply' === $mode ? '고쳤습니다' : '이렇게 바꿉니다 (아직 안 고쳤습니다)' ) . '</h1>';
	echo '<ul>'
	   . '<li>주소에 <code>?topic=결혼운</code> 이 있으면 <b>그 주제로 못 박고</b> 고르는 자리를 감춥니다</li>'
	   . '<li>미리보기 제목과 링크가 <b>그 주제로</b> 바뀝니다</li>'
	   . '<li>첫머리 말에서 <b>100자가 넘는 문단</b>을 접습니다 (「더 보기」로 펴집니다)</li>'
	   . '</ul>';

	echo '<table><tr style="color:#D4AF6A"><td>쪽</td><td>주제 단추</td>'
	   . '<td>미리보기</td><td>지금 들어 있는 판</td><td>할 일</td></tr>';

	$plan = array();
	foreach ( $SLUGS as $slug ) {
		$p = get_page_by_path( $slug );
		if ( ! $p ) {
			echo '<tr><td>' . esc_html( $slug ) . '</td><td colspan="4" class="no">쪽 없음</td></tr>';
			continue;
		}
		$c = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
		$nt  = substr_count( $c, 'name="ssg-topic"' );
		$smp = substr_count( $c, 'class="sample"' );
		$has = ( false !== strpos( $c, $s ) );
		$ver = '—';
		if ( $has ) {
			$ver = ( false !== strpos( $c, 'stella-doors1-v9' ) )
				? '<span class="ok">v5 (지금 판)</span>'
				: '<span class="no">옛 판</span>';
		}
		echo '<tr><td><b>' . esc_html( $slug ) . '</b></td>'
		   . '<td>' . ( $nt ? $nt . '개' : '<span class="no">없음</span>' ) . '</td>'
		   . '<td>' . ( $smp ? '있음' : '<span class="dim">없음</span>' ) . '</td>'
		   . '<td>' . $ver . '</td>'
		   . '<td>' . ( $has ? '새것으로 바꿈' : '새로 붙임' ) . '</td></tr>';
		if ( $nt > 0 ) { $plan[] = array( $p->ID, $slug, $c, $has ); }
	}
	echo '</table>';

	if ( ! $plan ) {
		echo '<h1 class="no">손댈 쪽을 못 찾았습니다. 아무것도 안 바꿨습니다.</h1>';
		exit;
	}

	if ( 'apply' !== $mode ) {
		echo '<p>정말 고치시려면 <code>/?stella_doors1=apply</code> 를 여세요.</p>';
		exit;
	}

	$done = 0;
	foreach ( $plan as $row ) {
		list( $pid, $slug, $c, $has ) = $row;
		if ( ! get_post_meta( $pid, $META, true ) ) {
			update_post_meta( $pid, $META, $c );
		}
		if ( $has ) {
			$a = strpos( $c, $s );
			$z = strpos( $c, $e );
			if ( false === $z || $z < $a ) { continue; }
			$z   = $z + strlen( $e );
			$new = substr( $c, 0, $a ) . ltrim( $b, "\n" ) . substr( $c, $z );
		} else {
			$new = $c . $b;
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		$done++;
	}
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<p class="ok">' . $done . ' 쪽을 고쳤습니다. 이제 다섯 쪽 모두 <b>v9</b> 입니다.</p>';
	echo '<p>이렇게 열어보세요.</p>';
	echo '<pre style="background:#000;padding:12px;border-left:3px solid #D4AF6A">'
	   . "/door-love/?topic=결혼운\n"
	   . "/door-career/?topic=이직운\n"
	   . "/door-love/            (고르는 자리가 그대로 있어야 맞습니다)</pre>";
	echo '<p class="ok">되돌리시려면 <code>/?stella_doors1=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
