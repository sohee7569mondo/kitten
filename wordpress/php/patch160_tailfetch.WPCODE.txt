function stella_tailfetch_html() {
	return <<<'STELLATAILFETCH'
<script>
/* ═══════════════════════════════════════════════════════════════
   STELLA SAJU — 맺음말 앞 두 장 (가벼운 판)   BUILD: TAIL-FETCH-8
   ───────────────────────────────────────────────────────────────
   글을 담고 있지 않습니다. 짝 조각에게 두 칸만 물어봅니다.
   두 글자짜리 논리기호를 쓰지 않고 if 를 겹쳐 썼습니다.
   주소에 붙이는 「그리고」 기호도 글자로 직접 쓰지 않습니다.
   ═══════════════════════════════════════════════════════════════ */
(function(){
  'use strict';

  var AMP=String.fromCharCode(38);
  var ASKAT='/?stella_tailx=1';

  /* ══ 어디서 멈췄나 ═══════════════════════════════════
     주소에 stella_tailsay 가 붙어 있을 때만 글자로 보여줍니다.
     붙지 않았으면 한 글자도 안 나옵니다. */
  var SAY = false;
  if(String(location.search).indexOf('stella_tailsay')>=0){ SAY=true; }
  var LOG=[];
  function say(t){
    LOG.push(t);
    if(!SAY){ return; }
    var box=document.getElementById('tailsayout');
    if(!box){ return; }
    box.textContent='A 조각이 나갔습니다 (TAIL-FETCH-8)\n'+LOG.join('\n');
  }

  var ANIMAL=['쥐','소','호랑이','토끼','용','뱀','말','양','원숭이','닭','개','돼지'];

  /* 짝 조각에서 받아온 두 칸이 여기 담깁니다. */
  var DATA=null, asked=0, dead=0;

  function ask(topic, sign, animal){
    asked=1;
    var u=ASKAT+AMP+'tp='+encodeURIComponent(topic)+
               AMP+'sg='+encodeURIComponent(sign)+
               AMP+'ji='+encodeURIComponent(animal);
    say('물어봅니다 : '+u);
    try{
      fetch(u, {credentials:'same-origin'}).then(function(r){
        say('답이 왔습니다 : ' + r.status);
        if(!r.ok){ dead=1; return null; }
        return r.json();
      }).then(function(j){
        if(!j){ dead=1; say('글이 비었습니다'); return; }
        DATA=j;
        var z='없음', d='없음';
        if(j.zo){ z=j.zo[0]; }
        if(j.dd){ d=j.dd[0]; }
        say('받은 주제 : '+j.topic);
        say('별자리 칸 : '+z);
        say('띠 칸 : '+d);
      }).catch(function(e){ dead=1; say('물어보다 탈났습니다 : '+e); });
    }catch(e){ dead=1; say('물어볼 수가 없습니다 : '+e); }
  }

  function two(n){ if(n<10){ return '0'+n; } return ''+n; }

  function renumber(){
    var box=document.getElementById('ssb');
    if(!box){ return; }
    var pages=box.getElementsByClassName('page'), i, f, n=0;
    for(i=0;i<pages.length;i++){
      n+=1;
      f=pages[i].getElementsByClassName('folio')[0];
      if(f){ f.textContent=two(n); }
    }
    var bkN=document.getElementById('bkN');
    if(bkN){ bkN.textContent=n+'쪽'; }
  }

  /* ══ 그림 스물넷 ══════════════════════════════════════
     소희 님이 별자리 열둘, 띠 열둘을 그려 올리셨습니다.
     올리실 때 이름에 .webp 가 들어가고 실제 파일은 png/jpg 라
     확장자가 두 번 붙어 있습니다. 주소는 그대로 맞춰 씁니다. */
  var PICUP='https://stellasaju.com/wp-content/uploads/2026/09/';
  var SIGNPIC={
    '양자리':'aries',      '황소자리':'taurus',   '쌍둥이자리':'gemini',
    '게자리':'cancer',     '사자자리':'leo',      '처녀자리':'virgo',
    '천칭자리':'libra',    '전갈자리':'scorpio',  '사수자리':'sagittarius',
    '염소자리':'capricorn','물병자리':'aquarius', '물고기자리':'pisces'
  };
  var JIPIC={
    '쥐띠':'rat',    '소띠':'ox',      '호랑이띠':'tiger', '토끼띠':'rabbit',
    '용띠':'dragon', '뱀띠':'snake',   '말띠':'horse',     '양띠':'goat',
    '원숭이띠':'monkey','닭띠':'rooster','개띠':'dog',     '돼지띠':'pig'
  };
  function pic(file, name){
    if(!file){ return ''; }
    return '<p class="tailpic"><img loading="lazy" decoding="async" '+
           'src="'+PICUP+file+'" alt="'+name+'"></p>';
  }
  function signPic(sign){
    var e=SIGNPIC[sign];
    if(!e){ return ''; }
    return pic('sign-'+e+'.webp.png', sign);
  }
  function jiPic(animal){
    var e=JIPIC[animal];
    if(!e){ return ''; }
    return pic('ji-'+e+'.webp.jpg', animal);
  }

  function sheet(html){
    return '<div class="page" data-part="tail">'+
           '<div class="tailbox">'+html+'</div>'+
           '<div class="folio"></div></div>';
  }

  /* ── 폭 ──────────────────────────────────────────────
     ★ 2026-09-12 — 여기서 폭을 재던 것을 뺐습니다.
     책의 첫 문단을 재어 따라가게 했더니, 좁은 문단 하나를 잡아
     덤 쪽이 290px 로 쪼그라들었습니다. 재는 방식이 잘못이었습니다.

     책 전체 폭은 여백 조각(patch_spacerun)이 쪽마다 900px 로 잡습니다.
     덤 두 쪽도 같은 「쪽」이라 그 폭을 저절로 따릅니다.
     여기서는 아무것도 건드리지 않습니다. */

  function ttl(head, sub){
    if(!sub){ return head; }
    return head+' · '+sub;
  }

  function build(d, sign, animal){
    var made=[];
    /* ★ 2026-09-12 소희 님 : 「덤으로 드리는 이야기」보다
       「태어난 띠로 다시 보면」을 노란 딱지에 넣었으면 좋겠어.
       그래서 딱지가 그 쪽의 이름을 답니다. 눈썹줄은 딱지가 대신합니다. */
    function badge(name){
      return '<p class="bonus">'+name+'</p>';
    }

    /* 하나 · 별자리 */
    if(d.zo){
      made.push(sheet(
        badge('별자리로 다시 보면')+
        signPic(sign)+
        '<h2>'+d.q+'</h2>'+
        '<p class="dek">여덟 글자로 보신 것을 하늘로 한 번 더 겹쳐 봅니다. '+
        '여기서는 누구나 아는 태양 자리로만 봅니다</p>'+
        '<h3>'+ttl(sign, d.zo[0])+'</h3>'+
        d.zo[1]));
    }

    /* 둘 · 띠 */
    if(d.dd){
      made.push(sheet(
        badge('태어난 띠로 다시 보면')+
        jiPic(animal)+
        '<h2>'+ttl(animal, d.dd[0])+'</h2>'+
        '<p class="dek">여덟 글자의 맨 앞 글자가 곧 띠입니다. '+
        '어릴 때부터 알고 계셨던 그 한 글자가 사실은 사주의 첫 칸이에요</p>'+
        d.dd[1]+
        '<p class="mini">앞에서 본 여덟 글자가 세밀한 결이라면, 띠는 그 아래 깔린 바탕색입니다. '+
        '겹치는 데가 있다면 그 자리가 그만큼 분명하다는 뜻입니다.</p>'));
    }
    return made;
  }

  function start(){
    var box=document.getElementById('ssb');
    if(!box){ say('책 상자(#ssb)가 아직 없습니다'); return false; }
    if(box.getAttribute('data-tail')==='1'){
      say('빗장이 이미 걸려 있습니다 — 옛 조각이 먼저 넣었습니다');
      return true;
    }

    var order=null;
    try{ order=JSON.parse(localStorage.getItem('stella_demo')); }catch(e){ order=null; }
    if(!order){ say('주문(stella_demo)이 없습니다 — 받으신 풀이에서 열어야 합니다'); return false; }

    var S=window.StellaSaju;
    if(!S){ say('엔진(StellaSaju)이 아직 없습니다'); return false; }
    if(!S.lastChart){ say('엔진에 lastChart 가 없습니다'); return false; }
    var chart=S.lastChart();
    if(!chart){ say('여덟 글자(chart)가 아직 없습니다'); return false; }

    var topic=order.topic;
    if(!topic){ say('주문에 주제가 없습니다'); return true; }

    var sign='';
    if(chart.stars){ if(chart.stars.sun){ sign=chart.stars.sun.name; } }
    var animal='';
    if(chart.pillars){ if(chart.pillars.year){
      var jz=chart.pillars.year.ji;
      if(typeof jz==='number'){ animal=ANIMAL[jz]+'띠'; }
    } }

    /* 짝 조각에게 두 칸을 물어봅니다. 아직 안 왔으면 다음에 다시 옵니다. */
    if(dead){ say('짝 조각(B)에게 물어보지 못했습니다'); return true; }
    if(!asked){ say('주제 '+topic+' · '+sign+' · '+animal); ask(topic, sign, animal); return false; }
    if(!DATA){ return false; }
    if(!DATA.zo){ if(!DATA.dd){ say('두 칸이 다 비었습니다 — 주제 이름이 안 맞습니다'); return true; } }

    var made=build(DATA, sign, animal);
    if(!made.length){ return true; }

    var pages=box.getElementsByClassName('page');
    if(!pages.length){ return false; }

    /* ── 맺음말 쪽을 찾습니다. 아직 없으면 다음에 다시 옵니다. ── */
    var anchor=null, i, eb;
    for(i=0;i<pages.length;i++){
      eb=pages[i].getElementsByClassName('eyebrow')[0];
      if(!eb){ continue; }
      if(eb.textContent==='맺음말'){ anchor=pages[i]; break; }
    }
    if(!anchor){
      var names=[], q;
      for(q=0;q<pages.length;q++){
        var e3=pages[q].getElementsByClassName('eyebrow')[0];
        if(e3){ names.push(e3.textContent); }
      }
      say('맺음말 쪽을 못 찾았습니다. 쪽 '+pages.length+'개, 눈썹줄 : '+names.slice(-12).join(' / '));
      return false;
    }

    /* ── 근거 묶음을 맨 뒤로 옮깁니다 ──────────────────────
       계산 자료가 맺음말과 편지 사이를 막지 않게 진짜 뒤로 보냅니다.
         … 마지막에 놓는 세 장
         별자리로 다시 보면 · 띠로 보면
         맺음말 · ○○님께
         뒤에 붙이는 것 · 근거 하나~여섯             */
    var all=[].slice.call(pages);
    var backs=[], k, p, cls, ebt;
    for(k=0;k<all.length;k++){
      p=all[k];
      cls=p.className;
      if(cls.indexOf('divider')>=0){
        if(p.innerHTML.indexOf('뒤에 붙이는 것')>=0){ backs.push(p); continue; }
      }
      ebt='';
      var e2=p.getElementsByClassName('eyebrow')[0];
      if(e2){ ebt=e2.textContent; }
      if(ebt){ if(ebt.indexOf('근거')===0){ backs.push(p); } }
    }

    if(backs.length>0){
      var tail=null;
      for(k=all.length-1;k>=0;k--){
        if(all[k].className.indexOf('farewell')>=0){ tail=all[k]; break; }
      }
      if(!tail){
        for(k=all.length-1;k>=0;k--){
          if(backs.indexOf(all[k])>=0){ continue; }
          tail=all[k]; break;
        }
      }
      if(tail){
        var host=tail.parentNode;
        for(k=0;k<backs.length;k++){
          if(tail.nextSibling){ host.insertBefore(backs[k], tail.nextSibling); }
          else { host.appendChild(backs[k]); }
          tail=backs[k];
        }
      }
    }

    /* ── 덤 두 쪽을 맺음말 앞에 ── */
    var hold=document.createElement('div');
    hold.innerHTML=made.join('');
    while(hold.firstChild){ anchor.parentNode.insertBefore(hold.firstChild, anchor); }

    box.setAttribute('data-tail','1');
    renumber();
    say('두 쪽을 넣었습니다 — 끝');
    return true;
  }

  /* 다른 조각들이 쪽을 다 넣을 때까지 기다립니다. 최대 20초. */
  var tries=0;
  function tick(){
    tries+=1;
    var done=false;
    try{ done=start(); }catch(e){ done=true; }
    if(done){ return; }
    if(tries>60){ say('스무 초를 기다렸지만 안 됐습니다'); return; }
    setTimeout(tick, 330);
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', function(){ setTimeout(tick, 400); });
  } else { setTimeout(tick, 400); }
})();
</script>
