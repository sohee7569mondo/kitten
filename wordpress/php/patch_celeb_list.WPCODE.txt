/* ════════════════════════════════════════════════════════════
   스타 사주 — 인물 목록을 감추고 이름 검색만 남깁니다
   ★ 2026-09-15 · 소희 님께

   소희 님 : 「스타 운세 / 리스트 안보이게 / 이름만 쓰면 우리가
              알아서 사주 주는거지」

   무엇을 하나
     「지금까지 올라온 인물」 칸(#cRoster)만 감춥니다.
     검색칸과 찾기 단추는 그대로 남고, 이름을 적으면 그대로 나옵니다.

   ★ 지우지 않고 감춥니다 — 스위치를 끄면 그 즉시 도로 나옵니다.
   ★ 쪽 글을 안 고칩니다. 그래서 제 사본이 낡아도 안 놓칩니다.
   ★ 울타리는 슬러그가 아니라 화면으로 칩니다 — 스타 사주 쪽에만
     있는 칸(#ssc 와 #cRoster)이 보일 때만 움직입니다.

   ── 못 찾았을 때 ──
   목록이 없으면 손님이 누가 올라가 있는지 모릅니다. 그래서 검색칸
   밑에 한 줄을 둡니다 — 「찾으시는 분이 없으면 알려주세요」.
   그 글은 아래 HINT 를 고치시면 바뀝니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ════════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {
	?>
<!-- stella 스타사주 목록 감추기 · 판 2026-09-15 -->
<style id="stella-celeb-list">
/* 목록만 감춥니다. 검색칸(.find)은 그대로 둡니다. */
#ssc #cRoster{ display:none !important; }
#ssc .celeb-hint{ margin:10px 2px 0; font-size:.82rem; line-height:1.8;
  opacity:.72; }
</style>
<script>
(function(){
  if(window.StellaCelebList){ return; }
  window.StellaCelebList = 1;

  var HINT = '이름을 적고 찾기를 눌러 주세요. 찾으시는 분이 없으면 '
           + '알려주시면 확인해서 올리겠습니다.';

  function go(){
    var root = document.getElementById('ssc');
    if(!root){ return 0; }
    var find = root.querySelector('.find');
    if(!find){ return 0; }
    if(root.querySelector('.celeb-hint')){ return 1; }
    var p = document.createElement('p');
    p.className = 'celeb-hint';
    p.textContent = HINT;
    if(find.parentNode){ find.parentNode.insertBefore(p, find.nextSibling); }
    return 1;
  }

  var tries = 0;
  function tick(){
    tries++;
    var n = 0;
    try{ n = go(); }catch(e){ return; }
    if(n > 0){ return; }
    if(tries < 20){ setTimeout(tick, 300); }
  }
  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', tick);
  } else { tick(); }
})();
</script>
	<?php
} );
