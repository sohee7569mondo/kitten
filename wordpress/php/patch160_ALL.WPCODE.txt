add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_wed'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_wed'] ) );
	$META = '_stella_wed_bak';

	$page = get_page_by_path( 'reading-start' );
	if ( ! $page ) { echo '질문 쪽(reading-start)을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "바이트\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

	echo "═══ 결혼운 · 미혼인지 기혼인지 ═══\n";
	echo '질문 쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "바이트\n\n";

	if ( false !== strpos( $c, "k:'wed'" ) ) {
		echo "이미 들어가 있습니다.\n";
		echo "다시 넣으시려면 ?stella_wed=undo 로 걷어낸 뒤 하세요.\n";
		exit;
	}

	$A = 'var QS=QSETS[SLUG];';
	$n = substr_count( $c, $A );
	if ( 1 !== $n ) {
		echo '★ 붙일 자리가 ' . $n . "군데입니다. 한 군데여야 합니다.\n";
		echo '  찾는 것 : ' . $A . "\n";
		exit;
	}
	echo "붙일 자리를 찾았습니다 — 한 군데.\n";

	$made = str_replace( $A, $A . stella_wed_new(), $c );
	echo number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $made ) ) . '바이트 (+'
	   . number_format( strlen( $made ) - strlen( $c ) ) . ")\n\n";

	if ( 'apply' !== $mode ) {
		echo "아직 안 바꿨습니다. 정말 넣으시려면 ?stella_wed=apply\n";
		exit;
	}

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		update_post_meta( $id, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $made ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "넣었습니다.\n";
	echo "확인 : 연성의 신에서 결혼운을 고르고 질문 화면을 여세요.\n";
	echo "       맨 앞에 「지금 어느 쪽이신가요?」가 나와야 합니다.\n";
	echo "되돌리려면 ?stella_wed=undo\n";
	exit;
}, 1 );

function stella_wed_new() {
	return <<<'WEDNEW'


  /* ══ 2026-09-13 · 결혼운은 먼저 여쭙니다 ═══════════════════
     같은 결혼운이라도 미혼과 기혼은 읽을 것이 아예 다릅니다.
     결혼운을 고르신 분께만 맨 앞에 한 줄 더 여쭙니다.
     다른 주제에는 아무것도 안 보입니다. */
  (function(){
    var t='';
    if(ORDER){ if(ORDER.topic){ t=String(ORDER.topic).replace(/^\s+/,'').replace(/\s+$/,''); } }
    if(t!=='결혼운'){ return; }
    QS = QS.slice(0);
    QS.unshift({
      t:'지금 어느 쪽이신가요?',
      s:'먼저 여쭐 것',
      k:'wed',
      a:'아직 결혼 전이에요',
      b:'이미 결혼했어요'
    });
  })();
WEDNEW;
}

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_against'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_against'] ) );
	$META = '_stella_against_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "바이트\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_open'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_open'] ) );
	$META = '_stella_open_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "바이트\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

	echo "═══ 여는 글 · 주제마다 ═══\n";
	echo '책 쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "바이트\n\n";

	if ( false !== strpos( $c, 'OPEN_TOPIC' ) ) {
		echo "이미 들어가 있습니다.\n";
		echo "다시 넣으시려면 ?stella_open=undo 로 걷어낸 뒤 하세요.\n";
		exit;
	}

	$jobs = array(
		array(
			'name' => '하나 · 주제별 여는 말 열여섯 벌',
			'find' => stella_open_f0(),
			'to'   => stella_open_t0(),
		),
		array(
			'name' => '둘 · 말상자가 주제 글을 읽게',
			'find' => stella_open_f1(),
			'to'   => stella_open_t1(),
		),
		array(
			'name' => '셋 · 「두 부분으로 나뉩니다」를 장 안내로',
			'find' => stella_open_f2(),
			'to'   => stella_open_t2(),
			'soft' => 1,
		),
	);

	/* 판 OPEN-2 (2026-09-13)
	   셋째는 네 줄짜리 긴 글자 일치라, 책이 손질되면 한 글자 차이로
	   어긋납니다. 그래서 셋째만 「못 찾아도 멈추지 않고 건너뜁니다」로
	   바꿉니다. 대신 그 자리에 지금 무슨 글이 있는지 찍어 보여드립니다.
	   그 글을 저에게 주시면 제가 다음 판에서 정확히 맞추겠습니다. */
	$bad  = 0;
	$skip = 0;
	$show = '';
	$todo = array();
	foreach ( $jobs as $j ) {
		$n = substr_count( $c, $j['find'] );
		if ( 1 === $n ) {
			printf( "  %-40s 찾음\n", $j['name'] );
			$todo[] = $j;
			continue;
		}
		printf( "  %-40s ★ %d군데\n", $j['name'], $n );
		if ( ! isset( $j['soft'] ) ) { $bad++; continue; }
		$skip++;
		/* 짧은 표시로 다시 찾아 그 자리 글을 떠 옵니다 */
		$at = strpos( $c, '이 책은 두 부분으로 나뉩니다' );
		if ( false !== $at ) { $show = substr( $c, max( 0, $at - 260 ), 1500 ); }
	}
	if ( $bad ) {
		echo "\n한 군데가 아닌 자리가 있어 아무것도 안 건드렸습니다.\n이 글을 그대로 주세요.\n";
		exit;
	}
	if ( 0 === count( $todo ) ) {
		echo "\n넣을 자리가 하나도 없습니다. 그만둡니다.\n";
		exit;
	}

	if ( $skip ) {
		echo "\n★ 셋째는 건너뜁니다 — 책이 손질되어 글자가 안 맞습니다.\n";
		echo "  하나와 둘은 그대로 넣을 수 있습니다.\n";
		if ( '' !== $show ) {
			echo "\n──── 지금 그 자리에 있는 글 (이대로 저에게 주세요) ────\n";
			echo $show;
			echo "\n──── 여기까지 ─────────────────────────────────────\n";
		} else {
			echo "\n  「이 책은 두 부분으로 나뉩니다」라는 말도 못 찾았습니다.\n";
			echo "  이미 누가 고쳐 두었거나 이 조각을 벌써 넣으셨을 수 있습니다.\n";
		}
	}

	$made = $c;
	foreach ( $todo as $j ) { $made = str_replace( $j['find'], $j['to'], $made ); }

	echo "\n" . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $made ) ) . '바이트 (+'
	   . number_format( strlen( $made ) - strlen( $c ) ) . ")\n\n";

	if ( 'apply' !== $mode ) {
		echo "아직 안 바꿨습니다. 정말 넣으시려면 ?stella_open=apply\n";
		exit;
	}

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		update_post_meta( $id, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $made ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "넣었습니다. 주제를 바꿔가며 여는 쪽을 보세요.\n";
	echo "되돌리려면 ?stella_open=undo\n";
	exit;
}, 1 );

	);

	$found = 0;
	foreach ( $jobs as $j ) {
		$pk = $j['page'];
		$n  = substr_count( $now[ $pk ]['c'], $j['anchor'] );
		if ( 1 !== $n ) {
			echo '★ ' . $j['name'] . ' — 붙일 자리가 ' . $n . "군데입니다. 한 군데여야 합니다.\n";
			echo '   찾는 것 : ' . $j['anchor'] . "\n";
			echo "\n한 자리라도 틀리면 반쪽만 고쳐져 저울이 어긋납니다. 그만둡니다.\n";
			exit;
		}
		$piece = stella_qdivorce_piece( $j['piece'] );
		if ( 'after' === $j['how'] )       { $new = $j['anchor'] . $piece; }
		elseif ( 'before' === $j['how'] )  { $new = $piece . $j['anchor']; }
		else                               { $new = $piece; }

		echo $j['name'] . " — 찾았습니다 · " . number_format( strlen( $piece ) ) . "바이트\n";
		if ( 'preview' !== $mode ) {
			$now[ $pk ]['c'] = str_replace( $j['anchor'], $new, $now[ $pk ]['c'] );
		}
		$found++;
	}

	echo "\n세 자리 모두 찾았습니다.\n";

	if ( 'preview' === $mode ) {
		echo "\n아직 아무것도 안 바꿨습니다.\n";
		echo "정말 넣으시려면 ?stella_qdivorce=apply 로 오세요.\n";
		exit;
	}

	/* ── 두 쪽을 같이 저장합니다 ────────────────────────── */
	foreach ( $pages as $p ) {
		$slot = $now[ $p[0] ];
		$was  = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $slot['id'] ) );
		$out  = $slot['c'] . "\n<!-- " . $MARK . " -->\n";
		update_post_meta( $slot['id'], $slot['meta'], $was );
		$wpdb->update( $wpdb->posts, array( 'post_content' => $out ), array( 'ID' => $slot['id'] ) );
		clean_post_cache( $slot['id'] );
		echo $slot['label'] . ' : ' . number_format( strlen( $was ) ) . ' → '
			. number_format( strlen( $out ) ) . '바이트 (+'
			. number_format( strlen( $out ) - strlen( $was ) ) . ")\n";
	}

	echo "\n넣었습니다. 두 쪽이 한 쌍으로 맞았습니다.\n";
	echo "되돌리기 : ?stella_qdivorce=undo (두 쪽을 같이 되돌립니다)\n";
	exit;
}, 1 );
