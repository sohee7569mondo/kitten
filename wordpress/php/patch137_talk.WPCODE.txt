/* ════════════════════════════════════════════════════════════
   벼리 말 덜어내기   ★ 2026-09-09 · 소희 님께
   (reading-start · 가디언이 묻습니다 · 137)

   ── 무엇을 하나 ───────────────────────────────────────
   ① 첫 화면 벼리 말 여섯 줄에서 넷을 감춥니다.          (6·7)

        남김  소희님을 살펴보기 전에, 몇 가지만 물을게요.
        뺌    답을 듣기 전에 내가 먼저 읽은 것과…
        뺌    두 가지를 나란히 적어 드리겠습니다.
        뺌    그래야 내가 별을 읽은 것인지…
        뺌    맞고 틀리고가 없는 질문입니다…
        남김  일곱 문항, 2분이면 끝납니다.

      ★ 뺀 셋은 「자료를 다 세웠습니다」 쪽에 똑같이 나옵니다.

   ② 「마지막으로, 세 장을 뽑아주세요」를 크게.           (9)

   ③ 그 옆 설명 넉 줄을 한 줄로.                        (10)
        남김  카드는 앞으로 여섯 달을 읽습니다.
              카드를 눌러 세 장을 고르세요

   ④ 「가디언」을 그 문의 이름으로 바꿉니다.              (11)
        가디언에게 보내기   ->  벼리에게 보내기
        가디언이 쓰고 있습니다 -> 벼리가 쓰고 있습니다

   ── 어떻게 하나 ───────────────────────────────────────
   글을 지우지 않습니다. 화면이 다 그려진 뒤에 감추기만 합니다.
   되돌리면 그대로 돌아옵니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_talk=1
   ② 정말 고치기 https://stellasaju.com/?stella_talk=apply
   ③ 되돌리기    https://stellasaju.com/?stella_talk=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_talk'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_talk_backup';
	$MARK = 'stella-talk-0909';
	$mode = sanitize_key( wp_unslash( $_GET['stella_talk'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.85;'
	   . 'padding:34px 24px;max-width:940px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.35rem;margin:0 0 18px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.88rem}a{color:#D4AF6A}li{margin:6px 0}'
	   . 'code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:11.5px/1.6 monospace;border-left:3px solid #D4AF6A}</style>';

	$page = get_page_by_path( 'reading-start' );
	if ( ! $page ) { echo '<h1 class="no">reading-start 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	$b  = "\n<!-- " . $MARK . " 시작 -->\n<style>\n";
	$b .= "/* 벼리 말 덜어내기 · 2026-09-09 */\n";
	$b .= ".stella-hide-0909{ display:none !important; }\n";
	$b .= ".stella-big-0909{ font-size:1.42em !important; font-weight:700 !important;\n";
	$b .= "  line-height:1.42 !important; }\n";
	$b .= ".stella-one-0909{ margin:10px 0 0 !important; }\n";
	$b .= "</style>\n";
	$b .= "<script>\n";
	$b .= "/* ★ 이 안에서 홑 앰퍼샌드를 쓰지 않습니다 (사이트 규칙). */\n";
	$b .= "(function(){\n";
	$b .= "  var HIDE = [\n";
	$b .= "    '답을 듣기 전에 내가 먼저 읽은 것과',\n";
	$b .= "    '두 가지를 나란히 적어 드리겠습니다',\n";
	$b .= "    '그래야 내가 별을 읽은 것인지',\n";
	$b .= "    '맞고 틀리고가 없는 질문입니다',\n";
	$b .= "    '사주는 태어날 때 정해진 것을 읽고',\n";
	$b .= "    '자기 결의 카드만 골라',\n";
	$b .= "    '뽑으신 세 장은 책 마지막 장에'\n";
	$b .= "  ];\n";
	$b .= "  var BIG = '세 장을 뽑아주세요';\n";
	$b .= "  var ONE = '카드는 앞으로 여섯 달을 읽습니다.';\n";
	$b .= "  var NAMES = ['벼리','마루','아람','미르','아라','아르카나'];\n";
	$b .= "\n";
	$b .= "  function root(){\n";
	$b .= "    var r = document.getElementById('ssq');\n";
	$b .= "    if(r){ return r; }\n";
	$b .= "    return document.body;\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  /* 그 말을 담은 가장 작은 덩어리를 찾습니다 */\n";
	$b .= "  function smallest(needle){\n";
	$b .= "    var all = root().querySelectorAll('p,li,h1,h2,h3,h4,div,span,button');\n";
	$b .= "    var best = null;\n";
	$b .= "    var i, t;\n";
	$b .= "    for(i = 0; i < all.length; i++){\n";
	$b .= "      t = all[i].textContent;\n";
	$b .= "      if(!t){ continue; }\n";
	$b .= "      if(t.indexOf(needle) < 0){ continue; }\n";
	$b .= "      if(best === null){ best = all[i]; continue; }\n";
	$b .= "      if(t.length < best.textContent.length){ best = all[i]; }\n";
	$b .= "    }\n";
	$b .= "    return best;\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  /* 이 문에서 말하는 이의 이름을 찾습니다 */\n";
	$b .= "  function whoName(){\n";
	$b .= "    var t = document.body.textContent || '';\n";
	$b .= "    var best = '';\n";
	$b .= "    var bn = 0;\n";
	$b .= "    var i, n, at;\n";
	$b .= "    for(i = 0; i < NAMES.length; i++){\n";
	$b .= "      n = 0;\n";
	$b .= "      at = t.indexOf(NAMES[i]);\n";
	$b .= "      while(at >= 0){\n";
	$b .= "        n = n + 1;\n";
	$b .= "        at = t.indexOf(NAMES[i], at + 1);\n";
	$b .= "        if(n > 40){ break; }\n";
	$b .= "      }\n";
	$b .= "      if(n > bn){ bn = n; best = NAMES[i]; }\n";
	$b .= "    }\n";
	$b .= "    if(bn < 2){ return ''; }\n";
	$b .= "    return best;\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  /* 받침이 있으면 이, 없으면 가 */\n";
	$b .= "  function josa(name){\n";
	$b .= "    var ch = name.charCodeAt(name.length - 1);\n";
	$b .= "    if(ch < 44032){ return '이'; }\n";
	$b .= "    if(ch > 55203){ return '이'; }\n";
	$b .= "    if((ch - 44032) % 28 === 0){ return '가'; }\n";
	$b .= "    return '이';\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  function swapGuardian(name){\n";
	$b .= "    if(!name){ return; }\n";
	$b .= "    var it = document.createTreeWalker(root(), NodeFilter.SHOW_TEXT, null);\n";
	$b .= "    var list = [], nd;\n";
	$b .= "    while(true){\n";
	$b .= "      nd = it.nextNode();\n";
	$b .= "      if(!nd){ break; }\n";
	$b .= "      list.push(nd);\n";
	$b .= "    }\n";
	$b .= "    var i, o, t;\n";
	$b .= "    for(i = 0; i < list.length; i++){\n";
	$b .= "      o = list[i].nodeValue;\n";
	$b .= "      if(!o){ continue; }\n";
	$b .= "      if(o.indexOf('가디언') < 0){ continue; }\n";
	$b .= "      t = o.split('가디언이').join(name + josa(name));\n";
	$b .= "      t = t.split('가디언에게').join(name + '에게');\n";
	$b .= "      t = t.split('가디언').join(name);\n";
	$b .= "      if(t !== o){ list[i].nodeValue = t; }\n";
	$b .= "    }\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  var done = false;\n";
	$b .= "  function run(){\n";
	$b .= "    var i, el;\n";
	$b .= "    for(i = 0; i < HIDE.length; i++){\n";
	$b .= "      el = smallest(HIDE[i]);\n";
	$b .= "      if(el){ el.classList.add('stella-hide-0909'); }\n";
	$b .= "    }\n";
	$b .= "    var big = smallest(BIG);\n";
	$b .= "    if(big){\n";
	$b .= "      big.classList.add('stella-big-0909');\n";
	$b .= "      if(!document.getElementById('stellaOne0909')){\n";
	$b .= "        var p = document.createElement('p');\n";
	$b .= "        p.id = 'stellaOne0909';\n";
	$b .= "        p.className = 'stella-one-0909';\n";
	$b .= "        p.textContent = ONE;\n";
	$b .= "        if(big.parentNode){ big.parentNode.insertBefore(p, big.nextSibling); }\n";
	$b .= "      }\n";
	$b .= "      done = true;\n";
	$b .= "    }\n";
	$b .= "    swapGuardian(whoName());\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  function start(){\n";
	$b .= "    run();\n";
	$b .= "    var k = 0;\n";
	$b .= "    var tick = setInterval(function(){\n";
	$b .= "      k = k + 1;\n";
	$b .= "      run();\n";
	$b .= "      if(k > 40){ clearInterval(tick); }\n";
	$b .= "    }, 500);\n";
	$b .= "  }\n";
	$b .= "\n";
	$b .= "  if(document.readyState === 'loading'){\n";
	$b .= "    document.addEventListener('DOMContentLoaded', start);\n";
	$b .= "  } else { start(); }\n";
	$b .= "})();\n";
	$b .= "</script>\n<!-- " . $MARK . " 끝 -->\n";

	if ( false !== strpos( $b, '&' ) ) {
		echo '<h1 class="no">앰퍼샌드가 섞여 있습니다. 넣지 않았습니다.</h1>';
		exit;
	}

	$s   = '<!-- ' . $MARK . ' 시작 -->';
	$e   = '<!-- ' . $MARK . ' 끝 -->';
	$has = ( false !== strpos( $c, $s ) );

	echo '<h1>' . ( 'apply' === $mode ? '고쳤습니다' : '이렇게 바꿉니다 (아직 안 고쳤습니다)' ) . '</h1>';
	echo '<ul>'
	   . '<li>벼리 말 여섯 줄에서 <b>넷을 감춥니다</b></li>'
	   . '<li>「마지막으로, 세 장을 뽑아주세요」를 <b>1.4배</b>로</li>'
	   . '<li>카드 설명 넉 줄을 <b>한 줄</b>로</li>'
	   . '<li>「가디언」을 <b>그 문의 이름</b>으로</li>'
	   . '</ul>';

	$check = array(
		'맞고 틀리고가 없는 질문입니다',
		'답을 듣기 전에 내가 먼저 읽은 것과',
		'세 장을 뽑아주세요',
		'가디언에게 보내기',
	);
	$t = '';
	foreach ( $check as $k ) {
		$n = substr_count( $c, $k );
		$t .= str_pad( $k, 34 ) . ( $n ? $n . ' 군데' : '못 찾음' ) . "\n";
	}
	echo '<pre>' . esc_html( $t ) . '</pre>';
	echo '<p class="dim">글을 지우지 않습니다. 다 그려진 뒤에 감추기만 합니다. '
	   . ( $has ? '이미 들어가 있는 덩어리를 <b>새것으로 바꿉니다</b>.' : '덩어리를 쪽 맨 뒤에 <b>새로 붙입니다</b>.' )
	   . '</p>';

	if ( 'apply' !== $mode ) {
		echo '<p>정말 고치시려면 <code>/?stella_talk=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}

	if ( $has ) {
		$a = strpos( $c, $s );
		$z = strpos( $c, $e );
		if ( false === $z || $z < $a ) {
			echo '<h1 class="no">표는 찾았는데 끝을 못 찾았습니다. 아무것도 안 바꿨습니다.</h1>';
			exit;
		}
		$z   = $z + strlen( $e );
		$new = substr( $c, 0, $a ) . ltrim( $b, "\n" ) . substr( $c, $z );
	} else {
		$new = $c . $b;
	}

	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<p class="ok">고쳤습니다. ' . number_format( strlen( $c ) ) . ' -> '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p><a href="/reading-start/?v=' . time() . '">열어보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_talk=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
