/* ════════════════════════════════════════════════════════════
   장 속표지 높이 — 한 단어 고치기 · 소희 님께

   무엇이 있었나
     앞서 넣은 규칙이 책 안에 잘 들어갔는데, 그 중 min-height
     한 줄만 다른 규칙에 밀리고 있었습니다. 재보니 이렇게 나왔습니다.

       새 규칙 있음 1군데
       display flex        ← 제 규칙이 이김
       min-height 0px      ← 이 줄만 짐

     어딘가에 min-height 를 0 으로 못박은 규칙이 있어서
     제 82vh 를 눌러버린 것입니다.

   무엇을 하나
     앞서 넣은 덩어리에서 두 줄에 !important 를 붙입니다.

       min-height:82vh;   →  min-height:82vh !important;
       min-height:88vh;   →  min-height:88vh !important;

     새 덩어리를 또 넣지 않습니다. 있던 것 두 글자만 고칩니다.

   ★ 먼저 ?stella_dvfull=apply 를 넣으신 뒤에 이것을 쓰세요.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_dvfix=1
   ② 정말 고치기 https://stellasaju.com/?stella_dvfix=apply
   ③ 되돌리기    https://stellasaju.com/?stella_dvfix=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_dvfix'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_dvfix_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_dvfix'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:880px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}a{color:#D4AF6A}'
	   . 'code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:12.5px/1.6 monospace;border-left:3px solid #D4AF6A}</style>';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '<h1 class="no">reading-book 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	$swap = array(
		'min-height:82vh;' => 'min-height:82vh !important;',
		'min-height:88vh;' => 'min-height:88vh !important;',
	);

	$found = 0;
	foreach ( $swap as $from => $to ) {
		$found += substr_count( $c, $from );
	}
	$already = substr_count( $c, 'min-height:82vh !important' );

	echo '<h1>' . ( 'apply' === $mode ? '고쳤습니다' : '고치면 이렇게 됩니다 (아직 안 고쳤습니다)' ) . '</h1>';

	if ( $already > 0 ) {
		echo '<h1 class="ok">이미 고쳐져 있습니다.</h1>';
		echo '<p class="dim">그래도 작게 나오면 캐시입니다. 주소 뒤에 <code>&v=2</code> 를 붙여 열어보세요.</p>';
		exit;
	}
	if ( 0 === $found ) {
		echo '<h1 class="no">고칠 줄을 못 찾았습니다.</h1>';
		echo '<p>먼저 <code>/?stella_dvfull=apply</code> 를 넣으셔야 합니다. '
		   . '아무것도 바꾸지 않았습니다.</p>';
		exit;
	}

	printf( '<p><b>%d</b> 줄에 <code>!important</code> 를 붙입니다.</p>', $found );
	echo '<pre>min-height:82vh;   →  min-height:82vh !important;' . "\n"
	   . 'min-height:88vh;   →  min-height:88vh !important;</pre>';
	echo '<p class="dim">재보니 <code>display flex</code> 는 제 규칙이 이겼는데 '
	   . '<code>min-height</code> 만 0px 로 눌려 있었습니다. 어딘가에 0 으로 못박은 '
	   . '규칙이 있다는 뜻이라, 같은 무게로 맞춰줍니다.</p>';

	if ( 'apply' !== $mode ) {
		echo '<p>정말 고치시려면 <code>/?stella_dvfix=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}
	$new = strtr( $c, $swap );
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<p class="ok">고쳤습니다.</p>';
	echo '<p>다시 재보세요 — <code>min-height</code> 가 <b>0px</b> 이 아니라 '
	   . '<b>1049px</b> 처럼 나와야 맞습니다.</p>';
	echo '<p><a href="/reading-book/?demo=door-health&stella_bk=1">책 열고 재보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_dvfix=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
