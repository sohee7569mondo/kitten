/* ══════════════════════════════════════════════════════════
   이미 본 주제를 또 사지 않게 — 문 여섯
   ★ 2026-09-10 · 소희 님께

   소희 님 : 「퇴사운 본 적 있는데 또 결제 되는 것도 문제일듯」

   맞습니다. 손님이 「아까 그거 다시 보려고」 누른 것뿐인데
   돈이 또 나가면 그 손님은 다시 안 옵니다.

   그런데 서버에는 이미 길이 나 있었습니다.

     GET  /wp-json/stella/v1/me       지난 주문 스무 건 (주제 · 날짜 · 아이디)
     POST /wp-json/stella/v1/reopen   그 책을 구슬 안 받고 다시 폅니다

   아무도 모르게 숨어 있던 것을 문 앞으로 꺼냅니다.
   이미 본 주제를 고르면 단추 위에 이렇게 뜹니다.

     ┌──────────────────────────────────────┐
     │ 퇴사운은(는) 9월 3일에 이미 보셨습니다.  │
     │  [ 그때 책 다시 펼치기 ]                │
     │  구슬이 들지 않습니다. 새로 보시려면    │
     │  아래 단추를 그대로 누르세요 —          │
     │  사주는 해가 바뀌면 흐름이 달라집니다.   │
     └──────────────────────────────────────┘

   ★ 서버 쪽은 한 줄도 안 건드립니다.
   ★ 문 여섯 쪽 끝에 덩어리 하나씩만 붙입니다. 안에 있는 글은
     한 글자도 안 건드립니다.

   해보기   https://stellasaju.com/?stella_again=preview
   붙이기   https://stellasaju.com/?stella_again=apply
   되돌리기 https://stellasaju.com/?stella_again=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_again'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode  = sanitize_text_field( wp_unslash( $_GET['stella_again'] ) );
	$META  = '_stella_again_bak';
	$SLUGS = array( 'door-love', 'door-career', 'door-health',
	                'door-fortune', 'door-astro', 'door-tarot' );

	if ( 'undo' === $mode ) {
		$n = 0;
		foreach ( $SLUGS as $slug ) {
			$p = get_page_by_path( $slug );
			if ( ! $p ) { continue; }
			$old = get_post_meta( $p->ID, $META, true );
			if ( '' === $old ) { continue; }
			$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $p->ID ) );
			clean_post_cache( $p->ID );
			delete_post_meta( $p->ID, $META );
			$n++;
		}
		echo $n . '쪽을 되돌렸습니다.';
		exit;
	}

	$add = <<<'STELLA_AGAIN_HTML'
<!-- wp:html -->
<script>
/* ══ 이미 본 주제를 또 사지 않게 ─ 2026-09-10 ══════════════════
   같은 주제를 다시 고르면, 결제하기 전에 알려줍니다.

     퇴사운은 9월 3일에 이미 보셨습니다.
     → 그때 책 다시 펼치기 (구슬 안 듭니다)

   서버에는 이미 길이 나 있었습니다.
     GET  /wp-json/stella/v1/me       지난 주문 스무 건
     POST /wp-json/stella/v1/reopen   구슬 안 받고 그 책을 다시 폅니다
   아무도 모르게 숨어 있던 것을 문 앞으로 꺼냅니다.

   서버 쪽은 한 줄도 안 건드립니다. ═══════════════════════════ */
(function(){
  if(!window.STELLA_USER){ return; }          /* 손님이 아직 안 들어오셨으면 */

  var go = document.getElementById('goBtn');
  if(!go){ return; }

  /* ── 빛깔 ─ 2026-09-10
     문이 밝은 종이인지 밤하늘인지에 따라 글씨가 묻히지 않도록,
     상자가 스스로 제 바탕을 가집니다. 어느 문에서든 읽힙니다. */
  var css = document.createElement('style');
  css.textContent =
    '#ssg .stre{ display:none; margin:0 0 14px; padding:15px 17px;' +
    '  border:0; border-radius:12px; background:#1B1636; color:#F5F1E8;' +
    '  font-size:.93rem; line-height:1.7;' +
    '  box-shadow:0 8px 22px -14px rgba(27,22,54,.65); }' +
    '#ssg .stre b{ color:#F0D9A8; }' +
    '#ssg .stre .again{ display:inline-block; margin-top:9px; padding:9px 17px;' +
    '  border-radius:999px; background:#F0D9A8; color:#1B1636;' +
    '  font-weight:800; text-decoration:none; cursor:pointer; font-size:.9rem; }' +
    '#ssg .stre .no{ display:block; margin-top:9px; font-size:.83rem;' +
    '  color:rgba(245,241,232,.78); }';
  document.head.appendChild(css);

  var box = document.createElement('div');
  box.className = 'stre';
  go.parentNode.insertBefore(box, go);

  var SEEN = {};   /* 주제 => { id, date } 가장 나중 것 */

  function pickedTopic(){
    var el = document.querySelector('input[name="ssg-topic"]:checked');
    return el ? el.value : '';
  }

  function jong(w){
    if(!w){ return -1; }
    var c = w.charCodeAt(w.length - 1);
    if(c < 0xAC00){ return -1; }
    if(c > 0xD7A3){ return -1; }
    return (c - 0xAC00) % 28;
  }
  function eunT(w){ return jong(w) > 0 ? '은' : '는'; }

  function today(){
    var d = new Date();
    function two(n){ return (n < 10 ? '0' : '') + n; }
    return d.getFullYear() + '-' + two(d.getMonth() + 1) + '-' + two(d.getDate());
  }

  function prettyDate(s){
    /* 2026-09-03 → 9월 3일 */
    if(String(s) === today()){ return '오늘'; }
    var m = String(s || '').split('-');
    if(m.length !== 3){ return String(s || ''); }
    return parseInt(m[1], 10) + '월 ' + parseInt(m[2], 10) + '일';
  }

  function draw(){
    var t = pickedTopic();
    var hit = t ? SEEN[t] : null;
    if(!hit){ box.style.display = 'none'; return; }
    box.innerHTML =
      '<b>' + t + '</b>' + eunT(t) + ' ' + prettyDate(hit.date) +
      (prettyDate(hit.date) === '오늘' ? ' 이미 보셨습니다.' : '에 이미 보셨습니다.') +
      '<br><a class="again" href="#">그때 책 다시 펼치기</a>' +
      '<span class="no">구슬이 들지 않습니다. 새로 보시려면 아래 단추를 그대로 누르세요 ' +
      '— 사주는 해가 바뀌면 흐름이 달라져서 새 풀이가 나옵니다.</span>';
    box.style.display = 'block';
    var a = box.querySelector('.again');
    a.addEventListener('click', function(e){
      e.preventDefault();
      a.textContent = '펼치는 중…';
      fetch('/wp-json/stella/v1/reopen', {
        method:'POST', headers:{'Content-Type':'application/json'},
        credentials:'same-origin', body:JSON.stringify({ id: hit.id })
      }).then(function(r){ return r.json(); })
        .then(function(b){
          if(b){ if(b.next){ location.href = b.next; return; } }
          location.href = '/reading-book/';
        })
        .catch(function(){ a.textContent = '다시 펼치기'; });
    });
  }

  fetch('/wp-json/stella/v1/me', { credentials:'same-origin' })
    .then(function(r){ return r.json(); })
    .then(function(b){
      if(!b){ return; }
      var rows = b.orders;
      if(!rows){ return; }
      if(!rows.length){ return; }
      var i, row;
      for(i = 0; i < rows.length; i++){
        row = rows[i];
        if(!row){ continue; }
        if(!row.topic){ continue; }
        SEEN[row.topic] = { id: row.id ? row.id : '', date: row.date ? row.date : '' };
      }
      var rs = document.querySelectorAll('input[name="ssg-topic"]');
      for(i = 0; i < rs.length; i++){
        rs[i].addEventListener('change', draw);
      }
      draw();
    })
    .catch(function(){});
})();
</script>
<!-- /wp:html -->
STELLA_AGAIN_HTML;

	echo "═══ 이미 본 주제 알려주기 ═══\n\n";

	$todo = array();
	foreach ( $SLUGS as $slug ) {
		$p = get_page_by_path( $slug );
		if ( ! $p ) { echo sprintf( "  %-14s ★ 쪽이 없습니다\n", $slug ); continue; }
		$c = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );

		if ( false !== strpos( $c, 'stella/v1/reopen' ) ) {
			echo sprintf( "  %-14s 이미 붙어 있습니다\n", $slug );
			continue;
		}
		if ( false === strpos( $c, "name=\"ssg-topic\"" ) ) {
			echo sprintf( "  %-14s ★ 주제 고르는 칸이 없습니다 — 건너뜁니다\n", $slug );
			continue;
		}
		if ( false === strpos( $c, "id=\"goBtn\"" ) ) {
			echo sprintf( "  %-14s ★ 단추를 못 찾았습니다 — 건너뜁니다\n", $slug );
			continue;
		}

		$todo[ $slug ] = array( $p->ID, $c );
		echo sprintf( "  %-14s 쪽 %d · %s글자\n", $slug, $p->ID, number_format( strlen( $c ) ) );
	}

	if ( ! $todo ) { echo "\n붙일 곳이 없습니다."; exit; }

	if ( 'preview' === $mode ) {
		echo "\n" . count( $todo ) . "쪽 끝에 덩어리 하나씩 붙입니다.\n";
		echo "안에 있는 글은 한 글자도 안 건드립니다.\n";
		echo "붙이시려면 ?stella_again=apply\n";
		exit;
	}

	foreach ( $todo as $slug => $row ) {
		list( $id, $c ) = $row;
		if ( '' === (string) get_post_meta( $id, $META, true ) ) {
			update_post_meta( $id, $META, $c );
		}
		$new = rtrim( $c ) . "\n\n" . $add . "\n";
		$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
		clean_post_cache( $id );
	}

	echo "\n" . count( $todo ) . "쪽에 붙였습니다.\n";
	echo "이미 보신 주제를 골라보세요. 단추 위에 안내가 뜹니다.\n";
	echo "되돌리려면 ?stella_again=undo\n";
	exit;
}, 1 );
