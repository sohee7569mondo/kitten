/* ════════════════════════════════════════════════════════════
   책을 종이로 바꿉니다 — 연한 바탕 · 진한 고딕 · 오행 다섯 빛깔

   지금 책은 이렇게 되어 있습니다.

     바탕      거의 검정 (#0B0817)
     글씨      고운바탕 · 명조
     여덟 글자 여덟 자가 전부 같은 금색

   정했던 것은 이것이었습니다.

     바탕      연한 색
     글씨      진한 고딕
     여덟 글자 목 · 화 · 토 · 금 · 수 를 색깔로 나눔

   세 번째가 제일 아깝습니다. 바로 다음 쪽에 오행 저울이 나오는데,
   손님이 「내 목이 어느 글자지」 하고 되짚을 수가 없습니다. 색이
   없으니까요. 그래서 여덟 글자와 오행 저울이 같은 색을 쓰게 했고,
   표 밑에 색깔 안내줄을 한 줄 넣었습니다.

     목 나무   푸른색   #1E7A4A
     화 불     붉은색   #C0392B
     토 흙     누런색   #96700F
     금 쇠     잿빛     #59636F
     수 물     검푸름   #1D3E77

   한자 여덟 자만 명조로 남겼습니다. 한자는 명조가 제 모양이고,
   본문 고딕하고 나란히 놓으면 「이건 옛 글자」라고 눈에 바로
   들어옵니다. 이것도 고딕으로 하고 싶으시면 말씀만 주세요.

   위쪽 코드는 한 줄도 지우지 않았습니다. 색과 글꼴을 스타일 맨
   끝에서 다시 정하는 방식이라, 그 덩어리만 걷어내면 원래 밤하늘
   책으로 그대로 돌아갑니다.

   페이지 내용을 wp_update_post 로 저장하면 워드프레스가 <script> 를
   지워버립니다. 그래서 데이터베이스에 바로 씁니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로 바꾸세요.
     기본값인 「사이트 전체 헤더」로 두면 아래 주소가 그냥 홈으로 넘어갑니다.

   ① 먼저 보기   https://stellasaju.com/?stella_paper=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_paper=apply
   ③ 되돌리기    https://stellasaju.com/?stella_paper=undo

   바꾸고 나서 볼 곳
     https://stellasaju.com/reading-book/?demo=door-love-2
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_paper'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$PAGE = 160;
	$META = '_stella_paper_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_paper'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $PAGE ) );
	if ( ! $row ) { echo '<h1>160쪽을 못 찾았습니다.</h1>'; exit; }

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $PAGE, $META, true );
		echo '<h1>되돌리기</h1>';
		if ( ! $old ) { echo '<p>되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $PAGE ) );
		clean_post_cache( $PAGE );
		delete_post_meta( $PAGE, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		echo '<p><a href="/reading-book/?demo=door-love-2">책 보기</a></p>';
		exit;
	}

	$c = $row->post_content;
	$before = strlen( $c );

	$jobs = array();

	$jobs[] = array(
		'name' => '① 고딕 글꼴 불러오기',
		'find' => '/family=Gowun\\+Batang:wght@400;700\\&amp;/u',
		'to'   => "family=Gowun+Batang:wght@400;700&amp;family=Noto+Sans+KR:wght@400;500;700&amp;",
	);

	$jobs[] = array(
		'name' => '② 종이 바탕 · 고딕 · 오행 빛깔 (CSS)',
		'find' => '/<\/style>\\s+<div\\s+id="ssb">/u',
		'to'   => "\n/* ══ 2026-09-07 · 종이책 ═══════════════════════════════════════\n   연한 바탕 · 진한 고딕 · 여덟 글자는 오행 다섯 빛깔로.\n   위에 있는 색과 글꼴을 여기서 다시 정합니다. 나중에 쓴 것이 이깁니다.\n   위쪽 코드는 한 줄도 지우지 않았으니, 이 덩어리만 걷어내면\n   원래 밤하늘 책으로 그대로 돌아갑니다. */\n#ssb{\n  --void:#E9E5DA;\n  --page:#FCFAF4;\n  --page-edge:#F1EDE1;\n  --gold:#8A6A24;\n  --gold-lite:#6B5219;\n  --speak:#5A43A4;\n  --speak-lite:#46318A;\n  --star:#171420;\n  --soft:rgba(23,20,32,.84);\n  --dim:rgba(23,20,32,.55);\n  --rule:rgba(138,106,36,.30);\n  --rule-s:rgba(90,67,164,.32);\n  --serif:'Pretendard','Noto Sans KR','Apple SD Gothic Neo','Malgun Gothic',system-ui,sans-serif;\n  --han:'Gowun Batang','Nanum Myeongjo',serif;\n  --e-mok:#1E7A4A;\n  --e-hwa:#C0392B;\n  --e-to:#96700F;\n  --e-geum:#59636F;\n  --e-su:#1D3E77;\n  font-weight:500;\n  letter-spacing:-.01em;\n}\n\n/* 연한 바탕에서는 밤하늘용 그림자가 때가 탄 것처럼 보입니다 */\n#ssb .page{\n  box-shadow:0 1px 3px rgba(31,24,12,.07), 0 14px 34px -22px rgba(31,24,12,.32);\n}\n\n/* 어두운 판에서만 맞던 바탕들 — 종이 쪽으로 바꿉니다 */\n#ssb .foldbar, #ssb details.readfold{ background:rgba(138,106,36,.055); }\n#ssb .foldbtn:hover{ background:var(--gold); color:#FCFAF4; }\n#ssb .voice{ background:rgba(138,106,36,.075); }\n#ssb tr.now td{ background:rgba(138,106,36,.11); }\n#ssb .card{ background:linear-gradient(180deg, rgba(138,106,36,.07), rgba(138,106,36,0)); }\n#ssb .keep{ background:rgba(138,106,36,.06); border-color:rgba(138,106,36,.28); }\n#ssb .quad-grid .you{ background:rgba(90,67,164,.10); }\n#ssb .bk-bar{ background:rgba(252,250,244,.94); }\n#ssb .el-bar{ background:rgba(23,20,32,.09); }\n\n/* 고딕은 명조보다 가늘어 보입니다. 제목과 강조를 한 단 올립니다 */\n#ssb h2, #ssb h3, #ssb .el-n, #ssb .cover h1, #ssb .divider h2{ font-weight:700; }\n#ssb em, #ssb .keep .k .kt, #ssb .fw-name{ font-weight:700; }\n\n/* ── 오행 다섯 빛깔 ─────────────────────\n   목은 나무라 푸르고, 화는 불이라 붉고, 토는 흙이라 누렇고,\n   금은 쇠라 잿빛이고, 수는 물이라 검푸릅니다.\n   여덟 글자와 오행 저울이 같은 색을 씁니다. 손님이 표를 보고\n   「내 목이 어느 글자지」를 색으로 되짚을 수 있게요. */\n#ssb [data-el=\"목\"]{color:var(--e-mok);}\n#ssb [data-el=\"화\"]{color:var(--e-hwa);}\n#ssb [data-el=\"토\"]{color:var(--e-to);}\n#ssb [data-el=\"금\"]{color:var(--e-geum);}\n#ssb [data-el=\"수\"]{color:var(--e-su);}\n\n#ssb .chart .han{\n  font-family:var(--han); font-weight:700; font-size:2rem;\n  letter-spacing:.09em; color:var(--star);\n}\n#ssb .chart .kor{ color:var(--dim); }\n@media(max-width:640px){ #ssb .chart .han{ font-size:1.5rem; } }\n\n#ssb .ellegend{\n  display:flex; flex-wrap:wrap; justify-content:center; gap:7px 18px;\n  margin:14px 0 0; font-family:var(--mono); font-size:.7rem; letter-spacing:.08em;\n}\n#ssb .ellegend span{ display:inline-flex; align-items:center; gap:5px; font-weight:700; }\n#ssb .ellegend span::before{\n  content:''; width:9px; height:9px; border-radius:50%; background:currentColor;\n}\n\n#ssb .el[data-el] .el-n{ color:inherit; }\n#ssb .el[data-el] .el-bar span{ background:currentColor; }\n#ssb .el[data-el] .el-v{ color:var(--dim); }\n#ssb .el.zero[data-el] .el-n, #ssb .el.zero[data-el] .el-v{ color:var(--dim); }\n#ssb .el.zero[data-el] .el-bar span{ background:rgba(23,20,32,.16); }\n\n@media print{\n  html, body{ background:#FFFFFF !important; }\n}\n</style>\n<div id=\"ssb\">\n",
	);

	$jobs[] = array(
		'name' => '③ 글자마다 오행 붙이기',
		'find' => '/var\\s+heads=\\[\'연주\',\'월주\',\'일주\',\'시주\'\\];/u',
		'to'   => "  var heads=['연주','월주','일주','시주'];\n  /* 2026-09-07 · 여덟 글자를 오행 빛깔로 나눠 칠합니다.\n     바깥의 GAN_EL/JI_EL 을 끌어오지 않고 여기서 따로 적습니다.\n     이 자리에서 그 표가 보이는지 확신할 수 없어서, 글자 자체로 찾습니다. */\n  var EL5={'甲':'목','乙':'목','丙':'화','丁':'화','戊':'토','己':'토',\n           '庚':'금','辛':'금','壬':'수','癸':'수',\n           '子':'수','丑':'토','寅':'목','卯':'목','辰':'토','巳':'화',\n           '午':'화','未':'토','申':'금','酉':'금','戌':'토','亥':'수'};\n  function elSpan(ch){\n    var e=EL5[ch] || '';\n    return '<span data-el=\"'+e+'\">'+ch+'</span>';\n  }\n",
	);

	$jobs[] = array(
		'name' => '④ 여덟 글자에 색 입히기 + 색깔 안내줄',
		'find' => '/cols\\.map\\(function\\(c,i\\)\\{\\s+if\\(!c\\)\\{\\s+return\\s+\'<div><div\\s+class="han"\\s+style="opacity:\\.25">—<\/div><div\\s+class="kor">시간\\s+모름<\/div><\/div>\';\\s+\\}\\s+return\\s+\'<div><div\\s+class="han">\'\\+c\\.han\\+\'<\/div><div\\s+class="kor">\'\\+c\\.kor\\+\'<\/div>\'\\+\\s+\'<div\\s+class="sip">\'\\+\\(sipTop\\[i\\]\\|\\|\'\'\\)\\+\'<br>\'\\+\\(sipBot\\[i\\]\\|\\|\'\'\\)\\+\'<\/div><\/div>\';\\s+\\}\\)\\.join\\(\'\'\\)\\+\\s+\'<\/div>\'\\+/u',
		'to'   => "      cols.map(function(c,i){\n        if(!c){ return '<div><div class=\"han\" style=\"opacity:.25\">—</div><div class=\"kor\">시간 모름</div></div>'; }\n        return '<div><div class=\"han\">'+elSpan(c.han.charAt(0))+elSpan(c.han.charAt(1))+'</div>'+\n          '<div class=\"kor\">'+c.kor+'</div>'+\n          '<div class=\"sip\">'+(sipTop[i]||'')+'<br>'+(sipBot[i]||'')+'</div></div>';\n      }).join('')+\n    '</div>'+\n    '<p class=\"ellegend\"><span data-el=\"목\">목 나무</span><span data-el=\"화\">화 불</span>'+\n    '<span data-el=\"토\">토 흙</span><span data-el=\"금\">금 쇠</span>'+\n    '<span data-el=\"수\">수 물</span></p>'+\n",
	);

	$jobs[] = array(
		'name' => '⑤ 오행 저울도 같은 색으로',
		'find' => '/return\\s+\'<div\\s+class="el\'\\+\\(f\\.count===0\\?\'\\s+zero\':\'\'\\)\\+\'">\'\\+\\s+\'<div\\s+class="el\\-n">\'\\+f\\.element\\+\'<\/div>\'\\+/u',
		'to'   => "        return '<div class=\"el'+(f.count===0?' zero':'')+'\" data-el=\"'+f.element+'\">'+\n          '<div class=\"el-n\">'+f.element+'</div>'+\n",
	);

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$ok = true;
	foreach ( $jobs as $j ) {
		$n = preg_match_all( $j['find'], $c );
		echo '<h2>' . esc_html( $j['name'] ) . '</h2>';
		if ( 1 === $n ) {
			echo '<p class="ok">찾았습니다 — 한 군데.</p>';
		} elseif ( 0 === $n ) {
			echo '<p class="no">못 찾았습니다. 이미 바꿨거나 코드가 달라졌습니다.</p>';
			$ok = false;
		} else {
			echo '<p class="no">' . intval( $n ) . '군데가 걸립니다. 한 군데여야 안전합니다.</p>';
			$ok = false;
		}
	}

	if ( ! $ok ) {
		echo '<p class="no">한 자리가 한 군데씩 걸릴 때만 바꿉니다. 아무것도 건드리지 않았습니다.</p>';
		exit;
	}

	$new = $c;
	foreach ( $jobs as $j ) {
		$new = preg_replace( $j['find'], str_replace( '\\', '\\\\', $j['to'] ), $new, 1 );
		if ( null === $new ) { echo '<p class="no">바꾸다가 멈췄습니다. 아무것도 저장하지 않았습니다.</p>'; exit; }
	}

	echo '<p>' . number_format( $before ) . ' → ' . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_paper=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $PAGE, $META, true ) ) {
		update_post_meta( $PAGE, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $PAGE ) );
	clean_post_cache( $PAGE );

	echo '<p class="ok">바꿨습니다.</p>';
	echo '<p>되돌리려면 <code>/?stella_paper=undo</code></p>';
	echo '<p><a href="/reading-book/?demo=door-love-2">책 보기</a></p>';
	exit;
} );
