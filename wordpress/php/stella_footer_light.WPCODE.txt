/* ════════════════════════════════════════════════════════════
   맨 아래 띠(푸터)도 흰 바탕으로   ★ 2026-09-08

   소희 님 :「푸터도 하얗게 가야 할 거 같아 거기만 어두워서 좀」

   맞습니다. 쪽은 다 종이빛인데 맨 아래만 남색이라 거기서
   뚝 끊깁니다.

   ── 왜 스니펫으로 하나 ─────────────────────────
   푸터는 쪽 안에 있는 글이 아니라 테마가 그리는 자리입니다.
   그래서 지금까지 만든 쪽 패치들이 닿지 않습니다.
   이건 사이트 전체에 색표 한 덩이를 얹는 방식으로 합니다.

   ── 되돌리기 ───────────────────────────────────
   이 스니펫을 끄기만 하면 됩니다. 쪽 내용을 하나도 안 건드리니
   따로 되돌릴 것이 없습니다.

   ── 하는 일 ────────────────────────────────────
   · 푸터 바탕을 종이빛(#F1EDE3)으로
   · 푸터 안의 어두운 바탕들을 걷어냅니다 (테마가 안쪽 덩이에
     색을 칠해두는 경우가 많습니다)
   · 글자와 링크를 진한 색으로
   · 위쪽 쪽과 갈리도록 가는 줄 하나

   ★ 가운데 스텔라사주 그림은 어두운 판이라 흰 바탕에서 네모로
     떠 보일 수 있습니다. 밝은 판 그림이 있으시면 그걸로 바꾸는
     것이 제일 깔끔합니다. 없으면 아래 「그림 테두리」 줄이
     모서리를 둥글게 다듬어 줍니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ════════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {

	if ( is_admin() ) { return; }

	?>
<style id="stella-footer-light">
/* ══ 2026-09-08 · 맨 아래 띠를 종이빛으로 ══════════════════ */

/* 테마마다 푸터를 부르는 이름이 달라서 몇 가지를 함께 겁니다 */
footer.wp-block-template-part,
.wp-site-blocks > footer,
.wp-block-template-part.site-footer,
body > footer,
#colophon,
.site-footer{
  background-color:#F1EDE3 !important;
  color:#4E4763 !important;
  border-top:1px solid #E2DACB !important;
}

/* 안쪽 덩이에 칠해둔 어두운 바탕을 걷어냅니다 */
footer.wp-block-template-part *,
.wp-site-blocks > footer *,
.site-footer *{
  background-color:transparent !important;
}

/* 글자와 링크 */
footer.wp-block-template-part,
footer.wp-block-template-part p,
footer.wp-block-template-part li,
footer.wp-block-template-part span,
.site-footer, .site-footer p, .site-footer li, .site-footer span{
  color:#4E4763 !important;
}
footer.wp-block-template-part a,
.site-footer a{
  color:#4E4763 !important;
  text-decoration:none;
}
footer.wp-block-template-part a:hover,
.site-footer a:hover{
  color:#6B4FB0 !important;
}

/* 어두운 판 그림이 네모로 떠 보이지 않게 모서리만 다듬습니다 */
footer.wp-block-template-part img,
.site-footer img{
  border-radius:10px;
}

/* 구분선이 어두운 색이면 옅게 */
footer.wp-block-template-part hr,
footer.wp-block-template-part .wp-block-separator,
.site-footer hr{
  border-color:#E2DACB !important;
  background-color:#E2DACB !important;
  opacity:1 !important;
}
</style>
	<?php
}, 100 );
