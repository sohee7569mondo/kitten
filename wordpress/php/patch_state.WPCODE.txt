   소희 님 물음 :
     「샘플 목록에 버전을 써놓을까? 우리가 헷갈리지 않게?
      안 써도 네가 알지?」

   솔직히 말씀드리면 — 저는 확실히는 모릅니다.
   제 저장소 사본은 사이트의 진짜 글보다 뒤처져 있을 때가 많고,
   오늘만 해도 두 번 틀렸습니다 (연애 문 넣는 칸 · 문 이름).

   그런데 손으로 적어둔 버전은 또 금방 낡습니다. 오늘 목차가
   그랬고 수비학이 그랬습니다. 적어둔 사람이 고치는 걸 잊으면
   거짓말이 됩니다.

   그래서 손으로 안 적고 기계가 적게 했습니다.

   패치를 넣을 때마다 그 쪽에 「바꾸기 전 글」을 하나씩 넣어둡니다
   (되돌리기에 쓰는 것입니다). 그 자국이 곧 「이 패치가 들어갔다」는
   표시입니다. 이 스니펫은 그 자국을 모아서 보여줍니다.

     나의 전자책 (160)   1,193,838 바이트 · 2026-09-08
       tail split amp div name paper say byline taste ev
       box voice ask head sip sum gap2 twice

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_state'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:960px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 6px}'
	   . 'h2{color:#D4AF6A;font-size:1rem;margin:22px 0 4px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}'
	   . '.tags{margin:4px 0 0}.tag{display:inline-block;background:rgba(212,175,106,.14);'
	   . 'color:#F0D9A8;border:1px solid rgba(212,175,106,.3);border-radius:20px;'
	   . 'padding:1px 10px;margin:3px 4px 0 0;font-size:.8rem;'
	   . 'font-family:ui-monospace,monospace}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	echo '<h1>어느 쪽에 무엇이 들어갔나</h1>';
	echo '<p class="dim">패치가 남긴 자국을 모은 것입니다. 손으로 적은 것이 아니라 '
	   . '틀릴 수가 없습니다. 이 화면은 읽기만 했습니다.</p>';

	$rows = $wpdb->get_results(
		"SELECT p.ID, p.post_title, p.post_modified, p.post_status,
		        LENGTH(p.post_content) AS len, m.meta_key
		 FROM {$wpdb->posts} p
		 JOIN {$wpdb->postmeta} m ON m.post_id = p.ID
		 WHERE p.post_type = 'page'
		   AND m.meta_key LIKE '\\_stella\\_%\\_backup'
		 ORDER BY p.ID ASC, m.meta_key ASC" );

	if ( ! $rows ) {
		echo '<p class="no">자국이 하나도 없습니다. 패치를 아직 안 넣으셨거나 '
		   . '되돌리기를 하신 것입니다.</p>';
		exit;
	}

	$byPage = array(); $count = array();
	foreach ( $rows as $r ) {
		$tag = $r->meta_key;
		$tag = preg_replace( '/^_stella_/', '', $tag );
		$tag = preg_replace( '/_backup$/', '', $tag );
		if ( ! isset( $byPage[ $r->ID ] ) ) {
			$byPage[ $r->ID ] = array(
				'title' => $r->post_title, 'mod' => $r->post_modified,
				'len' => $r->len, 'status' => $r->post_status, 'tags' => array() );
		}
		$byPage[ $r->ID ]['tags'][] = $tag;
		$count[ $tag ] = isset( $count[ $tag ] ) ? $count[ $tag ] + 1 : 1;
	}

	foreach ( $byPage as $id => $p ) {
		echo '<h2>' . esc_html( $p['title'] ) . ' <span class="dim">(' . intval( $id );
		if ( 'publish' !== $p['status'] ) { echo ' &middot; ' . esc_html( $p['status'] ); }
		echo ')</span></h2>';
		echo '<p class="dim">' . number_format( (int) $p['len'] ) . ' 바이트 &middot; 마지막 손질 '
		   . esc_html( substr( $p['mod'], 0, 16 ) ) . ' &middot; '
		   . count( $p['tags'] ) . '개</p>';
		echo '<p class="tags">';
		foreach ( $p['tags'] as $t ) { echo '<span class="tag">' . esc_html( $t ) . '</span>'; }
		echo '</p>';
	}

	ksort( $count );
	echo '<h2>패치별로 몇 쪽에 들어갔나</h2><p class="tags">';
	foreach ( $count as $t => $n ) {
		echo '<span class="tag">' . esc_html( $t ) . ' &middot; ' . intval( $n ) . '</span>';
	}
	echo '</p>';

	echo '<h2>정리</h2>';
	echo '<p class="ok">' . count( $byPage ) . '쪽에 ' . count( $rows ) . '개의 자국이 있습니다.</p>';
	echo '<p class="dim">되돌리기를 하면 그 자국도 같이 지워집니다. '
	   . '그래서 여기 보이는 것이 지금 사이트에 실제로 들어가 있는 것입니다.</p>';
	exit;
}, 0 );
