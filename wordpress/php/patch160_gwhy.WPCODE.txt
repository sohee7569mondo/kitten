	?>
<script>
(function(){
  if(window.__gwhyErr!==undefined){ return; }
  window.__gwhyErr=null;
  window.addEventListener('error', function(ev){
    if(window.__gwhyErr){ return; }
    var m=''; var where='';
    try{
      m=String((ev.error!==null)?((ev.error?ev.error.message:'')||ev.message||''):(ev.message||''));
      where=String(ev.filename||'')+' 줄 '+String(ev.lineno||'');
    }catch(e){}
    window.__gwhyErr={ msg:m, at:where, ln:(ev.lineno||0), file:String(ev.filename||'') };
  }, true);
  window.addEventListener('unhandledrejection', function(ev){
    if(window.__gwhyErr){ return; }
    var m='';
    try{ m=String(ev.reason?(ev.reason.message||ev.reason):''); }catch(e){}
    window.__gwhyErr={ msg:m, at:'(약속이 깨짐)' };
  }, true);
})();
</script>
	<?php
}, 1 );

add_action( 'wp_head', function () {

	if ( ! isset( $_GET['gwhy'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	$js = <<<'STELLAGWHY'
<script>
/* ═══ 궁합이 왜 안 나오나 (GWHY-3 · 살아 움직이는 띠) ═══ */
(function(){

  var AMP=String.fromCharCode(38);

  function el(id){ return document.getElementById(id); }
  function txt(e){
    if(!e){ return '(칸 없음)'; }
    var v=String(e.value===undefined?'':e.value);
    if(v===''){ return '(빈칸)'; }
    return v;
  }
  function mark(ok){ if(ok){ return '○'; } return '✗'; }
  function read(k){
    try{
      var v=localStorage.getItem(k);
      if(!v){ return null; }
      return JSON.parse(v);
    }catch(e){ return null; }
  }

  /* 지금 화면에 있는 p 로 시작하는 칸 이름을 다 모읍니다 */
  function pIds(){
    var out=[];
    try{
      var all=document.querySelectorAll('[id]');
      for(var i=0;i<all.length;i++){
        var id=all[i].id;
        if(id.charAt(0)==='p'){ out.push(id); }
      }
    }catch(e){}
    return out;
  }

  function collect(){
    var lines=[];
    try{ gather(lines); }
    catch(e){
      var m='';
      try{ m=String(e.message||e); }catch(e9){}
      lines.push('★★ 띠가 읽다가 멈췄습니다 — '+m);
      lines.push('(여기까지는 읽은 것입니다. 멈춘 자리가 곧 단서입니다.)');
    }
    return lines;
  }

  /* 탈이 난 줄을 보려고 쪽을 한 번 더 받아옵니다 (같은 주소라 안전합니다) */
  var SRC=null, tried=false;
  function grabSrc(){
    if(tried){ return; }
    tried=true;
    try{
      window.fetch(location.href, { credentials:'same-origin' })
        .then(function(r){ return r.text(); })
        .then(function(x){ SRC=String(x).split(String.fromCharCode(10)); })
        .catch(function(){});
    }catch(e){}
  }

  function gather(lines){
    var path=String(location.pathname||'');

    /* ── 주문 ── */
    var o=read('stella_demo');
    if(!o){
      lines.push('주문(stella_demo) 이 아예 없습니다');
    } else {
      var tp=String(o.topic===undefined?'':o.topic);
      lines.push('주문 있음 · 주제 = 「'+(tp===''?'(없음)':tp)+'」 · 문 = '+(o.guardian_slug||'(없음)'));
      var P=o.partner;
      if(!P){
        lines.push('★ 주문에 partner 가 없습니다');
      } else {
        lines.push('주문 partner ○ · 해='+(P.year||'?')+' 달='+(P.month||'?')+' 날='+(P.day||'?')+
                   ' 이름='+(P.alias||'(없음)'));
      }
    }

    /* ── 챙겨둔 것 · 체 ── */
    var st=read('stella_partner');
    if(st){
      lines.push('챙겨둔 상대(stella_partner) ○ · 해='+(st.year||'?')+' 달='+(st.month||'?')+
                 ' 날='+(st.day||'?')+' 이름='+(st.alias||'(없음)'));
    } else {
      lines.push('챙겨둔 상대(stella_partner) ✗');
    }
    lines.push('PFIX 체(StellaFillPartner) '+mark(typeof window.StellaFillPartner==='function'));

    var er=window.__gwhyErr;
    if(er){
      lines.push('★★ 이 쪽에서 처음 난 탈 — '+(er.msg||'(까닭 없음)')+'   ['+(er.at||'')+']');
      if(er.ln){
        grabSrc();
        if(SRC){
          var a=er.ln-2; if(a<1){ a=1; }
          var b=er.ln+2; if(b>SRC.length){ b=SRC.length; }
          lines.push('── 그 자리의 글 ──');
          for(var q=a;q<=b;q++){
            var one=String(SRC[q-1]||'');
            if(one.length>300){ one=one.slice(0,300)+' …'; }
            lines.push((q===er.ln?'★ ':'  ')+q+' | '+one);
          }
        } else {
          lines.push('(그 자리의 글을 받아오는 중…)');
        }
      }
    } else {
      lines.push('이 쪽에서 난 탈 없음');
    }

    /* ── 책 쪽 ── */
    if(path.indexOf('reading-book')>=0){
      var GC=window.StellaGunghapCalc, GT=window.StellaGunghapText;
      lines.push('셈법 Calc '+mark(!!GC)+'   본문 Text '+mark(!!GT));
      var why=[];
      if(!GC){ why.push('셈법 조각'); }
      if(!GT){ why.push('본문 조각'); }
      if(!o){ why.push('주문'); }
      if(o){
        if(String(o.topic||'')!=='궁합'){ why.push('주제가 궁합이 아님'); }
        var Q=o.partner||{};
        if(!Q.year){ why.push('partner.year'); }
        if(!Q.month){ why.push('partner.month'); }
        if(!Q.day){ why.push('partner.day'); }
      }
      if(why.length===0){ lines.push('★ 재료는 다 있습니다'); }
      else { lines.push('★ 없는 것 — '+why.join(' · ')); }

      /* ── 실제로 한 번 돌려봅니다 ── */
      if(why.length===0){
        var S=window.StellaSaju;
        if(!S){
          lines.push('★ 셈 바탕 StellaSaju 이 없습니다');
        } else {
          var pr=o.profile||{}, PP=o.partner||{};
          var loc=null, chart=null, yours=null, GK=null, html=null;
          var step='', bad='';
          try{
            step='① 태어난 곳 찾기';  loc=S.locate(pr.city, pr.country);
            step='② 내 원국';        chart=S.compute({ year:pr.year, month:pr.month, day:pr.day,
                                       hour:pr.hour, minute:pr.minute||0, sex:pr.sex,
                                       lon:loc.lon, lat:loc.lat, tz:loc.tz });
            step='③ 상대 원국';      yours=S.compute({ year:PP.year, month:PP.month, day:PP.day,
                                       hour:PP.hour, minute:PP.minute||0, sex:PP.sex,
                                       lon:loc.lon, lat:loc.lat, tz:loc.tz });
            step='④ 셈법 keys';      GK=GC.keys(chart, yours, { year:(new Date()).getFullYear() });
            step='⑤ 본문 build';     html=GT.build(GK, { rel:(PP.rel==='couple')?'couple':'lover',
                                       alias:PP.alias });
            step='';
          }catch(e){
            bad=String((e!==null)?(e.message||e):'');
          }
          if(step!==''){
            lines.push('★★ '+step+' 에서 멈췄습니다 — '+(bad===''?'(까닭 없음)':bad));
          } else {
            lines.push('돌려보니 다 지나갑니다 · 본문 '+String(html).length+'자');
            var gd=(GK.grade?GK.grade.name:'(구간 이름 없음)');
            var sc=GK.score||{};
            lines.push('총점 '+GK.total+' · '+gd+
                       ' · 성격'+sc.char+' 연애'+sc.love+
                       ' 결혼'+sc.marry+' 금전'+sc.money);
          }
        }
      }

      /* ── 화면이 정말 비어 있나 ── */
      var bk=el('bkBook'), be=el('bkEmpty');
      var inner=0;
      if(bk){
        inner=String(bk.innerHTML).length;
        var mineEcho=el('gwhyEcho');
        if(mineEcho){ inner=inner-String(mineEcho.outerHTML).length; }
        if(inner<0){ inner=0; }
      }
      lines.push('책 그릇 bkBook '+mark(!!bk)+
                 (bk?(' · 안에 '+inner+'자 (이 띠는 뺀 숫자)'):'')+
                 ' · 빈화면 bkEmpty '+(be?(be.style.display==='flex'?'떠 있음':'숨어 있음'):'(없음)'));

    }
  }

  var SKIN='background:#0B0817;color:#BCADEA;padding:14px 16px;'+
    'font-family:monospace;font-size:13px;line-height:1.8;'+
    'border:2px solid #9E2B50;white-space:normal;';

          echo.style.cssText=SKIN+'margin:20px;';
          echo.innerHTML=h;
          bk.appendChild(echo);
        }
      }
    }catch(e2){}
  }

    setInterval(draw, 2000);
  }

	   . esc_html( $slug ) . '</div>';
}, 99 );
