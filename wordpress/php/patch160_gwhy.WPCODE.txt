/* ══════════════════════════════════════════════════════════
   궁합이 왜 안 나오나 — 살아 움직이는 띠
                                        ★ 2026-09-13 · 소희 님께
                                        판 GWHY-3

   ── 판 3 에서 바뀐 것 ─────────────────────────────────
   판 2 의 띠는 쪽이 열릴 때 한 번만 찍었습니다. 그래서 소희 님이
   궁합을 고르기 전의 모습만 보였습니다 — 「상대 칸이 하나도 없음」이
   그래서 나온 것일 수 있습니다.

   판 3 은 **2초마다 다시 읽습니다.** 궁합을 고르고 칸을 채우는 대로
   띠가 따라 바뀝니다. 그래서 칸이 언제 생기는지 눈으로 보입니다.

   그리고 문 쪽에서는 세 가지를 더 봅니다 —
       지금 화면에 있는 p 로 시작하는 칸 이름을 **다 늘어놓습니다**
       pBase(상대 칸을 담는 그릇)가 있나
       쪽 글 안에 id=pYear 라는 글자가 있기는 한가

   ── 보는 법 ───────────────────────────────────────────
       문    https://stellasaju.com/door-love/?gwhy=1
             → 궁합을 고르고 남편분 생년월일을 채워 보세요.
               채우는 동안 띠가 바뀌는지 봐주시면 됩니다.
       책    https://stellasaju.com/reading-book/?gwhy=1

   ★ 로그인한 관리자에게만 보입니다. ?gwhy=1 이 없으면 아무 일도
     안 합니다.

   ── 붙이는 법 ─────────────────────────────────────────
   WPCode -> 이미 있는 patch160_gwhy 조각에 덮어쓰기 -> 저장
   ★ 위치(Location)는 「어디서나 실행 / Run Everywhere」
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ══════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {

	if ( ! isset( $_GET['gwhy'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	$js = <<<'STELLAGWHY'
<script>
/* ═══ 궁합이 왜 안 나오나 (GWHY-3 · 살아 움직이는 띠) ═══ */
(function(){

  function el(id){ return document.getElementById(id); }
  function txt(e){
    if(!e){ return '(칸 없음)'; }
    var v=String(e.value===undefined?'':e.value);
    if(v===''){ return '(빈칸)'; }
    return v;
  }
  function mark(ok){ if(ok){ return '○'; } return '✗'; }
  function read(k){
    try{
      var v=localStorage.getItem(k);
      if(!v){ return null; }
      return JSON.parse(v);
    }catch(e){ return null; }
  }

  /* 지금 화면에 있는 p 로 시작하는 칸 이름을 다 모읍니다 */
  function pIds(){
    var out=[];
    try{
      var all=document.querySelectorAll('[id]');
      for(var i=0;i<all.length;i++){
        var id=all[i].id;
        if(id.charAt(0)==='p'){ out.push(id); }
      }
    }catch(e){}
    return out;
  }

  function collect(){
    var lines=[];
    var path=String(location.pathname||'');

    /* ── 주문 ── */
    var o=read('stella_demo');
    if(!o){
      lines.push('주문(stella_demo) 이 아예 없습니다');
    } else {
      var tp=String(o.topic===undefined?'':o.topic);
      lines.push('주문 있음 · 주제 = 「'+(tp===''?'(없음)':tp)+'」 · 문 = '+(o.guardian_slug||'(없음)'));
      var P=o.partner;
      if(!P){
        lines.push('★ 주문에 partner 가 없습니다');
      } else {
        lines.push('주문 partner ○ · 해='+(P.year||'?')+' 달='+(P.month||'?')+' 날='+(P.day||'?')+
                   ' 이름='+(P.alias||'(없음)'));
      }
    }

    /* ── 챙겨둔 것 · 체 ── */
    var st=read('stella_partner');
    if(st){
      lines.push('챙겨둔 상대(stella_partner) ○ · 해='+(st.year||'?')+' 달='+(st.month||'?')+
                 ' 날='+(st.day||'?')+' 이름='+(st.alias||'(없음)'));
    } else {
      lines.push('챙겨둔 상대(stella_partner) ✗');
    }
    lines.push('PFIX 체(StellaFillPartner) '+mark(typeof window.StellaFillPartner==='function'));

    /* ── 책 쪽 ── */
    if(path.indexOf('reading-book')>=0){
      var GC=window.StellaGunghapCalc, GT=window.StellaGunghapText;
      lines.push('셈법 Calc '+mark(!!GC)+'   본문 Text '+mark(!!GT));
      var why=[];
      if(!GC){ why.push('셈법 조각'); }
      if(!GT){ why.push('본문 조각'); }
      if(!o){ why.push('주문'); }
      if(o){
        if(String(o.topic||'')!=='궁합'){ why.push('주제가 궁합이 아님'); }
        var Q=o.partner||{};
        if(!Q.year){ why.push('partner.year'); }
        if(!Q.month){ why.push('partner.month'); }
        if(!Q.day){ why.push('partner.day'); }
      }
      if(why.length===0){ lines.push('★ 다 있습니다 — 궁합 책이 나와야 합니다'); }
      else { lines.push('★ 없는 것 — '+why.join(' · ')); }
    }

    /* ── 문 쪽 ── */
    if(path.indexOf('door-love')>=0){
      var te=document.querySelector('input[name=ssg-topic]:checked');
      lines.push('고른 주제 = '+(te?te.value:'(아직 안 고름)'));

      var ids=pIds();
      lines.push('지금 화면의 p 칸 '+ids.length+'개 — '+(ids.length?ids.join(' · '):'(하나도 없음)'));
      lines.push('그릇 pBase '+mark(!!el('pBase')));

      var inMarkup=false;
      try{
        if(String(document.body.innerHTML).indexOf('pYear')>=0){ inMarkup=true; }
      }catch(e){}
      lines.push('쪽 글 안에 pYear 라는 글자가 있나 '+mark(inMarkup));

      lines.push('상대 칸 값 — 이름='+txt(el('pAlias'))+' 관계='+txt(el('pRel'))+
                 ' 성별='+txt(el('pSex'))+' 해='+txt(el('pYear'))+' 달='+txt(el('pMonth'))+
                 ' 날='+txt(el('pDay'))+' 시='+txt(el('pHour'))+' 분='+txt(el('pMin')));
    }

    return lines;
  }

  var bar=null;
  function draw(){
    if(!bar){
      bar=document.createElement('div');
      bar.style.cssText='position:fixed;top:0;left:0;right:0;z-index:999999;'+
        'background:#0B0817;color:#BCADEA;padding:10px 14px;'+
        'font-family:monospace;font-size:12px;line-height:1.7;'+
        'border-bottom:2px solid #9E2B50;max-height:48vh;overflow:auto;';
      document.body.insertBefore(bar, document.body.firstChild);
    }
    var lines=collect();
    var h='<b style=color:#fff>[GWHY] 궁합이 왜 안 나오나</b> '+
          '<span style=color:#7a7a8c>(2초마다 다시 읽습니다)</span><br>';
    for(var i=0;i<lines.length;i++){
      h+=lines[i].split('<').join('(').split('>').join(')')+'<br>';
    }
    bar.innerHTML=h;
  }

  function start(){
    draw();
    setInterval(draw, 2000);
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', start);
  } else {
    start();
  }

})();
</script>
STELLAGWHY;

	echo $js;
}, 99 );
