/* ══════════════════════════════════════════════════════════
   「1부 / 2부」 안내를 장 안내로 — 앵커 없이
                                        ★ 2026-09-13 · 소희 님께
                                        판 INTRO2-1

   ── 무엇이 남아 있었나 ────────────────────────────────
   patch160_open 은 셋을 고칩니다.
       하나 · 주제별 여는 말 표를 넣기          → 들어갔습니다
       둘   · 말상자가 그 표를 읽게              → 들어갔습니다
       셋   · 「1부 / 2부」 안내를 장 안내로     → **못 들어갔습니다**

   셋째는 네 줄짜리 긴 글자 맞추기였는데, 그 자리 글이 책 손질로
   바뀌어 있어 한 글자도 안 맞았습니다. 그래서 「0군데」였습니다.
   지금 손님이 읽는 안내는 이렇습니다 —

       이 책은 두 부분으로 나뉩니다. 1부는 … 2부는 …

   그런데 책은 이미 **장**으로 짜여 있습니다. 안내와 실제가
   어긋나 있는 자리입니다.

   ── 이번엔 앵커를 안 잡습니다 ─────────────────────────
   쪽 글을 고치지 않습니다. 책이 다 그려진 **화면에서** 그 문단을
   찾아 장 안내로 바꿔치기합니다. 글자가 한 자 달라도 상관없습니다.
   「이 책은 두 부분으로 나뉩니다」라는 말만 찾으면 됩니다.

   바꾸는 것은 그 문단과, 바로 뒤에 이어지는 같은 무리의 문단들뿐입니다.
   (「이렇게 나눈 데에는…」 · 「두 내용이 같은 곳도…」 · 작은 글씨 한 줄)
   그 뒤 문단이 그 무리가 아니면 거기서 멈춥니다.

   ── 무엇으로 바뀌나 ───────────────────────────────────
       사주가 말하는 ○○님 모습   — 태어난 때만 보고 썼습니다
       현실에서의 ○○님 모습      — 답해주신 일곱 가지를 읽고 다시
       [주제마다 다른 한 줄]       — 여기서부터가 물어보신 것
       마지막에 놓는 세 장         — 타로 세 장으로 여섯 달
       별자리와 띠로 다시 보면      — 같은 때를 다른 언어로
       맨 뒤에 셈의 근거
       그리고 마지막에 기억하실 세 가지

   가운데 한 줄은 주제마다 다릅니다 (열여섯 주제).
   결혼운은 미혼·기혼이 갈립니다 — topicKey 를 봅니다.

   ── 붙이는 법 ─────────────────────────────────────────
   WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.

   되돌리기가 없습니다 — 스니펫을 끄면 원래대로입니다.

   ── 검사한 것 ─────────────────────────────────────────
   php -l 통과 · 앰퍼샌드 0개 · node --check 통과
   지금 살아 있는 안내문 그대로를 놓고 실제로 바꿔봄 —
   그 무리만 지우고 뒤 문단은 그대로 둡니다
   ══════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {
	?>
<script>
/* ═══ 「1부 / 2부」 안내를 장 안내로 (INTRO2-1) ═══
   쪽 글은 안 건드립니다. 다 그려진 화면에서 그 문단만 바꿉니다. */
(function(){

  var HEAD='이 책은 두 부분으로 나뉩니다';

  /* 그 문단 뒤에 같이 딸려 있는 무리 — 이 말로 시작하면 같이 지웁니다 */
  var KIN=['이렇게 나눈 데에는','그렇게 나눈 이유가','두 내용이 같은 곳도',
           '두 쪽이 어긋나는','제가 전해드리고 싶은 말이','전해드리고 싶은 말이',
           '그리고 두 쪽이 어긋나는'];

  var MID={
    '연애운':'어떤 사람에게 마음이 가고, 그 마음을 어떻게 쓰시는지',
    '결혼운_미혼':'어떤 사람과 어떤 생활을 함께하고 싶은지',
    '결혼운_기혼':'지금의 결혼생활에서 무엇이 편하고 무엇이 힘든지',
    '결혼운':'어떤 사람과 어떤 생활을 함께하고 싶은지',
    '이별운':'당신이 어떻게 사랑을 끝내고, 마음을 정리하는 사람인지',
    '이혼운':'이 결혼에서 무엇이 오래 쌓였고, 이제 무엇을 정리해야 하는지',
    '재회운':'다시 만났을 때 무엇이 달라져야 하는지',
    '짝사랑':'당신의 마음이 지금 어디에 서 있고, 어디까지 다가가고 싶은지',
    '짝사랑운':'당신의 마음이 지금 어디에 서 있고, 어디까지 다가가고 싶은지',
    '궁합':'두 사람이 만나면 어디에서 가까워지고, 어디에서 부딪히는지',
    '취업운':'어떤 자리에서 당신의 능력이 제대로 쓰이고 인정받는지',
    '직장운':'어떤 자리에서 당신이 덜 지치고 오래 일하시는지',
    '이직운':'지금 자리를 떠나 무엇을 얻고, 어떤 자리로 옮겨가는지',
    '퇴사운':'지금까지 해온 일에서 무엇을 내려놓고 무엇을 가져가야 하는지',
    '퇴직운':'지금까지 해온 일에서 무엇을 내려놓고 무엇을 가져가야 하는지',
    '사업운':'당신은 무엇을 만들어야 돈이 움직이고, 어떻게 키워가는 사람인지',
    '재물운':'당신의 돈이 어디서 모이고, 어디에서 새는지'
  };

  var AMP=String.fromCharCode(38);
  function esc(s){
    s=String(s===undefined?'':s);
    s=s.split(AMP).join(AMP+'amp;');
    s=s.split('<').join(AMP+'lt;');
    s=s.split('>').join(AMP+'gt;');
    return s;
  }
  function read(k){
    try{
      var v=localStorage.getItem(k);
      if(!v){ return null; }
      return JSON.parse(v);
    }catch(e){ return null; }
  }

  function intro(){
    var o=read('stella_demo');
    var nm='손님', tp='';
    if(o){
      if(o.profile){ if(o.profile.name){ nm=o.profile.name; } }
      tp=String(o.topic===undefined?'':o.topic);
    }
    var k=tp;
    if(typeof window.topicKey==='string'){ if(window.topicKey){ k=window.topicKey; } }
    var mid=MID[k] || MID[tp] || '';
    nm=esc(nm);

    var h='<p>이 책은 이렇게 짜여 있습니다.</p>'+
      '<p><em>사주가 말하는 '+nm+'님 모습</em> — 태어난 때만 보고 썼습니다.<br>'+
      '<em>현실에서의 '+nm+'님 모습</em> — 답해주신 일곱 가지를 읽고 다시 썼습니다.';
    if(mid){ h+='<br><em>'+mid+'</em> — 여기서부터가 처음에 물어보신 것입니다.'; }
    h+='<br><em>마지막에 놓는 세 장</em> — 타로 세 장으로 앞으로 여섯 달을 봅니다.'+
      '<br><em>별자리와 띠로 다시 보면</em> — 같은 때를 다른 언어로 한 번 더 읽습니다.'+
      '<br>맨 뒤에는 셈의 근거를 모아 두었습니다.</p>'+
      '<p>그리고 마지막에 <em>기억하실 세 가지</em>를 적었습니다. '+
      '사주에서 읽은 것과 답해주신 것이 어긋나는 자리가 있다면, 그 자리도 숨기지 않고 적었습니다. '+
      '어느 쪽이 더 '+nm+'님다운지는 다 읽으신 뒤에 직접 정해 주세요.</p>'+
      '<p class="mini">전해드리고 싶은 말이 많아 양이 적지 않습니다. '+
      '오늘 한 번 읽으시고, 나중에 한 번 더 펼쳐보시면 더 좋습니다.</p>';
    return h;
  }

  function kin(el){
    if(!el){ return false; }
    if(String(el.tagName||'').toLowerCase()!=='p'){ return false; }
    var t=String(el.textContent||'').replace(/^\s+/,'');
    for(var i=0;i<KIN.length;i++){
      if(t.indexOf(KIN[i])===0){ return true; }
    }
    return false;
  }

  function swap(){
    var ps=document.querySelectorAll('p');
    var first=null, i;
    for(i=0;i<ps.length;i++){
      if(String(ps[i].textContent||'').indexOf(HEAD)>=0){ first=ps[i]; break; }
    }
    if(!first){ return false; }

    /* 같은 무리를 모읍니다 — 그 문단과, 뒤이어 딸린 것들 */
    var gone=[first];
    var nx=first.nextElementSibling;
    while(kin(nx)){
      gone.push(nx);
      nx=nx.nextElementSibling;
    }

    var box=document.createElement('div');
    box.innerHTML=intro();
    var parent=first.parentNode;
    if(!parent){ return false; }
    while(box.firstChild){ parent.insertBefore(box.firstChild, first); }
    for(i=0;i<gone.length;i++){
      if(gone[i].parentNode){ gone[i].parentNode.removeChild(gone[i]); }
    }
    return true;
  }

  /* 책은 나중에 그려집니다. 될 때까지 잠깐 지켜봅니다. */
  var tries=0;
  function watch(){
    var done=false;
    try{ done=swap(); }catch(e){ done=false; }
    tries++;
    if(done){ return; }
    if(tries>40){ return; }
    setTimeout(watch, 300);
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', watch);
  } else {
    watch();
  }

})();
</script>
	<?php
}, 20 );
