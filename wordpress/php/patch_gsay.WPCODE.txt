/* ════════════════════════════════════════════════════════════
   가디언 소개 글 줄이기 (다섯 문)   ★ 2026-09-09 · 소희 님께

   접기 단추를 그만두고, 글 자체를 짧게 다시 씁니다.
   첫 인사말(g-greet)은 손대지 않습니다.

   바꾸는 곳은 두 군데뿐입니다.
     소개            g-intro
     이렇게 읽습니다   g-method 안의 문단

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   미리보기   https://stellasaju.com/?stella_gsay=1
   바꾸기     https://stellasaju.com/?stella_gsay=apply
   되돌리기   https://stellasaju.com/?stella_gsay=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_gsay'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_gsay'] ) );
	$META = '_stella_gsay_bak_0909';

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="google" content="notranslate">'
	   . '<style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.75;'
	   . 'padding:28px 18px;max-width:900px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.2rem;margin:0 0 10px}'
	   . 'h2{color:#9AD1A5;font-size:1rem;margin:26px 0 6px;'
	   . 'border-top:1px solid rgba(255,255,255,.12);padding-top:14px}'
	   . '.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}'
	   . '.old{color:#8B849C}.new{color:#F2ECE0}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:12px/1.7 monospace;border-left:3px solid #D4AF6A}'
	   . '</style>';

	/* ── 새 글 ────────────────────────────────────────── */
	$W = array();

	$W['door-career'] = array(
		'intro'  => '마루는 타고난 재능과 때를 읽어 일과 돈의 흐름을 짚어요. '
			. '화려한 약속 대신, 지금 서 있는 자리에서 밟을 다음 한 걸음을 알려드립니다.',
		'method' => '<strong>사주명리학</strong>으로 타고난 기질과 때를 읽고, '
			. '<strong>서양점성술</strong>로 그 기질이 향하는 쪽을 봅니다. '
			. '둘이 같은 곳을 가리키는 지점이 풀이의 뼈대예요.',
	);

	$W['door-love'] = array(
		'intro'  => '벼리는 마음을 쓰는 결을 읽어 인연의 흐름을 짚어요. '
			. '상대의 속마음을 대신 말해주지는 않습니다. '
			. '다만 어떤 사람에게 끌리고 어디서 자꾸 같은 데 걸리는지를 알려드려요.',
		'method' => '<strong>사주명리학</strong>으로 타고난 인연의 결과 때를 읽고, '
			. '<strong>서양점성술</strong>로 마음이 어느 쪽으로 기우는지를 봅니다. '
			. '둘이 같은 곳을 가리키는 지점이 풀이의 뼈대예요.',
	);

	$W['door-health'] = array(
		'intro'  => '아람은 타고난 체질과 가족의 자리를 읽어 몸과 관계의 흐름을 짚어요. '
			. '병을 진단하지는 않습니다. '
			. '다만 어느 계절에 약해지고 누구를 짊어지고 있는지를 알려드려요.',
		'method' => '<strong>사주명리학</strong>으로 타고난 체질과 육친의 자리를 읽고, '
			. '<strong>서양점성술</strong>로 몸과 마음이 어느 쪽으로 기우는지를 봅니다. '
			. '둘이 같은 곳을 가리키는 지점이 풀이의 뼈대예요.',
	);

	$W['door-fortune'] = array(
		'intro'  => '미르는 십 년마다 바뀌는 대운과 해마다 도는 세운을 읽어 '
			. '인생 전체의 지도를 그려요. 한 해의 길흉을 점치기보다, '
			. '언제가 씨 뿌릴 때이고 언제가 거둘 때인지를 봅니다.',
		'method' => '<strong>사주명리학</strong>으로 대운과 세운의 흐름을 읽고, '
			. '<strong>서양점성술</strong>로 지나가는 별들이 어느 자리를 건드리는지를 봅니다. '
			. '둘이 같은 곳을 가리키는 지점이 풀이의 뼈대예요.',
	);

	$W['door-astro'] = array(
		'intro'  => '아라는 태어난 순간의 하늘을 그대로 펼쳐 '
			. '별들의 자리로 성격과 시기를 읽어요. '
			. '태양 별자리 하나로 사람을 나누지 않습니다. '
			. '상승궁과 달, 행성까지 겹쳐 봐야 한 사람이 나와요.',
		'method' => '<strong>서양점성술</strong>로 태어난 순간의 하늘을 세우고, '
			. '<strong>사주명리학</strong>으로 그 하늘이 땅에서 어떻게 풀리는지를 봅니다. '
			. '둘이 같은 곳을 가리키는 지점이 풀이의 뼈대예요.',
	);

	/* ── 자리 찾기 (정규식 안 씁니다) ──────────────────── */
	$span = function ( $c, $anchor, $need_p ) {
		$from = 0;
		while ( true ) {
			$at = strpos( $c, $anchor, $from );
			if ( false === $at ) { return false; }
			$from = $at + 1;

			/* g-method 뒤에 - 가 오면 g-method-t 입니다 */
			$nx = substr( $c, $at + strlen( $anchor ), 1 );
			if ( '-' === $nx ) { continue; }

			/* ★ 태그 안에 있는 것만 셉니다.
			   스타일표의 .g-method{ ... } 를 잘못 잡던 자리입니다.
			   뒤로 훑어서 < 를 먼저 만나면 태그 안, > 를 먼저 만나면 글 안입니다. */
			$lt = strrpos( substr( $c, 0, $at ), '<' );
			$rt = strrpos( substr( $c, 0, $at ), '>' );
			if ( false === $lt ) { continue; }
			if ( false !== $rt ) { if ( $rt > $lt ) { continue; } }

			$head = $at;
			if ( $need_p ) {
				$pt = strpos( $c, '<p', $at );
				if ( false === $pt ) { continue; }
				if ( $pt - $at > 400 ) { continue; }
				$head = $pt;
			}
			$gt = strpos( $c, '>', $head );
			if ( false === $gt ) { continue; }
			$end = strpos( $c, '</p>', $gt );
			if ( false === $end ) { continue; }
			if ( $end - $gt > 1500 ) { continue; }

			$seg = substr( $c, $gt + 1, $end - $gt - 1 );
			/* 스타일표 안이면 중괄호가 들어 있습니다 */
			if ( false !== strpos( $seg, '{' ) ) { continue; }
			if ( false !== strpos( $seg, '<p' ) ) { continue; }

			return array( $gt + 1, $end, $seg );
		}
	};

	$len = function ( $s ) {
		$s = trim( preg_replace( '/\s+/', ' ', wp_strip_all_tags( $s ) ) );
		if ( function_exists( 'mb_strlen' ) ) { return mb_strlen( $s, 'UTF-8' ); }
		return strlen( $s );
	};

	$plain = function ( $s ) {
		return trim( preg_replace( '/\s+/', ' ', wp_strip_all_tags( $s ) ) );
	};

	/* ── 되돌리기 ─────────────────────────────────────── */
	if ( 'undo' === $mode ) {
		echo '<h1>되돌립니다</h1>';
		$n = 0;
		foreach ( $W as $slug => $x ) {
			$p = get_page_by_path( $slug );
			if ( ! $p ) { continue; }
			$bak = get_post_meta( $p->ID, $META, true );
			if ( '' === $bak ) {
				echo '<p class="dim">' . esc_html( $slug ) . ' — 되돌릴 것 없음</p>';
				continue;
			}
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ),
				array( 'ID' => $p->ID ) );
			clean_post_cache( $p->ID );
			delete_post_meta( $p->ID, $META );
			echo '<p class="ok">' . esc_html( $slug ) . ' — 되돌렸습니다</p>';
			$n++;
		}
		if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		echo '<p class="ok">' . (int) $n . ' 쪽을 되돌렸습니다.</p>';
		exit;
	}

	/* ── 미리보기 / 바꾸기 ────────────────────────────── */
	$go = ( 'apply' === $mode );
	echo '<h1>가디언 소개 글 줄이기 — '
	   . ( $go ? '바꿉니다' : '미리보기 (아직 안 바꿈)' ) . '</h1>';

	$done = 0;
	foreach ( $W as $slug => $x ) {

		echo '<h2>' . esc_html( $slug ) . '</h2>';

		$p = get_page_by_path( $slug );
		if ( ! $p ) { echo '<p class="no">쪽 없음</p>'; continue; }

		$c = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
		$orig = $c;

		$rows = array(
			array( 'g-intro',  false, '소개',           $x['intro'] ),
			array( 'g-method', true,  '이렇게 읽습니다', $x['method'] ),
		);

		$out = '';
		$hit = 0;
		foreach ( $rows as $r ) {
			$s = $span( $c, $r[0], $r[1] );
			if ( ! $s ) {
				$out .= $r[2] . " — 자리를 못 찾았습니다\n\n";
				continue;
			}
			$old = $s[2];
			if ( $plain( $old ) === $plain( $r[3] ) ) {
				$out .= $r[2] . " — 이미 새 글입니다\n\n";
				continue;
			}
			$out .= $r[2] . '  ' . $len( $old ) . '자  ->  ' . $len( $r[3] ) . "자\n";
			$out .= '  전  ' . $plain( $old ) . "\n";
			$out .= '  후  ' . $plain( $r[3] ) . "\n\n";
			$c    = substr( $c, 0, $s[0] ) . $r[3] . substr( $c, $s[1] );
			$hit++;
		}

		echo '<pre>' . esc_html( $out ) . '</pre>';

		if ( ! $go || ! $hit ) { continue; }

		if ( '' === get_post_meta( $p->ID, $META, true ) ) {
			update_post_meta( $p->ID, $META, $orig );
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $c ),
			array( 'ID' => $p->ID ) );
		clean_post_cache( $p->ID );
		echo '<p class="ok">' . (int) $hit . ' 군데 바꿨습니다.</p>';
		$done++;
	}

	if ( $go ) {
		if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		echo '<h2>끝</h2><p class="ok">' . (int) $done . ' 쪽을 고쳤습니다.</p>';
		echo '<p class="dim">되돌리시려면 /?stella_gsay=undo 를 여세요.</p>';
	} else {
		echo '<h2>끝</h2><p class="dim">이대로 괜찮으시면 '
		   . '/?stella_gsay=apply 를 여세요.</p>';
	}
	exit;
} );
