/* ══════════════════════════════════════════════════════════
   책이 어디서 끊겼나 보기 · 소희 님께      2026-09-11 · 1판

   책이 안 나옵니다. 어제는 화면이 왼쪽으로 치우쳤고요.
   이 도구는 책이 온전한지만 봅니다. 아무것도 바꾸지 않습니다.

     https://stellasaju.com/?stella_q0911a=1

   ★ 이름을 stella_q0911a 로 바꿨습니다.
     stella_health 는 이미 다른 스니펫(강성의 신 주제별)이 쓰고 있어서
     그쪽이 먼저 대답해버렸습니다.

   나온 글을 통째로 붙여주세요.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_q0911a'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;
	$c  = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	echo "═══ 책 건강검진 ═══\n(스니펫 판 2026-09-11-a · 이 줄이 보여야 새 파일입니다)\n";
	echo '쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "바이트\n";
	echo '글자로는 ' . number_format( mb_strlen( $c, 'UTF-8' ) ) . "글자\n\n";

	$open  = substr_count( $c, '<script' );
	$close = substr_count( $c, '</script>' );
	echo "─── 짝이 맞는가 ───\n";
	echo sprintf( "  <script       %d개\n", $open );
	echo sprintf( "  </script>     %d개   %s\n", $close,
		( $open === $close ? '짝이 맞습니다' : '★ 짝이 안 맞습니다 — 잘렸을 수 있습니다' ) );
	echo sprintf( "  <style        %d개\n", substr_count( $c, '<style' ) );
	echo sprintf( "  </style>      %d개\n", substr_count( $c, '</style>' ) );
	echo sprintf( "  <div          %d개\n", substr_count( $c, '<div' ) );
	echo sprintf( "  </div>        %d개\n", substr_count( $c, '</div>' ) );
	echo sprintf( "  page(         %d개\n", substr_count( $c, 'page(' ) );

	echo "\n─── 맨 끝 300글자 (여기서 잘렸는지 보입니다) ───\n";
	$tail = mb_substr( $c, -300, null, 'UTF-8' );
	echo $tail . "\n";

	echo "\n─── 맨 앞 200글자 ───\n";
	echo mb_substr( $c, 0, 200, 'UTF-8' ) . "\n";

	echo "\n─── 잘린 흔적이 있는가 ───\n";
	$bad = 0;
	if ( '</script>' !== substr( rtrim( $c ), -9 ) ) {
		echo "  ★ 맨 끝이 </script> 로 안 끝납니다\n"; $bad++;
	} else {
		echo "  맨 끝이 </script> 로 잘 끝납니다\n";
	}
	if ( $open !== $close ) { $bad++; }
	if ( ! $bad ) { echo "  겉으로 보이는 문제는 없습니다\n"; }

	echo "\n─── 되돌릴 수 있는 판 ───\n";
	foreach ( array( '_stella_scale_stack', '_stella_title_stack',
	                 '_stella_b2_stack', '_stella_b3_stack',
	                 '_stella_fix_stack', '_stella_name_stack' ) as $k ) {
		$st = get_post_meta( $id, $k, true );
		$n  = is_array( $st ) ? count( $st ) : 0;
		if ( ! $n ) { continue; }
		echo '  ' . str_replace( array( '_stella_', '_stack' ), '', $k ) . ' — ' . $n . "판\n";
		foreach ( $st as $s ) {
			echo '      ' . $s['when'] . '  ' . number_format( strlen( $s['body'] ) ) . "바이트\n";
		}
	}

	echo "\n═══ 여기까지 · 위를 통째로 붙여주세요 ═══";
	exit;
}, 1 );
