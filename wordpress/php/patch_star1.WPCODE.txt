 1668).

	      안 맞습니다. 한 줄짜리만 맞아서 「17군데 중 4군데」가 났습니다.

	$mkpat = function ( $a ) {

		   실제로 한 군데가 그렇게 걸려 108자가 어긋났습니다. */
		$n = substr_count( $sub, $e[2] );
		if ( 1 === $n ) {
			$EDITS[ $ei ][4] = '그대로';
