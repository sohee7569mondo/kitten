/* ════════════════════════════════════════════════════════════
   「나의 인생길잡이」 값이 어디서 오는지 찾기 · 소희 님께

   0원이 어느 줄에서 나오는지 스니펫을 전부 뒤져서 찾아냅니다.
   아무것도 고치지 않습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 Run Everywhere →
              첫 줄 PHP 태그 아래에 붙여넣기 → 저장 → 활성

        https://stellasaju.com/?stella_patch=pricefind

   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_patch'] ) || 'pricefind' !== $_GET['stella_patch'] ) {
		return;
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		return;
	}

	global $wpdb;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:14px/1.8 -apple-system,system-ui,sans-serif;background:#111;color:#eee;padding:24px;max-width:1100px">';
	echo '<h1 style="color:#ffd76a">인생길잡이 값이 어디서 오나</h1>';

	/* ── 1. 지금 살아 있는 값들 ─────────────────────────── */
	echo '<h2 style="color:#9fd">1. 지금 돌고 있는 값</h2>';
	echo '<table cellpadding="8" style="border-collapse:collapse"><tr style="color:#ffd76a"><td>이름</td><td>값</td></tr>';
	foreach ( array( 'STELLA_TRIAL', 'STELLA_ORB_KRW', 'STELLA_PRICE_BASE', 'STELLA_PRICE_FULL', 'STELLA_PRICE_ONELINE', 'STELLA_PRICE_DEEP', 'STELLA_WELCOME_ORBS', 'STELLA_EVENT_ORBS', 'STELLA_EVENT_UNTIL' ) as $k ) {
		if ( defined( $k ) ) {
			$v = constant( $k );
			$v = is_bool( $v ) ? ( $v ? 'true' : 'false' ) : (string) $v;
			printf( '<tr style="border-top:1px solid #333"><td>%s</td><td style="color:#7bff9b"><b>%s</b></td></tr>', esc_html( $k ), esc_html( $v ) );
		} else {
			printf( '<tr style="border-top:1px solid #333"><td>%s</td><td style="color:#888">정해진 적 없음</td></tr>', esc_html( $k ) );
		}
	}
	echo '</table>';

	/* ── 2. 값을 세는 함수들이 뭐라고 답하나 ────────────── */
	echo '<h2 style="color:#9fd">2. 값을 세는 함수에게 직접 물어보기</h2>';
	echo '<table cellpadding="8" style="border-collapse:collapse"><tr style="color:#ffd76a"><td>물어본 것</td><td>대답</td></tr>';

	$ask = function ( $label, $fn ) {
		try {
			$v = $fn();
			printf( '<tr style="border-top:1px solid #333"><td>%s</td><td style="color:#7bff9b"><b>%s</b></td></tr>', esc_html( $label ), esc_html( is_scalar( $v ) ? (string) $v : wp_json_encode( $v, JSON_UNESCAPED_UNICODE ) ) );
		} catch ( \Throwable $e ) {
			printf( '<tr style="border-top:1px solid #333"><td>%s</td><td style="color:#ff7b7b">%s</td></tr>', esc_html( $label ), esc_html( $e->getMessage() ) );
		}
	};

	if ( function_exists( 'stella_po_bundle_krw' ) ) {
		$ask( "stella_po_bundle_krw('guide')  ← 점검창이 쓰는 것", function () { return stella_po_bundle_krw( 'guide' ); } );
		$ask( "stella_po_bundle_krw('door-guide')", function () { return stella_po_bundle_krw( 'door-guide' ); } );
	} else {
		echo '<tr><td>stella_po_bundle_krw</td><td style="color:#ff7b7b">그런 함수가 없습니다</td></tr>';
	}
	if ( function_exists( 'stella_price' ) ) {
		$ask( "stella_price(guide)  ← 구슬 개수", function () { return stella_price( array( 'slug' => 'guide', 'oneline' => '', 'deeps' => array() ) ); } );
		$ask( "stella_price(door-career)  ← 구슬 개수", function () { return stella_price( array( 'slug' => 'door-career', 'oneline' => '', 'deeps' => array() ) ); } );
	}
	if ( function_exists( 'stella_orb_balance' ) ) {
		$ask( '내 구슬 잔액', function () { return stella_orb_balance( get_current_user_id() ); } );
	}
	echo '</table>';

	/* ── 3. 스니펫 전부 뒤져서 값이 적힌 줄 찾기 ─────────── */
	echo '<h2 style="color:#9fd">3. 값이 적혀 있는 줄 (스니펫 전체에서 찾음)</h2>';

	$words = array(
		'STELLA_PRICE_FULL',
		'STELLA_PRICE_BASE',
		'STELLA_WELCOME_ORBS',
		'STELLA_EVENT_ORBS',
		'stella_po_bundle_krw',
		"'guide'",
		'인생길잡이',
	);

	$snips = $wpdb->get_results( "SELECT ID, post_title, post_status, post_content FROM {$wpdb->posts} WHERE post_type = 'wpcode' ORDER BY post_title" );

	$hits = 0;
	foreach ( (array) $snips as $s ) {
		$lines = preg_split( "/\r\n|\n|\r/", $s->post_content );
		$found = array();
		foreach ( $lines as $i => $line ) {
			foreach ( $words as $w ) {
				if ( false !== strpos( $line, $w ) ) {
					$found[] = array( $i + 1, $line );
					break;
				}
			}
		}
		if ( ! $found ) {
			continue;
		}
		$hits++;
		printf(
			'<h3 style="color:%s;margin-top:24px">▣ %s <span style="font-weight:400;font-size:.85em">(스니펫 %d · %s)</span></h3>',
			'publish' === $s->post_status ? '#7bff9b' : '#888',
			esc_html( $s->post_title ),
			$s->ID,
			'publish' === $s->post_status ? '켜짐' : '꺼짐'
		);
		echo '<pre style="white-space:pre-wrap;background:#000;padding:12px;border-left:3px solid #ffd76a;overflow-x:auto;font:13px/1.6 monospace">';
		$shown = 0;
		foreach ( $found as $f ) {
			printf( "%5d │ %s\n", $f[0], esc_html( rtrim( $f[1] ) ) );
			$shown++;
			if ( $shown >= 40 ) {
				echo "      │ … (그 아래로 더 있습니다)\n";
				break;
			}
		}
		echo '</pre>';
	}

	if ( ! $hits ) {
		echo '<p style="color:#ff7b7b">어느 스니펫에서도 못 찾았습니다. 테마 파일이나 wp-config.php 에 있을 수 있습니다.</p>';
	}

	/* ── 4. wp-config 에 정해뒀는지 ─────────────────────── */
	echo '<h2 style="color:#9fd">4. wp-config.php 에 있나</h2>';
	$cfg = ABSPATH . 'wp-config.php';
	if ( is_readable( $cfg ) ) {
		$txt   = (string) file_get_contents( $cfg );
		$lines = preg_split( "/\r\n|\n|\r/", $txt );
		$any   = false;
		echo '<pre style="white-space:pre-wrap;background:#000;padding:12px;border-left:3px solid #ffd76a;font:13px/1.6 monospace">';
		foreach ( $lines as $i => $line ) {
			if ( false !== strpos( $line, 'STELLA' ) ) {
				$any = true;
				/* 열쇠는 가립니다 */
				$line = preg_replace( '/([\'"])[A-Za-z0-9+\/=_-]{20,}\1/', '$1[가림]$1', $line );
				printf( "%5d │ %s\n", $i + 1, esc_html( rtrim( $line ) ) );
			}
		}
		if ( ! $any ) {
			echo '(STELLA 로 시작하는 줄이 없습니다)';
		}
		echo '</pre>';
	} else {
		echo '<p style="color:#888">wp-config.php 를 읽을 수 없습니다 (권한 문제라 정상일 수 있습니다).</p>';
	}

	echo '</body>';
	exit;
} );
