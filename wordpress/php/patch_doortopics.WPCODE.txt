//     대운은 없애고 2026신년운세 2027신년운세 이렇게 하기로 했고
//     이혼운 퇴사운은 배너를 만들라는거지?」
//   「ㅇㅇ 이혼운 퇴사운 갈라서 썼어」
//
// 하는 일
//   door-fortune   대운을 감춥니다
//                  신년운세를 2026년 운세 · 2027년 운세 둘로 가릅니다
//   door-health    부모운 · 형제운 · 자녀운을 감춥니다 (가족운 한 권에 듭니다)
//   door-love      이혼운을 넣습니다 (이별운 뒤)
//   door-career    퇴사운을 넣습니다 (이직운 뒤)
//
// 왜 지우지 않고 감추나
//   지우면 되돌리기 어렵고, 그 줄의 셈이 틀어질 수 있습니다.
//   감추면 스니펫을 끄는 것만으로 그대로 돌아옵니다.
//
// 넣는 주제는 책이 이미 압니다
//   이혼운  patch160_divorce 가 새 주제로 붙였습니다
//   퇴사운  주제별 표 조각 여럿에 이미 들어 있습니다
//   ★ 투자운은 아직 전용 글이 없어 건드리지 않았습니다 (원고를 쓰는 중)
//
// patch160_nydoor 와 부딪히지 않습니다
//   그 조각도 신년운세를 둘로 가릅니다. 이미 갈라져 있으면 물러납니다.
//
// 앵커를 안 잡습니다 — 쪽 원본은 한 글자도 안 고칩니다.
//
// 살펴보기  https://stellasaju.com/?stella_doortopics=1
//
// ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.
// ★ 크롬 번역을 끄고 붙여넣으세요.

add_action( 'wp_head', function () {

	$doors = array( 'door-love', 'door-career', 'door-health',
		'door-fortune', 'door-astro', 'door-tarot' );
	if ( ! is_page( $doors ) ) { return; }
	?>
<script>
(function(){
  if(window.StellaDoorTopics){ return; }
  window.StellaDoorTopics = 1;

  // 감출 주제 — 문마다
  var HIDE = {};
  HIDE['door-fortune'] = ['대운'];
  HIDE['door-health']  = ['부모운', '형제운', '자녀운'];

  // 둘로 가를 주제 — [ 문, 본디 값, 새 값 둘 ]
  var SPLIT = ['door-fortune', '신년운세', ['2026년 운세', '2027년 운세']];

  // 넣을 주제 — [ 문, 새 값, 이 주제 뒤에 ]
  var ADD = [
    ['door-love',   '이혼운', '이별운'],
    ['door-career', '퇴사운', '이직운']
  ];

  function slug(){
    var p = String(location.pathname).split('/');
    var i;
    for(i = p.length - 1; i >= 0; i--){
      if(p[i]){ return p[i]; }
    }
    return '';
  }

  function inputs(){
    return document.querySelectorAll('input[name="ssg-topic"]');
  }

  function labelOf(inp){
    var el = inp.parentNode;
    var up = 0;
    while(el){
      if(String(el.tagName).toLowerCase() === 'label'){ return el; }
      el = el.parentNode;
      up++;
      if(up > 4){ return null; }
    }
    return null;
  }

  function find(v){
    var r = inputs(), i;
    for(i = 0; i < r.length; i++){
      if(r[i].value === v){ return r[i]; }
    }
    return null;
  }

  var here = slug();
  var did = 0;

  // ① 감추기
  function hide(){
    var list = HIDE[here];
    if(!list){ return 0; }
    var n = 0, i, inp, lab;
    for(i = 0; i < list.length; i++){
      inp = find(list[i]);
      if(!inp){ continue; }
      lab = labelOf(inp);
      if(!lab){ continue; }
      if(lab.getAttribute('data-dt-hide') === '1'){ continue; }
      lab.setAttribute('data-dt-hide', '1');
      lab.style.display = 'none';
      inp.disabled = true;
      n++;
    }
    return n;
  }

  // ② 가르기 — 이미 갈라져 있으면 물러납니다
  function split(){
    if(here !== SPLIT[0]){ return 0; }
    if(find(SPLIT[2][0])){ return 0; }   // nydoor 가 이미 했습니다
    var old = find(SPLIT[1]);
    if(!old){ return 0; }
    var lab = labelOf(old);
    if(!lab){ return 0; }
    if(!lab.parentNode){ return 0; }

    var made = [], j, clone, node, span;
    for(j = 0; j < SPLIT[2].length; j++){
      clone = lab.cloneNode(true);
      node  = clone.querySelector('input[name="ssg-topic"]');
      span  = clone.querySelector('span');
      if(!node){ continue; }
      node.value = SPLIT[2][j];
      node.checked = false;
      node.disabled = false;
      if(span){ span.textContent = SPLIT[2][j]; }
      clone.style.display = '';
      clone.setAttribute('data-dt-new', '1');
      made.push(clone);
    }
    if(!made.length){ return 0; }
    for(j = 0; j < made.length; j++){
      lab.parentNode.insertBefore(made[j], lab);
    }
    lab.parentNode.removeChild(lab);
    return made.length;
  }

  // ③ 넣기
  function add(){
    var n = 0, i;
    for(i = 0; i < ADD.length; i++){
      if(ADD[i][0] !== here){ continue; }
      if(find(ADD[i][1])){ continue; }      // 이미 있습니다
      var after = find(ADD[i][2]);
      if(!after){ continue; }
      var lab = labelOf(after);
      if(!lab){ continue; }
      if(!lab.parentNode){ continue; }

      var clone = lab.cloneNode(true);
      var node  = clone.querySelector('input[name="ssg-topic"]');
      var span  = clone.querySelector('span');
      if(!node){ continue; }
      node.value = ADD[i][1];
      node.checked = false;
      node.disabled = false;
      if(span){ span.textContent = ADD[i][1]; }
      clone.style.display = '';
      clone.setAttribute('data-dt-new', '1');
      if(lab.nextSibling){ lab.parentNode.insertBefore(clone, lab.nextSibling); }
      else { lab.parentNode.appendChild(clone); }
      n++;
    }
    return n;
  }

  // ④ 골라진 것이 감춰졌으면 첫 보이는 것으로 옮깁니다
  //    안 그러면 손님 눈에는 아무것도 안 골라진 것처럼 보입니다.
  function refix(){
    var r = inputs(), i, first = null, ok = false;
    for(i = 0; i < r.length; i++){
      var lab = labelOf(r[i]);
      var shown = true;
      if(lab){ if(lab.style.display === 'none'){ shown = false; } }
      if(!shown){ continue; }
      if(!first){ first = r[i]; }
      if(r[i].checked){ ok = true; }
    }
    if(ok){ return; }
    if(!first){ return; }
    first.checked = true;
    try {
      var ev = document.createEvent('HTMLEvents');
      ev.initEvent('change', true, true);
      first.dispatchEvent(ev);
    } catch(e){}
  }

  function run(){
    if(!inputs().length){ return false; }
    hide();
    split();
    add();
    refix();
    did = 1;
    return true;
  }

  run();
  var n = 0;
  var t = setInterval(function(){
    n++;
    run();
    if(n > 25){ clearInterval(t); }
  }, 300);
})();
</script>
	<?php
} );

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_doortopics'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/plain; charset=utf-8' );
	echo "문 단추 정리 — 살펴보기\n";
	echo "═══════════════════════════════════════════\n\n";

	$plan = array(
		'door-fortune' => array( 'hide' => array( '대운' ),
			'split' => array( '신년운세', '2026년 운세 · 2027년 운세' ), 'add' => array() ),
		'door-health'  => array( 'hide' => array( '부모운', '형제운', '자녀운' ),
			'split' => null, 'add' => array() ),
		'door-love'    => array( 'hide' => array(), 'split' => null,
			'add' => array( array( '이혼운', '이별운' ) ) ),
		'door-career'  => array( 'hide' => array(), 'split' => null,
			'add' => array( array( '퇴사운', '이직운' ) ) ),
	);

	foreach ( $plan as $slug => $p ) {
		$page = get_page_by_path( $slug );
		echo $slug . "\n";
		if ( ! $page ) { echo "   쪽이 없습니다\n\n"; continue; }
		$c = (string) $page->post_content;

		// 쪽에 든 주제 값을 그대로 읽습니다 (정규식을 안 씁니다)
		$vs = array(); $at = 0;
		while ( true ) {
			$a = strpos( $c, 'name="ssg-topic" value="', $at );
			if ( false === $a ) { break; }
			$a = $a + strlen( 'name="ssg-topic" value="' );
			$b = strpos( $c, '"', $a );
			if ( false === $b ) { break; }
			$vs[] = substr( $c, $a, $b - $a );
			$at = $b;
		}
		echo '   지금  ' . implode( ' · ', $vs ) . "\n";

		foreach ( $p['hide'] as $h ) {
			echo '   감춤  ' . $h
				. ( in_array( $h, $vs, true ) ? '' : '   (쪽에 없습니다)' ) . "\n";
		}
		if ( $p['split'] ) {
			echo '   가름  ' . $p['split'][0] . '  →  ' . $p['split'][1];
			echo in_array( $p['split'][0], $vs, true ) ? '' : '   (쪽에 없습니다)';
			echo "\n";
		}
		foreach ( $p['add'] as $a2 ) {
			echo '   넣음  ' . $a2[0] . '  (' . $a2[1] . ' 뒤)';
			echo in_array( $a2[1], $vs, true ) ? '' : '   ★ 기댈 주제가 없습니다';
			echo "\n";
		}
		echo "\n";
	}

	echo "책이 새 주제를 아는지는 여기서 봅니다\n";
	echo '   ' . home_url( '/?stella_audit=1' ) . "\n";
	echo "   표마다 어느 주제가 들어 있는지 찍어 줍니다.\n\n";

	echo "눌러서 확인해 보세요\n";
	foreach ( array(
		'door-fortune' => array( '2026년 운세', '2027년 운세' ),
		'door-health'  => array( '가족운' ),
		'door-love'    => array( '이혼운' ),
		'door-career'  => array( '퇴사운' ),
	) as $slug => $ts ) {
		foreach ( $ts as $t ) {
			echo '   ' . home_url( '/' . $slug . '/?topic=' . $t ) . "\n";
		}
	}
	exit;
} );
