add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_wedsplit'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_wedsplit'] ) );
	$META = '_stella_wedsplit_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "바이트\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

	echo "═══ 결혼운 · 미혼과 기혼으로 ═══\n";
	echo '책 쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "바이트\n\n";

	/* 판 WEDSPLIT-2 (2026-09-13) — 넣었는지 보는 표시를 고쳤습니다.
	   전에는 「결혼운_미혼」이라는 글자를 봤는데, patch160_open 도 그 말을
	   넣습니다. 그래서 open 을 먼저 넣으면 이 조각이 자기가 이미 들어간
	   줄로 잘못 알고 멈췄습니다. 이제 이 조각만 만드는 줄을 봅니다. */
	if ( 1 === preg_match( '/var\s+topicKey\s*=\s*topic\s*;/u', $c ) ) {
		echo "이미 들어가 있습니다.\n";
		echo "다시 넣으시려면 ?stella_wedsplit=undo 로 걷어낸 뒤 하세요.\n";
		exit;
	}

	$jobs = array(
		array(
			'name' => '하나 · 미혼인지 기혼인지 가르는 스위치',
			'must' => true,
			'find' => stella_wed_f0(),
			'to'   => stella_wed_t0(),
		),
		array(
			'name' => '둘 · 저울 설명 두 벌 (4,000자씩)',
			'must' => true,
			'find' => stella_wed_f1(),
			'to'   => stella_wed_t1(),
		),
		array(
			'name' => '셋 · 조합 풀이 두 벌',
			'must' => true,
			'find' => stella_wed_f2(),
			'to'   => stella_wed_t2(),
		),
		array(
			'name' => '넷 · 편지 두 벌',
			'must' => true,
			'find' => stella_wed_f3(),
			'to'   => stella_wed_t3(),
		),
		array(
			'name' => '다섯 · 배웅 한 줄 두 벌',
			'must' => true,
			'find' => stella_wed_f4(),
			'to'   => stella_wed_t4(),
		),
		array(
			'name' => '여섯 · 1부 렌즈 두 벌',
			'must' => true,
			'find' => stella_wed_f5(),
			'to'   => stella_wed_t5(),
		),
		array(
			'name' => '일곱 · 저울 설명을 찾을 때 갈라 보기',
			'must' => true,
			'find' => stella_wed_f6(),
			'to'   => stella_wed_t6(),
		),
		array(
			'name' => '여덟 · 저울 제목',
			'must' => true,
			'find' => stella_wed_f7(),
			'to'   => stella_wed_t7(),
		),
		array(
			'name' => '아홉 · 저울 쪽 이름',
			'must' => true,
			'find' => stella_wed_f8(),
			'to'   => stella_wed_t8(),
		),
		array(
			'name' => '열 · 조합 풀이',
			'must' => true,
			'find' => stella_wed_f9(),
			'to'   => stella_wed_t9(),
		),
		array(
			'name' => '열하나 · 편지',
			'must' => true,
			'find' => stella_wed_f10(),
			'to'   => stella_wed_t10(),
		),
		array(
			'name' => '열둘 · 배웅 한 줄',
			'must' => true,
			'find' => stella_wed_f11(),
			'to'   => stella_wed_t11(),
		),
		array(
			'name' => '열셋 · 1부 렌즈를 찾을 때',
			'must' => true,
			'find' => stella_wed_f12(),
			'to'   => stella_wed_t12(),
		),
		array(
			'name' => '열넷 · 맺음말 세 가지 두 벌 (GAP 를 넣으셨을 때만)',
			'must' => false,
			'find' => stella_wed_f13(),
			'to'   => stella_wed_t13(),
		),
		array(
			'name' => '열다섯 · 맺음말이 주제를 찾을 때 (GAP 를 넣으셨을 때만)',
			'must' => false,
			'find' => stella_wed_f14(),
			'to'   => stella_wed_t14(),
		),
		array(
			'name' => '열여섯 · 거스르는 자리 (AGAINST 를 넣으셨을 때만)',
			'must' => false,
			'find' => stella_wed_f15(),
			'to'   => stella_wed_t15(),
		),
	);

	$bad  = 0;
	$skip = 0;
	foreach ( $jobs as $j ) {
		$n = substr_count( $c, $j['find'] );
		if ( 1 === $n ) { printf( "  %-46s 찾음\n", $j['name'] ); continue; }
		if ( ! $j['must'] ) {
			if ( 0 === $n ) { printf( "  %-46s 건너뜀\n", $j['name'] ); $skip++; continue; }
		}
		printf( "  %-46s ★ %d군데\n", $j['name'], $n );
		$bad++;
	}

	if ( $bad ) {
		echo "\n한 군데가 아닌 자리가 " . $bad . "개 있어 아무것도 안 건드렸습니다.\n";
		echo "이 글을 그대로 주세요.\n";
		exit;
	}

	$made = $c;
	foreach ( $jobs as $j ) {
		if ( 1 !== substr_count( $made, $j['find'] ) ) { continue; }
		$made = str_replace( $j['find'], $j['to'], $made );
	}

	echo "\n" . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $made ) ) . '바이트 (+'
	   . number_format( strlen( $made ) - strlen( $c ) ) . ")\n";
	if ( $skip ) { echo '건너뛴 자리 ' . $skip . "개 (GAP 이나 AGAINST 를 아직 안 넣으신 것입니다)\n"; }
	echo "\n";

	if ( 'apply' !== $mode ) {
		echo "아직 안 바꿨습니다. 정말 넣으시려면 ?stella_wedsplit=apply\n";
		exit;
	}

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		update_post_meta( $id, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $made ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "넣었습니다.\n";
	echo "확인 : 결혼운으로 미혼 한 번, 기혼 한 번 뽑아보세요.\n";
	echo "되돌리려면 ?stella_wedsplit=undo\n";
	exit;
}, 1 );

function stella_wed_t0() {
	return <<<'WEDT0'
var topic = order.topic || g.subject;

  /* ══ 2026-09-13 · 결혼운은 미혼과 기혼이 다른 책입니다 ═══════
     소희 님 : 「결혼운은 전부 미혼과 기혼을 나눠야 할 듯해」

     손님에게 보이는 이름(topic)은 그대로 「결혼운」으로 두고,
     표를 찾을 때 쓰는 이름(topicKey)만 갈라 둡니다.
     그래야 목차와 제목은 그대로이고 글만 갈립니다.

     답이 없는 옛 손님은 미혼으로 봅니다. */
  var topicKey = topic;
  if(topic==='결혼운'){
    var __wed='';
    if(answers){ if(answers.wed){ __wed=String(answers.wed); } }
    if(__wed.indexOf('이미')>=0){ topicKey='결혼운_기혼'; }
    else { topicKey='결혼운_미혼'; }
  }
WEDT0;
}
