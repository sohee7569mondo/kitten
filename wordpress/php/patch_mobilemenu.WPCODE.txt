     5  연예인 191명            -> 지금 사람 수로 (이제 537명)

/* ── 여기만 고치면 됩니다 ───────────────────────── */
function stella_mm_renames() {
	return array(
		'직성의 신' => '마루',
		'연성의 신' => '벼리',
		'강성의 신' => '아람',
		'세성의 신' => '미르',
		'성좌의 신' => '아라',
	);
}
function stella_mm_drops() {
	return array( '/lotto/', '/oracle/' );   /* 이 주소로 가는 줄을 뺍니다 */
}

/* 스타 사주에 지금 몇 명 있는지 — 한 번만 세고 기억해 둡니다 */
function stella_mm_star_count() {
	static $n = null;
	if ( null !== $n ) { return $n; }
	$n = 0;
	$p = get_page_by_path( 'star-saju' );
	if ( $p ) {
		global $wpdb;
		$c = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
		$n = substr_count( $c, '{"n":"' );
	}
	return $n;
}

global $stella_mm_log;
$stella_mm_log = array();

add_filter( 'render_block', function ( $html, $block ) {

	/* 이 메뉴인지 알아보는 표시 — 설명글과 링크가 같이 있는 덩어리 */
	if ( false === strpos( $html, 'class="sub"' ) ) { return $html; }
	if ( false === strpos( $html, '<a href="' ) ) { return $html; }

	global $stella_mm_log;
	$before = $html;

	/* 1) 줄 빼기 — 그 주소로 가는 <a>...</a> 를 통째로 */
	foreach ( stella_mm_drops() as $href ) {
		while ( true ) {
			$at = strpos( $html, 'href="' . $href . '"' );
			if ( false === $at ) { break; }
			$s = strrpos( substr( $html, 0, $at ), '<a ' );
			if ( false === $s ) { break; }
			$e = strpos( $html, '</a>', $at );
			if ( false === $e ) { break; }
			$e = $e + 4;
			/* <a> 를 감싼 <li> 가 있으면 그것까지 걷습니다 */
			$li = strrpos( substr( $html, 0, $s ), '<li' );
			if ( false !== $li ) {
				$mid = substr( $html, $li, $s - $li );
				if ( false === strpos( $mid, '</li>' ) ) {
					$le = strpos( $html, '</li>', $e );
					if ( false !== $le ) {
						if ( $le - $e < 40 ) { $s = $li; $e = $le + 5; }
					}
				}
			}
			$stella_mm_log[] = '뺌   ' . $href;
			$html = substr( $html, 0, $s ) . substr( $html, $e );
		}
	}

	/* 2) 옛 가디언 이름 바꾸기 */
	foreach ( stella_mm_renames() as $old => $new ) {
		$k = substr_count( $html, $old );
		if ( ! $k ) { continue; }
		$html = str_replace( $old, $new, $html );
		$stella_mm_log[] = '이름 ' . $old . ' -> ' . $new . '  (' . $k . '군데)';
	}

	/* 3) 타로 -> 스텔라 타로
	   아르카나 문으로 가는 줄의 이름표만 손댑니다. 이름표 뒤에
	   설명글(<span class="sub">)이 바로 붙어 있어서 「>타로<」로는
	   안 잡힙니다. 주소로 자리를 잡고 그 뒤 글자를 봅니다. */
	$at = strpos( $html, 'href="/door-tarot/"' );
	if ( false !== $at ) {
		$gt = strpos( $html, '>', $at );
		if ( false !== $gt ) {
			$head = substr( $html, $gt + 1, 24 );
			$head = ltrim( $head );
			if ( 0 === strpos( $head, '스텔라 타로' ) ) {
				/* 이미 되어 있습니다 */
			} elseif ( 0 === strpos( $head, '타로' ) ) {
				$k = strpos( $html, '타로', $gt );
				$html = substr( $html, 0, $k ) . '스텔라 타로'
				      . substr( $html, $k + strlen( '타로' ) );
				$stella_mm_log[] = '이름 타로 -> 스텔라 타로';
			}
		}
	}

	/* 4) 연예인 몇 명 — 지금 수로 */
	$at = strpos( $html, '연예인 ' );
	if ( false !== $at ) {
		$e = strpos( $html, '명', $at );
		if ( false !== $e ) {
			if ( $e - $at < 30 ) {
				$cnt = stella_mm_star_count();
				if ( $cnt > 0 ) {
					$was  = substr( $html, $at, $e - $at + 3 );
					$now  = '연예인 ' . number_format( $cnt ) . '명';
					$html = substr( $html, 0, $at ) . $now . substr( $html, $e + 3 );
					$stella_mm_log[] = '숫자 ' . trim( $was ) . ' -> ' . $now;
				}
			}
		}
	}

	if ( $html === $before ) { $stella_mm_log[] = '이 덩어리에서는 바꿀 것이 없었습니다'; }
	return $html;
}, 10, 2 );

/* ── 무엇을 했는지 보여주기 ─────────────────────── */
add_action( 'wp_footer', function () {

	if ( ! isset( $_GET['stella_mobilemenu'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $stella_mm_log;
	$log = is_array( $stella_mm_log ) ? $stella_mm_log : array();

	echo '<div style="position:relative;z-index:99999;background:#12101B;'
	   . 'color:#F2ECE0;padding:20px 16px;font:13px/1.8 monospace;'
	   . 'white-space:pre-wrap;border-top:3px solid #D4AF6A">';
	echo "모바일 메뉴에서 한 일  " . count( $log ) . "가지\n";
	echo "(메뉴를 한 번 펼치고 다시 보세요)\n\n";
	if ( ! count( $log ) ) {
		echo "아무것도 못 찾았습니다.\n";
		echo "이 메뉴는 class=\"sub\" 를 안 쓰는 모양입니다.\n";
	} else {
		foreach ( $log as $s ) { echo '   ' . esc_html( $s ) . "\n"; }
	}
	echo "\n스타 사주에 지금 들어 있는 사람  "
	   . number_format( stella_mm_star_count() ) . "명\n";
	echo '</div>';
}, 999 );
