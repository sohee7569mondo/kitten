if ( ! function_exists( 'stella_umog_bits' ) ) {
	function stella_umog_bits() {
		if ( ! isset( $_GET['sc'] ) ) { return null; }
		$rel = isset( $_GET['rel'] ) ? preg_replace( '/[^a-z_]/', '', (string) $_GET['rel'] ) : 'lover';
		if ( ! $rel ) { $rel = 'lover'; }
		return array(
			'sc'  => max( 0, min( 99, intval( $_GET['sc'] ) ) ),
			'rel' => $rel,
			'a'   => isset( $_GET['a'] ) ? max( 0, min( 40, intval( $_GET['a'] ) ) ) : 0,
			'b'   => isset( $_GET['b'] ) ? max( 0, min( 30, intval( $_GET['b'] ) ) ) : 0,
			'c'   => isset( $_GET['c'] ) ? max( 0, min( 30, intval( $_GET['c'] ) ) ) : 0,
		);
	}
}

/* ── 가로 카드 이름 ───────────────────────────────────── */
if ( ! function_exists( 'stella_umw_name' ) ) {
	function stella_umw_name( $p, $shape = 'w' ) {
		$v = substr( md5( (string) get_option( 'stella_um_bg', '' )
			. wp_json_encode( get_option( 'stella_um_bg_map', array() ) )
			. 'wide-2026-09-17' ), 0, 6 );
		return $shape . $p['sc'] . '-' . $p['rel'] . '-' . $p['a'] . '-' . $p['b'] . '-'
			. $p['c'] . '-' . $v . '.jpg';
	}
}

if ( ! function_exists( 'stella_umw_tall' ) ) {
	/* 완성된 세로 카드를 구합니다 — 파일이 있으면 파일, 없으면 그리게 부릅니다 */
	function stella_umw_tall( $p ) {
		if ( ! function_exists( 'stella_um_card_name' ) ) { return null; }
		$up   = wp_upload_dir();
		$dir  = $up['basedir'] . '/uandme/card';
		$file = $dir . '/' . stella_um_card_name( $p['sc'], $p['rel'], $p['a'], $p['b'], $p['c'] );
		if ( file_exists( $file ) ) {
			$im = @imagecreatefrompng( $file );
			if ( $im ) { return $im; }
		}
		$u = home_url( '/um-card/' . $p['sc'] . '-' . $p['rel'] . '-'
			. $p['a'] . '-' . $p['b'] . '-' . $p['c'] );
		$r = wp_remote_get( $u, array( 'timeout' => 40, 'sslverify' => false ) );
		if ( is_wp_error( $r ) ) { return null; }
		$bd = wp_remote_retrieve_body( $r );
		if ( strlen( $bd ) < 500 ) { return null; }
		$im = @imagecreatefromstring( $bd );
		return $im ? $im : null;
	}
}

/* ════════════════════════════════════════════════════════════
   /um-tall/ · 소희 님 카드를 **그대로** 작게

   소희 님 「배경없이 우리 카드만 보이게 하면안돼? 글씨가 작아
            카드에 누구랑 누구 궁합인지 이름이 빠져있네」

   ④번이 그린 세로 카드(1080x1920 PNG · 1.19MB)를 그대로 읽어
   폭 800 JPEG 로 내보냅니다. 덧대는 것도 자르는 것도 없습니다.
   두 사람 이름은 카드에 이미 들어 있습니다.

       확인 : https://stellasaju.com/um-tall/79-lover-30-12-21

   ★ i0.wp.com CDN 은 안 씁니다 — 「remote data could not be
     fetched」로 원본을 못 가져왔습니다. 우리가 직접 그려 냅니다.
   ════════════════════════════════════════════════════════════ */
add_action( 'init', function () {

	$uri = isset( $_SERVER['REQUEST_URI'] ) ? $_SERVER['REQUEST_URI'] : '';
	if ( false === strpos( $uri, '/um-tall/' ) ) { return; }
	if ( ! preg_match( '#/um-tall/([0-9]{1,2})-([a-z_]{2,14})-([0-9]{1,2})-([0-9]{1,2})-([0-9]{1,2})(?:-([A-Za-z0-9_-]{1,140}))?#', $uri, $m ) ) { return; }
	if ( ! function_exists( 'stella_um_card_name' ) ) { return; }

	$sc  = max( 0, min( 99, intval( $m[1] ) ) );
	$rel = $m[2];
	$a   = max( 0, min( 40, intval( $m[3] ) ) );
	$b   = max( 0, min( 30, intval( $m[4] ) ) );
	$c   = max( 0, min( 30, intval( $m[5] ) ) );
	$nm  = isset( $m[6] ) ? $m[6] : '';

	$up  = wp_upload_dir();
	$dir = $up['basedir'] . '/uandme/card';
	if ( ! is_dir( $dir ) ) { wp_mkdir_p( $dir ); }

	$src = $dir . '/' . stella_um_card_name( $sc, $rel, $a, $b, $c, $nm );
	$out = $dir . '/t' . basename( $src, '.png' ) . '.jpg';

	$fresh = current_user_can( 'manage_options' );
	if ( isset( $_GET['new'] ) ) { $fresh = true; }

	if ( file_exists( $out ) ) {
		if ( ! $fresh ) {
			header( 'Content-Type: image/jpeg' );
			header( 'Content-Length: ' . (int) filesize( $out ) );
			header( 'Cache-Control: public, max-age=604800' );
			readfile( $out );
			exit;
		}
	}

	/* 원본이 없으면 ④번에게 그리게 합니다 (이름 꼬리까지 그대로) */
	if ( ! file_exists( $src ) ) {
		$u = home_url( '/um-card/' . $sc . '-' . $rel . '-' . $a . '-' . $b . '-' . $c );
		if ( $nm ) { $u .= '-' . $nm; }
		wp_remote_get( $u, array( 'timeout' => 40, 'sslverify' => false ) );
	}
	if ( ! file_exists( $src ) ) {
		status_header( 404 );
		header( 'Content-Type: text/plain; charset=utf-8' );
		echo '카드 원본이 아직 없습니다 : ' . basename( $src );
		exit;
	}

	$im = @imagecreatefrompng( $src );
	if ( ! $im ) {
		status_header( 404 );
		header( 'Content-Type: text/plain; charset=utf-8' );
		echo '카드를 못 읽었습니다.';
		exit;
	}
	$w0 = imagesx( $im );
	$h0 = imagesy( $im );
	$W  = 800;
	if ( $w0 < $W ) { $W = $w0; }
	$H  = (int) round( $W * $h0 / max( 1, $w0 ) );
	$sm = imagecreatetruecolor( $W, $H );
	imagefilledrectangle( $sm, 0, 0, $W, $H, imagecolorallocate( $sm, 255, 255, 255 ) );
	imagecopyresampled( $sm, $im, 0, 0, 0, 0, $W, $H, $w0, $h0 );
	imagedestroy( $im );

	ob_start();
	imagejpeg( $sm, null, 88 );
	imagedestroy( $sm );
	$jpg = ob_get_clean();
	@file_put_contents( $out, $jpg );

	header( 'Content-Type: image/jpeg' );
	header( 'Content-Length: ' . strlen( $jpg ) );
	if ( $fresh ) { header( 'Cache-Control: no-store, no-cache, must-revalidate, max-age=0' ); }
	else { header( 'Cache-Control: public, max-age=604800' ); }
	echo $jpg;
	exit;

}, 1 );

/* ════════════════════════════════════════════════════════════
   정사각형 판 (1080 x 1080)

   소희 님 : 「자동으로 아래가 짤리게 하면안되나? 정사각형으로 해?」

   ★ 아래를 자르면 맨 밑 「나도 해보기」가 잘려 들어올 곳을 모릅니다.
     그래서 카드를 통째로 가운데 놓고 빈 자리는 같은 카드를 흐리게
     늘려 채웁니다 (인스타 방식). 아무것도 안 잘립니다.
   ════════════════════════════════════════════════════════════ */
if ( ! function_exists( 'stella_umw_sq' ) ) {
	function stella_umw_sq( $p ) {

		$S  = 1080;
		$im = imagecreatetruecolor( $S, $S );
		imagefilledrectangle( $im, 0, 0, $S, $S,
			imagecolorallocate( $im, 239, 233, 248 ) );

		$tall = stella_umw_tall( $p );
		if ( ! $tall ) { return $im; }

		$tw = imagesx( $tall );
		$th = imagesy( $tall );

		/* ── 바탕 : 같은 카드를 꽉 채워 자른 뒤 흐리게 ──
		   작게 줄였다 다시 키우면 빠르고 곱게 흐려집니다. */
		$cw = $S;
		$ch = (int) round( $S * $th / max( 1, $tw ) );
		if ( $ch < $S ) { $ch = $S; $cw = (int) round( $S * $tw / max( 1, $th ) ); }
		$tiny = imagecreatetruecolor( 48, max( 1, (int) round( 48 * $ch / max( 1, $cw ) ) ) );
		imagecopyresampled( $tiny, $tall, 0, 0, 0, 0,
			imagesx( $tiny ), imagesy( $tiny ), $tw, $th );
		imagecopyresampled( $im, $tiny, (int) round( ( $S - $cw ) / 2 ),
			(int) round( ( $S - $ch ) / 2 ), 0, 0, $cw, $ch,
			imagesx( $tiny ), imagesy( $tiny ) );
		imagedestroy( $tiny );

		/* 하얀 막을 한 겹 덮어 가운데 카드가 도드라지게 */
		$veil = imagecolorallocatealpha( $im, 246, 241, 252, 46 );
		imagefilledrectangle( $im, 0, 0, $S, $S, $veil );

		/* ── 가운데 : 카드를 통째로 (안 자릅니다) ── */
		$nh = 1004;
		$nw = (int) round( $nh * $tw / max( 1, $th ) );
		if ( $nw > $S - 120 ) { $nw = $S - 120; $nh = (int) round( $nw * $th / max( 1, $tw ) ); }
		$px = (int) round( ( $S - $nw ) / 2 );
		$py = (int) round( ( $S - $nh ) / 2 );

		$sh = imagecolorallocatealpha( $im, 70, 55, 115, 104 );
		imagefilledrectangle( $im, $px + 7, $py + 11, $px + $nw + 7, $py + $nh + 11, $sh );
		imagecopyresampled( $im, $tall, $px, $py, 0, 0, $nw, $nh, $tw, $th );
		imagedestroy( $tall );

		return $im;
	}
}

if ( ! function_exists( 'stella_umog_pic' ) ) {
	function stella_umog_pic( $p, $shape = '' ) {

		if ( ! $shape ) { $shape = stella_umog_shape(); }
		$tag  = ( 'sq' === $shape ) ? 'sq' : 'wide';
		$key  = ( 'sq' === $shape ) ? 'sq' : 'w';
		$size = ( 'sq' === $shape ) ? array( 1080, 1080 ) : array( 1200, 630 );

		$live = home_url( '/um-' . $tag . '/' . $p['sc'] . '-' . $p['rel'] . '-'
			. $p['a'] . '-' . $p['b'] . '-' . $p['c'] );

		$up   = wp_upload_dir();
		$dir  = $up['basedir'] . '/uandme/card';
		$file = $dir . '/' . stella_umw_name( $p, $key );

		/* ⑤번이 찾던 이름으로 세로 카드 복사본을 둡니다 (카톡 올리기용) */
		if ( function_exists( 'stella_um_card_name' ) ) {
			$tn = $dir . '/' . stella_um_card_name( $p['sc'], $p['rel'], $p['a'], $p['b'], $p['c'] );
			$lk = $dir . '/' . $p['sc'] . '-' . $p['rel'] . '-' . $p['a'] . '-'
				. $p['b'] . '-' . $p['c'] . '.png';
			if ( file_exists( $tn ) ) {
				if ( ! file_exists( $lk ) ) { @copy( $tn, $lk ); }
			}
		}
