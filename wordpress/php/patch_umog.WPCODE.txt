/* ── 글자를 폭에 맞게 줄로 나눕니다 (GD 는 스스로 안 합니다) ── */
if ( ! function_exists( 'stella_umw_wrap' ) ) {
	function stella_umw_wrap( $txt, $ttf, $size, $maxw, $maxline = 3 ) {
		$txt = trim( (string) $txt );
		if ( '' === $txt ) { return array(); }
		if ( ! $ttf ) { return array( $txt ); }

		$words = preg_split( '/\s+/u', $txt );
		$lines = array();
		$cur   = '';
		foreach ( $words as $w ) {
			$try = ( '' === $cur ) ? $w : ( $cur . ' ' . $w );
			$bb  = @imagettfbbox( $size, 0, $ttf, $try );
			$wd  = $bb ? ( $bb[2] - $bb[0] ) : ( mb_strlen( $try ) * $size );
			if ( $wd <= $maxw ) { $cur = $try; continue; }
			if ( '' !== $cur ) { $lines[] = $cur; $cur = $w; continue; }
			/* 한 낱말이 통째로 넘칠 때는 글자로 자릅니다 */
			$acc = '';
			$len = mb_strlen( $w );
			for ( $i = 0; $i < $len; $i++ ) {
				$ch = mb_substr( $w, $i, 1 );
				$bb2 = @imagettfbbox( $size, 0, $ttf, $acc . $ch );
				$w2  = $bb2 ? ( $bb2[2] - $bb2[0] ) : 0;
				if ( $w2 > $maxw ) { if ( '' !== $acc ) { $lines[] = $acc; $acc = ''; } }
				$acc .= $ch;
			}
			$cur = $acc;
		}
		if ( '' !== $cur ) { $lines[] = $cur; }
		if ( count( $lines ) > $maxline ) { $lines = array_slice( $lines, 0, $maxline ); }
		return $lines;
	}
}

/* ════════════════════════════════════════════════════════════
   가로 카드 그리기 — 1200 x 630

   ★ 소희 님이 그리신 **세로 카드를 그대로 오른쪽에 얹습니다.**
     다시 그리지 않습니다. 잘리지도 않습니다.
     왼쪽 빈 자리에 점수와 글을 크게 놓아 작은 화면에서도 읽힙니다.
   ════════════════════════════════════════════════════════════ */
if ( ! function_exists( 'stella_umw_tall' ) ) {
	/* 완성된 세로 카드를 구합니다 — 파일이 있으면 파일, 없으면 그리게 부릅니다 */
	function stella_umw_tall( $p ) {
		if ( ! function_exists( 'stella_um_card_name' ) ) { return null; }
		$up   = wp_upload_dir();
		$dir  = $up['basedir'] . '/uandme/card';
		$file = $dir . '/' . stella_um_card_name( $p['sc'], $p['rel'], $p['a'], $p['b'], $p['c'] );
		if ( file_exists( $file ) ) {
			$im = @imagecreatefrompng( $file );
			if ( $im ) { return $im; }
		}
		$u = home_url( '/um-card/' . $p['sc'] . '-' . $p['rel'] . '-'
			. $p['a'] . '-' . $p['b'] . '-' . $p['c'] );
		$r = wp_remote_get( $u, array( 'timeout' => 40, 'sslverify' => false ) );
		if ( is_wp_error( $r ) ) { return null; }
		$bd = wp_remote_retrieve_body( $r );
		if ( strlen( $bd ) < 500 ) { return null; }
		$im = @imagecreatefromstring( $bd );
		return $im ? $im : null;
	}
}

if ( ! function_exists( 'stella_umw_draw' ) ) {
	function stella_umw_draw( $p ) {

		$W = 1200; $H = 630;
		$im = imagecreatetruecolor( $W, $H );
		imagealphablending( $im, true );

		$bg    = imagecolorallocate( $im, 244, 240, 251 );
		$ink   = imagecolorallocate( $im, 63, 58, 82 );
		$purp  = imagecolorallocate( $im, 107, 91, 168 );
		$soft  = imagecolorallocate( $im, 155, 139, 208 );
		$white = imagecolorallocate( $im, 255, 255, 255 );
		imagefilledrectangle( $im, 0, 0, $W, $H, $bg );

		/* 장식 — 연한 동그라미 둘 */
		$c1 = imagecolorallocatealpha( $im, 198, 178, 240, 92 );
		$c2 = imagecolorallocatealpha( $im, 244, 190, 214, 96 );
		imagefilledellipse( $im, 1085, 66, 300, 300, $c1 );
		imagefilledellipse( $im, 44, 600, 250, 250, $c2 );

		/* 글꼴 — ④번이 받아 둔 것을 빌려 씁니다 */
		$lg = array();
		$fNum  = function_exists( 'stella_um_font' ) ? stella_um_font( 'num', $lg )     : '';
		$fBlk  = function_exists( 'stella_um_font' ) ? stella_um_font( 'gothic', $lg )  : '';
		$fBold = function_exists( 'stella_um_font' ) ? stella_um_font( 'gothicb', $lg ) : '';
		$fBody = function_exists( 'stella_um_font' ) ? stella_um_font( 'dodum', $lg )   : '';
		if ( ! $fBlk )  { $fBlk  = $fBold; }
		if ( ! $fBody ) { $fBody = $fBold; }
		if ( ! $fNum )  { $fNum  = $fBold; }

		/* ── 오른쪽 · 소희 님 세로 카드를 그대로 ── */
		$tall = stella_umw_tall( $p );
		$rightX = $W - 48;
		if ( $tall ) {
			$tw = imagesx( $tall );
			$th = imagesy( $tall );
			$nh = 566;
			$nw = (int) round( $nh * $tw / max( 1, $th ) );
			if ( $nw > 360 ) { $nw = 360; $nh = (int) round( $nw * $th / max( 1, $tw ) ); }
			$px = $W - 48 - $nw;
			$py = (int) round( ( $H - $nh ) / 2 );
			/* 그림자 한 겹 */
			$sh = imagecolorallocatealpha( $im, 90, 70, 140, 108 );
			imagefilledrectangle( $im, $px + 5, $py + 8, $px + $nw + 5, $py + $nh + 8, $sh );
			imagecopyresampled( $im, $tall, $px, $py, 0, 0, $nw, $nh, $tw, $th );
			imagedestroy( $tall );
			$rightX = $px;
		}

		/* ── 왼쪽 글 ── */
		$x    = 64;
		$maxw = $rightX - $x - 44;
		if ( $maxw < 240 ) { $maxw = 240; }

		$relko = array(
			'lover' => '연인', 'married' => '부부', 'friend' => '친구',
			'sibling' => '형제자매', 'parent_child' => '부모와 자녀',
			'boss_sub' => '상사와 부하', 'partner' => '동업자',
		);
		$rk = isset( $relko[ $p['rel'] ] ) ? $relko[ $p['rel'] ] : '우리';

		$nick = ''; $sum = ''; $l1 = ''; $l2 = '';
		if ( function_exists( 'stella_um_nick' ) ) {
			$n = stella_um_nick( $p['sc'], $p['rel'] );
			if ( is_array( $n ) ) {
				$nick = isset( $n[0] ) ? $n[0] : '';
				$l1   = isset( $n[2] ) ? $n[2] : '';
				$l2   = isset( $n[3] ) ? $n[3] : '';
				$sum  = isset( $n[5] ) ? $n[5] : '';
			}
		}
		if ( ! $nick ) {
			$nick = function_exists( 'stella_um_tier' ) ? stella_um_tier( $p['sc'] ) : '우리 궁합';
		}

		/* 관계 딱지 — 요약을 한 줄로 합칩니다 */
		$cap = $rk . ' 궁합';
		if ( $sum ) { $cap .= '  ·  ' . $sum; }
		$cl = stella_umw_wrap( $cap, $fBold, 27, $maxw, 1 );
		stella_umw_text( $im, $fBold, 27, $x, 104, $soft, $cl ? $cl[0] : $cap );

		/* 점수 */
		$sTxt = (string) $p['sc'];
		stella_umw_text( $im, $fNum, 92, $x, 210, $purp, $sTxt );
		$bb = @imagettfbbox( 92, 0, $fNum, $sTxt );
		$sw = $bb ? ( $bb[2] - $bb[0] ) : ( strlen( $sTxt ) * 55 );
		stella_umw_text( $im, $fBold, 34, $x + $sw + 12, 210, $purp, '점' );

		$y = 238;

		/* 별명 — 가장 큰 글씨 */
		$ns = 50;
		$nl = stella_umw_wrap( $nick, $fBlk, $ns, $maxw, 2 );
		if ( count( $nl ) > 1 ) {
			$ns = 42;
			$nl = stella_umw_wrap( $nick, $fBlk, $ns, $maxw, 2 );
		}
		$y += $ns + 24;
		foreach ( $nl as $ln ) {
			stella_umw_text( $im, $fBlk, $ns, $x, $y, $ink, $ln );
			$y += $ns + 14;
		}

		/* 본문 두 문장 */
		$y += 18;
		$body = trim( $l1 . ' ' . $l2 );
		$bl   = stella_umw_wrap( $body, $fBody, 26, $maxw, 3 );
		foreach ( $bl as $ln ) {
			stella_umw_text( $im, $fBody, 26, $x, $y, $purp, $ln );
			$y += 40;
		}

		/* 발치 알약 */
		$foot = 'stellasaju.com/uandme';
		$fb   = @imagettfbbox( 23, 0, $fBold, $foot );
		$fw   = $fb ? ( $fb[2] - $fb[0] ) : 300;
		$py1  = $H - 84;
		$py2  = $H - 36;
		$pill = imagecolorallocate( $im, 107, 91, 168 );
		imagefilledrectangle( $im, $x, $py1, $x + $fw + 52, $py2, $pill );
		stella_umw_text( $im, $fBold, 23, $x + 26, $py2 - 16, $white, $foot );

		return $im;
	}
}

/* ── /um-wide/<점수-관계-사주-별자리-띠> ────────────────── */
add_action( 'init', function () {

	$uri = isset( $_SERVER['REQUEST_URI'] ) ? $_SERVER['REQUEST_URI'] : '';
	if ( false === strpos( $uri, '/um-wide/' ) ) { return; }
	if ( ! preg_match( '#/um-wide/([0-9]{1,2})-([a-z_]{2,14})-([0-9]{1,2})-([0-9]{1,2})-([0-9]{1,2})#', $uri, $m ) ) { return; }

	$p = array(
		'sc'  => max( 0, min( 99, intval( $m[1] ) ) ),
		'rel' => $m[2],
		'a'   => max( 0, min( 40, intval( $m[3] ) ) ),
		'b'   => max( 0, min( 30, intval( $m[4] ) ) ),
		'c'   => max( 0, min( 30, intval( $m[5] ) ) ),
	);

	$up   = wp_upload_dir();
	$dir  = $up['basedir'] . '/uandme/card';
	if ( ! is_dir( $dir ) ) { wp_mkdir_p( $dir ); }
	$file = $dir . '/' . stella_umw_name( $p, 'w' );

	$fresh = current_user_can( 'manage_options' );
	if ( isset( $_GET['new'] ) ) { $fresh = true; }

	if ( file_exists( $file ) ) {
		if ( ! $fresh ) {
			header( 'Content-Type: image/jpeg' );
			header( 'Content-Length: ' . (int) filesize( $file ) );
			header( 'Cache-Control: public, max-age=604800' );
			readfile( $file );
			exit;
		}
	}

	$im = stella_umw_draw( $p );
	ob_start();
	imagejpeg( $im, null, 86 );
	imagedestroy( $im );
	$jpg = ob_get_clean();

	@file_put_contents( $file, $jpg );

	header( 'Content-Type: image/jpeg' );
	header( 'Content-Length: ' . strlen( $jpg ) );
	if ( $fresh ) { header( 'Cache-Control: no-store, no-cache, must-revalidate, max-age=0' ); }
	else { header( 'Cache-Control: public, max-age=604800' ); }
	echo $jpg;
	exit;

}, 1 );

/* ════════════════════════════════════════════════════════════
   정사각형 판 (1080 x 1080)

   소희 님 : 「자동으로 아래가 짤리게 하면안되나? 정사각형으로 해?」

   ★ 아래를 자르면 맨 밑 「나도 해보기」가 잘려 들어올 곳을 모릅니다.
     그래서 카드를 통째로 가운데 놓고 빈 자리는 같은 카드를 흐리게
     늘려 채웁니다 (인스타 방식). 아무것도 안 잘립니다.
   ════════════════════════════════════════════════════════════ */
if ( ! function_exists( 'stella_umw_sq' ) ) {
	function stella_umw_sq( $p ) {

		$S  = 1080;
		$im = imagecreatetruecolor( $S, $S );
		imagefilledrectangle( $im, 0, 0, $S, $S,
			imagecolorallocate( $im, 239, 233, 248 ) );

		$tall = stella_umw_tall( $p );
		if ( ! $tall ) { return $im; }

		$tw = imagesx( $tall );
		$th = imagesy( $tall );

		/* ── 바탕 : 같은 카드를 꽉 채워 자른 뒤 흐리게 ──
		   작게 줄였다 다시 키우면 빠르고 곱게 흐려집니다. */
		$cw = $S;
		$ch = (int) round( $S * $th / max( 1, $tw ) );
		if ( $ch < $S ) { $ch = $S; $cw = (int) round( $S * $tw / max( 1, $th ) ); }
		$tiny = imagecreatetruecolor( 48, max( 1, (int) round( 48 * $ch / max( 1, $cw ) ) ) );
		imagecopyresampled( $tiny, $tall, 0, 0, 0, 0,
			imagesx( $tiny ), imagesy( $tiny ), $tw, $th );
		imagecopyresampled( $im, $tiny, (int) round( ( $S - $cw ) / 2 ),
			(int) round( ( $S - $ch ) / 2 ), 0, 0, $cw, $ch,
			imagesx( $tiny ), imagesy( $tiny ) );
		imagedestroy( $tiny );

		/* 하얀 막을 한 겹 덮어 가운데 카드가 도드라지게 */
		$veil = imagecolorallocatealpha( $im, 246, 241, 252, 46 );
		imagefilledrectangle( $im, 0, 0, $S, $S, $veil );

		/* ── 가운데 : 카드를 통째로 (안 자릅니다) ── */
		$nh = 1004;
		$nw = (int) round( $nh * $tw / max( 1, $th ) );
		if ( $nw > $S - 120 ) { $nw = $S - 120; $nh = (int) round( $nw * $th / max( 1, $tw ) ); }
		$px = (int) round( ( $S - $nw ) / 2 );
		$py = (int) round( ( $S - $nh ) / 2 );

		$sh = imagecolorallocatealpha( $im, 70, 55, 115, 104 );
		imagefilledrectangle( $im, $px + 7, $py + 11, $px + $nw + 7, $py + $nh + 11, $sh );
		imagecopyresampled( $im, $tall, $px, $py, 0, 0, $nw, $nh, $tw, $th );
		imagedestroy( $tall );

		return $im;
	}
}
