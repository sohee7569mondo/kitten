/* ════════════════════════════════════════════════════════════

/* ── 사람 수 세기 ─────────────────────────────────────
   하루에 한 번만 세고 담아 둡니다. 쪽 글이 크기 때문입니다.
   사람을 더 넣으신 뒤 바로 보시려면 ?stella_starn=1 을 여세요 —
   그때는 다시 셉니다. */
if ( ! function_exists( 'stella_star_count' ) ) {
	function stella_star_count( $fresh = false ) {
		$key = 'stella_star_n';
		if ( ! $fresh ) {
			$n = (int) get_transient( $key );
			if ( $n > 0 ) { return $n; }
		}
		$p = get_page_by_path( 'star-saju' );
		if ( ! $p ) { return 0; }
		$c = (string) $p->post_content;
		/* patch_star2 가 넣는 꼴 */
		$n = substr_count( $c, '{"n":"' );
		if ( $n < 1 ) {
			/* 쪽을 손으로 고치셨을 때를 위한 다른 꼴 */
			$n = substr_count( $c, 'name:' );
		}
		if ( $n > 0 ) { set_transient( $key, $n, DAY_IN_SECONDS ); }
		return $n;
	}
}

/* ── ① 발췌의 숫자를 바로잡습니다 ─────────────────────
   DB 는 안 건드립니다. 보여줄 때만 갈아 끼웁니다.
   스니펫을 끄면 그대로 돌아옵니다. */
add_filter( 'get_the_excerpt', function ( $text, $post = null ) {
	if ( ! $post ) { return $text; }
	$id = is_object( $post ) ? (int) $post->ID : (int) $post;
	$p  = get_page_by_path( 'star-saju' );
	if ( ! $p ) { return $text; }
	if ( $id !== (int) $p->ID ) { return $text; }
	$n = stella_star_count();
	if ( $n < 1 ) { return $text; }
	return preg_replace( '/[0-9][0-9,]*\s*명/u', number_format( $n ) . '명', $text, 1 );
}, 20, 2 );

<script>
(function(){
  if(window.StellaStarN){ return; }
  window.StellaStarN = 1;

  var NOW = '<?php echo esc_js( number_format( $n ) ); ?>';

  /* 「숫자+명」을 찾아 바꿉니다. 사람 수를 말하는 자리만 고르려고
     앞에 「연예인」이 붙은 것만 봅니다 — 다른 쪽의 숫자를 건드리면
     안 되기 때문입니다. */
  function fix(t){
    var s = String(t);
    var at = s.indexOf('연예인');
    if(at < 0){ return null; }
    var i = at + 3;
    while(i < s.length){ if(s.charAt(i) === ' '){ i++; } else { break; } }
    var j = i;
    while(j < s.length){
      var ch = s.charAt(j);
      if(ch >= '0'){ if(ch <= '9'){ j++; continue; } }
      if(ch === ','){ j++; continue; }
      break;
    }
    if(j === i){ return null; }
    if(s.charAt(j) !== '명'){ return null; }
    var old = s.slice(i, j);
    if(old === NOW){ return null; }
    return s.slice(0, i) + NOW + s.slice(j);
  }

  function walk(){
    var n = 0;
    try{
      var w = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, null, false);
      var list = [], t;
      while((t = w.nextNode())){
        if(String(t.nodeValue).indexOf('연예인') >= 0){ list.push(t); }
      }
      var i;
      for(i = 0; i < list.length; i++){
        var nu = fix(list[i].nodeValue);
        if(nu !== null){ list[i].nodeValue = nu; n++; }
      }
    }catch(e){}
    return n;
  }

  function go(){
    if(!document.body){ return; }
    walk();
    /* 나중에 그려지는 자리도 한 번 더 봅니다 */
    setTimeout(walk, 800);
    setTimeout(walk, 2500);
  }
  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', go);
  } else { go(); }
})();
</script>
