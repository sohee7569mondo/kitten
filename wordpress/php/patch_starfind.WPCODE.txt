	function stella_star_count( $fresh = false ) {
		$key = 'stella_star_n';
		if ( ! $fresh ) {
			$n = (int) get_transient( $key );
			if ( $n > 0 ) { return $n; }
		}
		$p = get_page_by_path( 'star-saju' );
		if ( ! $p ) { return 0; }
		$c = (string) $p->post_content;
		$n = substr_count( $c, '{"n":"' );
		if ( $n < 1 ) { $n = substr_count( $c, 'name:' ); }
		if ( $n > 0 ) { set_transient( $key, $n, DAY_IN_SECONDS ); }
		return $n;
	}
}

add_filter( 'get_the_excerpt', function ( $text, $post = null ) {
	if ( ! $post ) { return $text; }
	$id = is_object( $post ) ? (int) $post->ID : (int) $post;
	$p  = get_page_by_path( 'star-saju' );
	if ( ! $p ) { return $text; }
	if ( $id !== (int) $p->ID ) { return $text; }
	$n = stella_star_count();
	if ( $n < 1 ) { return $text; }
	return preg_replace( '/[0-9][0-9,]*\s*명/u', number_format( $n ) . '명', $text, 1 );
}, 20, 2 );

(function(){

     워드프레스가 홑 역빗금을 한 겹 벗겨내면 정규식이 망가집니다. */
  var SP = ' ' + String.fromCharCode(9) + String.fromCharCode(10) + String.fromCharCode(13);

  function tidy(v){
    var s = String(v === null ? '' : v), out = '', i, ch;
    for(i = 0; i < s.length; i++){
      ch = s.charAt(i);
      if(SP.indexOf(ch) >= 0){ ch = ' '; }
      out += ch;
    }
    while(out.indexOf('  ') >= 0){ out = out.split('  ').join(' '); }
    return out.trim();
  }

  function squash(v){
    var s = String(v === null ? '' : v), out = '', i, ch;
    for(i = 0; i < s.length; i++){
      ch = s.charAt(i);
      if(SP.indexOf(ch) >= 0){ continue; }
      out += ch;
    }
    return out;
  }

  function txt(el){

  function findRoster(){

    for(i = 0; i < cnt.length; i++){
