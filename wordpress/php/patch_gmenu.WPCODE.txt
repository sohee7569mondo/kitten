/* ══════════════════════════════════════════════════════════
   메뉴에 남은 옛 이름을 바꿉니다
   ★ 2026-09-12 · 소희 님께

   소희 님 — 「매뉴 옆에 벼리 마루 미르 이렇게 넣어야 할 거
              같은데 아직 살아 있음」

   맞습니다. 카드(받으신 풀이)는 이미 새 이름인데
   메뉴 옆 회색 글씨만 옛 이름으로 남아 있습니다.

       직업 · 재물   직성의 신   ← 이 자리
       연애 · 결혼   연성의 신
       건강 · 가족   강성의 신
       흐름 · 시기   세성의 신
       별자리        성좌의 신

   ── 바꾸는 이름 ───────────────────────────────────────
       직성의 신 → 마루      세성의 신 → 미르
       연성의 신 → 벼리      성좌의 신 → 아라
       강성의 신 → 아람

     중간에 잠깐 썼던 이름도 같이 잡습니다.
       일을 짓는 신 → 마루    때를 아는 신 → 미르
       인연을 잇는 신 → 벼리  별을 읽는 신 → 아라
       몸을 지키는 신 → 아람

     「아르카나」는 그대로 둡니다. 이미 제 이름입니다.

   ── 어디를 뒤지나 ─────────────────────────────────────
     메뉴가 한 군데에만 있지 않습니다. 그래서 네 곳을 다 봅니다.
       nav_menu_item     손으로 만든 메뉴 (이름과 설명)
       wp_navigation     블록 메뉴
       wp_template_part  머리말 · 바닥글
       wp_block          따로 떼어둔 덩어리
     글(페이지·포스트)은 건드리지 않습니다. 메뉴만입니다.

   ── 지키는 것 ─────────────────────────────────────────
   · 먼저 보기에서는 한 글자도 안 바꿉니다. 어디에 몇 개인지만 셉니다.
   · 바꾸기 전 판을 따로 담아둡니다.
     ★ update_option 으로 담습니다 — 역슬래시를 벗겨내지 않는 자리입니다.
     (update_post_meta 로 담았다가 책이 깨졌습니다. 같은 실수를 안 합니다.)
   · 두 번 눌러도 한 번만 바뀝니다. 옛 이름이 없으면 그냥 물러납니다.

   먼저 보기 https://stellasaju.com/?stella_gmenu=look
   바꾸기    https://stellasaju.com/?stella_gmenu=apply
   되돌리기  https://stellasaju.com/?stella_gmenu=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_gmenu'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_gmenu'] ) );
	$HOLD = 'stella_gmenu_hold';

	$MAP = array(
		'직성의 신'     => '마루',
		'연성의 신'     => '벼리',
		'강성의 신'     => '아람',
		'세성의 신'     => '미르',
		'성좌의 신'     => '아라',
		'일을 짓는 신'  => '마루',
		'인연을 잇는 신' => '벼리',
		'몸을 지키는 신' => '아람',
		'때를 아는 신'  => '미르',
		'별을 읽는 신'  => '아라',
	);

	/* ── 되돌리기 ───────────────────────────────────── */
	if ( 'undo' === $mode ) {
		$held = get_option( $HOLD, array() );
		if ( ! is_array( $held ) ) { $held = array(); }
		if ( empty( $held ) ) { echo '되돌릴 것이 없습니다.'; exit; }
		$n = 0;
		foreach ( $held as $row ) {
			$wpdb->update( $wpdb->posts, array(
				'post_title'   => $row['t'],
				'post_content' => $row['c'],
			), array( 'ID' => (int) $row['id'] ) );
			clean_post_cache( (int) $row['id'] );
			$n = $n + 1;
		}
		delete_option( $HOLD );
		wp_cache_flush();
		echo $n . "군데를 옛 이름으로 되돌렸습니다.\n";
		exit;
	}

	$rows = $wpdb->get_results(
		"SELECT ID, post_type, post_title, post_content
		   FROM {$wpdb->posts}
		  WHERE post_type IN ('nav_menu_item','wp_navigation','wp_template_part','wp_block')
		    AND post_status NOT IN ('trash','auto-draft')
		  ORDER BY post_type, ID" );

	if ( ! $rows ) { echo '메뉴를 하나도 못 찾았습니다.'; exit; }

	/* 어디에 옛 이름이 있는지 모읍니다 */
	$hits = array();
	foreach ( $rows as $r ) {
		$t = (string) $r->post_title;
		$c = (string) $r->post_content;
		$nt = $t;
		$nc = $c;
		$found = array();
		foreach ( $MAP as $old => $new ) {
			$k = substr_count( $t, $old ) + substr_count( $c, $old );
			if ( 0 === $k ) { continue; }
			$found[] = $old . ' → ' . $new . ' (' . $k . ')';
			$nt = str_replace( $old, $new, $nt );
			$nc = str_replace( $old, $new, $nc );
		}
		if ( empty( $found ) ) { continue; }
		$hits[] = array(
			'id'    => (int) $r->ID,
			'type'  => $r->post_type,
			't'     => $t,
			'c'     => $c,
			'nt'    => $nt,
			'nc'    => $nc,
			'found' => $found,
		);
	}

	/* ── 먼저 보기 ──────────────────────────────────── */
	if ( 'look' === $mode ) {
		echo "═══ 메뉴 안의 옛 이름 ═══════════════════\n";
		echo '  뒤진 자리 : ' . count( $rows ) . "개\n";
		echo '  걸린 자리 : ' . count( $hits ) . "개\n\n";
		if ( empty( $hits ) ) {
			echo "옛 이름이 하나도 없습니다.\n";
			echo "메뉴 옆 글씨가 다른 데서 오는 모양입니다. 말씀해 주세요.\n";
			exit;
		}
		foreach ( $hits as $h ) {
			echo '· ' . $h['type'] . ' #' . $h['id'] . '  「' . $h['t'] . "」\n";
			foreach ( $h['found'] as $f ) { echo '    ' . $f . "\n"; }
			if ( $h['t'] !== $h['nt'] ) {
				echo '    이름 : ' . $h['t'] . '  →  ' . $h['nt'] . "\n";
			}
			if ( $h['c'] !== $h['nc'] ) {
				if ( strlen( $h['c'] ) < 200 ) {
					echo '    설명 : ' . trim( $h['c'] ) . '  →  ' . trim( $h['nc'] ) . "\n";
				} else {
					echo '    안쪽 글 ' . number_format( strlen( $h['c'] ) ) . "자 안에 있습니다\n";
				}
			}
			echo "\n";
		}
		echo "바꾸시려면 ?stella_gmenu=apply\n";
		exit;
	}

	if ( 'apply' !== $mode ) { echo '?stella_gmenu=look 부터 보세요.'; exit; }

	if ( empty( $hits ) ) { echo '바꿀 것이 없습니다. 이미 새 이름입니다.'; exit; }

	/* ── 바꾸기 ─────────────────────────────────────── */
	$held = array();
	foreach ( $hits as $h ) {
		$held[] = array( 'id' => $h['id'], 't' => $h['t'], 'c' => $h['c'] );
	}
	update_option( $HOLD, $held, false );

	$n = 0;
	foreach ( $hits as $h ) {
		$wpdb->update( $wpdb->posts, array(
			'post_title'   => $h['nt'],
			'post_content' => $h['nc'],
		), array( 'ID' => $h['id'] ) );
		clean_post_cache( $h['id'] );
		$n = $n + 1;
	}
	wp_cache_flush();

	echo $n . "군데를 새 이름으로 바꿨습니다.\n\n";
	foreach ( $hits as $h ) {
		echo '  ' . $h['type'] . ' #' . $h['id'] . '  ' . $h['nt'] . "\n";
	}
	echo "\n메뉴를 열어보세요. 되돌리려면 ?stella_gmenu=undo\n";
	exit;
}, 1 );
