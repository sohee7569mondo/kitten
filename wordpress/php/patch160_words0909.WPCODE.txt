/* ════════════════════════════════════════════════════════════
   책 글 고치기 · 네 군데   ★ 2026-09-09 · 소희 님께

   ★ 실제 글자를 찾아보고 다시 만들었습니다.

   ── 고치는 것 ─────────────────────────────────────────
   ① 「네 자리가 한 사람이 되면」  (앞엣것 하나만)
      -> 「네 개의 별이 의미하는 것」

      ★ 찾아보니 이건 사주 기둥 넷이 아니라
        태양 · 달 · 상승궁 · 천정 — 별 넷이었습니다.
        소희 님이 처음 말씀하신 쪽이 맞았습니다.
        뒤엣것(아라 「더 깊이」)은 그대로 둡니다.        (22)

   ② 「다섯 저울을 읽는 법」
      -> 「왜 나는 사주와 다른 답을 했지?」               (32)
      ★ 글은 그대로 둡니다. 접는 단추 이름만 바꿉니다.

   ③ 「여덟 글자에서 제일 센 것」
      -> 「여덟 글자에서 제일 많은 것」                    (41)
      ★ 이 자리는 일곱 글자를 열 자리에 하나씩 넣어
        세는 곳이라, 「많은 것」이 사실에 맞습니다.

   ④ 책 가운데의 점성술 권유를 한 줄로 줄입니다.         (24)

        전  사주는 태어난 때를 흙의 언어로 읽고,
            별자리는 같은 때를 빛의 언어로 읽습니다.
            여기서는 세 자리만 짚었어요.
            더 보고 싶으시면 그 이야기가 따로 있습니다.

        후  여기까지 읽고 사주보다 별자리의 해설이
            더 맞다고 생각하시면,
            점성술에서 더 자세히 볼 수 있습니다.

   ── 안 하는 것 ────────────────────────────────────────
   찾은 게 없으면 아무것도 안 바꿉니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_w0909=1
   ② 정말 고치기 https://stellasaju.com/?stella_w0909=apply
   ③ 되돌리기    https://stellasaju.com/?stella_w0909=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_w0909'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_w0909_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_w0909'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.85;'
	   . 'padding:34px 24px;max-width:940px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.35rem;margin:0 0 18px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.88rem}a{color:#D4AF6A}'
	   . 'table{border-collapse:collapse;width:100%}'
	   . 'td{padding:8px 10px;border-bottom:1px solid rgba(255,255,255,.09);'
	   . 'font-size:.9rem;vertical-align:top}'
	   . 'code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}'
	   . '</style>';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '<h1 class="no">reading-book 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	$swap = array(

		/* ① 별 넷 — 앞엣것 하나. mh+= 로 시작하는 쪽이 달자리 문단입니다 */
		"mh+='<h3>네 자리가 한 사람이 되면</h3>';"
			=> "mh+='<h3>네 개의 별이 의미하는 것</h3>';",

		/* ② 접는 단추 이름 */
		'<summary>다섯 저울을 읽는 법</summary>'
			=> '<summary>왜 나는 사주와 다른 답을 했지?</summary>',

		/* ③ 센 것 -> 많은 것 */
		'<h2>여덟 글자에서 제일 센 것</h2>'
			=> '<h2>여덟 글자에서 제일 많은 것</h2>',

		/* ④ 가운데 권유를 한 줄로 */
		'사주는 태어난 때를 흙의 언어로 읽고, 별자리는 같은 때를 빛의 언어로 읽습니다.'
			=> '여기까지 읽고 사주보다 별자리의 해설이 더 맞다고 생각하시면,',
		'여기서는 세 자리만 짚었어요. 더 보고 싶으시면 그 이야기가 따로 있습니다.'
			=> '점성술에서 더 자세히 볼 수 있습니다.',
	);

	$rows  = '';
	$total = 0;
	foreach ( $swap as $from => $to ) {
		$n = substr_count( $c, $from );
		$total += $n;
		$rows .= '<tr><td>' . ( $n ? '<b>' . $n . '</b>' : '<span class="no">0</span>' )
		       . '</td><td>' . esc_html( mb_substr( $from, 0, 60 ) ) . '</td>'
		       . '<td>-></td><td><b>' . esc_html( mb_substr( $to, 0, 60 ) ) . '</b></td></tr>';
	}

	echo '<h1>' . ( 'apply' === $mode ? '고쳤습니다' : '이렇게 바꿉니다 (아직 안 고쳤습니다)' ) . '</h1>';
	echo '<table>' . $rows . '</table>';

	if ( 0 === $total ) {
		echo '<h1 class="no">바꿀 말을 하나도 못 찾았습니다.</h1>';
		echo '<p>아무것도 안 바꿨습니다.</p>';
		exit;
	}

	echo '<p class="dim">0 으로 나온 줄은 건너뜁니다. 나머지만 바꿉니다.</p>';

	if ( 'apply' !== $mode ) {
		echo '<p>정말 고치시려면 <code>/?stella_w0909=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}

	$new = strtr( $c, $swap );

	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<p class="ok">고쳤습니다. ' . number_format( strlen( $c ) ) . ' -> '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p><a href="/reading-book/?demo=door-love&v=' . time() . '">책 열어보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_w0909=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
