/* ════════════════════════════════════════════════════════════
   「나의 사주와 별자리」 목차를 지금 책에 맞춥니다   ★ 2026-09-08

   소희 님 말씀 :
     「당신이 아무 말도 하기 전에 — 이거 문구 바꾼 거 같은데,
      사주가 보는 당신의 모습 (책엔 이름으로 들어가고)」
     「이 부분 수정하면 우리도 한번 정리될 거 같아」

   맞습니다. 목차가 옛날 책을 가리키고 있었습니다.

   ── 무엇이 어긋나 있었나 ────────────────────────
   책은 그동안 「1부 · 2부」에서 「제 1 장 ~ 제 6 장」으로 바뀌었고
   장마다 속표지가 붙었습니다. 그런데 목차는 아직 1부 · 2부입니다.

     08  당신이 아무 말도 하기 전에   1부   ← 책에는 이 제목이 없습니다
     10  당신이 답해주신 것           2부
     17  가디언이 드리는 마지막 말          ← 「가디언」은 우리끼리 쓰는 말

   책의 지금 속표지는 이렇습니다 (되살린 사이트 글에서 그대로 읽었습니다) —

     제 1 장  사주가 말하는 ○○님 모습    여덟 글자를 보고 전해드립니다
     제 2 장  답에서 말하는 ○○님 모습    일곱 가지를 보고 전해드립니다
     제 3 장  현실에서의 ○○님 모습
     제 4 장  (고르신 주제)
     제 5 장  (주제 안에서 더 깊이 · 타로를 뽑으셨으면 카드 세 장)
     제 6 장  언제 움직이면 되는가

   ── 새 목차 ────────────────────────────────────
   장을 그대로 옮겼습니다. 소희 님이 주신 말을 8번에 넣었습니다.
   목차 쪽에는 이름이 없으니 「당신」으로 두고, 책에서는 그대로
   손님 이름이 들어갑니다.

   「가디언」은 두 군데(02 · 17)에서 「쓴 이」로 바꿉니다.
   책 표지의 칸이 이미 「쓴 이」로 바뀌어 있습니다.

   제 4 · 5 · 6 장은 문과 주제에 따라 달라져서, 목차 아래에
   그렇다고 한 줄 적었습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_toc=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_toc=apply
   ③ 되돌리기    https://stellasaju.com/?stella_toc=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_toc'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_toc_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_toc'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$page = get_page_by_path( 'guide' );
	if ( ! $page ) { echo '<h1 class="no">guide 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	/* 2026-09-08 · 두 번째 판 — 통째로 갈아끼우지 않고 줄 하나씩 바꿉니다.
	   통째로 잡는 방법은 사이트에서 안 걸렸습니다. 줄 하나씩이면
	   어느 줄이 걸리고 어느 줄이 안 걸리는지 화면에 그대로 나옵니다.
	   못 걸린 줄이 있어도 나머지는 그대로 들어갑니다. */

	$jobs = array(

		array( '02 · 가디언 → 쓴 이',
			'<div class="t">여는 글 — 가디언이 직접</div>',
			'<div class="t">여는 글 — 쓴 이가 직접</div>' ),

		array( '08 · 제목을 지금 책에 맞춤',
			'<div class="t">당신이 아무 말도 하기 전에</div><div class="g">1부</div>',
			'<div class="t">사주가 보는 당신의 모습</div><div class="g">제 1 장</div>' ),

		array( '09 · 1부 → 제 1 장',
			'<div class="t">주제의 언어로 옮기면</div><div class="g">1부</div>',
			'<div class="t">주제의 언어로 옮기면</div><div class="g">제 1 장</div>' ),

		array( '10 · 제목과 꼬리표',
			'<div class="t">당신이 답해주신 것</div><div class="g">2부</div>',
			'<div class="t">답에서 말하는 당신의 모습</div><div class="g">제 2 장</div>' ),

		array( '11 · 2부 → 제 2 장',
			'<div class="t">적어주신 말</div><div class="g">2부</div>',
			'<div class="t">적어주신 말</div><div class="g">제 2 장</div>' ),

		array( '12 · 2부 → 제 3 장',
			'<div class="t">어긋나는 자리 — 두 저울</div><div class="g">2부</div>',
			'<div class="t">어긋나는 자리 — 두 저울</div><div class="g">제 3 장</div>' ),

		array( '13 · 2부 → 제 3 장',
			'<div class="t">어긋나는 자리 — 남은 셋</div><div class="g">2부</div>',
			'<div class="t">어긋나는 자리 — 남은 셋</div><div class="g">제 3 장</div>' ),

		array( '17 · 가디언 → 쓴 이',
			'<div class="t">가디언이 드리는 마지막 말</div>',
			'<div class="t">쓴 이가 드리는 마지막 말</div>' ),

		array( '꼬리 · 장마다 달라진다는 한 줄',
			'<p>PDF 로 저장해 두시면',
			'<p class="sub" style="margin-top:14px">제 4 · 5 · 6 장은 고르신 문과 주제에 '
			. '따라 달라집니다. 책 안에서는 장 제목에 손님 이름이 들어갑니다 — '
			. '「사주가 보는 덕선님의 모습」처럼요.</p>'
			. "\n      " . '<p>PDF 로 저장해 두시면' ),
	);

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$new = $c; $hit = 0; $miss = 0;

	foreach ( $jobs as $j ) {
		$n = substr_count( $new, $j[1] );
		echo '<h2>' . esc_html( $j[0] ) . '</h2>';
		if ( 1 === $n ) {
			$new = str_replace( $j[1], $j[2], $new );
			echo '<p class="ok">찾았습니다 — 한 군데.</p>';
			$hit++;
		} elseif ( 0 === $n ) {
			echo '<p class="no">못 찾았습니다. 이미 바꿨거나 글이 다릅니다.</p>';
			$miss++;
		} else {
			echo '<p class="no">' . intval( $n ) . '군데입니다. 이 줄은 건너뜁니다.</p>';
			$miss++;
		}
	}

	if ( ! $hit ) {
		echo '<h2>정리</h2><p class="no">걸린 줄이 하나도 없습니다. 아무것도 안 바꿨습니다.</p>';
		echo '<h2>지금 목차가 어떻게 생겼나</h2>';
		$at = strpos( $c, '<div class="toc">' );
		if ( false === $at ) {
			echo '<p class="no">「class="toc"」 자체가 없습니다. 이 쪽이 맞나요?</p>';
		} else {
			echo '<pre style="white-space:pre-wrap;color:#C9C2D8;font-size:.82rem;'
			   . 'background:rgba(255,255,255,.04);padding:10px 12px;border-radius:6px">'
			   . esc_html( substr( $c, $at, 1400 ) ) . '</pre>';
		}
		exit;
	}

	if ( ! $apply ) {
		echo '<h2>정리</h2><p>' . intval( $hit ) . '줄 바꿉니다 &middot; '
		   . intval( $miss ) . '줄 못 찾음</p>';
		echo '<p>정말 바꾸시려면 <code>/?stella_toc=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );

	echo '<h2>정리</h2>';
	echo '<p class="ok">' . intval( $hit ) . '줄 바꿨습니다 &middot; '
	   . intval( $miss ) . '줄 못 찾음 &middot; '
	   . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p><a href="/guide/">목차 열어보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_toc=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
