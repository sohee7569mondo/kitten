/* ════════════════════════════════════════════════════════════
   「나의 사주와 별자리」 목차를 지금 책 그대로   ★ 2026-09-08 · 3판

   소희 님이 사이트에서 뽑아주신 책의 제목 마흔셋을 그대로 옮겼습니다.
   짐작이 아니라 사이트에 들어 있는 글에서 읽은 것입니다.

   ── 앞판이 왜 안 걸렸나 ────────────────────────
   자리를 정규식으로 잡았는데, 글에 UTF-8 이 아닌 바이트가 하나만
   섞여 있어도 /u 정규식은 조용히 실패합니다. 이번 판은 정규식을
   아예 안 씁니다 — strpos 로 앞뒤만 찾아서 그 사이를 갈아끼웁니다.
   실패할 구석이 없습니다.

   ── 무엇이 달라지나 ────────────────────────────
   앞의 목차는 「1부 · 2부」 열여덟 줄이었습니다. 지금 책은
   장이 여섯이고, 제 4장부터는 고르신 문마다 글이 통째로 다릅니다.
   그래서 목차를 둘로 나눕니다 —

     ① 어느 문에서나 같은 것   (표지 · 여는 글 · 근거 다섯 · 제1~3장 · 맺음)
     ② 고르신 문에 따라        (제 4 · 5 · 6 장 — 문마다 네댓 꼭지씩)

   소희 님이 주신 말을 제1장에 넣었습니다 —
     「사주가 보는 당신의 모습」 (책에서는 손님 이름이 들어갑니다)
   「가디언」은 「쓴 이」로 바꿉니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_toc=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_toc=apply
   ③ 되돌리기    https://stellasaju.com/?stella_toc=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_toc'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_toc_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_toc'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}.dim{color:#8B849C;font-size:.86rem}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$page = get_page_by_path( 'guide' );
	if ( ! $page ) { echo '<h1 class="no">guide 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	/* ── 어느 문에서나 같은 것 ── */
	$same = array(
		array( '표지',     '이름과 여덟 글자',            '' ),
		array( '여는 글',  '쓴 이가 직접',                '' ),
		array( '근거 하나','시각을 먼저 바로잡습니다',     '' ),
		array( '근거 둘',  '여덟 글자',                   '' ),
		array( '근거 셋',  '기운의 저울',                 '' ),
		array( '근거 넷',  '십 년마다 바뀌는 판',          '' ),
		array( '근거 다섯','태어나던 밤의 하늘',           '' ),
		array( '제 1 장',  '사주가 보는 당신의 모습',
			'여덟 글자에서 제일 센 것 &middot; 비어 있는 자리 &middot; '
			. '여덟 글자가 서로 부딪히는 자리' ),
		array( '제 2 장',  '답에서 말하는 당신의 모습',
			'답해주신 것 &middot; 적어주신 말' ),
		array( '제 3 장',  '현실에서의 당신의 모습',
			'타고난 성향과 지금의 생활을 나란히' ),
		array( '맺음말',   '세 가지만 기억하세요',
			'쓴 이가 드리는 마지막 말 &middot; 이 책이 어떻게 나왔는지 &middot; 배웅' ),
	);

	/* ── 고르신 문에 따라 · 제 4 · 5 · 6 장 ──
	   사이트에 들어 있는 글에서 제목을 그대로 읽어 옮겼습니다. */
	$doors = array(
		array( '마루', '일과 돈',
			'나에게 맞는 일의 결 &middot; 돈이 들어오고 나가는 자리 &middot; '
			. '올해의 시기 흐름 &middot; 그래서 언제 움직이고 언제 기다리는지' ),
		array( '벼리', '연애와 결혼',
			'올해 인연의 흐름 &middot; 나는 왜 늘 비슷한 사람에게 끌릴까 &middot; '
			. '나는 왜 늘 같은 데서 걸릴까 &middot; 그래서 언제 말하고 언제 기다리는지' ),
		array( '아람', '건강과 가족',
			'타고난 체질의 결 &middot; 올해 조심할 계절 &middot; 가족 안에서 나의 자리 &middot; '
			. '누구를 짊어지고 누구에게 기대는지 &middot; 그래서 무엇부터 챙기시면 되는지' ),
		array( '미르', '흐름과 시기',
			'앞으로 십 년의 대운 &middot; 지금 판과 다음 판 &middot; 올해의 달별 흐름 &middot; '
			. '인생의 큰 전환점 &middot; 그래서 올해 남은 달은' ),
		array( '아라', '점성술',
			'달자리 — 아무도 안 볼 때의 당신 &middot; 태어나던 밤, 달은 이만큼 차 있었습니다 &middot; '
			. '별과 여덟 글자가 겹치는 자리 &middot; 두 언어를 한 문장으로 &middot; '
			. '그래서 어떤 결로 사는 사람인지' ),
		array( '아르카나', '스텔라타로',
			'켈틱 크로스 열 장 &middot; 나머지 다섯 장 &middot; 카드가 말하는 시기 &middot; '
			. '사주와 겹쳐 읽기 &middot; 앞으로 여섯 달 &middot; 세 장이 하는 말 &middot; '
			. '셋을 겹쳐 놓으면' ),
	);

	$toc  = '<div class="toc">' . "\n";
	foreach ( $same as $r ) {
		$toc .= '        <div class="r"><div class="p">' . $r[0] . '</div>'
		      . '<div class="t">' . $r[1]
		      . ( $r[2] ? '<span class="d2">' . $r[2] . '</span>' : '' )
		      . '</div><div class="g"></div></div>' . "\n";
	}
	$toc .= '      </div>' . "\n"
	      . '      <h2 style="margin-top:44px">고르신 문에 따라</h2>' . "\n"
	      . '      <p class="sub">제 4 &middot; 5 &middot; 6 장은 문마다 글이 통째로 다릅니다</p>' . "\n"
	      . '      <div class="toc">' . "\n";
	foreach ( $doors as $d ) {
		$toc .= '        <div class="r"><div class="p">' . $d[0] . '</div>'
		      . '<div class="t">' . $d[1]
		      . '<span class="d2">' . $d[2] . '</span>'
		      . '</div><div class="g"></div></div>' . "\n";
	}
	$toc .= '      </div>' . "\n      ";

	/* 작은 줄 하나를 위한 색표. 이미 있으면 두 번 안 넣습니다. */
	/* 왼쪽 칸이 3.4rem 이라 「근거 다섯」 「아르카나」가 안 들어갑니다. 넓힙니다. */
	$css = '<style>'
	     . '#ssi .toc .d2{display:block;margin-top:4px;color:var(--ink-dim);'
	     . 'font-size:.86rem;line-height:1.75;}'
	     . '#ssi .toc .r{grid-template-columns:5.6rem 1fr auto;align-items:start;}'
	     . '#ssi .toc .p{white-space:nowrap;}'
	     . '@media(max-width:560px){#ssi .toc .r{grid-template-columns:1fr;gap:2px;}'
	     . '#ssi .toc .p{opacity:1;}}'
	     . '</style>' . "\n      ";

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	/* ── 정규식을 안 씁니다. 앞뒤를 글자로 찾습니다. ── */
	$a = strpos( $c, '<div class="toc">' );
	$b = strpos( $c, '<p>PDF 로 저장해' );

	if ( false === $a || false === $b || $b <= $a ) {
		echo '<p class="no">목차의 앞뒤를 못 찾았습니다.</p>';
		echo '<p class="dim">「&lt;div class=&quot;toc&quot;&gt;」 '
		   . ( false === $a ? '없음' : intval( $a ) . '번째 글자' ) . ' &middot; '
		   . '「&lt;p&gt;PDF 로 저장해」 '
		   . ( false === $b ? '없음' : intval( $b ) . '번째 글자' ) . '</p>';
		exit;
	}

	$new = substr( $c, 0, $a ) . $css . $toc . substr( $c, $b );

	/* 열여덟 장 어쩌고 하던 줄도 같이 고칩니다 */
	$oldsub = '<p class="sub">열여덟 장. 카드를 뽑지 않으시면 열여섯 장</p>';
	$newsub = '<p class="sub">앞의 열한 꼭지는 어느 문에서나 같습니다. '
	        . '제 4 장부터가 고르신 문에 따라 달라집니다.</p>';
	$subok = ( false !== strpos( $new, $oldsub ) );
	if ( $subok ) { $new = str_replace( $oldsub, $newsub, $new ); }

	echo '<h2>바뀐 목차</h2>';
	echo '<div style="background:#FFFDF9;color:#221C33;padding:18px 20px;border-radius:8px;'
	   . 'font-size:.92rem;line-height:1.9">' . $css . $toc . '</div>';
	echo '<p class="dim">「열여덟 장」 안내줄 : ' . ( $subok ? '같이 고칩니다' : '못 찾음 (그대로 둡니다)' ) . '</p>';
	echo '<p class="dim">' . number_format( strlen( $c ) ) . ' → '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_toc=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );

	echo '<p class="ok">바꿨습니다.</p>';
	echo '<p><a href="/guide/">목차 열어보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_toc=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
