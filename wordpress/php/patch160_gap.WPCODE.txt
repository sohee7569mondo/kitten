       짝사랑  「말할까 말까 며칠째 망설이실 때」
               「이 마음을 언제까지 들고 갈 것인가?」
       이별운  「그 사람 소식이 궁금해 손이 가실 때」
               「이것을 아직 들고 갈 것인가, 이제 내려놓을 것인가?」

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_gap'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_gap'] ) );
	$META = '_stella_gap_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	/* ══ 판 GAP-7 (2026-09-13) · undo 를 두 단계로 ══════════════
	   이 조각의 백업은 **GAP 을 처음 apply 한 그 순간의 책**입니다.
	   그 뒤에 다른 조각을 여럿 넣으셨다면, undo 는 그것들까지 통째로
	   되돌립니다. 그래서 크기를 먼저 보여드리고 한 번 더 물어봅니다. */
	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$now = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
		echo "═══ 되돌리기 · 먼저 크기를 보세요 ═══\n\n";
		echo '  지금 책   ' . number_format( strlen( $now ) ) . "자\n";
		echo '  백업      ' . number_format( strlen( $back ) ) . "자\n";
		$gap = strlen( $now ) - strlen( $back );
		echo '  차이      ' . number_format( $gap ) . "자\n\n";
		if ( $gap > 50000 ) {
			echo "★★ 위험합니다. 차이가 너무 큽니다.\n\n";
			echo "   이 백업은 GAP 을 처음 넣으신 그때의 책입니다.\n";
			echo "   그 뒤에 넣으신 다른 조각들(결혼운 미혼·기혼, 이혼운,\n";
			echo "   거스르는 자리, 여는 말 …)이 **전부 사라집니다.**\n\n";
			echo "   되돌리지 마세요. 벼리에게 이 화면을 보여주세요.\n";
			exit;
		}
		echo "차이가 크지 않습니다. 정말 되돌리시려면\n";
		echo "  ?stella_gap=undo-really\n";
		exit;
	}

	if ( 'undo-really' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "자\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

	echo "═══ 맺음말 · 세 가지 ═══\n";
	echo '책 쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "자\n\n";

	/* 판 GAP-7 · 이미 들어간 경우에도 preview 는 **지금 글을 찍어 줍니다.**
	   막아버리면 무엇이 들어 있는지 볼 길이 없어서요. apply 만 막습니다. */
	$already = ( false !== strpos( $c, '맺음말 세 가지를 다시 씁니다' ) );
	if ( $already ) {
		echo "★ 이미 어떤 판이 들어가 있습니다.\n";
		if ( 'apply' === $mode ) {
			/* 판 GAP-7 · 덮어써도 됩니다.
			   이 조각은 앞뒤 표시 사이 **그 구간만** 바꿉니다. 다른 조각이
			   넣은 글은 건드리지 않습니다. 그리고 덮어쓸 때 백업을 **지금
			   책으로 갱신**합니다 — 옛 백업(오늘 작업 전 책)을 그대로 두면
			   되돌리기가 계속 위험하니까요. */
			echo "  그 구간만 새 판으로 덮어씁니다. 다른 글은 안 건드립니다.\n";
			echo "  되돌리기용 백업을 **지금 책으로 갱신**합니다.\n\n";
		} else {
			echo "  지금 그 자리에 무엇이 있는지 아래에 찍어 드립니다.\n\n";
		}
		$KEEPHAS = ( false !== strpos( $c, 'var KEEPTOPIC' ) );
		echo '  KEEPTOPIC 표 : ' . ( $KEEPHAS ? '있음' : '없음 — 주제별로 안 갈립니다' ) . "\n";
		$KTHAS = ( false !== strpos( $c, 'KEEPTOPIC[topic' ) );
		echo '  그 표를 읽는 줄 : ' . ( $KTHAS ? '있음' : '없음' ) . "\n\n";
	}
	if ( false !== strpos( $c, '어긋난 자리를 제대로 씁니다' ) ) {
		echo "먼저 넣으신 GAP-1 이 들어가 있습니다.\n";
		echo "?stella_gap=undo 로 걷어낸 다음 다시 여세요.\n";
		exit;
	}

	/* ── 앞뒤 표시로 자리를 찾습니다 ─────────────────
	   글자 하나하나를 맞추지 않습니다. 가운데 글은 손질될 수 있고,
	   실제로 손질되어 있었습니다. 이름은 잘 안 바뀝니다. */
	$A = "var keep='<div class=\"keep\">';";
	$B = "keep+='</div>';";

	$na = substr_count( $c, $A );
	if ( 1 !== $na ) {
		echo '★ 앞 표시가 ' . $na . "군데입니다. 한 군데여야 합니다.\n";
		echo "  찾는 것 : " . $A . "\n";
		exit;
	}

	$p1 = strpos( $c, $A );
	$p2 = strpos( $c, $B, $p1 + strlen( $A ) );
	if ( false === $p2 ) {
		echo "★ 뒤 표시를 못 찾았습니다.\n";
		echo "  찾는 것 : " . $B . "\n";
		exit;
	}
	$p2 += strlen( $B );

	$find = substr( $c, $p1, $p2 - $p1 );
	$to   = stella_gap_new();

	echo '바꿀 자리를 찾았습니다 — ' . number_format( strlen( $find ) ) . "자.\n\n";

	/* ★ 지금 그 자리에 있는 글을 그대로 보여드립니다 */
	if ( 'apply' !== $mode ) {
		echo "───── 지금 그 자리에 있는 글 ─────────────────────\n";
		echo $find . "\n";
		echo "───── 여기까지 ───────────────────────────────────\n\n";
	}

	$made = str_replace( $find, $to, $c );
	echo number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $made ) ) . '자 (+'
	   . number_format( strlen( $made ) - strlen( $c ) ) . ")\n\n";

	if ( 'apply' !== $mode ) {
		echo "아직 안 바꿨습니다. 정말 넣으시려면 ?stella_gap=apply\n";
		exit;
	}

	/* 판 GAP-7 · 백업을 늘 지금 책으로 갱신합니다.
	   옛 백업을 그대로 두면 되돌리기가 오늘 작업까지 날립니다. */
	update_post_meta( $id, $META, $c );
	$wpdb->update( $wpdb->posts, array( 'post_content' => $made ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "넣었습니다. 책을 열어 맺음말을 보세요.\n";
	echo "되돌리려면 ?stella_gap=undo\n";
	exit;
}, 1 );

function stella_gap_new() {
	return <<<'GAPNEW'
  /* ══ 2026-09-12 · 맺음말 세 가지를 다시 씁니다 ═══════════════
     소희 님이 손으로 고쳐 오신 원고를 그대로 옮깁니다.
     바뀌는 것은 대괄호 자리(계산값)뿐이고, 문장은 소희 님 것입니다.

         첫째  내 방식     ← 가장 크게 기운 저울
         둘째  관계·어긋남 ← 사주와 답이 갈린 자리
         셋째  지금의 시간 ← 지금 지나는 대운

     ★ 갈래를 반드시 둘씩 둡니다. 어긋난 자리가 없는 분과 있는 분,
       기운 분과 고른 분, 대운 안에 계신 분과 아직 아닌 분.
       한쪽만 쓰면 다른 쪽 손님에게 거짓말이 됩니다. */

  var keep='<div class="keep">';
  var leaned=false;
  if(leanAxis){ if(leanAxis.lean){ leaned=true; } }

  /* ── 첫째 · 내 방식 ─────────────────────────────────── */
