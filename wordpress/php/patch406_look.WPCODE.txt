add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_look406'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_look406'] ) );
	$META = '_stella_look406_bak';

	$page = get_page_by_path( 'star-saju' );

		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	if ( false !== strpos( $c, 'srMoreFill' ) ) {
		echo "이미 붙어 있습니다.\n먼저 ?stella_look406=undo 로 되돌리세요.";
		exit;
	}
	if ( false === strpos( $c, 'srMeBox' ) ) {
		echo "먼저 patch406_me 를 붙이셔야 합니다 (?stella_me406=apply).";
		exit;
	}

	/* ── 붙일 자리 ────────────────────────────────── */
	$Y = "'<div><div class=\"han\">'+p.y+'</div></div>'+";
	$M = "'<div><div class=\"han\">'+p.m+'</div></div>'+";

	$FN  = '  function show(p, known){';
	$HOOK = "$('svX').addEventListener('click', shut);";

	$anchors = array(
		'연주' => $Y, '월주' => $M, '일주' => $D2,
		'오행 막대' => $BAR, '별자리 앞' => $SEA,
		'함수 앞' => $FN, '단추 매는 곳' => $HOOK,
	);
	$bad = false;
	foreach ( $anchors as $nm => $a ) {
		$n = substr_count( $c, $a );
		echo sprintf( "%-12s %d군데\n", $nm, $n );
		if ( 1 !== $n ) { $bad = true; }
	}
	if ( $bad ) {
		echo "\n한 군데가 아니어서 아무것도 안 바꿨습니다.";
		exit;
	}

	$js = <<<'STELLA_LOOK_JS'

  /* ══ 여덟 글자를 색으로 · 태어난 계절 · 신강 신약 ─ 2026-09-09 ══ */

  var SR_EL5 = {

  function srHan(two){
    var i, ch, e, out = '<div class="han">';

      ch = two.charAt(i);
      e  = SR_EL5[ch] ? SR_EL5[ch] : '';
      out += '<span data-el="' + e + '">' +
             (e ? '<i>' + e + '</i>' : '') + ch +
             (SR_SOUND[ch] ? '<b>' + SR_SOUND[ch] + '</b>' : '') +
             '</span>';
    }
    return out + '</div>';
  }

  /* 태어난 계절 · 자기 힘 */
  function srMore(p){
    var out = '';
    var myEl = SR_GAN[p.g] ? SR_GAN[p.g].el : '';
    if(!myEl){ return ''; }

    /* ── 태어난 계절 ── */
    var mj = p.m.charAt(1);
    var se = SR_SEASON[mj];
    if(se){
      var sEl = SR_SEASON_EL[se];
      var rel;
      if(sEl === myEl){
        rel = '<b>제 계절에 태어났습니다.</b> 힘이 넉넉합니다. ' +
              '하고 싶은 대로 밀고 나갈 바탕이 있습니다.';
      } else if(SR_GEN[sEl] === myEl){
        rel = '<b>나를 밀어 주는 계절입니다.</b> 시작할 때 손이 잘 붙습니다. ' +
              '도와주는 사람이 잘 나타납니다.';
      } else if(SR_CTL[sEl] === myEl){
        rel = '<b>나를 누르는 계절입니다.</b> 어릴 때부터 참는 법을 먼저 배웁니다. ' +
              '그래서 일찍 단단해집니다.';
      } else if(SR_GEN[myEl] === sEl){
        rel = '<b>내 것을 내주는 계절입니다.</b> 주변에 쏟는 것이 많습니다. ' +
              '쉬는 자리를 스스로 만들어야 합니다.';
      } else {
        rel = '<b>내가 다스리는 계절입니다.</b> 할 일이 눈에 먼저 보입니다. ' +
              '대신 늘 바쁩니다.';
      }
      out += '<h3>태어난 계절 — ' + se + '</h3>' +
             '<p>' + SR_SEASON_LINE[se] + ' ' +
             ga(SR_GAN[p.g].ko) + ' ' + se + '을 만났습니다. ' + rel + '</p>';
    }

    /* ── 자기 힘이 센가 ── */
    var mine = 0, other = 0, e;
    for(e in p.el){
      if(!p.el.hasOwnProperty(e)){ continue; }
      if(e === myEl){ mine += p.el[e]; continue; }
      if(e === SR_BORN[myEl]){ mine += p.el[e]; continue; }
      other += p.el[e];
    }
    var head, body;
    if(mine > other){
      head = '자기 힘이 센 쪽입니다';
      body = '남의 말보다 자기 판단을 믿습니다. 밀어붙이는 힘이 있어서 ' +
             '혼자서도 일이 됩니다. 대신 꺾이는 법을 늦게 배웁니다. ' +
             '<b>내보내는 쪽에서 풀립니다</b> — 만들어 내놓는 일, 돈 쓰는 일, ' +
             '책임을 맡는 일. 더 받아들이려 하면 오히려 막힙니다.';

      head = '기대는 힘이 있는 쪽입니다';
      body = '혼자보다 사람 속에서 잘 됩니다. 눈치가 빠르고 잘 맞춰 줍니다. ' +

             '같이 하는 일. 혼자 다 지려 하면 무너집니다.';
    } else {
      head = '한쪽으로 안 기운 쪽입니다';
      body = '흔치 않습니다. 밀 때 밀고 물러설 때 물러섭니다. ' +
             '대신 뚜렷한 색이 없어 보일 때가 있습니다. ' +
             '<b>어느 쪽으로도 갈 수 있어서</b> 무엇을 고르느냐가 더 중요해집니다.';
    }
    out += '<h3>자기 힘 — ' + head + '</h3>' +
           '<div class="sr-scale">' +
             '<span>내 편 <b>' + mine + '자</b> (' + myEl + '·' + SR_BORN[myEl] + ')</span>' +
             '<span>나머지 <b>' + other + '자</b></span>' +
           '</div>' +
           '<p>' + body + '</p>';

    return out;
  }
