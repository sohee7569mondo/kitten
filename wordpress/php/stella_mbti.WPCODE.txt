  var LOGKEY = 'stella_lab_log';
  function today(){
    var d = new Date();
    function p(n){ return (n < 10 ? '0' : '') + n; }
    return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate());
  }
  function logRead(){
    try{
      var r = localStorage.getItem(LOGKEY);
      if(!r){ return []; }
      var a = JSON.parse(r);
      return (a instanceof Array) ? a : [];
    }catch(e){ return []; }
  }
  function logWrite(a){
    try{ localStorage.setItem(LOGKEY, JSON.stringify(a.slice(-60))); }catch(e){}
  }
  function logLast(t){
    var a = logRead(), i;
    for(i=a.length-1;i>=0;i--){ if(a[i].t === t){ return a[i]; } }
    return null;
  }
  function logAdd(t, r){
    var a = logRead(), last = null, i;
    for(i=a.length-1;i>=0;i--){ if(a[i].t === t){ last = a[i]; break; } }
    if(last){ if(last.r === r){
      last.l = today();
      last.n = (last.n || 1) + 1;
      logWrite(a);
      return null;
    } }
    a.push({ t:t, r:r, d:today(), l:today(), n:1 });
    logWrite(a);
    return last;
  }
