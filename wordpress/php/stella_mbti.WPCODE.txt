  function pickWin(){
    /* 고를 기회가 많은 쪽이 유리해지지 않게 기회 수로 나눕니다 */
    var chance = {}, i, best = '', bv = -1, k;
    for(i=0;i<T.q.length;i++){
      chance[T.q[i][3]] = (chance[T.q[i][3]] || 0) + 1;
      chance[T.q[i][4]] = (chance[T.q[i][4]] || 0) + 1;
    }
    for(k in T.out){
      if(!Object.prototype.hasOwnProperty.call(T.out, k)){ continue; }
      var v = (pick[k] || 0) / (chance[k] || 1);
      if(v > bv){ bv = v; best = k; }
    }
    return best;
  }
