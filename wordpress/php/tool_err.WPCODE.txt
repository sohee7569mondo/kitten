     WPCode → 새 스니펫 → PHP Snippet → 「어디서나 실행」 → 저장 → 활성

   ★ 위치(Location)는 「어디서나 실행 / Run Everywhere」

add_action( 'wp_head', function () {

	if ( ! current_user_can( 'manage_options' ) ) { return; }
	if ( ! is_page( 'reading-book' ) ) { return; }

	?>
<script>
(function(){
  var LOG = [];
  function paint(){
    var d = document.getElementById('stella-err');
    if(!d){
      if(!document.body){ return; }
      d = document.createElement('div');
      d.id = 'stella-err';
      d.style.cssText = 'position:fixed;left:0;right:0;top:0;z-index:2147483647;'
        + 'background:#b3261e;color:#fff;padding:14px 18px;font:13px/1.7 system-ui;'
        + 'white-space:pre-wrap;word-break:break-all;max-height:60vh;overflow:auto;'
        + 'box-shadow:0 2px 16px rgba(0,0,0,.45)';
      document.body.appendChild(d);
    }
    d.textContent = '판 ERR-2 (2026-09-11)\n\n' + LOG.join('\n\n');
  }
  function box(msg){ LOG.push(msg); paint(); }

  window.addEventListener('error', function(e){
    var m = '【책이 멈춘 자리】\n' + (e.message || '(까닭 모름)');
    if(e.filename){ m += '\n파일 : ' + e.filename; }
    m += '\n줄 ' + (e.lineno || '?') + ' · 칸 ' + (e.colno || '?');
    if(e.error && e.error.stack){
      m += '\n\n' + String(e.error.stack).split('\n').slice(0,5).join('\n');
    }
    box(m);
  }, true);

  window.addEventListener('unhandledrejection', function(e){
    box('【기다리다 멈춘 자리】\n' + String(e.reason));
  });

  function check(){
    if(LOG.length){ return; }
    var b = document.getElementById('ssb');
    if(!b){
      box('【알림】 책을 담는 자리(#ssb)를 못 찾았습니다.\n'
        + '몸통에 있는 상자 이름이 바뀌었거나, 책 코드가 아예 안 실려 있습니다.');
      return;
    }
    var h = b.querySelectorAll('h2').length;
    var p = b.querySelectorAll('.page, section, article').length;
    if(h === 0){
      box('【알림】 오류는 안 났는데 장이 하나도 안 그려졌습니다.\n'
        + 'h2 ' + h + '개 · 칸 ' + p + '개\n'
        + '#ssb 안 글자수 : ' + (b.textContent || '').length + '\n\n'
        + '앞쪽 300글자 :\n' + (b.innerHTML || '').slice(0, 300));
    }
  }
  if(document.readyState === 'complete'){ setTimeout(check, 3000); }
  else { window.addEventListener('load', function(){ setTimeout(check, 3000); }); }
})();
</script>
	<?php
}, 1 );
