/* ════════════════════════════════════════════════════════════
   한 줄 질문 답변을 네 문단으로

   손님이 적어주신 한 줄 질문에 지금은 한 문단으로 답합니다.
   소희 님이 새로 쓰신 글로 네 문단이 됩니다.

   ① 물으신 것 — [질문] — 에 대해서는, 앞의 1부와 2부에서 이미
      답을 위한 재료를 충분히 살펴보았습니다. 종합해 보면 사주는
      [가장 크게 기운 쪽] 쪽에 가깝습니다.
   ② 사주가 「이렇게 하세요」라고 대신 정해주지는 않습니다. 다만
      어떤 선택이 타고난 기질과 지금의 흐름에 더 자연스러운지는
      살펴볼 수 있습니다.
   ③ (기운 쪽에 따라 달라지는 조언 — 원래 있던 글 그대로)
   ④ 그러니 이 일을 두고 고민하고 계시다면, 「해도 될까」를 묻기보다
      「어떤 방식으로 할까」를 생각해 보셨으면 합니다.

   ④번이 이 고침의 핵심입니다. 손님은 대개 「해도 될까요」를 적어
   보내십니다. 사주는 그 물음에 답할 수 없고, 답하는 척하면 거짓이
   됩니다. 그래서 답 대신 물음을 바꿔 드립니다 — 사주가 실제로
   답할 수 있는 물음으로요.

   ②번도 같은 뜻입니다. 대신 정해주지 않는다고 먼저 밝히고 나서
   조언으로 갑니다. 밝히지 않고 조언만 하면 점쟁이 말이 됩니다.

   여섯 문 모두 같은 자리를 씁니다. 기운 저울이 없는 분께는 원래처럼
   「어느 쪽으로 정하셔도 크게 무리가 없습니다」가 ③에 들어갑니다.

   ★ patch160_paper · today · box · voice 가 먼저 들어가 있어야 합니다.

   페이지 내용을 wp_update_post 로 저장하면 워드프레스가 <script> 를
   지워버립니다. 그래서 데이터베이스에 바로 씁니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로 바꾸세요.

   ① 먼저 보기   https://stellasaju.com/?stella_ask=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_ask=apply
   ③ 되돌리기    https://stellasaju.com/?stella_ask=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_ask'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$PAGE = 160;
	$META = '_stella_ask_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_ask'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $PAGE ) );
	if ( ! $row ) { echo '<h1>160쪽을 못 찾았습니다.</h1>'; exit; }

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $PAGE, $META, true );
		echo '<h1>되돌리기</h1>';
		if ( ! $old ) { echo '<p>되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $PAGE ) );
		clean_post_cache( $PAGE );
		delete_post_meta( $PAGE, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		echo '<p><a href="/reading-book/?demo=door-love-2">책 보기</a></p>';
		exit;
	}

	$c = $row->post_content;
	$before = strlen( $c );

	$jobs = array();

	$jobs[] = array(
		'name' => '① 한 줄 질문 답변을 네 문단으로',
		'find' => '/\\(order\\.oneline\\s+\\?\\s+\'<hr><p>물으신\\s+것\\s+—\\s+<em>\'\\+esc\\(order\\.oneline\\)\\+\'<\/em>\\s+—\\s+에\\s+대해서는,\\s+\'\\+\\s+\'앞의\\s+1부와\\s+2부가\\s+이미\\s+답의\\s+재료를\\s+다\\s+꺼내\\s+놓았습니다\\.\\s+\'\\+\\s+\'가장\\s+크게\\s+기운\\s+저울이\\s+<em>\'\\+esc\\(topLean\\(\\)\\)\\+\'<\/em>\\s+쪽이니,\\s+\'\\+\\s+leanAdvice\\(\\)\\s+:\\s+\'\'\\),\\s+null,\\s+\'two\'\\);/u',
		'to'   => "      (order.oneline\n        /* 2026-09-07 · 소희 님이 새로 쓰신 네 문단.\n           한 문단으로 끝내던 자리를 넷으로 폅니다.\n           차례 — 재료는 이미 다 봤다 → 기운 쪽은 이렇다 →\n           그래도 사주가 대신 정해주지는 않는다 → 그러니 물음을 바꾸자.\n           마지막이 이 고침의 핵심입니다. 손님은 대개 「해도 될까요」를\n           적어 보내시는데, 사주는 그 물음에 답할 수 없습니다.\n           답하는 척하는 대신 답할 수 있는 물음으로 옮겨 드립니다. */\n        ? '<hr><p>물으신 것 — <em>'+esc(order.oneline)+'</em> — 에 대해서는, '+\n          '앞의 1부와 2부에서 이미 답을 위한 재료를 충분히 살펴보았습니다.</p>'+\n          '<p>그 내용을 종합해 보면, 가장 크게 기운 저울이 '+\n          '<em>'+esc(topLean())+'</em> 쪽입니다. '+ leanAdvice()+\n          '<p>물론 사주가 「이렇게 하세요」라고 대신 정해주지는 않습니다. '+\n          '다만 어떤 선택이 타고난 기질과 지금의 흐름에 더 자연스러운지는 '+\n          '살펴볼 수 있습니다.</p>'+\n          '<p>그러니 이 일을 두고 고민하고 계시다면, <em>「해도 될까」</em>를 묻기보다 '+\n          '<em>「어떤 방식으로 할까」</em>를 생각해 보셨으면 합니다. '+\n          '그 질문이 당신의 사주에는 더 잘 맞습니다.</p>'\n        : ''), null, 'two');\n",
	);

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$ok = true;
	foreach ( $jobs as $j ) {
		$n = preg_match_all( $j['find'], $c );
		echo '<h2>' . esc_html( $j['name'] ) . '</h2>';
		if ( 1 === $n ) {
			echo '<p class="ok">찾았습니다 — 한 군데.</p>';
		} elseif ( 0 === $n ) {
			echo '<p class="no">못 찾았습니다. 이미 바꿨거나 코드가 달라졌습니다.</p>';
			$ok = false;
		} else {
			echo '<p class="no">' . intval( $n ) . '군데가 걸립니다. 한 군데여야 안전합니다.</p>';
			$ok = false;
		}
	}

	if ( ! $ok ) {
		echo '<p class="no">한 자리가 한 군데씩 걸릴 때만 바꿉니다. 아무것도 건드리지 않았습니다.</p>';
		exit;
	}

	$new = $c;
	foreach ( $jobs as $j ) {
		$new = preg_replace( $j['find'], str_replace( '\\', '\\\\', $j['to'] ), $new, 1 );
		if ( null === $new ) { echo '<p class="no">바꾸다가 멈췄습니다. 아무것도 저장하지 않았습니다.</p>'; exit; }
	}

	echo '<p>' . number_format( $before ) . ' → ' . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_ask=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $PAGE, $META, true ) ) {
		update_post_meta( $PAGE, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $PAGE ) );
	clean_post_cache( $PAGE );

	echo '<p class="ok">바꿨습니다.</p>';
	echo '<p>되돌리려면 <code>/?stella_ask=undo</code></p>';
	echo '<p><a href="/reading-book/?demo=door-love-2">책 보기</a></p>';
	exit;
} );
