/* ── 카드 파일과 그 CDN 주소 ───────────────────────────
   카카오가 우리 서버에서 그림을 못 가져오는 듯하여, 카드를 진짜 파일로
   남기고 워드프레스닷컴 그림 CDN(i0.wp.com) 주소를 알려줍니다.
   이 CDN 은 소희 님 사이트 그림을 이미 나르고 있는 곳입니다.

   담기는 것은 점수뿐입니다. 이름도 생년월일도 들어가지 않습니다. */
if ( ! function_exists( 'stella_um_card_name' ) ) {
	function stella_um_card_name( $score, $rel, $a, $b, $c, $n = '' ) {
		$base = intval( $score ) . '-' . preg_replace( '/[^a-z_]/', '', $rel ) . '-' .
		        intval( $a ) . '-' . intval( $b ) . '-' . intval( $c );
		if ( $n ) { $base .= '-' . preg_replace( '/[^A-Za-z0-9_-]/', '', $n ); }
		return $base . '.png';
	}
}
if ( ! function_exists( 'stella_um_card_url' ) ) {
	function stella_um_card_url( $score, $rel, $a, $b, $c ) {
		$up   = wp_upload_dir();
		$name = stella_um_card_name( $score, $rel, $a, $b, $c );
		$url  = $up['baseurl'] . '/uandme/card/' . $name;
		/* CDN 을 거치면 카카오가 잘 가져갑니다 */
		$bare = preg_replace( '#^https?://#', '', $url );
		return 'https://i0.wp.com/' . $bare . '?ssl=1';
	}
}

/* ── 점수로 등급 이름 알아내기 ─────────────────────────
   주소에 한글을 싣지 않으려고 서버도 같은 표를 갖습니다.
   유앤미 화면(uandme-tiers.js)과 같은 구간입니다. */
if ( ! function_exists( 'stella_um_tier' ) ) {
	function stella_um_tier( $n ) {
		if ( $n >= 95 ) { return '천생연분'; }
		if ( $n >= 90 ) { return '완전 찰떡궁합'; }
		if ( $n >= 80 ) { return '환상의 케미'; }
		if ( $n >= 70 ) { return '좋은 궁합'; }
		if ( $n >= 60 ) { return '무난한 사이'; }
		if ( $n >= 50 ) { return '노력형 궁합'; }
		if ( $n >= 40 ) { return '위태위태'; }
		if ( $n >= 30 ) { return '친구가 나을지도'; }
		if ( $n >= 20 ) { return '재고요망'; }
		return '이번 생은 예능감';
	}
}

/* ── 글꼴 한 벌 마련하기 ────────────────────────────────
   서버에 한글 글꼴이 없어서 한 번만 받아 uploads 에 둡니다.
   받는 곳을 여러 군데 두고 되는 데까지 가 봅니다. */
if ( ! function_exists( 'stella_um_font_path' ) ) {
	function stella_um_font_path() {
		$up = wp_upload_dir();
		return $up['basedir'] . '/uandme/jua.ttf';
	}
}
if ( ! function_exists( 'stella_um_font_try' ) ) {
	/* 한 곳에서 받아 봅니다. 무슨 일이 있었는지 글로 돌려줍니다. */
	function stella_um_font_try( $url, &$log ) {

		$r = wp_remote_get( $url, array(
			'timeout'    => 30,
			'user-agent' => 'Mozilla/4.0 (compatible; MSIE 6.0; Windows NT 5.1)',
		) );
		if ( is_wp_error( $r ) ) {
			$log[] = array( $url, '못 감', $r->get_error_message() );
			return '';
		}
		$code = wp_remote_retrieve_response_code( $r );
		$body = wp_remote_retrieve_body( $r );
		$log[] = array( $url, '응답 ' . $code, strlen( $body ) . ' 바이트' );
		if ( 200 !== intval( $code ) ) { return ''; }

		/* 글꼴 파일이면 앞 네 글자가 0x00010000 이거나 true/OTTO 입니다 */
		$head = substr( $body, 0, 4 );
		if ( strlen( $body ) > 40000 ) {
			if ( "\x00\x01\x00\x00" === $head || 'true' === $head || 'ttcf' === $head ) {
				return $body;
			}
		}
		/* 글꼴이 아니면 안에서 ttf 주소를 찾아 다시 가 봅니다 */
		if ( preg_match( '#https://[^)\'"\s]+\.ttf#', $body, $m ) ) {
			$log[] = array( $m[0], '안에서 찾음', '따라갑니다' );
			return stella_um_font_try( $m[0], $log );
		}
		$log[] = array( $url, '글꼴 아님', substr( strip_tags( $body ), 0, 120 ) );
		return '';
	}
}
if ( ! function_exists( 'stella_um_font' ) ) {
	function stella_um_font( &$log = null ) {

		if ( null === $log ) { $log = array(); }
		$ttf = stella_um_font_path();

		if ( file_exists( $ttf ) ) {
			if ( filesize( $ttf ) > 40000 ) {
				$log[] = array( $ttf, '이미 있음', filesize( $ttf ) . ' 바이트' );
				return $ttf;
			}
		}
		$dir = dirname( $ttf );
		if ( ! is_dir( $dir ) ) { wp_mkdir_p( $dir ); }

		$where = array(
			'https://cdn.jsdelivr.net/gh/google/fonts@main/ofl/jua/Jua-Regular.ttf',
			'https://raw.githubusercontent.com/google/fonts/main/ofl/jua/Jua-Regular.ttf',
			'https://fonts.googleapis.com/css?family=Jua',
			'https://cdn.jsdelivr.net/gh/google/fonts@main/ofl/notosanskr/NotoSansKR%5Bwght%5D.ttf',
			'https://cdn.jsdelivr.net/gh/google/fonts@main/ofl/gothica1/GothicA1-Bold.ttf',
		);
		foreach ( $where as $u ) {
			$raw = stella_um_font_try( $u, $log );
			if ( ! $raw ) { continue; }
			$ok = @file_put_contents( $ttf, $raw );
			$log[] = array( $ttf, $ok ? '저장함' : '저장 못 함', strlen( $raw ) . ' 바이트' );
			if ( $ok ) { return $ttf; }
		}
		return '';
	}
}

   https://stellasaju.com/?um_font=1                 (관리자만)
   https://stellasaju.com/?um_font=1&again=1         다시 받기 */

	if ( ! isset( $_GET['um_font'] ) ) { return; }

	if ( isset( $_GET['again'] ) ) { @unlink( stella_um_font_path() ); }

	$log  = array();
	$font = stella_um_font( $log );

	echo '<table style="width:100%;border-collapse:collapse;font-size:12.5px">';
	foreach ( $log as $row ) {
		echo '<tr style="border-bottom:1px solid #333">';
		echo '<td style="padding:7px;color:#8cf;word-break:break-all">' . esc_html( $row[0] ) . '</td>';

		echo '<p>' . esc_html( $font ) . '</p>';
		echo '<p>아래 그림에 한글이 제대로 보이면 다 된 것입니다.</p>';

	exit;
} );

/* ── 그림 그리기 ──────────────────────────────────────── */
add_action( 'init', function () {

	/* 두 가지 주소를 다 받습니다.
	     예쁜 주소 : /um-card/63-lover-30-12-24
	     예전 주소 : /?um_card=1&s=63&r=lover&a=30&b=12&c=24
	   .png 를 붙이면 웹서버가 그림 파일로 여겨 가로채 404 를 냅니다. */
	$want = isset( $_GET['um_card'] );
	if ( ! $want ) {
		$uri = isset( $_SERVER['REQUEST_URI'] ) ? $_SERVER['REQUEST_URI'] : '';
		if ( preg_match( '#/um-card/(\d{1,2})-([a-z_]{2,14})-(\d{1,2})-(\d{1,2})-(\d{1,2})(?:-([A-Za-z0-9_-]{1,140}))?#', $uri, $mm ) ) {
			$_GET['s'] = $mm[1];
			$_GET['r'] = $mm[2];
			$_GET['a'] = $mm[3];
			$_GET['b'] = $mm[4];
			$_GET['c'] = $mm[5];
			if ( isset( $mm[6] ) ) { $_GET['n'] = $mm[6]; }
			$want = true;
		}
	}
	if ( ! $want ) { return; }

	$W = 800; $H = 400;

	$score = isset( $_GET['s'] ) ? max( 0, min( 99, intval( $_GET['s'] ) ) ) : 0;
	$a     = isset( $_GET['a'] ) ? max( 0, min( 40, intval( $_GET['a'] ) ) ) : 0;
	$b     = isset( $_GET['b'] ) ? max( 0, min( 30, intval( $_GET['b'] ) ) ) : 0;
	$c     = isset( $_GET['c'] ) ? max( 0, min( 30, intval( $_GET['c'] ) ) ) : 0;
	$rel   = isset( $_GET['r'] ) ? sanitize_key( $_GET['r'] ) : 'lover';
	$tier  = isset( $_GET['t'] ) ? wp_strip_all_tags( wp_unslash( $_GET['t'] ) ) : stella_um_tier( $score );
	if ( mb_strlen( $tier ) > 14 ) { $tier = mb_substr( $tier, 0, 14 ); }
	/* Jua 가 그릴 줄 아는 글자만 남깁니다 — 한글·영문·숫자·기본 부호.
	   그러지 않으면 네모난 두부가 나옵니다. */
	$keep = '';
	$len  = mb_strlen( $tier );
	for ( $i = 0; $i < $len; $i++ ) {
		$ch = mb_substr( $tier, $i, 1 );
		$cp = mb_ord( $ch, 'UTF-8' );
		if ( false === $cp ) { continue; }
		if ( $cp >= 0x20 ) { if ( $cp <= 0x7E ) { $keep .= $ch; continue; } }
		if ( $cp >= 0xAC00 ) { if ( $cp <= 0xD7A3 ) { $keep .= $ch; continue; } }
		if ( $cp >= 0x3131 ) { if ( $cp <= 0x318E ) { $keep .= $ch; continue; } }
		$keep .= ' ';
	}
	$tier = trim( $keep );

	/* 두 사람 이름 — 주소에 담을 때 뒤죽박죽 글자로 싸 두었습니다.
	   「지은|민호」를 base64 로 싸서 실어 보냅니다. */
	$who = '';
	if ( isset( $_GET['n'] ) ) {
		$raw = strtr( sanitize_text_field( wp_unslash( $_GET['n'] ) ), '-_', '+/' );
		$pad = strlen( $raw ) % 4;
		if ( $pad ) { $raw .= str_repeat( '=', 4 - $pad ); }
		$dec = base64_decode( $raw, true );
		if ( false !== $dec ) {
			$dec = wp_strip_all_tags( $dec );
			if ( mb_check_encoding( $dec, 'UTF-8' ) ) {
				$two = explode( '|', $dec );
				$one = isset( $two[0] ) ? mb_substr( trim( $two[0] ), 0, 8 ) : '';
				$to  = isset( $two[1] ) ? mb_substr( trim( $two[1] ), 0, 8 ) : '';
				if ( $one ) { if ( $to ) { $who = $one . ' X ' . $to; } }
			}
		}
	}

	$relko = array(
		'lover' => '연인', 'married' => '부부', 'friend' => '친구',
		'sibling' => '형제자매', 'parent_child' => '부모와 자녀',
		'boss_sub' => '상사와 부하', 'partner' => '동업자',
	);
	$ko = isset( $relko[ $rel ] ) ? $relko[ $rel ] : '연인';

	/* 관계마다 다른 아이가 나옵니다 */
	$pics = array(
		'lover'        => 'animal_2785140635-01-300px.png',
		'married'      => 'animal_2785140635-02-300px.png',
		'friend'       => 'animal_2785140635-03-300px.png',
		'sibling'      => 'animal_2785140635-04-300px.png',
		'parent_child' => 'animal_2785140635-05-300px.png',
		'boss_sub'     => 'animal_2785140635-06-300px.png',
		'partner'      => 'animal_2785140635-07-300px.png',
	);
	$left = isset( $pics[ $rel ] ) ? $pics[ $rel ] : $pics['lover'];

	/* 오른쪽은 그때그때 다른 아이가 나옵니다 — 스물아홉 중 하나 */
	$pool = array( 'uandme_cha-02-300px.png' );
	$i = 1;
	while ( $i <= 6 )  { $pool[] = sprintf( 'animal-%02d-300px.png', $i ); $i++; }
	$i = 1;
	while ( $i <= 11 ) { $pool[] = sprintf( 'animal_2785140636-%02d-300px.png', $i ); $i++; }
	$i = 1;
	while ( $i <= 12 ) { $pool[] = sprintf( 'animal_2785140635-%02d-300px.png', $i ); $i++; }
	$right = $pool[ wp_rand( 0, count( $pool ) - 1 ) ];

	$im = imagecreatetruecolor( $W, $H );
	imagealphablending( $im, true );
	imagesavealpha( $im, true );

	/* 파스텔 바탕 — 왼쪽 위에서 오른쪽 아래로 */
	for ( $y = 0; $y < $H; $y++ ) {
		$t = $y / ( $H - 1 );
		$r1 = 251 + ( 255 - 251 ) * $t;
		$g1 = 246 + ( 237 - 246 ) * $t;
		$b1 = 255 + ( 226 - 255 ) * $t;
		$col = imagecolorallocate( $im, intval( $r1 ), intval( $g1 ), intval( $b1 ) );
		imagefilledrectangle( $im, 0, $y, $W, $y, $col );
	}
	$ink  = imagecolorallocate( $im, 63, 58, 82 );
	$pur  = imagecolorallocate( $im, 107, 91, 168 );
	$soft = imagecolorallocate( $im, 167, 155, 196 );
	$sub  = imagecolorallocate( $im, 124, 115, 146 );

	/* 아이 둘 — 좌우 끝에 하나씩 */
	$up = wp_upload_dir();
	/* 키(높이)를 기준으로 맞춥니다. 상자에 맞추면 세로로 긴 아이가
	   작아 보입니다. 너비에는 울타리를 둬 옆으로 퍼지지 않게 합니다. */
	$paste = function ( $file, $cx, $cy, $tall, $wide ) use ( $im, $up ) {
		$p = $up['basedir'] . '/2026/09/' . $file;
		if ( ! file_exists( $p ) ) { $p = $up['basedir'] . '/2026/08/' . $file; }
		if ( ! file_exists( $p ) ) { return; }
		$src = @imagecreatefrompng( $p );
		if ( ! $src ) { return; }
		/* 둘레의 빈 곳을 떼어냅니다. 아이마다 여백이 달라서 그냥 맞추면
		   어떤 아이는 작아 보입니다. */
		if ( function_exists( 'imagecropauto' ) ) {
			$cut = @imagecropauto( $src, IMG_CROP_SIDES );
			if ( false !== $cut ) { imagedestroy( $src ); $src = $cut; }
		}
		$sw = imagesx( $src ); $sh = imagesy( $src );
		if ( ! $sw ) { imagedestroy( $src ); return; }
		$k  = min( $tall / $sh, $wide / $sw );
		$dw = intval( $sw * $k ); $dh = intval( $sh * $k );
		imagecopyresampled( $im, $src, intval( $cx - $dw / 2 ), intval( $cy - $dh / 2 ),
			0, 0, $dw, $dh, $sw, $sh );
		imagedestroy( $src );
	};
	$paste( $left,  104, 212, 224, 196 );
	$paste( $right, 696, 212, 224, 196 );

	$font = stella_um_font();

	if ( $font ) {
		/* 가운데 정렬 · 자리가 좁으면 글씨를 줄여 넣습니다 */
		$mid = function ( $size, $y, $text, $col, $max ) use ( $im, $font ) {
			if ( '' === $text ) { return; }
			$w = 0;
			while ( $size > 8 ) {
				$bx = imagettfbbox( $size, 0, $font, $text );
				$w  = $bx[2] - $bx[0];
				if ( $w <= $max ) { break; }
				$size = $size - 1;
			}
			imagettftext( $im, $size, 0, intval( ( 800 - $w ) / 2 ), $y, $col, $font, $text );
		};
		/* 카톡에서는 카드가 작게 보이므로 글씨를 큼직하게 씁니다 */
		/* 좌우 아이가 x 6~202, 598~794 를 차지하므로 글씨는 390 안쪽으로 */
		$mid( 21,  58, '유앤미 - ' . $ko . ' 궁합', $soft, 380 );
		if ( $who ) { $mid( 34, 104, $who, $ink, 380 ); }
		$mid( 116, 230, (string) $score,          $pur,  380 );
		$mid( 21,  264, '점',                      $soft, 380 );
		if ( $tier ) { $mid( 46, 320, $tier, $ink, 390 ); }
		$mid( 21, 360, '사주 ' . $a . ' / 별자리 ' . $b . ' / 띠 ' . $c, $sub, 390 );
		$mid( 16, 388, 'stellasaju.com/uandme',   $soft, 390 );
	} else {
		/* 글꼴을 못 받았을 때 — 숫자만이라도 큼직하게.
		   작게 그린 뒤 늘려 붙입니다. */
		$s  = (string) $score;
		$tw = imagefontwidth( 5 ) * strlen( $s );
		$th = imagefontheight( 5 );
		$tmp = imagecreatetruecolor( $tw, $th );
		$bg  = imagecolorallocate( $tmp, 251, 246, 255 );
		imagefill( $tmp, 0, 0, $bg );
		imagecolortransparent( $tmp, $bg );
		imagestring( $tmp, 5, 0, 0, $s, imagecolorallocate( $tmp, 107, 91, 168 ) );
		$k = 9;
		imagecopyresampled( $im, $tmp, intval( 400 - $tw * $k / 2 ), intval( 210 - $th * $k / 2 ),
			0, 0, intval( $tw * $k ), intval( $th * $k ), $tw, $th );
		imagedestroy( $tmp );
		imagestring( $im, 3, 330, 300, 'stellasaju.com/uandme', $soft );
	}

	/* 크롤러가 그림을 가져갈 수 있게 아주 평범한 그림처럼 내보냅니다.
	   nocache_headers() 는 Expires 를 1984년으로 박아 두는데, 그러면
	   카카오처럼 그림을 자기 쪽에 담아두는 서비스가 담기를 포기합니다.
	   X-Robots-Tag: noindex 도 「가져가지 말라」는 뜻으로 읽힙니다.
	   둘 다 뺍니다. */
	ob_start();
	imagepng( $im );
	imagedestroy( $im );
	$png = ob_get_clean();

	/* 파일로도 남겨 둡니다. 다음부터는 CDN 이 이 파일을 나릅니다. */
	$up  = wp_upload_dir();
	$dir = $up['basedir'] . '/uandme/card';
	if ( ! is_dir( $dir ) ) { wp_mkdir_p( $dir ); }
	$file = $dir . '/' . stella_um_card_name( $score, $rel, $a, $b, $c,
		isset( $_GET['n'] ) ? $_GET['n'] : '' );
	if ( ! file_exists( $file ) ) { @file_put_contents( $file, $png ); }

	header( 'Content-Type: image/png' );
	header( 'Content-Length: ' . strlen( $png ) );
	header( 'Cache-Control: public, max-age=604800' );
	header( 'Accept-Ranges: none' );
	header_remove( 'Expires' );
	header_remove( 'Pragma' );
	header_remove( 'X-Robots-Tag' );
	echo $png;
	exit;
} );

/* ── 링크를 어디에 붙여도 우리 카드가 뜨게 ─────────────
   스레드·X·디스코드·페이스북은 페이지의 og:image 를 봅니다. 카톡도
   링크만 붙여넣으면 이걸 봅니다. 유앤미 결과 링크에 점수가 실려
   있으므로, 그 점수로 그린 카드를 og:image 로 내겁니다. */
add_action( 'wp_head', function () {

	if ( ! function_exists( 'stella_um_here' ) ) { return; }
	if ( ! stella_um_here() ) { return; }
	if ( ! isset( $_GET['sc'] ) ) { if ( ! isset( $_GET['s'] ) ) { return; } }

	$raw   = isset( $_GET['sc'] ) ? $_GET['sc'] : ( isset( $_GET['s'] ) ? $_GET['s'] : 0 );
	$score = max( 0, min( 99, intval( $raw ) ) );
	$a     = isset( $_GET['a'] ) ? max( 0, min( 40, intval( $_GET['a'] ) ) ) : 0;
	$b     = isset( $_GET['b'] ) ? max( 0, min( 30, intval( $_GET['b'] ) ) ) : 0;
	$c     = isset( $_GET['c'] ) ? max( 0, min( 30, intval( $_GET['c'] ) ) ) : 0;
	$rel   = isset( $_GET['rel'] ) ? sanitize_key( $_GET['rel'] ) : 'lover';
	$tier  = isset( $_GET['t'] ) ? wp_strip_all_tags( wp_unslash( $_GET['t'] ) ) : stella_um_tier( $score );
	if ( mb_strlen( $tier ) > 14 ) { $tier = mb_substr( $tier, 0, 14 ); }

	$img = home_url( '/um-card/' . $score . '-' . $rel . '-' . $a . '-' . $b . '-' . $c );

	$title = '우리 ' . $score . '점' . ( $tier ? ' - ' . $tier : '' );
	$desc  = '사주 ' . $a . '/40, 별자리 ' . $b . '/30, 띠 ' . $c . '/30. 생일만 넣으면 바로 나와요';

	$host = isset( $_SERVER['HTTP_HOST'] ) ? $_SERVER['HTTP_HOST'] : 'stellasaju.com';
	$uri  = isset( $_SERVER['REQUEST_URI'] ) ? $_SERVER['REQUEST_URI'] : '/uandme/';
	$here = ( is_ssl() ? 'https://' : 'http://' ) . $host . $uri;

	echo "\n<!-- 유앤미 -->\n";
	echo '<meta property="og:type" content="website">' . "\n";
	echo '<meta property="og:title" content="' . esc_attr( $title ) . '">' . "\n";
	echo '<meta property="og:description" content="' . esc_attr( $desc ) . '">' . "\n";
	echo '<meta property="og:image" content="' . esc_url( $img ) . '">' . "\n";
	echo '<meta property="og:image:width" content="800">' . "\n";
	echo '<meta property="og:image:height" content="400">' . "\n";
	echo '<meta property="og:url" content="' . esc_url( $here ) . '">' . "\n";
	echo '<meta name="twitter:card" content="summary_large_image">' . "\n";
	echo '<meta name="twitter:title" content="' . esc_attr( $title ) . '">' . "\n";
	echo '<meta name="twitter:description" content="' . esc_attr( $desc ) . '">' . "\n";
	echo '<meta name="twitter:image" content="' . esc_url( $img ) . '">' . "\n";

}, 1 );

/* 젯팩이 스스로 내보내는 og 태그를 우리 것으로 바꿔 끼웁니다.
   그러지 않으면 회색 스텔라사주 그림이 같이 나가 그쪽이 뽑힙니다. */

	$raw   = isset( $_GET['sc'] ) ? $_GET['sc'] : ( isset( $_GET['s'] ) ? $_GET['s'] : 0 );
	$score = max( 0, min( 99, intval( $raw ) ) );
	$a     = isset( $_GET['a'] ) ? max( 0, min( 40, intval( $_GET['a'] ) ) ) : 0;
	$b     = isset( $_GET['b'] ) ? max( 0, min( 30, intval( $_GET['b'] ) ) ) : 0;
	$c     = isset( $_GET['c'] ) ? max( 0, min( 30, intval( $_GET['c'] ) ) ) : 0;
	$rel   = isset( $_GET['rel'] ) ? sanitize_key( $_GET['rel'] ) : 'lover';
	$tier  = isset( $_GET['t'] ) ? wp_strip_all_tags( wp_unslash( $_GET['t'] ) ) : stella_um_tier( $score );
	if ( mb_strlen( $tier ) > 14 ) { $tier = mb_substr( $tier, 0, 14 ); }

	$img = home_url( '/um-card/' . $score . '-' . $rel . '-' . $a . '-' . $b . '-' . $c );

	$tags['og:title']        = '우리 ' . $score . '점' . ( $tier ? ' - ' . $tier : '' );
	$tags['og:description']  = '사주 ' . $a . '/40, 별자리 ' . $b . '/30, 띠 ' . $c . '/30. 생일만 넣으면 바로 나와요';

	$tags['og:image:width']  = 800;
	$tags['og:image:height'] = 400;
	$tags['twitter:card']    = 'summary_large_image';
	$tags['twitter:image']   = $img;

	return $tags;
}, 20 );

/* ── 크롤러가 보는 것을 그대로 들여다보기 ───────────────
   https://stellasaju.com/?um_og=1        (관리자만)

   서버가 스스로 결과 페이지를 「스레드인 척」 「카톡인 척」 두 번 읽어서
   og 태그가 실제로 나가는지, 그림 주소가 살아 있는지 보여줍니다. */
add_action( 'init', function () {

	if ( ! isset( $_GET['um_og'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:14px/1.7 system-ui;background:#111;color:#eee;padding:22px;max-width:900px">';
	echo '<h1 style="color:#ffd76a">크롤러가 보는 것</h1>';

	$page = home_url( '/uandme/?r=TEST&sc=63&a=30&b=12&c=24&rel=lover&t=' . rawurlencode( '무난한 사이' ) );
	$img  = home_url( '/um-card/63-lover-30-12-24' );

	$agents = array(
		'스레드·페이스북' => 'facebookexternalhit/1.1 (+http://www.facebook.com/externalhit_uatext.php)',
		'카카오톡'        => 'facebookexternalhit/1.1; kakaotalk-scrap/1.0; +https://devtalk.kakao.com/t/scrap/33984',
		'트위터'          => 'Twitterbot/1.0',
	);

	echo '<h2 style="color:#9fd">① 결과 페이지</h2>';
	echo '<p style="color:#888;font-size:12px;word-break:break-all">' . esc_html( $page ) . '</p>';

	foreach ( $agents as $who => $ua ) {

		echo '<h3 style="color:#8cf;margin-top:20px">' . esc_html( $who ) . '</h3>';

		$r = wp_remote_get( $page, array(
			'timeout'     => 25,
			'user-agent'  => $ua,
			'redirection' => 3,
		) );
		if ( is_wp_error( $r ) ) {
			echo '<p style="color:#f88">못 읽음 — ' . esc_html( $r->get_error_message() ) . '</p>';
			continue;
		}
		$code = wp_remote_retrieve_response_code( $r );
		$body = wp_remote_retrieve_body( $r );
		echo '<p>응답 <b style="color:' . ( 200 === intval( $code ) ? '#9f9' : '#f88' ) . '">' . intval( $code ) .
		     '</b> · 글 길이 ' . number_format( strlen( $body ) ) . '</p>';

		$n = preg_match_all( '#<meta[^>]*(?:property|name)=["\'](og:[^"\']*|twitter:[^"\']*)["\'][^>]*>#i', $body, $m );
		if ( ! $n ) {
			echo '<p style="color:#f88">og 태그가 하나도 없습니다</p>';
		} else {
			echo '<div style="background:#000;padding:10px;border-radius:8px;font-size:11.5px;' .
			     'white-space:pre-wrap;word-break:break-all;max-height:280px;overflow:auto">';
			foreach ( $m[0] as $tag ) {
				$hot = ( false !== strpos( $tag, 'um_card' ) );
				echo '<div style="color:' . ( $hot ? '#9f9' : '#999' ) . '">' . esc_html( $tag ) . '</div>';
			}
			echo '</div>';
			echo '<p style="font-size:12px;color:#888">초록색이 우리 점수 카드입니다. 회색만 있으면 옛 그림이 나가는 중입니다.</p>';
		}
	}

	echo '<h2 style="color:#9fd;margin-top:28px">② 카드 그림 자체</h2>';
	echo '<p style="color:#888;font-size:12px;word-break:break-all">' . esc_html( $img ) . '</p>';
	foreach ( $agents as $who => $ua ) {
		$r = wp_remote_get( $img, array( 'timeout' => 30, 'user-agent' => $ua, 'redirection' => 3 ) );
		if ( is_wp_error( $r ) ) {
			echo '<p style="color:#f88">' . esc_html( $who ) . ' — 못 받음 · ' . esc_html( $r->get_error_message() ) . '</p>';
			continue;
		}
		$code = wp_remote_retrieve_response_code( $r );
		$ct   = wp_remote_retrieve_header( $r, 'content-type' );
		$len  = strlen( wp_remote_retrieve_body( $r ) );
		$good = ( 200 === intval( $code ) );
		if ( false === strpos( (string) $ct, 'image' ) ) { $good = false; }
		echo '<p style="color:' . ( $good ? '#9f9' : '#f88' ) . '">' . esc_html( $who ) . ' — ' .
		     intval( $code ) . ' · ' . esc_html( $ct ) . ' · ' . number_format( $len ) . ' 바이트</p>';
	}

	echo '<hr style="margin:26px 0;border-color:#333">';
	echo '<p style="color:#888">이 화면을 통째로 찍어서 보내주세요.</p>';
	exit;
} );

/* ── 유앤미 화면에서 ?s= 를 검색으로 보지 않게 ──────────
   워드프레스는 ?s= 를 검색어로 읽어 「찾은 글 없음」 404 를 냅니다.
   크롤러는 404 페이지의 og 를 버리므로 카드가 뜨지 않았습니다.
   새 링크는 sc= 를 쓰지만, 이미 남에게 보낸 옛 링크도 살려 둡니다. */
add_action( 'parse_query', function ( $q ) {

	if ( ! $q->is_main_query() ) { return; }
	if ( ! function_exists( 'stella_um_here' ) ) { return; }
	if ( ! stella_um_here() ) { return; }
	if ( ! isset( $q->query_vars['s'] ) ) { return; }

	unset( $q->query_vars['s'] );
	$q->is_search = false;
	$q->is_404    = false;
} );
