/* ── 카드 파일과 그 CDN 주소 ───────────────────────────
   카카오가 우리 서버에서 그림을 못 가져오는 듯하여, 카드를 진짜 파일로
   남기고 워드프레스닷컴 그림 CDN(i0.wp.com) 주소를 알려줍니다.
   이 CDN 은 소희 님 사이트 그림을 이미 나르고 있는 곳입니다.

   담기는 것은 점수뿐입니다. 이름도 생년월일도 들어가지 않습니다. */
if ( ! function_exists( 'stella_um_card_name' ) ) {
	function stella_um_card_name( $score, $rel, $a, $b, $c, $n = '' ) {
		$base = intval( $score ) . '-' . preg_replace( '/[^a-z_]/', '', $rel ) . '-' .
		        intval( $a ) . '-' . intval( $b ) . '-' . intval( $c );
		if ( $n ) { $base .= '-' . preg_replace( '/[^A-Za-z0-9_-]/', '', $n ); }
		/* 판이나 글이 바뀌면 이름도 바뀌게 합니다. 안 그러면 한 번 만든
		   그림이 그대로 남아 새 판이 안 보입니다 — 2026-09-15 */
		$base .= '-v' . substr( md5( (string) get_option( 'stella_um_bg', '' ) .
		                             wp_json_encode( get_option( 'stella_um_bg_map', array() ) ) .
		                             wp_json_encode( get_option( 'stella_um_pos_map', array() ) ) .
		                             '2026-09-15' ), 0, 6 );
		return $base . '.png';
	}
}
if ( ! function_exists( 'stella_um_card_url' ) ) {
	function stella_um_card_url( $score, $rel, $a, $b, $c ) {
		$up   = wp_upload_dir();
		$name = stella_um_card_name( $score, $rel, $a, $b, $c );
		$url  = $up['baseurl'] . '/uandme/card/' . $name;
		/* CDN 을 거치면 카카오가 잘 가져갑니다 */
		$bare = preg_replace( '#^https?://#', '', $url );
		return 'https://i0.wp.com/' . $bare . '?ssl=1';
	}
}

	function stella_um_font_try( $url, &$log ) {

		$r = wp_remote_get( $url, array(
			'timeout'    => 30,
			'user-agent' => 'Mozilla/4.0 (compatible; MSIE 6.0; Windows NT 5.1)',
		) );
		if ( is_wp_error( $r ) ) {
			$log[] = array( $url, '못 감', $r->get_error_message() );
			return '';
		}
		$code = wp_remote_retrieve_response_code( $r );
		$body = wp_remote_retrieve_body( $r );
		$log[] = array( $url, '응답 ' . $code, strlen( $body ) . ' 바이트' );
		if ( 200 !== intval( $code ) ) { return ''; }

		/* 글꼴 파일이면 앞 네 글자가 0x00010000 이거나 true/OTTO 입니다 */
		$head = substr( $body, 0, 4 );
		if ( strlen( $body ) > 40000 ) {
			if ( "\x00\x01\x00\x00" === $head || 'true' === $head || 'ttcf' === $head ) {
				return $body;
			}
		}
		/* 글꼴이 아니면 안에서 ttf 주소를 찾아 다시 가 봅니다 */
		if ( preg_match( '#https://[^)\'"\s]+\.ttf#', $body, $m ) ) {
			$log[] = array( $m[0], '안에서 찾음', '따라갑니다' );
			return stella_um_font_try( $m[0], $log );
		}
		$log[] = array( $url, '글꼴 아님', substr( strip_tags( $body ), 0, 120 ) );
		return '';
	}
}
if ( ! function_exists( 'stella_um_font' ) ) {
	function stella_um_font( $which = 'jua', &$log = null ) {

		/* 예전처럼 $log 만 넘기던 자리도 그대로 받아줍니다 */
		if ( is_array( $which ) ) { $which = 'jua'; }
		if ( null === $log ) { $log = array(); }

		$kinds = stella_um_font_kinds();
		if ( ! isset( $kinds[ $which ] ) ) { $which = 'jua'; }
		$ttf = stella_um_font_path( $which );

		if ( file_exists( $ttf ) ) {
			if ( filesize( $ttf ) > 40000 ) {
				$log[] = array( $ttf, '이미 있음', filesize( $ttf ) . ' 바이트' );
				return $ttf;
			}
		}
		$dir = dirname( $ttf );
		if ( ! is_dir( $dir ) ) { wp_mkdir_p( $dir ); }

		foreach ( $kinds[ $which ]['where'] as $u ) {
			$raw = stella_um_font_try( $u, $log );
			if ( ! $raw ) { continue; }
			$ok = @file_put_contents( $ttf, $raw );
			$log[] = array( $ttf, $ok ? '저장함' : '저장 못 함', strlen( $raw ) . ' 바이트' );
			if ( $ok ) { return $ttf; }
		}
		/* 못 받으면 Jua 로 물러납니다 — 카드는 늘 나와야 합니다 */
		if ( 'jua' !== $which ) { return stella_um_font( 'jua', $log ); }
		return '';
	}
}

   실타래와 바늘이, 톰과 제리 칸에는 고양이와 쥐가 나오게요.

   넣는 법 (관리자만 · 주소창에 한 번씩)

     기본 판   ?stella_um_bg=0625r.jpg          파일 이름만 주셔도 됩니다
     구간 판   ?stella_um_bg=70|0625r.jpg       70~74 칸에만

     한 칸 빼기 ?stella_um_bg=70|off
     다 빼기   ?stella_um_bg=off

   판이 하나도 없으면 예전처럼 서버가 직접 그립니다.
   그러니 판을 못 넣으셔도 카드는 늘 나옵니다. */
if ( ! function_exists( 'stella_um_bg_open' ) ) {
	function stella_um_bg_open( $u ) {
		if ( ! $u ) { return null; }
		if ( 'off' === $u ) { return null; }

		$raw = '';
		$up  = wp_upload_dir();
		/* 우리 사이트 그림이면 파일을 바로 엽니다 — 빠르고 탈이 없습니다 */
		if ( 0 === strpos( $u, $up['baseurl'] ) ) {
			$path = $up['basedir'] . substr( $u, strlen( $up['baseurl'] ) );
			if ( file_exists( $path ) ) { $raw = (string) @file_get_contents( $path ); }
		}
		if ( ! $raw ) {
			$r = wp_remote_get( $u, array( 'timeout' => 20 ) );
			if ( is_wp_error( $r ) ) { return null; }
			if ( 200 !== intval( wp_remote_retrieve_response_code( $r ) ) ) { return null; }
			$raw = wp_remote_retrieve_body( $r );
		}
		if ( strlen( $raw ) < 1000 ) { return null; }
		$im = @imagecreatefromstring( $raw );
		if ( ! $im ) { return null; }
		return $im;
	}
}

/* 파일 이름만 주셔도 미디어에서 찾아 줍니다.
   주소를 통째로 넣으면 「https://」의 빗금 때문에 값이 잘리는 일이
   있어서, 이름만으로도 되게 했습니다 — 2026-09-15
     ?stella_um_bg=0625r.jpg */
if ( ! function_exists( 'stella_um_bg_find' ) ) {
	function stella_um_bg_find( $name ) {
		$name = basename( trim( $name ) );
		if ( '' === $name ) { return ''; }
		$up = wp_upload_dir();

		/* ① 미디어 서랍에서 이름으로 찾습니다 */
		$hit = get_posts( array(
			'post_type'      => 'attachment',
			'post_status'    => 'inherit',
			'posts_per_page' => 20,
			's'              => $name,
		) );
		foreach ( $hit as $one ) {
			$u = wp_get_attachment_url( $one->ID );
			if ( ! $u ) { continue; }
			if ( basename( $u ) === $name ) { return $u; }
		}

		/* ② 그래도 없으면 올린 달 서랍을 뒤집니다 */
		$year = (int) gmdate( 'Y' );
		$y = $year + 1;
		while ( $y >= $year - 2 ) {
			$m = 12;
			while ( $m >= 1 ) {
				$rel  = sprintf( '/%04d/%02d/%s', $y, $m, $name );
				if ( file_exists( $up['basedir'] . $rel ) ) {
					return $up['baseurl'] . $rel;
				}
				$m--;
			}
			$y--;
		}
		return '';
	}
}

	function stella_um_pos( $score ) {
		/* 기본 자리 — 말풍선 가운데 x · 가운데 y · 안쪽 너비 (비율) */
		$d = array( 0.501, 0.668, 0.70 );

		$b = stella_um_band_of( intval( $score ) );

			if ( is_array( $map[ $b ] ) ) {
				$g = $map[ $b ];
				/* 예전에 여섯 개로 넣어둔 것은 뒤 셋만 씁니다 */
				if ( 6 === count( $g ) ) { $g = array( $g[3], $g[4], $g[5] ); }
				if ( 3 === count( $g ) ) { $d = $g; }
			}
		}
		return array(
			'bx' => (float) $d[0], 'by' => (float) $d[1], 'bw' => (float) $d[2],
		);
	}
