/* ════════════════════════════════════════════════════════════
   표지 다음에 「차 례」 쪽 한 장 · 소희 님께

   소희 님 :「난 한 페이지로 보이는지 알았어. 사람들도 그래야
             이걸 다 봐준다는 느낌이 있지 않을까?」

   맞습니다. 맨 위 띠의 「목차」 단추는 기능이고,
   표지를 넘기자마자 펼쳐지는 차례 쪽은 약속입니다.
   제1장부터 뒤에 붙이는 것까지 쭉 보이면
   「이만큼을 다 봐주는구나」가 눈으로 옵니다.
   값을 치르기 전에 견본을 보는 사람에게 특히 그렇습니다.

   ── 하는 일 ────────────────────────────────────
   표지 바로 뒤에 차례 쪽을 한 장 끼웁니다.
     · 장 속표지는 굵게, 그 아래 쪽들은 들여씁니다
     · 오른쪽에 쪽번호, 사이는 점선으로 잇습니다
     · 목록은 책이 다 그려진 뒤에 그 책을 읽어서 만듭니다.
       문마다 쪽이 달라도 저절로 맞습니다.
     · PDF 로 받으실 때도 같이 나갑니다 — 파일에는 단추가
       없으니 차례 쪽이 더 중요합니다.

   ★ 글을 만드는 코드는 한 줄도 안 건드립니다.
   ★ 맨 위 띠의 「목차」 단추는 그대로 둡니다. 둘 다 있습니다.
   ★ 이 안에서 홑 앰퍼샌드를 쓰지 않습니다 (사이트 규칙).
     글자를 붙일 때 innerHTML 대신 textContent 만 씁니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_tocpage=1
   ② 정말 넣기   https://stellasaju.com/?stella_tocpage=apply
   ③ 되돌리기    https://stellasaju.com/?stella_tocpage=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_tocpage'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_tocpage_backup';
	$MARK = '2026-09-08 · 차례 쪽';
	$mode = sanitize_key( wp_unslash( $_GET['stella_tocpage'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:880px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}'
	   . 'h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '<h1 class="no">reading-book 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	if ( false !== strpos( $c, $MARK ) ) {
		echo '<h1 class="no">이미 들어가 있습니다.</h1>';
		echo '<p><code>/?stella_tocpage=undo</code> 로 걷어낸 뒤 다시 넣으세요.</p>';
		exit;
	}

	$TAIL = <<<'TAILX'

<style>
/* ══ 2026-09-08 · 차례 쪽 ═══════════════════════════════════ */
#ssb .tocpage .tocp-eye{
  font-size:.72rem; letter-spacing:.38em; color:var(--dim);
  text-align:center; margin:0 0 34px;
}
#ssb .tocpage .tocp-list{ max-width:30rem; margin:0 auto; }
#ssb .tocpage .tocp-ch,
#ssb .tocpage .tocp-sub{ display:flex; align-items:baseline; gap:9px; }
#ssb .tocpage .tocp-ch{
  font-family:var(--serif); color:var(--star); font-size:1.02rem;
  margin-top:26px; padding-top:13px; border-top:1px solid var(--rule);
}
#ssb .tocpage .tocp-ch:first-child{ margin-top:0; padding-top:0; border-top:0; }
#ssb .tocpage .tocp-sub{
  padding-left:15px; color:var(--soft); font-size:.92rem; margin-top:10px;
}
#ssb .tocpage .t{ flex:0 1 auto; }
#ssb .tocpage .d{
  flex:1 1 auto; min-width:18px; border-bottom:1px dotted var(--rule);
  transform:translateY(-4px);
}
#ssb .tocpage .n{
  flex:0 0 auto; font-family:var(--mono); font-size:.72rem; color:var(--dim);
}
</style>
<script>
/* ══ 2026-09-08 · 차례 쪽 ══
   책이 다 그려진 뒤에 그 책을 읽어서 표지 뒤에 한 장을 끼웁니다.
   ★ 홑 앰퍼샌드를 쓰지 않습니다 (사이트 규칙).
     그래서 글자는 innerHTML 이 아니라 textContent 로만 붙입니다. */
(function(){
  var tries = 0;

  function line(isCh, head, text, no){
    var d = document.createElement('div');
    d.className = isCh ? 'tocp-ch' : 'tocp-sub';

    var t = document.createElement('span');
    t.className = 't';
    if(isCh){
      var b = document.createElement('b');
      b.textContent = head;
      t.appendChild(b);
      t.appendChild(document.createTextNode(' ' + text));
    } else {
      t.textContent = text;
    }

    var dot = document.createElement('i');
    dot.className = 'd';

    var n = document.createElement('span');
    n.className = 'n';
    n.textContent = no;

    d.appendChild(t);
    d.appendChild(dot);
    d.appendChild(n);
    return d;
  }

  function build(){
    var box = document.getElementById('ssb');
    if(!box){ return false; }
    if(document.getElementById('ssTocPage')){ return true; }

    var pages = box.querySelectorAll('.page');
    if(pages.length < 5){ return false; }

    var cover = box.querySelector('.page.cover');
    if(!cover){
      var j;
      for(j = 0; j < pages.length; j++){
        if(pages[j].querySelector('h1')){ cover = pages[j]; break; }
      }
    }
    if(!cover){ return false; }

    var list = document.createElement('div');
    list.className = 'tocp-list';

    var made = 0;
    var i;
    for(i = 0; i < pages.length; i++){
      var pg = pages[i];
      if(pg === cover){ continue; }

      var f  = pg.querySelector('.folio');
      var no = f ? f.textContent.trim() : '';
      var dv = pg.querySelector('.dvno');
      var h2 = pg.querySelector('h2');

      if(dv){
        list.appendChild(line(true, dv.textContent.trim(),
                              h2 ? h2.textContent.trim() : '', no));
        made = made + 1;
        continue;
      }
      if(h2){
        list.appendChild(line(false, '', h2.textContent.trim(), no));
        made = made + 1;
        continue;
      }
      var ey = pg.querySelector('.eyebrow');
      if(ey){
        list.appendChild(line(false, '', ey.textContent.trim(), no));
        made = made + 1;
      }
    }
    if(made < 4){ return false; }

    var eye = document.createElement('p');
    eye.className = 'tocp-eye';
    eye.textContent = '차 례';

    var folio = document.createElement('div');
    folio.className = 'folio';

    var pageEl = document.createElement('div');
    pageEl.className = 'page tocpage';
    pageEl.id = 'ssTocPage';
    pageEl.appendChild(eye);
    pageEl.appendChild(list);
    pageEl.appendChild(folio);

    if(cover.nextSibling){
      cover.parentNode.insertBefore(pageEl, cover.nextSibling);
    } else {
      cover.parentNode.appendChild(pageEl);
    }
    return true;
  }

  var timer = setInterval(function(){
    tries = tries + 1;
    if(build()){ clearInterval(timer); return; }
    if(tries > 90){ clearInterval(timer); }
  }, 300);
})();
</script>
TAILX;

	$new = $c . $TAIL;

	echo '<h1>' . ( 'apply' === $mode ? '넣었습니다' : '넣으면 이렇게 됩니다 (아직 안 넣었습니다)' ) . '</h1>';
	echo '<h2>표지 다음에 「차 례」 쪽 한 장</h2>';
	echo '<p class="ok">장 속표지는 굵게 · 그 아래 쪽은 들여쓰기 · 오른쪽에 쪽번호 · 사이는 점선</p>';
	echo '<p class="dim">목록은 책이 다 그려진 뒤에 <b>그 책을 읽어서</b> 만듭니다. '
	   . '문마다 쪽이 달라도 저절로 맞고, 나중에 장을 더하셔도 따로 적을 것이 없습니다.</p>';
	echo '<p class="dim">맨 위 띠의 「목차」 단추는 그대로 둡니다. '
	   . '단추는 읽는 중에 건너뛰는 용도, 차례 쪽은 처음에 두께를 보여주는 용도입니다.</p>';
	echo '<p class="dim">PDF 로 받으실 때도 같이 나갑니다. 파일에는 단추가 없으니까요.</p>';
	echo '<p class="dim">' . number_format( strlen( $c ) ) . ' → '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( 'apply' !== $mode ) {
		echo '<p>정말 넣으시려면 <code>/?stella_tocpage=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );

	echo '<p class="ok">넣었습니다.</p>';
	echo '<p><a href="/reading-book/?demo=door-love">샘플 책에서 보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_tocpage=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
