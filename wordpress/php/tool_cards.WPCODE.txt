/* ════════════════════════════════════════════════════════════
   홈 카드 사진 고르기 · 소희 님께

   지금 홈의 카드들이 짝이 안 맞습니다 — 결혼운 자리에 헤어지는
   장면이 들어가 있는 식입니다.

   주소를 저에게 하나씩 보내실 필요 없습니다.
   이 화면에서 카드마다 그림을 골라 저장하시면 끝납니다.
   고르는 동안 오른쪽에 그림이 바로 뜹니다.

   ★ 파일 이름은 아무래도 됩니다. 이름을 바꾸실 필요 없습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 「어디서나 실행」 → 저장 → 활성

        https://stellasaju.com/?stella_cards=1

   ★ 되돌리기
        https://stellasaju.com/?stella_cards=1&undo=1
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_cards'] ) ) { return; }
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	$PID  = 62;
	$META = '_stella_cards_backup';

	/* 카드를 찾아 하나씩 뜯어봅니다 */
	$scan = function ( $c ) {
		$out = array();
		$pos = 0;
		while ( ( $p = strpos( $c, '<a class="ask-card"', $pos ) ) !== false ) {
			$e = strpos( $c, '</a>', $p );
			if ( false === $e ) { break; }
			$blk = substr( $c, $p, $e - $p );

			$src = '';
			$sp  = strpos( $blk, 'src="' );
			if ( false !== $sp ) {
				$sp += 5;
				$sq  = strpos( $blk, '"', $sp );
				if ( false !== $sq ) { $src = substr( $blk, $sp, $sq - $sp ); }
			}

			$tag = '';
			$tp  = strpos( $blk, '<span class="tag">' );
			if ( false !== $tp ) {
				$tp += strlen( '<span class="tag">' );
				$tq  = strpos( $blk, '</span>', $tp );
				if ( false !== $tq ) { $tag = substr( $blk, $tp, $tq - $tp ); }
			}

			$q  = '';
			$qp = strpos( $blk, '<div class="q">' );
			if ( false !== $qp ) {
				$qp += strlen( '<div class="q">' );
				$qq  = strpos( $blk, '</div>', $qp );
				if ( false !== $qq ) { $q = substr( $blk, $qp, $qq - $qp ); }
			}

			$out[] = array(
				'at'  => $p,
				'end' => $e,
				'src' => $src,
				'tag' => $tag,
				'q'   => $q,
			);
			$pos = $e + 4;
		}
		return $out;
	};

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $PID ) );
	if ( ! $row ) { wp_die( '62번 페이지를 못 읽었습니다.', '스텔라', array( 'response' => 200 ) ); }
	$c = $row->post_content;

	/* ── 되돌리기 ── */
	if ( isset( $_GET['undo'] ) ) {
		$bak = get_post_meta( $PID, $META, true );
		header( 'Content-Type: text/html; charset=utf-8' );
		echo '<meta charset="utf-8"><body style="background:#111;color:#eee;'
			. 'font:15px/1.8 system-ui,sans-serif;padding:24px">';
		if ( $bak ) {
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $PID ) );
			delete_post_meta( $PID, $META );
			clean_post_cache( $PID );
			if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
			echo '<h1 style="color:#7bff9b">되돌렸습니다.</h1>';
		} else {
			echo '<h1 style="color:#888">사본이 없습니다.</h1>';
		}
		echo '</body>';
		exit;
	}

	$cards = $scan( $c );
	$saved = 0;

	/* ── 저장 ── */
	if ( isset( $_POST['stella_cards_save'] ) ) {
		check_admin_referer( 'stella_cards' );

		$pick = isset( $_POST['pic'] ) ? (array) $_POST['pic'] : array();
		$edits = array();

		foreach ( $cards as $i => $cd ) {
			if ( empty( $pick[ $i ] ) ) { continue; }
			$aid = (int) $pick[ $i ];
			if ( $aid < 1 ) { continue; }
			$url = wp_get_attachment_url( $aid );
			if ( ! $url ) { continue; }
			if ( $url === $cd['src'] ) { continue; }

			/* 지금 쓰는 주소가 사진 가속 주소면 같은 모양으로 맞춥니다 */
			$new = $url;
			if ( false !== strpos( $cd['src'], 'i0.wp.com' ) ) {
				$bare = preg_replace( '#^https?://#', '', $url );
				$new  = 'https://i0.wp.com/' . $bare . '?resize=640%2C960';
			}

			$blk = substr( $c, $cd['at'], $cd['end'] - $cd['at'] );
			$sp  = strpos( $blk, 'src="' );
			if ( false === $sp ) { continue; }
			$sp += 5;
			$sq  = strpos( $blk, '"', $sp );
			if ( false === $sq ) { continue; }

			$edits[] = array( $cd['at'] + $sp, $sq - $sp, $new );
			$saved++;
		}

		if ( $edits ) {
			if ( ! get_post_meta( $PID, $META, true ) ) {
				update_post_meta( $PID, $META, $c );
			}
			$new_c = $c;
			for ( $k = count( $edits ) - 1; $k >= 0; $k-- ) {
				list( $at, $len, $txt ) = $edits[ $k ];
				$new_c = substr( $new_c, 0, $at ) . $txt . substr( $new_c, $at + $len );
			}
			$wpdb->update( $wpdb->posts, array( 'post_content' => $new_c ), array( 'ID' => $PID ) );
			clean_post_cache( $PID );
			if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
			$c     = $new_c;
			$cards = $scan( $c );
		}
	}

	/* ── 미디어 서랍의 그림들 ── */
	$atts = get_posts( array(
		'post_type'      => 'attachment',
		'post_mime_type' => 'image',
		'post_status'    => 'inherit',
		'posts_per_page' => 300,
		'orderby'        => 'date',
		'order'          => 'DESC',
	) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">';
	echo '<style>body{background:#12101B;color:#F2ECE0;'
		. 'font:15px/1.7 -apple-system,"Apple SD Gothic Neo",system-ui,sans-serif;'
		. 'padding:22px 16px;max-width:1000px;margin:0 auto}'
		. 'h1{color:#D4AF6A;font-size:1.3rem;margin:0 0 6px}'
		. '.dim{color:#8B849C;font-size:.88rem}'
		. '.card{display:flex;gap:14px;align-items:flex-start;'
		. 'border-top:1px solid #2A2438;padding:16px 0}'
		. '.thumb{width:74px;height:110px;object-fit:cover;border-radius:8px;'
		. 'background:#221D33;flex:0 0 auto}'
		. '.mid{flex:1 1 auto;min-width:0}'
		. '.tag{display:inline-block;background:#2A2438;color:#E8CD96;'
		. 'border-radius:20px;padding:2px 11px;font-size:.8rem}'
		. '.q{margin:7px 0 9px;font-size:.95rem}'
		. 'select{width:100%;max-width:420px;padding:9px;border-radius:7px;'
		. 'background:#1B172A;color:#F2ECE0;border:1px solid #3A3350;font:inherit}'
		. 'button{background:#D4AF6A;color:#1A1230;border:0;border-radius:8px;'
		. 'padding:13px 26px;font:inherit;font-weight:700;cursor:pointer}'
		. '.ok{color:#9AD1A5}</style>';
	echo '<body><h1>홈 카드 사진 고르기</h1>';
	echo '<p class="dim">카드마다 그림을 고르면 왼쪽 그림이 바로 바뀝니다. '
		. '다 고르신 뒤 맨 아래 저장을 누르세요. 파일 이름은 아무래도 됩니다.</p>';

	if ( $saved ) {
		printf( '<p class="ok">%d 장을 바꿨습니다.</p>', $saved );
	}
	if ( ! $cards ) {
		echo '<p style="color:#E5A0A0">카드를 못 찾았습니다.</p></body>';
		exit;
	}

	echo '<form method="post">';
	wp_nonce_field( 'stella_cards' );
	printf( '<p class="dim">카드 %d 장</p>', count( $cards ) );

	foreach ( $cards as $i => $cd ) {
		printf(
			'<div class="card"><img class="thumb" id="pv%d" src="%s" alt=""><div class="mid">'
			. '<span class="tag">%s</span><div class="q">%s</div>'
			. '<select name="pic[%d]" data-pv="pv%d"><option value="">그대로 두기</option>',
			$i, esc_url( $cd['src'] ), esc_html( $cd['tag'] ), esc_html( $cd['q'] ), $i, $i
		);
		foreach ( $atts as $a ) {
			$u = wp_get_attachment_url( $a->ID );
			printf(
				'<option value="%d" data-u="%s">%s</option>',
				$a->ID, esc_url( $u ), esc_html( basename( (string) $u ) )
			);
		}
		echo '</select></div></div>';
	}

	echo '<p style="margin:26px 0 40px"><button type="submit" name="stella_cards_save" value="1">'
		. '저장하기</button></p></form>';

	echo '<script>'
		. 'document.querySelectorAll("select").forEach(function(s){'
		. ' s.addEventListener("change",function(){'
		. '  var o=s.options[s.selectedIndex];'
		. '  var u=o.getAttribute("data-u");'
		. '  if(!u){ return; }'
		. '  var img=document.getElementById(s.getAttribute("data-pv"));'
		. '  if(img){ img.src=u; }'
		. ' });'
		. '});</script>';

	echo '<p class="dim">되돌리려면 ?stella_cards=1&undo=1</p></body>';
	exit;
}, 0 );
