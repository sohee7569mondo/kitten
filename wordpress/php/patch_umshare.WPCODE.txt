<script>
(function () {
  if (window.StellaUmShare) { return; }
  window.StellaUmShare = '2026-09-17';

  var LABEL = '사진으로 보내기';

  /* 이 기기가 사진 보내기를 할 줄 아나 — 짐작 안 하고 물어봅니다 */
  function canPhoto() {
    if (!navigator.share) { return '이 브라우저는 보내기를 모릅니다'; }
    if (!navigator.canShare) { return '이 브라우저는 사진 보내기를 모릅니다'; }
    try {
      var probe = new File([new Blob([new Uint8Array([1])], { type: 'image/png' })],
        'p.png', { type: 'image/png' });
      if (navigator.canShare({ files: [probe] })) { return ''; }
      return '이 기기는 사진 보내기를 막아 두었습니다';
    } catch (e) { return '사진을 만들 수 없습니다'; }
  }

  function btnCss(main) {
    var s = 'width:100%;border:0;border-radius:15px;padding:14px 0;margin-top:11px;'
      + 'font-size:14.5px;font-weight:700;font-family:inherit;cursor:pointer;';
    if (main) { s += 'background:#6B5BA8;color:#fff;'; }
    else { s += 'background:rgba(255,255,255,.14);color:#fff;'; }
    return s;
  }

  /* 사진을 미리 만들어 둡니다 — 누르는 순간 바로 보내야 아이폰이 막지 않습니다 */
  function prime(img, hold) {
    var src = img.getAttribute('src');
    if (!src) { return; }
    fetch(src).then(function (r) { return r.blob(); }).then(function (bl) {
      try { hold.file = new File([bl], 'uandme.png', { type: 'image/png' }); }
      catch (e) { hold.file = null; }
    }).catch(function () { hold.file = null; });
  }

  function words() {
    var t = document.title || '유앤미';
    var b = document.querySelector('meta[property="og:title"]');
    if (b) { if (b.content) { t = b.content; } }
    return t + ' — 생일만 넣으면 바로 나와요';
  }

  function add(sheet) {
    if (sheet.querySelector('[data-umphoto]')) { return; }
    var img = sheet.querySelector('img');
    if (!img) { return; }
    var box = img.parentNode;
    if (!box) { return; }

    var why = canPhoto();
    if (why) {
      var note = document.createElement('div');
      note.setAttribute('data-umphoto', 'no');
      note.style.cssText = 'color:#fff;opacity:.6;font-size:12.5px;margin-top:12px;line-height:1.7';
      note.textContent = why + ' — 아래에서 그림을 저장해 올려주세요';
      box.appendChild(note);
      return;
    }

    var hold = { file: null };
    prime(img, hold);

    var b = document.createElement('button');
    b.setAttribute('data-umphoto', '1');
    b.style.cssText = btnCss(true);
    b.textContent = LABEL;

    b.addEventListener('click', function () {
      if (hold.file) { fire(hold.file, b); return; }
      /* 아직 안 만들어졌으면 그 자리에서 만듭니다 */
      b.textContent = '사진 준비 중…';
      fetch(img.getAttribute('src')).then(function (r) { return r.blob(); })
        .then(function (bl) {
          var f = null;
          try { f = new File([bl], 'uandme.png', { type: 'image/png' }); } catch (e) { f = null; }
          if (!f) { b.textContent = '사진을 못 만들었어요'; return; }
          hold.file = f;
          b.textContent = LABEL;
          fire(f, b);
        }).catch(function () { b.textContent = '사진을 못 만들었어요'; });
    });

    /* 사진 바로 밑, 다른 단추들보다 위에 놓습니다 */
    if (img.nextSibling) { box.insertBefore(b, img.nextSibling); }
    else { box.appendChild(b); }
  }

  function fire(file, b) {
    var data = { files: [file], text: words() };
    try {
      navigator.share(data).then(function () {
        b.textContent = '보냈어요';
        setTimeout(function () { b.textContent = LABEL; }, 2500);
      }).catch(function () { b.textContent = LABEL; });
    } catch (e) { b.textContent = '보내기가 안 열렸어요'; }
  }

  function look() {
    var s = document.getElementById('um-sheet');
    if (s) { add(s); }
  }

  if (document.body) { watch(); }
  else { document.addEventListener('DOMContentLoaded', watch); }

  function watch() {
    look();
    try {
      var mo = new MutationObserver(look);
      mo.observe(document.body, { childList: true, subtree: true });
    } catch (e) {
      setInterval(look, 700);
    }
  }
})();

		exit;

	printf( '<p style="font-size:13px">보기 : <a style="color:#ffd76a" href="%s">%s</a></p>',
		esc_url( $plain ), esc_html( $plain ) );

	$r = wp_remote_get( $plain, array( 'timeout' => 40, 'sslverify' => false ) );
	if ( is_wp_error( $r ) ) {
		printf( '<p style="color:#ff7b7b">못 불렀습니다 — %s</p>',
			esc_html( $r->get_error_message() ) );
	} else {
		$code = wp_remote_retrieve_response_code( $r );
		$ct   = wp_remote_retrieve_header( $r, 'content-type' );
		$bd   = wp_remote_retrieve_body( $r );
		$sz   = @getimagesizefromstring( $bd );
		printf( '<p>상태 <b style="color:%s">%d</b> · 형식 <b>%s</b> · <b>%s</b>바이트 · 크기 <b style="color:%s">%s</b></p>',
			( 200 === intval( $code ) ) ? '#7bff9b' : '#ff7b7b', intval( $code ),
			esc_html( $ct ? $ct : '(없음)' ), number_format( strlen( $bd ) ),
			$sz ? '#7bff9b' : '#ff7b7b',
			$sz ? ( $sz[0] . ' x ' . $sz[1] ) : '그림이 아닙니다' );
		if ( $sz ) {
			printf( '<img src="%s" style="max-width:220px;border-radius:12px;background:#fff">',
				esc_url( $plain ) );
		}
	}

	/* ── 파일이 남았나 ── */
	echo '<h2 style="color:#9fd;font-size:16px">② 만들어 둔 카드 파일</h2>';
	$up  = wp_upload_dir();
	$dir = $up['basedir'] . '/uandme/card';
	if ( ! is_dir( $dir ) ) {
		echo '<p style="color:#ff7b7b">폴더가 아직 없습니다 (' . esc_html( $dir ) . ')</p>';
	} else {
		$fs = glob( $dir . '/*.png' );
		if ( ! $fs ) { $fs = array(); }
		printf( '<p><b>%d</b>장 있습니다.</p>', count( $fs ) );
		$show = array_slice( $fs, 0, 6 );
		echo '<ul style="font-size:12.5px;color:#9a9">';
		foreach ( $show as $f ) {
			printf( '<li>%s — %s바이트</li>', esc_html( basename( $f ) ),
				number_format( (int) @filesize( $f ) ) );
		}
		echo '</ul>';
	}

	/* ── ④ 가 저장하는 이름 vs ⑤ 가 찾는 이름 ── */
	echo '<h2 style="color:#9fd;font-size:16px">③ 카톡에 올리는 길</h2>';

		printf( '<p>④번이 <b>저장하는</b> 이름 &nbsp; <code style="color:#9fd">%s</code></p>',

		printf( '<p>⑤번이 <b>찾는</b> 이름 &nbsp;&nbsp;&nbsp; <code style="color:%s">%s</code></p>',
