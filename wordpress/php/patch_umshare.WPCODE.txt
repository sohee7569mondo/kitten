<script>
(function () {
  if (window.StellaUmShare) { return; }
  window.StellaUmShare = '2026-09-17b';

  var LABEL = '사진으로 보내기';
  var diagLine = null;

  function say(t) {
    if (!diagLine) { return; }
    diagLine.style.display = 'block';
    diagLine.textContent = t;
  }

  /* 이 기기가 사진 보내기를 할 줄 아나 — 짐작 안 하고 물어봅니다 */
  function whyNot() {
    if (!navigator.share) { return '이 브라우저는 보내기(share)를 모릅니다'; }
    if (!navigator.canShare) { return '이 브라우저는 사진 보내기(canShare)를 모릅니다'; }
    try {
      var probe = new File([new Blob([new Uint8Array([1])], { type: 'image/png' })],
        'p.png', { type: 'image/png' });
      if (navigator.canShare({ files: [probe] })) { return ''; }
      return '이 기기가 사진 보내기를 막아 두었습니다';
    } catch (e) { return '사진을 만들 수 없습니다 (' + (e.name || e) + ')'; }
  }

  /* 사진 만들기 — 길 두 가지. 하나가 막히면 다른 하나로. */
  function toFile(img, ok, bad) {
    var src = img.getAttribute('src') || '';

    function viaCanvas() {
      try {
        var c = document.createElement('canvas');
        c.width = img.naturalWidth || img.width || 800;
        c.height = img.naturalHeight || img.height || 1200;
        c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
        c.toBlob(function (bl) {
          if (!bl) { bad('그림판에서 사진이 안 나옵니다'); return; }
          wrap(bl, '그림판');
        }, 'image/png');
      } catch (e2) { bad('그림판 ' + (e2.name || e2)); }
    }

    function wrap(bl, how) {
      try {
        var f = new File([bl], 'uandme.png', { type: 'image/png' });
        ok(f, how, bl.size || 0);
      } catch (e3) { bad('사진 봉투 ' + (e3.name || e3)); }
    }

    if (!src) { bad('그림 주소가 비어 있습니다'); return; }

    fetch(src).then(function (r) {
      if (!r.ok) { throw new Error('HTTP ' + r.status); }
      return r.blob();
    }).then(function (bl) { wrap(bl, '가져오기'); })
      .catch(function () { viaCanvas(); });
  }

  function words() {
    var t = '유앤미 궁합';
    var b = document.querySelector('meta[property="og:title"]');
    if (b) { if (b.content) { t = b.content; } }
    return t + ' — 생일만 넣으면 바로 나와요';
  }

  function add(sheet) {
    if (sheet.querySelector('[data-umphoto]')) { return; }
    var img = sheet.querySelector('img');
    if (!img) { return; }
    var box = img.parentNode;
    if (!box) { return; }

    /* 멈춘 자리를 적을 줄 — 탈이 났을 때만 보입니다 */
    diagLine = document.createElement('div');
    diagLine.setAttribute('data-umdiag', '1');
    diagLine.style.cssText = 'display:none;color:#ffd76a;font-size:12px;line-height:1.7;'

      + 'background:rgba(0,0,0,.28);border-radius:10px;padding:8px 10px';

    var no = whyNot();
    if (no) {
      var note = document.createElement('div');
      note.setAttribute('data-umphoto', 'no');
      note.style.cssText = 'color:#fff;opacity:.62;font-size:12.5px;margin-top:12px;line-height:1.7';
      note.textContent = no + ' — 아래 「그림 저장하기」로 저장해서 올려주세요';
      if (img.nextSibling) { box.insertBefore(note, img.nextSibling); }
      else { box.appendChild(note); }
      return;
    }

    var hold = { file: null, how: '', size: 0 };
    /* 미리 만들어 둡니다 — 누르는 순간 바로 보내야 아이폰이 안 막습니다 */
    toFile(img, function (f, how, sz) { hold.file = f; hold.how = how; hold.size = sz; },
      function (w) { hold.file = null; hold.how = 'x'; hold.size = 0; say('미리 만들기 실패 : ' + w); });

    var b = document.createElement('button');
    b.setAttribute('data-umphoto', '1');
    b.style.cssText = 'width:100%;border:0;border-radius:15px;padding:14px 0;margin-top:11px;'
      + 'font-size:14.5px;font-weight:700;font-family:inherit;cursor:pointer;'
      + 'background:#6B5BA8;color:#fff';
    b.textContent = LABEL;

    b.addEventListener('click', function () {
      if (hold.file) { fire(hold.file, b, hold.how, hold.size); return; }
      b.textContent = '사진 만드는 중…';
      toFile(img, function (f, how, sz) {
        b.textContent = LABEL;
        hold.file = f;
        fire(f, b, how, sz);
      }, function (w) {
        b.textContent = LABEL;
        say('사진을 못 만들었습니다 : ' + w
          + '  ·  그림 주소 ' + String(img.getAttribute('src')).slice(0, 34));
      });
    });

    if (img.nextSibling) { box.insertBefore(b, img.nextSibling); }
    else { box.appendChild(b); }
    box.insertBefore(diagLine, b.nextSibling);
  }

  function fire(file, b, how, size) {
    var data = { files: [file] };
    /* 글을 같이 넣으면 거절하는 기기가 있습니다 — 넣어보고 안 되면 뺍니다 */
    var withText = { files: [file], text: words() };
    var use = data;
    try { if (navigator.canShare(withText)) { use = withText; } } catch (e0) { use = data; }

    try {

      p.then(function () {
        b.textContent = '보냈어요';
        setTimeout(function () { b.textContent = LABEL; }, 2500);
      }).catch(function (err) {
        var nm = (err ? (err.name || '') : '');
        if (nm === 'AbortError') { return; }          /* 사람이 닫은 것 */
        say('보내기가 멈췄습니다 : ' + nm + ' ' + (err ? (err.message || '') : '')
          + '  ·  사진 ' + how + ' ' + Math.round(size / 1024) + 'KB');
      });
    } catch (e) {
      say('보내기를 못 열었습니다 : ' + (e.name || '') + ' ' + (e.message || '')
        + '  ·  사진 ' + how + ' ' + Math.round(size / 1024) + 'KB');
    }
  }

  function look() {
    var s = document.getElementById('um-sheet');
    if (s) { add(s); }
  }

  function watch() {
    look();
    try {
      var mo = new MutationObserver(look);
      mo.observe(document.body, { childList: true, subtree: true });
    } catch (e) { setInterval(look, 700); }
  }

  if (document.body) { watch(); }
  else { document.addEventListener('DOMContentLoaded', watch); }
})();

			$sz ? '#7bff9b' : '#ff7b7b',
			$sz ? ( $sz[0] . ' x ' . $sz[1] ) : '그림이 아닙니다' );
		if ( $sz ) {
			printf( '<img src="%s" style="max-width:220px;border-radius:12px;background:#fff">',
				esc_url( $plain ) );
		}
