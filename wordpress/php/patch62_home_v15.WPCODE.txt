		   . 'STELLASAJU_LOVE-1~6, STELLASAJU_WORK-1~6 이 필요합니다.</p>';
		echo '<p>홈을 진짜로 바꾸시려면 <code>/?stella_home=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $PID, $META, true );
		if ( ! $old ) {
			echo '<p class="no">되돌릴 것이 없습니다.</p>';
			exit;
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $PID ) );
		clean_post_cache( $PID );
		delete_post_meta( $PID, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $PID ) );
	if ( ! $row ) {
		echo '<h1 class="no">홈(62)을 못 읽었습니다.</h1>';
		exit;
	}
	$c = $row->post_content;

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';
	echo '<p>지금 홈 : ' . number_format( strlen( $c ) ) . ' 바이트<br>'
	   . '새 홈    : ' . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( false !== strpos( $c, 'BUILD v15' ) ) {
		echo '<p class="no">이미 v15 입니다. 그래도 다시 쓰시려면 apply 를 여세요.</p>';
	}

	if ( ! $apply ) {
		echo '<p>초안으로 먼저 보시려면 <code>/?stella_home=preview</code>,<br>'
		   . '정말 바꾸시려면 <code>/?stella_home=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $PID, $META, true ) ) {
		update_post_meta( $PID, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $PID ) );
	clean_post_cache( $PID );

	echo '<p class="ok">홈을 바꿨습니다. <a href="/">홈 열어보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_home=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
