/* ══════════════════════════════════════════════════════════
   결제하고 돌아오면 왔던 문으로 — 마이페이지 하나만 고칩니다
   ★ 2026-09-10 · 소희 님께

   소희 님 : 「연애운을 눌러서 결제를 하는데 결제하면
   마이페이지인가에 들어가서 헷갈리더라」

   맞습니다. 지금은 이렇게 돼 있습니다.

     문(연애운)  →  구슬 모자람  →  /mypage/?charge=1
                 →  결제창       →  /mypage/?paid=1
                 →  ...끝. 손님은 마이페이지에 덩그러니 남습니다.

   돈을 낸 사람은 연애운을 보러 온 것입니다. 여기서 잃으면
   결제까지 다 하고도 물건을 안 받고 나가는 셈입니다.

   고치는 법
     마이페이지가 「어디서 왔는지」를 기억했다가, 결제가 끝나면
     되돌아갈 길을 맨 위에 크게 보여줍니다. 결제를 마친 표시가
     있으면 4초 뒤 저절로 돌아갑니다 (「여기 있을게요」로 멈춤).

   ★ 문 여섯 쪽은 한 글자도 안 건드립니다. 서버 쪽도 안 건드립니다.
     마이페이지 끝에 덩어리 하나를 붙일 뿐입니다.
     어디서 왔는지는 브라우저가 알려주는 referrer 로 압니다.

   ★ 기억은 이 기계 안에만, 30분만 남습니다. 서버로 안 보냅니다.

   해보기   https://stellasaju.com/?stella_backto=preview
   붙이기   https://stellasaju.com/?stella_backto=apply
   되돌리기 https://stellasaju.com/?stella_backto=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_backto'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_backto'] ) );
	$META = '_stella_backto_bak';

	$page = get_page_by_path( 'mypage' );
	if ( ! $page ) { echo '마이페이지가 없습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	if ( false !== strpos( $c, 'stella_back' ) ) {
		echo "이미 붙어 있습니다.\n먼저 ?stella_backto=undo 로 되돌리세요.";
		exit;
	}

	$add = <<<'STELLA_BACK_HTML'
<!-- wp:html -->
<script>
/* ══ 결제하고 돌아오면 왔던 문으로 ─ 2026-09-10 ══════════════
   문에서 「구슬이 모자랍니다」 → 마이페이지로 보내집니다.
   결제가 끝나도 마이페이지에 그대로 남습니다. 손님은 자기가
   무엇을 하려 했는지 잊습니다.

   그래서 마이페이지가 「어디서 왔는지」를 기억했다가
   되돌아갈 자리를 크게 보여줍니다.

   문 여섯 쪽은 한 글자도 안 건드립니다. 이 쪽 하나로 끝납니다.
   기억은 이 기계 안(localStorage)에만, 서른 해 아니라 30분만
   남습니다. ═══════════════════════════════════════════════ */
(function(){
  var KEY = 'stella_back';
  var LIVE = 30 * 60 * 1000;   /* 30분 */

  var DOORS = {
    'door-love'   : '연애운',
    'door-career' : '직업운',
    'door-health' : '건강운',
    'door-fortune': '재물운',
    'door-astro'  : '점성술',
    'door-tarot'  : '스텔라 타로'
  };

  function read(){
    try {
      var s = localStorage.getItem(KEY);
      if(!s){ return null; }
      var o = JSON.parse(s);
      if(!o){ return null; }
      if(!o.u){ return null; }
      if(Date.now() - (o.t || 0) > LIVE){ return null; }
      return o;
    } catch(e){ return null; }
  }
  function write(u){
    try { localStorage.setItem(KEY, JSON.stringify({ u:u, t:Date.now() })); } catch(e){}
  }
  function clear(){
    try { localStorage.removeItem(KEY); } catch(e){}
  }
  function jong(w){
    if(!w){ return -1; }
    var c = w.charCodeAt(w.length - 1);
    if(c < 0xAC00){ return -1; }
    if(c > 0xD7A3){ return -1; }
    return (c - 0xAC00) % 28;
  }
  function eulT(w){ return jong(w) > 0 ? '을' : '를'; }
  function roT(w){ var j = jong(w); return (j <= 0 || j === 8) ? '로' : '으로'; }

  function nameOf(u){
    var k;
    for(k in DOORS){
      if(!DOORS.hasOwnProperty(k)){ continue; }
      if(u.indexOf('/' + k) >= 0){ return DOORS[k]; }
    }
    return '보시던 자리';
  }

  var q = location.search || '';

  /* ── 1. 문에서 충전하러 온 참이면, 온 자리를 적어 둡니다 ── */
  if(q.indexOf('charge=1') >= 0){
    var r = document.referrer || '';
    if(r.indexOf(location.origin) === 0){
      if(r.indexOf('/door-') >= 0){ write(r); }
    }
  }

  /* ── 2. 적어 둔 자리가 있으면 되돌아갈 길을 보여줍니다 ── */
  var back = read();
  if(!back){ return; }
  if(q.indexOf('charge=1') >= 0){ return; }   /* 방금 온 참이면 아직 아닙니다 */

  var nm = nameOf(back.u);

  /* 결제를 마치고 돌아온 표시가 있으면 저절로 넘어갑니다 */
  var done = false;
  if(q.indexOf('paid=1') >= 0){ done = true; }
  if(q.indexOf('imp_success=true') >= 0){ done = true; }
  if(q.indexOf('success=true') >= 0){ done = true; }

  var css = document.createElement('style');
  css.textContent =
    '.stbk{ position:fixed; left:0; right:0; top:0; z-index:99999;' +
    '  background:#1B1636; color:#FFFFFF; padding:14px 16px;' +
    '  display:flex; align-items:center; justify-content:center;' +
    '  gap:14px; flex-wrap:wrap; font-size:15px; line-height:1.5;' +
    '  box-shadow:0 6px 20px -8px rgba(0,0,0,.6); }' +
    '.stbk b{ color:#FFD98A; }' +
    '.stbk a.go{ background:#FFD98A; color:#1B1636; font-weight:800;' +
    '  padding:10px 20px; border-radius:999px; text-decoration:none;' +
    '  white-space:nowrap; }' +
    '.stbk button.stay{ background:none; border:0; color:rgba(255,255,255,.72);' +
    '  font-size:13px; text-decoration:underline; cursor:pointer; padding:6px; }' +
    '@media(max-width:520px){ .stbk{ font-size:14px; padding:12px 12px; gap:10px; } }';
  document.head.appendChild(css);

  var bar = document.createElement('div');
  bar.className = 'stbk';
  bar.innerHTML =
    '<span id="stbkMsg">' +
      (done ? '구슬이 들어왔습니다. <b>' + nm + '</b> 이어서 보시겠어요?'
            : '<b>' + nm + '</b>' + eulT(nm) + ' 보시던 중이었습니다.') +
    '</span>' +
    '<a class="go" id="stbkGo" href="#">' + nm + ' 이어서 보기 →</a>' +
    '<button type="button" class="stay" id="stbkStay">여기 있을게요</button>';
  document.body.appendChild(bar);
  document.body.style.paddingTop = bar.offsetHeight + 'px';

  function leave(){
    clear();
    location.href = back.u;
  }
  document.getElementById('stbkGo').addEventListener('click', function(e){
    e.preventDefault();
    leave();
  });

  var timer = null;
  document.getElementById('stbkStay').addEventListener('click', function(){
    if(timer){ clearInterval(timer); }
    clear();
    bar.parentNode.removeChild(bar);
    document.body.style.paddingTop = '';
  });

  if(done){
    var left = 4;
    var msg = document.getElementById('stbkMsg');
    timer = setInterval(function(){
      left = left - 1;
      if(left <= 0){
        clearInterval(timer);
        leave();
        return;
      }
      msg.innerHTML = '구슬이 들어왔습니다. <b>' + left + '초</b> 뒤에 <b>' + nm + '</b>' + roT(nm) + ' 돌아갑니다.';
    }, 1000);
  }
})();
</script>
<!-- /wp:html -->
STELLA_BACK_HTML;

	$new = rtrim( $c ) . "\n\n" . $add . "\n";

	echo "마이페이지 (쪽 " . $id . ")\n";
	echo '글자 수  ' . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $new ) ) . "\n";

	if ( 'preview' === $mode ) {
		echo "\n끝에 덩어리 하나를 붙입니다. 안에 있는 글은 안 건드립니다.\n";
		echo "붙이시려면 ?stella_backto=apply\n";
		exit;
	}

	update_post_meta( $id, $META, $c );
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "\n붙였습니다.\n\n시험하는 법\n";
	echo "  1. 문(연애운)에서 구슬이 모자란 상태로 「운세 보기」\n";
	echo "  2. 마이페이지로 넘어갑니다 (여기서는 아직 아무것도 안 나옵니다)\n";
	echo "  3. 결제하고 돌아오면 맨 위에 「연애운 이어서 보기」가 뜹니다\n";
	echo "\n되돌리려면 ?stella_backto=undo\n";
	exit;
}, 1 );
