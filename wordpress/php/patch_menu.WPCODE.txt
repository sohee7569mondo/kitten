/* ════════════════════════════════════════════════════════════
   메뉴 손보기 · 소희 님께

   ① 「이번주 로또 번호 추첨」을 메뉴에서 뺍니다
      (쪽은 그대로 둡니다. 메뉴에서만 사라집니다.)
   ② 「나의 사주풀이」 → 「나의 사주풀이(기본)」

   ★ 메뉴가 어디에 저장되어 있든 찾습니다 —
     블록 메뉴(wp_navigation) · 템플릿 조각 · 재사용 블록 ·
     옛날 메뉴(nav_menu_item) 네 군데를 다 살핍니다.

   ★ 미리보기에서 무엇을 어떻게 바꿀지 그대로 보여줍니다.
     한 군데도 못 찾으면 아무것도 안 바꿉니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 「어디서나 실행」 → 저장 → 활성

   ① 미리보기   https://stellasaju.com/?stella_menu=1
   ② 적용       https://stellasaju.com/?stella_menu=1&go=1
   되돌리기     https://stellasaju.com/?stella_menu=1&undo=1
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_menu'] ) ) { return; }
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	$META = '_stella_menu_backup';
	$go   = isset( $_GET['go'] );
	$undo = isset( $_GET['undo'] );

	$OLD  = '나의 사주풀이';
	$NEW  = '나의 사주풀이(기본)';
	$DROP = '로또';

	$safe = function ( $s ) {
		if ( function_exists( 'mb_convert_encoding' ) ) {
			$s = mb_convert_encoding( $s, 'UTF-8', 'UTF-8' );
		}
		return esc_html( $s );
	};

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:14px/1.75 -apple-system,system-ui,sans-serif;'
		. 'background:#111;color:#eee;padding:24px;max-width:1000px">';
	printf(
		'<h1 style="color:#ffd76a">메뉴 손보기 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);
	echo '<p style="color:#ff9b9b">브라우저 번역을 꺼주세요.</p>';

	/* ── 되돌리기 ── */
	if ( $undo ) {
		$ids = $wpdb->get_col( $wpdb->prepare(
			"SELECT post_id FROM {$wpdb->postmeta} WHERE meta_key = %s", $META ) );
		$n = 0;
		foreach ( $ids as $pid ) {
			$bak = get_post_meta( $pid, $META, true );
			if ( ! is_array( $bak ) ) { continue; }
			$wpdb->update(
				$wpdb->posts,
				array(
					'post_content' => $bak['c'],
					'post_title'   => $bak['t'],
				),
				array( 'ID' => $pid )
			);
			delete_post_meta( $pid, $META );
			clean_post_cache( $pid );
			$n++;
		}
		if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		printf( '<h2 style="color:#7bff9b">%d 군데를 되돌렸습니다.</h2></body>', $n );
		exit;
	}

	$rows = $wpdb->get_results(
		"SELECT ID, post_title, post_type, post_status, post_content
		   FROM {$wpdb->posts}
		  WHERE post_type IN ('wp_navigation','wp_template_part','wp_block','nav_menu_item')
		  ORDER BY post_type, ID"
	);

	$plan = array();

	foreach ( $rows as $r ) {
		$c     = $r->post_content;
		$t     = $r->post_title;
		$did   = array();
		$cuts  = array();

		/* ① 로또가 든 메뉴 링크 덩어리를 통째로 걷어냅니다 */
		$pos = 0;
		while ( ( $s = strpos( $c, '<!-- wp:navigation-link', $pos ) ) !== false ) {
			$e = strpos( $c, '/-->', $s );
			if ( false === $e ) { break; }
			$e   = $e + 4;
			$blk = substr( $c, $s, $e - $s );
			if ( false !== strpos( $blk, $DROP ) ) {
				$cuts[] = array( $s, $e - $s, $blk );
			}
			$pos = $e;
		}
		for ( $i = count( $cuts ) - 1; $i >= 0; $i-- ) {
			list( $at, $len, $blk ) = $cuts[ $i ];
			$b = $at;
			while ( $b > 0 && ( ' ' === $c[ $b - 1 ] || "\t" === $c[ $b - 1 ] || "\n" === $c[ $b - 1 ] ) ) { $b--; }
			$c   = substr( $c, 0, $b ) . substr( $c, $at + $len );
			$did[] = '로또 링크 뺌';
		}

		/* ② 이름 바꾸기 — 이미 (기본)이 붙어 있으면 건너뜁니다 */
		if ( false === strpos( $c, $NEW ) ) {
			if ( false !== strpos( $c, $OLD ) ) {
				$c     = str_replace( $OLD, $NEW, $c );
				$did[] = '이름 바꿈';
			}
		}

		/* 옛날 메뉴는 제목 자체가 이름입니다 */
		if ( 'nav_menu_item' === $r->post_type ) {
			if ( $t === $OLD ) { $t = $NEW; $did[] = '제목 바꿈'; }
		}

		if ( ! $did ) { continue; }
		$plan[] = array(
			'id'   => $r->ID,
			'ttl'  => $r->post_title,
			'type' => $r->post_type,
			'did'  => $did,
			'cuts' => $cuts,
			'c'    => $c,
			't'    => $t,
			'oc'   => $r->post_content,
			'ot'   => $r->post_title,
		);
	}

	if ( ! $plan ) {
		echo '<h2 style="color:#ff7b7b">고칠 것을 못 찾았습니다.</h2>';
		echo '<p>메뉴가 다른 곳에 저장되어 있을 수 있습니다. 아무것도 바꾸지 않았습니다.</p>';
		echo '<h2 style="color:#9fd">지금 있는 메뉴들</h2><table cellpadding="7" style="border-collapse:collapse">';
		foreach ( $rows as $r ) {
			printf(
				'<tr style="border-top:1px solid #333"><td>%d</td><td>%s</td>'
				. '<td style="color:#888">%s</td><td style="color:#888">%s 글자</td></tr>',
				$r->ID, esc_html( $r->post_title ), esc_html( $r->post_type ),
				number_format( strlen( $r->post_content ) )
			);
		}
		echo '</table></body>';
		exit;
	}

	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
	echo '<tr style="color:#ffd76a"><td>ID</td><td>이름</td><td>종류</td><td>하는 일</td></tr>';
	foreach ( $plan as $p ) {
		printf(
			'<tr style="border-top:1px solid #333"><td><b>%d</b></td><td>%s</td>'
			. '<td style="color:#888">%s</td><td style="color:#7bff9b">%s</td></tr>',
			$p['id'], esc_html( $p['ttl'] ), esc_html( $p['type'] ),
			esc_html( implode( ' · ', $p['did'] ) )
		);
		foreach ( $p['cuts'] as $cut ) {
			printf(
				'<tr><td colspan="4"><pre style="white-space:pre-wrap;background:#000;padding:9px;'
				. 'border-left:3px solid #ff7b7b;font:12px/1.5 monospace">뺄 것 · %s</pre></td></tr>',
				$safe( $cut[2] )
			);
		}
	}
	echo '</table>';

	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다. 아무것도 바뀌지 않았습니다.</h2>';
		echo '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_menu=1&go=1">'
			. '→ 이대로 바꾸기</a></p></body>';
		exit;
	}

	foreach ( $plan as $p ) {
		if ( ! get_post_meta( $p['id'], $META, true ) ) {
			update_post_meta( $p['id'], $META, array( 'c' => $p['oc'], 't' => $p['ot'] ) );
		}
		$wpdb->update(
			$wpdb->posts,
			array(
				'post_content' => $p['c'],
				'post_title'   => $p['t'],
			),
			array( 'ID' => $p['id'] )
		);
		clean_post_cache( $p['id'] );
	}
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<h2 style="color:#7bff9b">바꿨습니다</h2>';
	echo '<p style="font-size:17px"><a style="color:#ffd76a" href="/">홈에서 메뉴 확인하기</a></p>';
	echo '<p style="color:#888">되돌리려면 ?stella_menu=1&undo=1</p></body>';
	exit;
} );
