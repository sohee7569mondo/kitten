/* ════════════════════════════════════════════════════════════
   책에서 「복사하기」   ★ 2026-09-08

   소희 님 :「복사하기도 만들어줘」

   손님이 마음에 남은 대목을 복사해서 친구에게 보냅니다.
   광고비 한 푼 안 들이고 퍼집니다. 그리고 보내는 사람은
   자기 이야기를 나누는 것이라 기분도 좋습니다.

   ── 어디에 붙나 ────────────────────────────────
   책 전부가 아니라 **마음에 남는 두 자리**에만 붙입니다.

     ○○님, 세 가지만 기억하세요      맺음말
     ○○님께                          마지막 편지

   꼭지마다 붙이면 단추가 스무 개가 됩니다. 그러면 아무도
   안 누릅니다. 옮겨 적고 싶어지는 자리에만 놓습니다.

   ── 복사되는 모양 ──────────────────────────────
     하은님, 세 가지만 기억하세요

     (그 쪽 글 그대로)

     — 스텔라사주 · stellasaju.com

   맨 아래 한 줄이 같이 갑니다. 카톡으로 옮겨져도 어디서 온
   글인지 남습니다. 이게 이 단추를 다는 진짜 이유입니다.

   ── 어떻게 붙나 ────────────────────────────────
   글을 만드는 코드는 한 줄도 안 건드립니다. 책이 다 그려진
   뒤에 그 두 쪽을 찾아 단추를 얹습니다. 되돌리면 그대로
   돌아옵니다.

   ★ PDF 로 저장할 때는 단추가 안 나갑니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_copy=1
   ② 정말 넣기   https://stellasaju.com/?stella_copy=apply
   ③ 되돌리기    https://stellasaju.com/?stella_copy=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_copy'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_copy_backup';
	$MARK = '2026-09-08 · 복사하기';
	$mode = sanitize_key( wp_unslash( $_GET['stella_copy'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:880px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}.dim{color:#8B849C;font-size:.86rem}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '<h1 class="no">reading-book 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	if ( false !== strpos( $c, $MARK ) ) {
		echo '<h1 class="no">이미 들어가 있습니다.</h1>';
		echo '<p><code>/?stella_copy=undo</code> 로 걷어낸 뒤 다시 넣으세요.</p>';
		exit;
	}

	$TAIL = <<<'TAILX'

<style>
/* ══ 2026-09-08 · 복사하기 ═════════════════════════════════ */
#ssb .cpwrap{ margin:30px 0 0; text-align:center; }
#ssb .cpgo{
  display:inline-flex; align-items:center; gap:8px;
  padding:11px 22px; border-radius:8px; cursor:pointer;
  border:1px solid var(--rule); background:transparent;
  font:inherit; font-size:.9rem; color:var(--soft); line-height:1;
  transition:border-color .15s ease, color .15s ease, background .15s ease;
}
#ssb .cpgo:hover{ border-color:var(--gold); color:var(--star); }
#ssb .cpgo.on{ border-color:var(--gold); color:var(--gold); background:rgba(138,106,36,.07); }
#ssb .cpgo .ic{ font-size:1rem; line-height:1; }
@media print{ #ssb .cpwrap{ display:none !important; } }
</style>
<script>
/* ══ 2026-09-08 · 복사하기 ══
   마음에 남는 두 쪽에만 붙입니다. 글을 만드는 코드는 안 건드리고,
   다 그려진 뒤에 찾아서 얹습니다.
   ★ 이 안에서 홑 앰퍼샌드를 쓰지 않습니다 (사이트 규칙). */
(function(){
  var WANT = ['세 가지만 기억하세요', '님께'];
  var TAIL = '— 스텔라사주 · stellasaju.com';
  var tries = 0;

  function want(t){
    var i;
    for(i = 0; i < WANT.length; i++){
      if(t.indexOf(WANT[i]) >= 0){ return true; }
    }
    return false;
  }

  function copy(text, btn){
    function done(){
      btn.classList.add('on');
      btn.querySelector('.tx').textContent = '복사했습니다';
      setTimeout(function(){
        btn.classList.remove('on');
        btn.querySelector('.tx').textContent = '복사하기';
      }, 2200);
    }
    if(navigator.clipboard){
      navigator.clipboard.writeText(text).then(done).catch(function(){ old(); });
      return;
    }
    old();
    function old(){
      var ta = document.createElement('textarea');
      ta.value = text;
      ta.style.cssText = 'position:fixed;top:-9999px;left:-9999px';
      document.body.appendChild(ta);
      ta.select();
      try{ document.execCommand('copy'); done(); }catch(e){}
      document.body.removeChild(ta);
    }
  }

  function put(){
    var pages = document.querySelectorAll('#ssb .page');
    if(!pages.length){ return false; }
    var made = 0;
    var i;
    for(i = 0; i < pages.length; i++){
      var pg = pages[i];
      if(pg.getAttribute('data-cp') === '1'){ made = made + 1; continue; }
      var h = pg.querySelector('h2');
      if(!h){ continue; }
      if(!want(h.textContent)){ continue; }
      pg.setAttribute('data-cp','1');

      var wrap = document.createElement('div');
      wrap.className = 'cpwrap';
      var btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'cpgo';
      btn.innerHTML = '<span class="ic">⧉</span><span class="tx">복사하기</span>';
      wrap.appendChild(btn);

      (function(page, button){
        button.addEventListener('click', function(){
          var t = page.innerText || '';
          t = t.replace('복사하기', '').replace('복사했습니다', '');
          t = t.replace(/\n{3,}/g, '\n\n').trim();
          copy(t + '\n\n' + TAIL, button);
        });
      })(pg, btn);

      var folio = pg.querySelector('.folio');
      if(folio){ pg.insertBefore(wrap, folio); }
      else { pg.appendChild(wrap); }
      made = made + 1;
    }
    return made > 0;
  }

  var timer = setInterval(function(){
    tries = tries + 1;
    if(put()){ clearInterval(timer); return; }
    if(tries > 80){ clearInterval(timer); }
  }, 300);
})();
</script>
TAILX;

	$new = $c . $TAIL;

	echo '<h1>' . ( 'apply' === $mode ? '넣었습니다' : '넣으면 이렇게 됩니다 (아직 안 넣었습니다)' ) . '</h1>';
	echo '<h2>어디에</h2>';
	echo '<p class="ok">「○○님, 세 가지만 기억하세요」와 「○○님께」 두 쪽에만.</p>';
	echo '<p class="dim">꼭지마다 붙이면 단추가 스무 개가 됩니다. 그러면 아무도 안 누릅니다.</p>';
	echo '<h2>복사되는 모양</h2>';
	echo '<pre style="white-space:pre-wrap;background:rgba(255,255,255,.04);padding:12px 14px;'
	   . 'border-radius:8px;font-size:.86rem;line-height:1.8;color:#D8D2E8">'
	   . "하은님, 세 가지만 기억하세요\n\n(그 쪽 글 그대로)\n\n— 스텔라사주 · stellasaju.com</pre>";
	echo '<p class="dim">맨 아래 한 줄이 같이 갑니다. 카톡으로 옮겨져도 어디서 온 글인지 남습니다. '
	   . '이게 이 단추를 다는 진짜 이유입니다.</p>';
	echo '<p class="dim">' . number_format( strlen( $c ) ) . ' → '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( 'apply' !== $mode ) {
		echo '<p>정말 넣으시려면 <code>/?stella_copy=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );

	echo '<p class="ok">넣었습니다.</p>';
	echo '<p><a href="/reading-book/?demo=door-love">샘플 책에서 맨 끝으로 가보기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_copy=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
