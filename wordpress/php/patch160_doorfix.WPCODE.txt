/* ══════════════════════════════════════════════════════════
   문 단추를 정리하고, 주제가 책까지 전해지게 합니다
                                        ★ 2026-09-13 · 소희 님께
                                        판 DOORFIX-5

   ── 앞선 판이 알려준 것 (그래서 이렇게 고쳤습니다) ────
   ① 운세 문 단추는 **다섯**이고 이름 안에 가운뎃점이 있습니다 —
        신년운세 · 삼재 · 인생총운 · 초년 · 인생총운 · 중년 ·
        인생총운 · 말년
      즉 값이 이렇습니다 : 「신년운세」「삼재」
        「인생총운 · 초년」「인생총운 · 중년」「인생총운 · 말년」
      제 사본의 「대운 · 올해 총운 · 전환기 · 평생운」은 이미
      없어진 옛 이름이었습니다.
   ② 책에 `door-health` 라는 글자가 **두 군데** 있습니다.
      그래서 DOOR_TOPICS **표 안에서만** 바꿉니다.
   ③ 살아 있는 DOOR_TOPICS 에는 door-love 에 이혼운·결혼운(미혼)·
      결혼운(기혼)이 이미 들어 있습니다. 통째로 바꾸면 그것을
      덮어씁니다. **필요한 줄만 손댑니다.**

   ── 어떻게 바뀌나 ─────────────────────────────────────
   건강 문   여섯 → 둘
       건강운          →  체질·건강운
       체질            →  뺍니다 (체질·건강운에 합쳐집니다)
       가족운          →  ★ 그대로 둡니다. 이 단추는 손대지 않습니다.
       부모 · 자녀 · 형제운 → 뺍니다 (가족운 ②③④ 로 들어갑니다)

   운세 문   다섯 → 넷      ★ 소희 님이 정하신 차례 그대로
       2026년 운세 · 2027년 운세 · 인생총운 · 삼재

       어느 자리를 무엇으로 바꾸는지 —
       신년운세        →  2026년 운세 · 2027년 운세  (둘로)
       삼재            →  뺍니다 (맨 뒤로 옮깁니다)
       인생총운 · 초년  →  인생총운
       인생총운 · 중년  →  뺍니다   ┐ 셋을 인생총운 한 권으로 합칩니다.
       인생총운 · 말년  →  삼재     ┘ 초년만으로는 책이 얇습니다(337자)

       「삼재」는 이름을 안 바꿉니다 — 소희 님이 그렇게 적으셨습니다.

   책 표     door-health 줄을 바꾸고, door-fortune 줄을 더합니다
       'door-health':['체질·건강운','가족운']
       'door-fortune':['2026년 운세','2027년 운세','인생총운','삼재']

   ── 앵커 ──────────────────────────────────────────────
   단추는 한 줄에 하나라 단추마다 한 줄 정규식으로 잡습니다.
   DOOR_TOPICS 는 **표 덩어리를 먼저 찾고 그 안에서만** 줄을 고칩니다.
   ★ 열한 자리 가운데 하나라도 못 찾으면 아무것도 안 바꿉니다.

   ── 하는 법 ───────────────────────────────────────────
   ① 먼저 보기   https://stellasaju.com/?stella_doorfix=preview
   ② 정말 넣기   https://stellasaju.com/?stella_doorfix=apply
   ③ 되돌리기    https://stellasaju.com/?stella_doorfix=undo

   ★ 이미 넣어두신 조각에 **덮어쓰기** 하세요 (새로 만들면 겹칩니다).
   ★ 위치(Location)는 「어디서나 실행 / Run Everywhere」

   ── 검사한 것 ─────────────────────────────────────────
   php -l 통과 · 앰퍼샌드 0개 · 살아 있는 모양으로 만들어 걸어봄
   ══════════════════════════════════════════════════════════ */

function stella_doorfix_btn( $value ) {
	return '/<label class="topic"><input type="radio" name="ssg-topic" value="'
	     . preg_quote( $value, '/' ) . '"[^>]*>\s*<span>[^<]*<\/span>\s*<\/label>/u';
}
function stella_doorfix_make( $value, $checked ) {
	return '<label class="topic"><input type="radio" name="ssg-topic" value="' . $value . '"'
	     . ( $checked ? ' checked' : '' ) . '><span>' . $value . '</span></label>';
}

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_doorfix'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_doorfix'] ) );

	$PAGES = array(
		array( 'door-health',  '_stella_doorfix_bak_h', '건강 문' ),
		array( 'door-fortune', '_stella_doorfix_bak_f', '운세 문' ),
		array( 'reading-book', '_stella_doorfix_bak_b', '책 쪽' ),
	);

	echo "═══ 문 단추 정리 · 판 DOORFIX-5 ═══\n\n";

	if ( 'undo' === $mode ) {
		foreach ( $PAGES as $p ) {
			$pg = get_page_by_path( $p[0] );
			if ( ! $pg ) { continue; }
			$back = get_post_meta( (int) $pg->ID, $p[1], true );
			if ( '' === $back ) { echo $p[2] . " — 되돌릴 것이 없습니다\n"; continue; }
			$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => (int) $pg->ID ) );
			clean_post_cache( (int) $pg->ID );
			delete_post_meta( (int) $pg->ID, $p[1] );
			echo $p[2] . ' — 되돌렸습니다 (' . number_format( strlen( $back ) ) . "글자)\n";
		}
		exit;
	}

	$C = array();
	foreach ( $PAGES as $p ) {
		$pg = get_page_by_path( $p[0] );
		if ( ! $pg ) { echo $p[2] . ' 쪽을 못 찾았습니다.'; exit; }
		$id = (int) $pg->ID;
		$C[ $p[0] ] = array( 'id' => $id,
			'body' => (string) $wpdb->get_var( $wpdb->prepare(
				"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) ) );
		echo $p[2] . ' ' . $id . ' · ' . number_format( strlen( $C[ $p[0] ]['body'] ) ) . "글자\n";
	}
	echo "\n";

	echo "── 지금 달려 있는 단추 ──\n";
	foreach ( array( 'door-health' => '건강 문', 'door-fortune' => '운세 문' ) as $slug => $label ) {
		$got = preg_match_all( '/name="ssg-topic" value="([^"]*)"/u', $C[ $slug ]['body'], $mm );
		echo '  ' . $label . ' (' . (int) $got . ') :';
		if ( $got ) { foreach ( $mm[1] as $v ) { echo "\n      「" . $v . "」"; } }
		echo "\n";
	}
	echo "\n";

	$bad = 0;

	/* ── 건강 문 ── */
	echo "── 건강 문 단추 ──\n";
	$h     = $C['door-health']['body'];
	$hJobs = array(
		array( '건강운', stella_doorfix_make( '체질·건강운', true ) ),
		array( '체질',   '' ),
		array( '부모운', '' ),
		array( '자녀운', '' ),
		array( '형제운', '' ),
	);
	$hMade = $h;
	foreach ( $hJobs as $j ) {
		$pat = stella_doorfix_btn( $j[0] );
		$n   = preg_match_all( $pat, $hMade, $m );
		if ( 1 !== $n ) { printf( "  ✗ %-8s %d군데\n", $j[0], $n ); $bad++; continue; }
		printf( "  ○ %-8s %s\n", $j[0], ( '' === $j[1] ? '뺍니다' : '→ 체질·건강운' ) );
		$hMade = preg_replace( $pat, $j[1], $hMade, 1 );
	}

	/* ── 운세 문 ── */
	echo "\n── 운세 문 단추 ──\n";
	$f  = $C['door-fortune']['body'];
	$sp = "\n            ";
	/* 소희 님이 정하신 차례 — 2026 · 2027 · 인생총운 · 삼재
	   단추는 있던 자리에 그대로 그려지므로, 자리를 보고 짝을 지었습니다. */
	$fNew = stella_doorfix_make( '2026년 운세', true ) . $sp
	      . stella_doorfix_make( '2027년 운세', false );
	$fJobs = array(
		array( '신년운세',        $fNew,                                  '→ 2026년 운세 · 2027년 운세' ),
		array( '삼재',            '',                                     '뺍니다 (맨 뒤로 옮깁니다)' ),
		array( '인생총운 · 초년', stella_doorfix_make( '인생총운', false ), '→ 인생총운' ),
		array( '인생총운 · 중년', '',                                     '뺍니다' ),
		array( '인생총운 · 말년', stella_doorfix_make( '삼재', false ),    '→ 삼재' ),
	);
	$fMade = $f;
	foreach ( $fJobs as $j ) {
		$pat = stella_doorfix_btn( $j[0] );
		$n   = preg_match_all( $pat, $fMade, $m );
		if ( 1 !== $n ) { printf( "  ✗ %-14s %d군데\n", $j[0], $n ); $bad++; continue; }
		printf( "  ○ %-14s %s\n", $j[0], $j[2] );
		$fMade = preg_replace( $pat, $j[1], $fMade, 1 );
	}

	/* ── 책 표 · 덩어리를 찾고 그 안에서만 ── */
	echo "\n── 책의 DOOR_TOPICS ──\n";
	$b     = $C['reading-book']['body'];
	$bMade = $b;
	$all   = '/var\s+DOOR_TOPICS\s*=\s*\{.*?\}\s*;/su';
	$nAll  = preg_match_all( $all, $b, $mA );
	if ( 1 !== $nAll ) {
		echo '  ✗ 표를 ' . $nAll . "군데 찾았습니다\n";
		$bad++;
	} else {
		$tab = $mA[0][0];
		echo "  지금 표 :\n" . $tab . "\n\n";
		$ph  = '/\'door-health\'\s*:\s*\[[^\]]*\]/u';
		$nh  = preg_match_all( $ph, $tab, $mh );
		if ( 1 !== $nh ) {
			echo '  ✗ 표 안의 door-health 줄 ' . $nh . "군데\n";
			$bad++;
		} else {
			$tabNew = preg_replace( $ph,
				"'door-health':['체질·건강운','가족운'],\n"
				. "    'door-fortune':['2026년 운세','2027년 운세','인생총운','삼재']",
				$tab, 1 );
			echo "  ○ door-health   " . $mh[0][0] . "\n";
			echo "                  -> 체질·건강운 · 가족운\n";
			echo "  ○ door-fortune  줄을 새로 더합니다\n";
			echo "\n  바꾼 뒤 표 :\n" . $tabNew . "\n";
			$bMade = str_replace( $tab, $tabNew, $b );
		}
	}

	if ( $bad ) {
		echo "\n★ 못 찾은 자리가 " . $bad . "군데 있어 아무것도 안 바꿨습니다.\n";
		echo "이 화면을 그대로 주세요.\n";
		exit;
	}

	echo "\n── 크기 ──\n";
	echo '  건강 문 ' . number_format( strlen( $h ) ) . ' -> ' . number_format( strlen( $hMade ) ) . "\n";
	echo '  운세 문 ' . number_format( strlen( $f ) ) . ' -> ' . number_format( strlen( $fMade ) ) . "\n";
	echo '  책 쪽   ' . number_format( strlen( $b ) ) . ' -> ' . number_format( strlen( $bMade ) ) . "\n\n";

	if ( 'apply' !== $mode ) {
		echo "아직 안 바꿨습니다. 정말 넣으시려면 ?stella_doorfix=apply\n";
		exit;
	}

	$put = array(
		array( 'door-health',  $h, $hMade ),
		array( 'door-fortune', $f, $fMade ),
		array( 'reading-book', $b, $bMade ),
	);
	foreach ( $put as $i => $one ) {
		$id = $C[ $one[0] ]['id'];
		update_post_meta( $id, $PAGES[ $i ][1], wp_slash( $one[1] ) );
		$wpdb->update( $wpdb->posts, array( 'post_content' => $one[2] ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		echo $PAGES[ $i ][2] . " 넣었습니다\n";
	}

	echo "\n두 문을 열어 단추를 봐주세요 — 건강 문 둘, 운세 문 넷입니다.\n";
	echo "그리고 체질·건강운을 골라 책을 한 번 뽑아보세요.\n";
	exit;
} );
