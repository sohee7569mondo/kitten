/* ══════════════════════════════════════════════════════════
   문 단추를 정리하고, 주제가 책까지 전해지게 합니다
                                        ★ 2026-09-13 · 소희 님께
                                        판 DOORFIX-3

   ── 왜 필요한가 ───────────────────────────────────────
   가. door-health · door-fortune 은 단추는 있는데 책의 DOOR_TOPICS
       표에 없습니다. 그래서 손님이 골라도 **주제가 책에 전해지지
       않고** 문 단위 글 하나가 나옵니다.
   나. 단추가 여섯씩인데, 여섯을 다 쓰려면 책 여섯 권을 따로 써야
       합니다. 소희 님 구조를 보면 이미 한 권 안에 들어가 있습니다 —
       가족운 ②③④ 가 부모·형제·자녀이고, 체질·건강운 ①②③ 이 체질과
       건강을 갈라 봅니다. 따로 팔 필요가 없습니다.

   ── 어떻게 바뀌나 ─────────────────────────────────────
   door-health   여섯  →  둘
       건강운 · 체질            →  체질·건강운   (합칩니다)
       가족운                   →  가족운        (단추를 안 건드립니다)
       부모운 · 자녀운 · 형제운  →  가족운 ②③④ 로 들어갑니다

   ★ 판 3 — 운세 문은 이번에 안 건드립니다
     판 2 를 돌려보니 운세 문 단추 넷(대운 · 올해 총운 · 전환기 ·
     평생운)이 **0군데**였습니다. 살아 있는 쪽의 단추 이름이 제
     사본과 다릅니다. 짐작으로 고치면 또 헛돕니다.
     그래서 이 판은 **지금 단추 이름을 그대로 찍어 보여주고**,
     건강 문만 고칩니다. 운세 문은 그 이름을 보고 다음 판에서.

   그리고 책의 DOOR_TOPICS 에서 door-health 줄만 바꿉니다.
   (판 2 는 표를 통째로 바꾸려 했는데, 살아 있는 표에는 door-love 에
    이혼운·결혼운(미혼)·결혼운(기혼)이 이미 들어 있었습니다.
    통째로 바꾸면 그것을 제 옛 목록으로 덮어쓸 뻔했습니다.
    **한 줄만 바꿉니다.**)

   ── 앵커 ──────────────────────────────────────────────
   단추는 **한 줄에 하나**입니다. 그래서 단추마다 한 줄 정규식으로
   잡습니다. 공백을 흡수하고 checked 가 있든 없든 걸립니다.
   DOOR_TOPICS 는 표 하나를 통째로 바꿉니다 — 지금 무엇이 들어
   있는지 preview 가 그대로 찍어 보여드립니다.

   ★ 여섯 자리 가운데 하나라도 못 찾으면 **아무것도 안 바꿉니다.**
     단추만 바뀌고 표가 안 바뀌면 손님이 고른 주제가 책에서 막힙니다.

   ── 하는 법 ───────────────────────────────────────────
   ① 먼저 보기   https://stellasaju.com/?stella_doorfix=preview
   ② 정말 넣기   https://stellasaju.com/?stella_doorfix=apply
   ③ 되돌리기    https://stellasaju.com/?stella_doorfix=undo

   ★ 새 조각입니다. WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.

   ── 검사한 것 ─────────────────────────────────────────
   php -l 통과 · 앰퍼샌드 0개 · 사본에 실제로 걸어 열두 자리 확인
   ══════════════════════════════════════════════════════════ */

function stella_doorfix_btn( $value ) {
	return '/<label class="topic"><input type="radio" name="ssg-topic" value="'
	     . preg_quote( $value, '/' ) . '"[^>]*>\s*<span>[^<]*<\/span>\s*<\/label>/u';
}
function stella_doorfix_make( $value, $checked ) {
	return '<label class="topic"><input type="radio" name="ssg-topic" value="' . $value . '"'
	     . ( $checked ? ' checked' : '' ) . '><span>' . $value . '</span></label>';
}

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_doorfix'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_doorfix'] ) );

	$PAGES = array(
		array( 'door-health',  '_stella_doorfix_bak_h', '건강 문' ),
		array( 'door-fortune', '_stella_doorfix_bak_f', '운세 문' ),
		array( 'reading-book', '_stella_doorfix_bak_b', '책 쪽' ),
	);
	/* 판 3 — 실제로 고치는 것은 건강 문과 책 쪽 둘뿐입니다.
	   운세 문은 지금 단추 이름을 보여주기만 합니다. */
	$FIX = array(
		array( 'door-health',  '_stella_doorfix_bak_h', '건강 문' ),
		array( 'reading-book', '_stella_doorfix_bak_b', '책 쪽' ),
	);

	echo "═══ 문 단추 정리 · 판 DOORFIX-3 ═══\n\n";

	if ( 'undo' === $mode ) {
		foreach ( $FIX as $p ) {
			$pg = get_page_by_path( $p[0] );
			if ( ! $pg ) { continue; }
			$back = get_post_meta( (int) $pg->ID, $p[1], true );
			if ( '' === $back ) { echo $p[2] . " — 되돌릴 것이 없습니다\n"; continue; }
			$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => (int) $pg->ID ) );
			clean_post_cache( (int) $pg->ID );
			delete_post_meta( (int) $pg->ID, $p[1] );
			echo $p[2] . ' — 되돌렸습니다 (' . number_format( strlen( $back ) ) . "글자)\n";
		}
		exit;
	}

	/* ── 읽기 ── */
	$C = array();
	foreach ( $PAGES as $p ) {
		$pg = get_page_by_path( $p[0] );
		if ( ! $pg ) { echo $p[2] . ' 쪽을 못 찾았습니다.'; exit; }
		$id = (int) $pg->ID;
		$C[ $p[0] ] = array(
			'id'   => $id,
			'body' => (string) $wpdb->get_var( $wpdb->prepare(
				"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) ),
		);
		echo $p[2] . ' ' . $id . ' · ' . number_format( strlen( $C[ $p[0] ]['body'] ) ) . "글자\n";
	}
	echo "\n";

	/* ── 지금 단추가 무엇인지 그대로 찍습니다 ── */
	echo "── 지금 달려 있는 단추 ──\n";
	foreach ( array( 'door-health' => '건강 문', 'door-fortune' => '운세 문' ) as $slug => $label ) {
		$got = preg_match_all( '/name="ssg-topic" value="([^"]*)"/u', $C[ $slug ]['body'], $mm );
		echo '  ' . $label . ' (' . (int) $got . ') : '
		   . ( $got ? implode( ' · ', $mm[1] ) : '(없음)' ) . "\n";
	}
	echo "\n";

	$bad = 0;

	/* ── 건강 문 : 여섯 → 둘 ── */
	echo "── 건강 문 단추 ──\n";
	$h    = $C['door-health']['body'];
	/* ★ 가족운 단추는 **그냥 둡니다.** 새로 만들어 넣으면 원래 것과
	   둘이 되어 「2군데」로 걸립니다 (사본에 걸어보고 알았습니다). */
	$hNew  = stella_doorfix_make( '체질·건강운', true );
	$hJobs = array(
		array( '건강운', $hNew ),
		array( '체질',   '' ),
		array( '부모운', '' ),
		array( '자녀운', '' ),
		array( '형제운', '' ),
	);
	$hMade = $h;
	foreach ( $hJobs as $j ) {
		$pat = stella_doorfix_btn( $j[0] );
		$n   = preg_match_all( $pat, $hMade, $m );
		if ( 1 !== $n ) { printf( "  ✗ %-8s %d군데\n", $j[0], $n ); $bad++; continue; }
		printf( "  ○ %-8s %s\n", $j[0], ( '' === $j[1] ? '뺍니다' : '→ 체질·건강운' ) );
		$hMade = preg_replace( $pat, $j[1], $hMade, 1 );
	}

	/* ── 운세 문 : 이번 판은 손대지 않습니다 ── */
	echo "── 운세 문 ──\n";
	echo "  이번 판은 건드리지 않습니다. 위의 「지금 달려 있는 단추」를\n";
	echo "  그대로 보내주시면 다음 판에서 넷으로 줄이겠습니다.\n\n";
	$fMade = $C['door-fortune']['body'];

	/* ── 책 : DOOR_TOPICS 의 door-health 한 줄만 ── */
	echo "── 책의 DOOR_TOPICS ──\n";
	$b   = $C['reading-book']['body'];
	$all = '/var\s+DOOR_TOPICS\s*=\s*\{.*?\}\s*;/su';
	if ( preg_match( $all, $b, $mA ) ) {
		echo "  지금 표 :\n" . $mA[0] . "\n\n";
	}
	$pat = '/\'door-health\'\s*:\s*\[[^\]]*\]/u';
	$n   = preg_match_all( $pat, $b, $m );
	$bMade = $b;
	if ( 1 !== $n ) {
		echo '  ✗ door-health 줄 ' . $n . "군데\n";
		$bad++;
	} else {
		$to = "'door-health':['체질·건강운','가족운']";
		echo '  ○ door-health   ' . $m[0][0] . "\n";
		echo "                  -> " . $to . "\n";
		$bMade = preg_replace( $pat, $to, $b, 1 );
	}
	echo "\n";

	if ( $bad ) {
		echo "\n★ 못 찾은 자리가 " . $bad . "군데 있어 아무것도 안 바꿨습니다.\n";
		echo "이 화면을 그대로 주세요.\n";
		exit;
	}

	echo "── 크기 ──\n";
	echo '  건강 문 ' . number_format( strlen( $h ) ) . ' -> ' . number_format( strlen( $hMade ) ) . "\n";
	echo '  책 쪽   ' . number_format( strlen( $b ) ) . ' -> ' . number_format( strlen( $bMade ) ) . "\n\n";

	if ( 'apply' !== $mode ) {
		echo "아직 안 바꿨습니다. 정말 넣으시려면 ?stella_doorfix=apply\n";
		exit;
	}

	$put = array(
		array( 'door-health',  $h,  $hMade ),
		array( 'reading-book', $b,  $bMade ),
	);
	foreach ( $put as $i => $one ) {
		$slug = $one[0];
		$id   = $C[ $slug ]['id'];
		update_post_meta( $id, $FIX[ $i ][1], wp_slash( $one[1] ) );
		$wpdb->update( $wpdb->posts, array( 'post_content' => $one[2] ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		echo $FIX[ $i ][2] . " 넣었습니다\n";
	}

	echo "\n건강 문을 열어 단추가 둘인지 봐주세요.\n";
	echo "그리고 체질·건강운을 골라 책을 한 번 뽑아보세요.\n";
	exit;
} );
