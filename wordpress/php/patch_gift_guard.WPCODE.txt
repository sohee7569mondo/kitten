/* 같은 인터넷 자리에서 하루에 선물을 받을 수 있는 횟수 */
if ( ! defined( 'STELLA_GIFT_PER_IP_DAY' ) ) {
	define( 'STELLA_GIFT_PER_IP_DAY', 3 );
}

/** 지금 접속한 사람의 인터넷 자리 */
function stella_gift_ip() {
	$ip = '';
	if ( ! empty( $_SERVER['HTTP_X_FORWARDED_FOR'] ) ) {
		$parts = explode( ',', (string) $_SERVER['HTTP_X_FORWARDED_FOR'] );
		$ip    = trim( $parts[0] );
	}
	if ( '' === $ip && ! empty( $_SERVER['REMOTE_ADDR'] ) ) {
		$ip = (string) $_SERVER['REMOTE_ADDR'];
	}
	$ip = preg_replace( '/[^0-9a-fA-F:.]/', '', $ip );
	return substr( $ip, 0, 45 );
}

/** 그 자리에서 최근 하루 동안 선물이 몇 번 나갔나 */
function stella_gift_count( $ip ) {
	if ( '' === $ip ) {
		return 0;
	}
	$rec = get_transient( 'stella_gift_' . md5( $ip ) );
	if ( ! is_array( $rec ) ) {
		return 0;
	}
	$cut  = time() - DAY_IN_SECONDS;
	$keep = array();
	foreach ( $rec as $t ) {
		if ( (int) $t > $cut ) {
			$keep[] = (int) $t;
		}
	}
	return count( $keep );
}

/** 한 번 나갔다고 적어둡니다 */
function stella_gift_mark( $ip ) {
	if ( '' === $ip ) {
		return;
	}
	$key = 'stella_gift_' . md5( $ip );
	$rec = get_transient( $key );
	$rec = is_array( $rec ) ? $rec : array();
	$cut = time() - DAY_IN_SECONDS;

	$keep = array();
	foreach ( $rec as $t ) {
		if ( (int) $t > $cut ) {
			$keep[] = (int) $t;
		}
	}
	$keep[] = time();
	set_transient( $key, array_slice( $keep, -20 ), DAY_IN_SECONDS );
}


/* ── 새 계정이 만들어지는 순간 판단합니다 ──
   이 갈고리는 구슬을 넣기 바로 전에 울립니다. */
add_action( 'user_register', function ( $uid ) {

	$ip = stella_gift_ip();
	update_user_meta( $uid, 'stella_signup_ip', $ip );
	update_user_meta( $uid, 'stella_signup_at', time() );

	/* 관리자가 시험하는 것은 세지 않습니다 */
	if ( is_user_logged_in() && current_user_can( 'manage_options' ) ) {
		return;
	}

	if ( stella_gift_count( $ip ) >= (int) STELLA_GIFT_PER_IP_DAY ) {
		$GLOBALS['stella_gift_block'] = true;
		update_user_meta( $uid, 'stella_gift_blocked', 1 );
		return;
	}

	stella_gift_mark( $ip );
	update_user_meta( $uid, 'stella_gift_ok', 1 );
}, 1 );


/* ── 막기로 했으면 이번 요청 동안 구슬 장부에 아무것도 못 적습니다 ──
   워드프레스가 메타를 적기 직전에 물어보는 자리입니다.
   null 이 아닌 값을 돌려주면 적기를 건너뜁니다. */
add_filter( 'update_user_metadata', function ( $check, $uid, $key, $value, $prev ) {
	if ( empty( $GLOBALS['stella_gift_block'] ) ) {
		return $check;
	}
	if ( 'stella_orb_lots' !== $key ) {
		return $check;
	}
	return true;   /* 적은 척하고 넘어갑니다 */
}, 10, 5 );

add_filter( 'add_user_metadata', function ( $check, $uid, $key, $value, $unique ) {
	if ( empty( $GLOBALS['stella_gift_block'] ) ) {
		return $check;
	}
	if ( 'stella_orb_lots' !== $key ) {
		return $check;
	}
	return true;
}, 10, 5 );


/* ── 들여다보기 ── */
add_action( 'init', function () {

	if ( ! isset( $_GET['stella_gift'] ) ) {
		return;
	}
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:15px/1.8 -apple-system,system-ui,sans-serif;'
		. 'background:#111;color:#eee;padding:24px;max-width:980px">';
	echo '<h1 style="color:#ffd76a">가입 선물 지키기</h1>';

	printf(
		'<p>한 인터넷 자리에서 하루에 선물 <b style="color:#9fd">%d</b>번까지.<br>'
		. '지금 소희 님 자리 : <code>%s</code> · 최근 하루 <b>%d</b>번</p>',
		(int) STELLA_GIFT_PER_IP_DAY,
		esc_html( stella_gift_ip() ),
		stella_gift_count( stella_gift_ip() )
	);

	$rows = $wpdb->get_results(
		"SELECT u.ID, u.user_email, u.user_registered,
		        MAX(CASE WHEN m.meta_key='stella_signup_ip'     THEN m.meta_value END) AS ip,
		        MAX(CASE WHEN m.meta_key='stella_gift_blocked'  THEN m.meta_value END) AS blocked,
		        MAX(CASE WHEN m.meta_key='stella_gift_ok'       THEN m.meta_value END) AS ok
		   FROM {$wpdb->users} u
		   LEFT JOIN {$wpdb->usermeta} m ON m.user_id = u.ID
		  GROUP BY u.ID
		  ORDER BY u.user_registered DESC
		  LIMIT 40"
	);

	echo '<h2 style="color:#9fd">최근 가입하신 분들</h2>';
	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
	echo '<tr style="color:#ffd76a"><td>ID</td><td>메일</td><td>들어온 날</td>'
		. '<td>인터넷 자리</td><td>선물</td><td>구슬</td></tr>';
	foreach ( $rows as $r ) {
		$bal = function_exists( 'stella_orb_balance' ) ? stella_orb_balance( $r->ID ) : '-';
		$gift = $r->blocked ? '<span style="color:#ff7b7b">막음</span>'
			: ( $r->ok ? '<span style="color:#7bff9b">줌</span>' : '<span style="color:#888">-</span>' );
		printf(
			'<tr style="border-top:1px solid #333"><td>%d</td><td>%s</td><td style="color:#888">%s</td>'
			. '<td><code>%s</code></td><td>%s</td><td>%s</td></tr>',
			$r->ID,
			esc_html( $r->user_email ),
			esc_html( substr( (string) $r->user_registered, 0, 16 ) ),
			esc_html( (string) $r->ip ),
			$gift,
			esc_html( (string) $bal )
		);
	}
	echo '</table>';

	echo '<h2 style="color:#9fd">한 자리 풀어주기</h2>';
	echo '<p style="color:#888">진짜 손님이 막혔을 때만 쓰세요. 그 자리의 셈을 0으로 돌립니다.</p>';
	if ( isset( $_GET['free'] ) ) {
		$ip = preg_replace( '/[^0-9a-fA-F:.]/', '', (string) $_GET['free'] );
		if ( '' !== $ip ) {
			delete_transient( 'stella_gift_' . md5( $ip ) );
			printf( '<p style="color:#7bff9b">%s 를 풀었습니다.</p>', esc_html( $ip ) );
		}
	}
	echo '<form method="get"><input type="hidden" name="stella_gift" value="1">'
		. '<input name="free" placeholder="123.45.67.89" '
		. 'style="padding:8px;font:15px monospace;width:240px">'
		. ' <button style="padding:8px 14px">풀어주기</button></form>';

	echo '</body>';
	exit;
} );
