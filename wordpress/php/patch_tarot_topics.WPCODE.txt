/* ════════════════════════════════════════════════════════════
   스텔라타로 · 주제 여섯을 바꾸고, 카드가 주제마다 달라지게

   소희 님 —
     「스텔라 타로도 연애 직업 건강 운세로 갈까?
       그래야 꺼내는 카드가 다르지 않을까?」

   찾아보니 두 가지가 걸려 있었습니다.

   ── 하나 · 카드가 주제로 안 갈렸습니다 ─────────────
   열 장 가운데 세 장은 손님이 질문 화면에서 직접 고르십니다.
   그건 원래 다릅니다. 나머지 일곱 장은 이렇게 정해집니다.

       씨앗 = 생년월일시 + 오늘 날짜

   주제가 없습니다. 그래서 같은 날 연애로 보나 일로 보나
   그 일곱 장이 똑같이 나왔습니다. 씨앗에 주제를 더합니다.
   그러면 주제마다 열 장이 전부 달라집니다.

   ── 둘 · 주제 이름 여섯 ────────────────────────────
       오늘의 선택 · 그 사람 마음 · 일의 성패 · 시기 · 숨은 변수 · 예 아니오
         ↓
       연애 · 직업 · 건강 · 올해운 · 오늘의 선택 · 예 아니오

   앞의 넷은 사주 · 점성술과 같은 말로 맞춥니다. 손님은
   「무엇에 대해」를 먼저 정하고 「어떤 방법으로」를 나중에 고르니까요.
   뒤의 둘은 남깁니다 — 「예 아니오」는 타로만 할 수 있는 것이고,
   「오늘의 선택」은 오늘 하루짜리라 다른 문에 없는 자리입니다.

   ★ 이름을 바꿔도 해설이 사라지지 않는지 먼저 확인했습니다.
     타로 해설표(CARD_BRIDGE · COMBO_TOPIC)는 주제 이름을 열쇠로
     쓰지만, 그 표는 DOOR_TOPICS 에 있는 문에만 걸립니다.
     지금 DOOR_TOPICS 에는 door-love 와 door-career 둘뿐이라
     타로 문은 애초에 그 표를 타지 않습니다. 그래서 안전합니다.

   ── 두 쪽을 건드립니다 ─────────────────────────────
     페이지 160 (전자책)  — 씨앗 한 줄
     door-tarot (주문 쪽) — 주제 여섯

   둘 다 <script> 가 들어 있어서 wp_update_post 로 저장하면
   워드프레스가 지워버립니다. 데이터베이스에 바로 씁니다.

   ★ 한 군데라도 못 찾으면 아무것도 안 바꿉니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로 바꾸세요.

   ① 먼저 보기   https://stellasaju.com/?stella_tarot=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_tarot=apply
   ③ 되돌리기    https://stellasaju.com/?stella_tarot=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_tarot'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_tarot_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_tarot'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	/* ── 어느 쪽을 건드리는지 ──────────────────────────── */
	$book = 160;
	$door = get_page_by_path( 'door-tarot' );

	if ( ! $door ) {
		echo '<h1 class="no">door-tarot 쪽을 못 찾았습니다.</h1>';
		exit;
	}
	$doorId = $door->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$n = 0;
		foreach ( array( $book, $doorId ) as $pid ) {
			$old = get_post_meta( $pid, $META, true );
			if ( ! $old ) {
				echo '<p>' . intval( $pid ) . ' — 되돌릴 것이 없습니다.</p>';
				continue;
			}
			$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
			clean_post_cache( $pid );
			delete_post_meta( $pid, $META );
			echo '<p class="ok">' . intval( $pid ) . ' — '
			   . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
			$n++;
		}
		echo '<p>' . intval( $n ) . '쪽을 되돌렸습니다.</p>';
		exit;
	}

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	/* ── ① 전자책 · 씨앗에 주제 더하기 ────────────────── */
	$bookRow = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $book ) );
	if ( ! $bookRow ) {
		echo '<p class="no">전자책(160)을 못 읽었습니다.</p>';
		exit;
	}
	$bc = $bookRow->post_content;

	$seedFind = "    var seed=(P.year||0)*10007+(P.month||0)*331+(P.day||0)*97\n"
	          . "            +(P.hour||0)*13+(P.minute||0)*7\n"
	          . "            +seedNow.getFullYear()*1103+(seedNow.getMonth()+1)*37+seedNow.getDate()*3+11;";

	$seedTo = <<<'JS'
    /* 2026-09-08 · 씨앗에 주제를 더합니다.
       전에는 생년월일시와 오늘 날짜만 씨앗이었습니다. 그래서
       같은 날 연애로 보나 일로 보나 일곱 장이 똑같았어요.
       주제를 글자 하나씩 더해 수 하나로 만들고 씨앗에 섞습니다.
       이제 주제마다 열 장이 전부 달라집니다.
       같은 주제로 같은 날 다시 오시면 여전히 같은 열 장입니다 —
       그건 그대로 두는 것이 맞습니다. 하루에도 뽑을 때마다 달라지면
       카드가 아니라 요행이 되니까요. */
    var tSeed=0, tStr=String(topic||'');
    for(var tk=0; tk<tStr.length; tk++){
      tSeed=(tSeed*31+tStr.charCodeAt(tk))%99991;
    }
    var seed=(P.year||0)*10007+(P.month||0)*331+(P.day||0)*97
            +(P.hour||0)*13+(P.minute||0)*7
            +seedNow.getFullYear()*1103+(seedNow.getMonth()+1)*37+seedNow.getDate()*3+11
            +tSeed*769;
JS;

	echo '<h2>① 전자책 160 · 카드 씨앗</h2>';
	$sn = substr_count( $bc, $seedFind );
	if ( 1 !== $sn ) {
		echo '<p class="no">씨앗 자리를 ' . intval( $sn ) . '군데 찾았습니다. '
		   . '딱 한 군데여야 합니다. 아무것도 바꾸지 않았습니다.</p>';
		exit;
	}
	echo '<p class="ok">한 군데 — 씨앗에 주제를 더합니다.</p>';

	/* ── ② 주문 쪽 · 주제 여섯 ────────────────────────── */
	$doorRow = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $doorId ) );
	if ( ! $doorRow ) {
		echo '<p class="no">door-tarot 내용을 못 읽었습니다.</p>';
		exit;
	}
	$dc = $doorRow->post_content;

	$topicFind = '#<label class="topic"><input type="radio" name="ssg-topic" value="오늘의 선택".*?예 아니오</span></label>#s';

	$topicTo = '<label class="topic"><input type="radio" name="ssg-topic" value="연애" checked><span>연애</span></label>' . "\n"
	         . '            <label class="topic"><input type="radio" name="ssg-topic" value="직업"><span>직업</span></label>' . "\n"
	         . '            <label class="topic"><input type="radio" name="ssg-topic" value="건강"><span>건강</span></label>' . "\n"
	         . '            <label class="topic"><input type="radio" name="ssg-topic" value="올해운"><span>올해운</span></label>' . "\n"
	         . '            <label class="topic"><input type="radio" name="ssg-topic" value="오늘의 선택"><span>오늘의 선택</span></label>' . "\n"
	         . '            <label class="topic"><input type="radio" name="ssg-topic" value="예 아니오"><span>예 아니오</span></label>';

	echo '<h2>② 주문 쪽 door-tarot · 주제 여섯</h2>';
	$tn = preg_match_all( $topicFind, $dc );
	if ( 1 !== $tn ) {
		echo '<p class="no">주제 칸을 ' . intval( $tn ) . '군데 찾았습니다. '
		   . '딱 한 군데여야 합니다. 아무것도 바꾸지 않았습니다.</p>';
		exit;
	}
	echo '<p class="ok">한 군데 — <b>연애 · 직업 · 건강 · 올해운 · 오늘의 선택 · 예 아니오</b></p>';

	if ( ! $apply ) {
		echo '<h2>정리</h2>';
		echo '<p>두 자리 다 찾았습니다. 정말 바꾸시려면 '
		   . '<code>/?stella_tarot=apply</code> 를 여세요.</p>';
		exit;
	}

	/* ── 백업하고 쓰기 ────────────────────────────────── */
	$bn = str_replace( $seedFind, $seedTo, $bc );
	$dn = preg_replace( $topicFind, str_replace( '\\', '\\\\', $topicTo ), $dc, 1 );

	if ( null === $dn ) {
		echo '<p class="no">주제 칸을 바꾸다가 멈췄습니다. 아무것도 안 바꿨습니다.</p>';
		exit;
	}

	if ( ! get_post_meta( $book, $META, true ) ) {
		update_post_meta( $book, $META, $bc );
	}
	if ( ! get_post_meta( $doorId, $META, true ) ) {
		update_post_meta( $doorId, $META, $dc );
	}

	$wpdb->update( $wpdb->posts, array( 'post_content' => $bn ), array( 'ID' => $book ) );
	clean_post_cache( $book );
	$wpdb->update( $wpdb->posts, array( 'post_content' => $dn ), array( 'ID' => $doorId ) );
	clean_post_cache( $doorId );

	echo '<h2>정리</h2>';
	echo '<p class="ok">전자책 160 : ' . number_format( strlen( $bc ) ) . ' → '
	   . number_format( strlen( $bn ) ) . ' 바이트</p>';
	echo '<p class="ok">door-tarot : ' . number_format( strlen( $dc ) ) . ' → '
	   . number_format( strlen( $dn ) ) . ' 바이트</p>';
	echo '<p>확인해 보세요 — <a href="/door-tarot/">타로 문 열기</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_tarot=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
