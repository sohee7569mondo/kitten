  function batchim(w){
    w=String(w||'');
    if(!w.length){ return false; }
    var c=w.charCodeAt(w.length-1);
    if(c<0xAC00){ return false; }
    if(c>0xD7A3){ return false; }
    return ((c-0xAC00)%28)!==0;
  }
