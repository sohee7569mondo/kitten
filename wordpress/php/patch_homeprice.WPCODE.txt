   홈에는 값 얘기가 한 자도 없었습니다. 이제 무료라는 것을
   가장 먼저 말합니다 — 유입에는 이것보다 강한 문구가 없습니다.
   정가 9,000원에 줄을 그어 값어치를 함께 보입니다.

   ★ 짜이 있습니다 — 조각 patch_freeall
     그것이 실제로 값을 0 으로 만듭니다. 이 띠는 말만 합니다.
     띠만 붙이고 그것을 안 하시면 손님이 「무료」를 보고 들어가
     「구슬이 모자라요」로 막힙니다. 둘을 같이 하셔야 합니다.

   놓는 자리
   「스텔라사주보기」 제목 바로 아래 — 물음 카드 열두 장 앞입니다.

   앵커를 안 잡습니다
   id 가 saju 인 자리를 찾아 그 머리말 뒤에 띠 하나를 끼웁니다.
   쪽 원본은 한 글자도 안 고칩니다 — 끄면 그대로 돌아갑니다.

   여는 기간을 접으실 때는 아래 네 줄만 고치시면 됩니다.

   ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.

add_action( 'wp_head', function () {

	if ( ! is_front_page() ) { return; }

	/* ── 고치실 곳은 여기뿐입니다 ───────────────────────── */
	$WAS  = '9,000원';   /* 정가 — 가운데 줄을 그어 보입니다 */
	$NOW  = '무료';      /* 지금 내실 값 */
	$ORB  = '가입하지 않으셔도 됩니다';
	$SAY  = '오픈 기간에는 여섯 문을 마음껏 여셔도 값을 안 받습니다. '
		. '글은 줄이지 않았습니다 — 풀이 한 편이 책 한 권 그대로입니다.';
	$GO   = '2026년 운세';   /* 단추가 데려가는 주제 */
	$GAP  = 52;            /* 띄 아래 여백(px) — 더 띄우시려면 이 숫자만 */
	?>
<style id="stella-homeprice-css">
#hp-band{ margin:18px 0 52px; padding:18px 22px; border-radius:14px;
  background:var(--plum-soft,#F0EBFB); border:1px solid #DED3F4;
  display:flex; flex-wrap:wrap; align-items:center; gap:10px 18px; }
#hp-band .hp-kick{ font-size:.82rem; letter-spacing:.04em; font-weight:700;
  color:var(--plum,#6B4FB0); background:#fff; border-radius:999px;
  padding:5px 12px; border:1px solid #DED3F4; }
#hp-band .hp-price{ display:flex; align-items:baseline; gap:9px; }
#hp-band .hp-was{ color:var(--ink-dim,#8B849C); text-decoration:line-through;
  font-size:1rem; }
#hp-band .hp-now{ color:#C4453A; font-weight:800; font-size:1.42rem;
  letter-spacing:-.01em; }
#hp-band .hp-orb{ color:var(--ink-dim,#8B849C); font-size:.82rem; }
#hp-band .hp-say{ color:var(--ink-soft,#4E4763); font-size:.92rem;
  line-height:1.7; flex:1 1 260px; min-width:200px; }
#hp-band .hp-say b{ color:var(--ink,#221C33); font-weight:700; }
#hp-band .hp-go{ display:inline-block; text-decoration:none; font-weight:700;
  font-size:.92rem; color:#fff; background:var(--plum,#6B4FB0);
  border-radius:999px; padding:11px 20px; white-space:nowrap; }
#hp-band .hp-go:hover{ background:#5A3F9C; }
@media(max-width:640px){
  #hp-band{ padding:16px; gap:9px 14px; }
  #hp-band .hp-now{ font-size:1.26rem; }
  #hp-band .hp-go{ width:100%; text-align:center; }
}
</style>
<script>
(function(){
  if(window.StellaHomePrice){ return; }
  window.StellaHomePrice = 1;

  var WAS = <?php echo wp_json_encode( $WAS ); ?>;
  var NOW = <?php echo wp_json_encode( $NOW ); ?>;
  var ORB = <?php echo wp_json_encode( $ORB ); ?>;
  var SAY = <?php echo wp_json_encode( $SAY ); ?>;
  var GO  = <?php echo wp_json_encode( $GO ); ?>;
  var ADMIN = <?php echo current_user_can( 'manage_options' ) ? '1' : '0'; ?>;
  var GAP   = <?php echo (int) $GAP; ?>;

  function band(){
    var d = document.createElement('div');
    d.id = 'hp-band';

    var kick = document.createElement('span');
    kick.className = 'hp-kick';
    kick.textContent = '오픈기념';
    d.appendChild(kick);

    var p = document.createElement('div');
    p.className = 'hp-price';
    var w = document.createElement('span');
    w.className = 'hp-was';
    w.textContent = WAS;
    var n = document.createElement('span');
    n.className = 'hp-now';
    n.textContent = NOW;
    var o = document.createElement('span');
    o.className = 'hp-orb';
    o.textContent = ORB;
    p.appendChild(w); p.appendChild(n); p.appendChild(o);
    d.appendChild(p);

    var s = document.createElement('div');
    s.className = 'hp-say';
    s.textContent = SAY;
    d.appendChild(s);

    var a = document.createElement('a');
    a.className = 'hp-go';
    a.setAttribute('href', '/door-fortune/?topic=' + encodeURIComponent(GO));
    a.textContent = GO + '부터 보기 ' + String.fromCharCode(8250);
    d.appendChild(a);

    /* ★ 여백을 인라인으로 못 박습니다.
       살아 있는 홈에는 제 사본에 없는 CSS 가 있을 수 있어
       시험에서는 먹던 margin 이 살아서는 질 수 있습니다.
       인라인 + important 는 스타일표가 이길 수 없습니다. */
    d.style.setProperty('margin-top', '18px', 'important');
    d.style.setProperty('margin-bottom', GAP + 'px', 'important');

    return d;
  }

  /* ★ 관리자에게만 — 띄가 어디 들어갔고 여백이 얼마인지
     짐작하지 말고 재서 찍습니다. 손님에게는 안 보입니다. */
  function ruler(d){
    if(!ADMIN){ return; }
    if(document.getElementById('hp-ruler')){ return; }
    var box = document.createElement('div');
    box.id = 'hp-ruler';
    box.style.cssText = 'margin:6px 0 0;padding:7px 11px;border-radius:7px;'
      + 'background:#FFF6D8;border:1px solid #E2CE8A;color:#5B4A12;'
      + 'font-size:12px;line-height:1.7;word-break:break-all';

    function say(){
      var prev = d.previousElementSibling;
      var next = d.nextElementSibling;
      var r  = d.getBoundingClientRect();
      var cs = window.getComputedStyle(d);
      var t = '띄 자리 — 앞: ' + (prev ? (prev.className || prev.tagName) : '(없음)');
      t += '  ·  뒤: ' + (next ? (next.className || next.tagName) : '(없음)');
      if(prev){
        t += '  ·  위 여백 ' + Math.round(r.top - prev.getBoundingClientRect().bottom) + 'px';
      }
      if(next){
        t += '  ·  아래 여백 ' + Math.round(next.getBoundingClientRect().top - r.bottom) + 'px';
      }
      t += '  ·  재서 놓은 margin ' + cs.marginTop + ' / ' + cs.marginBottom;
      box.textContent = t;
    }

    /* ★ 띄 밖에 두면 자 자신이 여백을 밀어내 재는 것을
       망칩니다. 띄 안에 넣어 바깥 여백을 안 건드리게 합니다. */
    box.style.flex = '1 1 100%';
    d.appendChild(box);
    say();
    setTimeout(say, 900);
    window.addEventListener('resize', say);
  }

  function put(){
    if(document.getElementById('hp-band')){ return true; }
    var sec = document.getElementById('saju');
    if(!sec){ return false; }

    var d = band();

    /* ① 머리말(.secthead) 뒤가 제자리입니다 */
    var head = sec.querySelector('.secthead');
    if(head){
      if(head.parentNode){
        head.parentNode.insertBefore(d, head.nextSibling);
        ruler(d);
        return true;
      }
    }
    /* ② 머리말을 못 찾으면 첫 줄머리(.rowhead) 앞에 */
    var row = sec.querySelector('.rowhead');
    if(row){
      if(row.parentNode){
        row.parentNode.insertBefore(d, row);
        ruler(d);
        return true;
      }
    }
    /* ③ 그리도 안 되면 상자 맨 앞 */
    var wrap = sec.querySelector('.wrap');
    if(!wrap){ return false; }
    wrap.insertBefore(d, wrap.firstChild);
    ruler(d);
    return true;
  }

  if(put()){ return; }
  var n = 0;
  var t = setInterval(function(){
    n++;
    if(put()){ clearInterval(t); return; }
    if(n > 40){ clearInterval(t); }
  }, 200);
})();
</script>
	<?php
} );
