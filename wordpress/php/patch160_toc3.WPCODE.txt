/* ══════════════════════════════════════════════════════════
   차례 첫 줄에서 「여는 장」 빼기
   ★ 2026-09-10 · 소희 님께

   소희 님 : 「목차에서 여는 장 사주를 읽기 전에 →
   여는 장을 빼고 사주를 읽기 전에만」

   지금 차례 첫 줄이 이렇게 만들어집니다.

     var HEAD  = '여는 장';
     var TITLE = '사주를 읽기 전에';
     s.textContent = HEAD;                       ← 여는 장
     t.appendChild(... ' ' + TITLE);             ← 사주를 읽기 전에

   HEAD 를 비웁니다. 다른 묶음 이름(1부 · 2부 …)은 딴 자리에서
   만들어지니 그대로 남습니다.

     전   여는 장 사주를 읽기 전에 ......... 02
     후   사주를 읽기 전에 ................. 02

   해보기   https://stellasaju.com/?stella_toc3=preview
   붙이기   https://stellasaju.com/?stella_toc3=apply
   되돌리기 https://stellasaju.com/?stella_toc3=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_toc3'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_toc3'] ) );
	$META = '_stella_toc3_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	$OLD = "var HEAD = '여는 장';";
	$NEW = "var HEAD = '';";

	$n   = substr_count( $c, $OLD );
	$got = substr_count( $c, $NEW );

	echo "═══ 차례 첫 줄 ═══\n\n";
	echo '쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "글자\n\n";

	if ( 0 === $n ) {
		if ( $got > 0 ) { echo "이미 빠져 있습니다."; exit; }
		echo "★ 자리를 못 찾았습니다.\n";
		echo "  찾던 말 : " . $OLD . "\n";
		echo "  patch160_toc2 를 아직 안 붙이셨거나, 글이 달라졌습니다.\n";
		exit;
	}
	if ( 1 !== $n ) {
		echo "★ " . $n . "군데입니다. 하나여야 해서 아무것도 안 바꿨습니다.";
		exit;
	}

	echo "  찾음\n";
	echo "  전  여는 장 사주를 읽기 전에\n";
	echo "  후  사주를 읽기 전에\n";

	if ( 'preview' === $mode ) {
		echo "\n붙이시려면 ?stella_toc3=apply\n";
		exit;
	}

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		update_post_meta( $id, $META, $c );
	}
	$wpdb->update( $wpdb->posts,
		array( 'post_content' => str_replace( $OLD, $NEW, $c ) ),
		array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "\n고쳤습니다. 책에서 목차를 눌러 보세요.\n";
	echo "되돌리려면 ?stella_toc3=undo\n";
	exit;
}, 1 );
