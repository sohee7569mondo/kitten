/* ══════════════════════════════════════════════════════════
   1부 십성 — 다섯을 다 내보냅니다 (없는 것까지)
                                        ★ 2026-09-13 · 소희 님께
                                        판 LENS5-1

   소희 님 : 「내가 가지고 있는거 가지고 있지 않은 십성이 모두 있어야
              도움이 될거 같은데」

   ── 지금은 어떻게 되어 있나 ───────────────────────────
   1부의 십성 자리가 이렇게 짜여 있습니다 —

       groupRank.slice(0,3).map( … )

   여덟 글자를 세어 **많은 순으로 위 세 개만** 내보냅니다.
   그래서 주제별로 갈라 써둔 글이 있어도 **둘은 늘 안 나갑니다.**
   그리고 **한 자리도 없는 십성**은 아예 줄에 서지도 못합니다 —
   `it.groups` 에 그 이름 자체가 없기 때문입니다.

   ── 무엇으로 바뀌나 ───────────────────────────────────
   다섯을 고정으로 세워 놓고, 많은 순으로 줄 세워 **다 내보냅니다.**

       비겁이 3자리          ← 있는 것 (지금과 같습니다)
       재성이 2자리
       관성이 1자리
       ───────────────────
       여덟 글자에 한 자리도 없는 것       ← 여기서부터
       식상이 한 자리도 없습니다
       인성이 한 자리도 없습니다

   없는 것에는 한 줄을 앞세웁니다 —
   「없다는 것은 모자라다는 뜻이 아닙니다. 그 결이 당신에게 낯설다는
    뜻이에요. 낯선 자리는 몸에 배어 있지 않아 애써야 나오지만,
    애쓴 만큼 크게 늘어나는 자리이기도 합니다.」

   풀이 글은 **이미 주제별로 갈라 둔 그 글 그대로**입니다. 새로 쓰는
   것이 아니라, 써두고 안 나가던 것을 마저 내보내는 것입니다.

   ── ★ 여러 줄을 한 번에 찾습니다 (그 까닭) ────────────
   집 규칙은 「한 줄만 찾기」입니다. 그런데 이 자리는 네 줄이
   한 덩어리로 맞물려 있어 한 줄만 바꾸면 나머지가 어긋납니다.

       groupRank.slice(0,3).map(function(r){
         var txt=g.lens[r.g]; if(!txt) return '';
         return '(h3)'+r.g+'이 '+r.n+'자리(/h3)(p)'+txt+'(/p)';
       }).join('')

   그래서 **공백을 다 흡수하는 정규식**으로 「slice(0,3).map(」 에서
   「}).join('')」 까지를 한 덩이로 잡습니다. 들여쓰기나 줄바꿈이
   달라도 걸립니다. 글자 그대로 맞추지 않습니다.

   ── 하는 법 ───────────────────────────────────────────
   ① 먼저 보기   https://stellasaju.com/?stella_lens5=preview
                 (지금 그 자리 글을 그대로 찍어 보여드립니다)
   ② 정말 넣기   https://stellasaju.com/?stella_lens5=apply
   ③ 되돌리기    https://stellasaju.com/?stella_lens5=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.

   ── 검사한 것 ─────────────────────────────────────────
   php -l 통과 · 앰퍼샌드 0개 · node --check 통과
   node 로 실제 사주 셈을 넣어 돌려봄 — 다섯이 많은 순으로 나오고,
   없는 것이 뒤에 따로 서고, 글이 없는 십성은 건너뜁니다
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_lens5'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_lens5'] ) );
	$META = '_stella_lens5_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	echo "═══ 1부 십성 · 다섯을 다 내보냅니다 ═══\n\n";

	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "글자\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

	echo '책 쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "글자\n\n";

	if ( false !== strpos( $c, 'stellaLensFiveMake' ) ) {
		echo "이미 들어가 있습니다.\n";
		echo "다시 넣으시려면 ?stella_lens5=undo 로 걷어낸 뒤 하세요.\n";
		exit;
	}

	/* 공백을 다 흡수합니다 — 들여쓰기·줄바꿈이 달라도 걸립니다 */
	$pat = '/groupRank\s*\.\s*slice\(\s*0\s*,\s*3\s*\)\s*\.\s*map\('
	     . '\s*function\s*\(\s*r\s*\)\s*\{.*?\}\s*\)\s*\.\s*join\(\s*\'\'\s*\)/su';

	$hit = preg_match( $pat, $c, $m, PREG_OFFSET_CAPTURE );
	if ( ! $hit ) {
		echo "★ 그 자리를 못 찾았습니다. 아무것도 안 바꿨습니다.\n\n";
		$at = strpos( $c, 'groupRank' );
		if ( false !== $at ) {
			echo "── 지금 groupRank 가 있는 자리 (이대로 저에게 주세요) ──\n";
			echo substr( $c, max( 0, $at - 200 ), 1200 );
			echo "\n── 여기까지 ──\n";
		} else {
			echo "groupRank 라는 말조차 없습니다. 책이 많이 달라졌습니다.\n";
		}
		exit;
	}

	echo "── 지금 그 자리 글 ──\n";
	echo $m[0][0] . "\n\n";

	$to = stella_lens5_code();

	echo "── 바꿀 글 ──\n";
	echo $to . "\n\n";

	$made = preg_replace_callback( $pat, function () use ( $to ) { return $to; }, $c, 1 );
	if ( null === $made ) { echo '바꾸다 탈이 났습니다. 아무것도 안 바꿨습니다.'; exit; }

	echo '지금 ' . number_format( strlen( $c ) ) . '글자  ->  바꾼 뒤 '
	   . number_format( strlen( $made ) ) . "글자\n\n";

	if ( 'apply' !== $mode ) {
		echo "아직 안 바꿨습니다. 정말 넣으시려면 ?stella_lens5=apply\n";
		exit;
	}

	/* 백업은 지금 쪽으로 갱신합니다 (옛 백업으로 되돌리면 다른 조각이 날아갑니다) */
	update_post_meta( $id, $META, wp_slash( $c ) );

	$wpdb->update( $wpdb->posts, array( 'post_content' => $made ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "넣었습니다.\n";
	echo "책을 열어 확인해 주세요 (Ctrl+Shift+R)\n";
	exit;
} );

function stella_lens5_code() {
	/* 책을 그리는 덩어리 안에 끼우는 한 마디입니다.
	   it 와 g 는 그 자리에 이미 있으니 그대로 넘깁니다.
	   함수 자체는 아래 wp_head 조각이 온 쪽이 아는 이름으로 답니다. */
	return <<<'LENS5'
window.stellaLensFiveMake(it.groups, g.lens)
LENS5;
}

/* ── 셈하는 함수는 온 쪽이 아는 이름으로 따로 답니다 ────
   책 덩어리 안에 함수를 새로 만들려면 앵커가 하나 더 필요합니다.
   그래서 부르는 한 마디만 끼우고, 함수는 여기서 답니다. */
add_action( 'wp_head', function () {
	?>
<script>
/* ═══ 1부 십성 다섯 (LENS5-1) ═══
   책 안에서 부를 수 있도록 온 쪽이 아는 이름으로 답니다.
   it 와 g 는 부를 때 넘겨받지 않고, 책이 만들어 둔 것을 그대로 씁니다. */
(function(){
  var FIVE=['비겁','식상','재성','관성','인성'];

  window.stellaLensFiveMake=function(groups, lens){
    var all=[], i, r, txt, out='', zero=0;
    for(i=0;i<FIVE.length;i++){
      all.push({ g:FIVE[i], n:(groups[FIVE[i]]||0) });
    }
    all.sort(function(a,b){ return b.n-a.n; });

    for(i=0;i<all.length;i++){
      r=all[i];
      txt=lens[r.g];
      if(!txt){ continue; }
      if(r.n>0){
        out+='<h3>'+r.g+'이 '+r.n+'자리</h3><p>'+txt+'</p>';
        continue;
      }
      if(zero===0){
        out+='<hr><p class="dek">여덟 글자에 한 자리도 없는 것</p>';
      }
      zero++;
      out+='<h3>'+r.g+'이 한 자리도 없습니다</h3>'+
        '<p class="mini">없다는 것은 모자라다는 뜻이 아닙니다. '+
        '그 결이 당신에게 낯설다는 뜻이에요. 낯선 자리는 몸에 배어 있지 않아 '+
        '애써야 나오지만, 애쓴 만큼 크게 늘어나는 자리이기도 합니다. '+
        '아래는 그 결이 이 주제에서 무엇을 뜻하는지입니다.</p>'+
        '<p>'+txt+'</p>';
    }
    return out;
  };
})();
</script>
	<?php
}, 1 );
