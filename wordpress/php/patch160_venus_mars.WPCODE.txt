  /* ── 금성과 화성이 이루는 각 ─────────────────────────────── */
  function aspectOf(a,b){
    var d=Math.abs(wrap(a-b)); if(d>180){ d=360-d; }
    if(d<=8){ return { n:'겹침', d:d,
      say:'금성과 화성이 같은 자리에 겹쳐 있습니다. 끌리는 것과 밀어붙이는 것이 같은 방향이에요. 좋아하면 바로 움직이는 쪽이라 마음이 헷갈리는 일이 적습니다. 대신 한번 아니라고 정하면 그것도 그 자리에서 끝납니다.' }; }
    if(d>=172){ return { n:'마주 봄', d:d,
      say:'금성과 화성이 정면으로 마주 보고 있습니다. 끌리는 쪽과 실제로 움직이는 쪽이 반대예요. 좋아하는 사람 앞에서 오히려 퉁명스러워지거나, 갖고 싶은 것 앞에서 먼저 물러서는 일이 잦습니다. 이건 성격이 이상해서가 아니라 두 자리가 서로 반대편에 있어서입니다.' }; }
    if(d>=112){ if(d<=128){ return { n:'순한 각', d:d,
      say:'금성과 화성이 순하게 맞물려 있습니다. 좋아하는 것과 밀어붙이는 힘이 서로 거들어 줍니다. 원하는 것을 얻는 일에 무리한 힘이 안 듭니다. 대신 너무 쉬워서 절실함이 덜한 편이에요.' }; } }
    if(d>=82){ if(d<=98){ return { n:'부딪히는 각', d:d,
      say:'금성과 화성이 직각으로 부딪힙니다. 원하는 것과 그것을 얻는 방식이 서로 안 맞아요. 좋아해서 다가갔는데 상대는 밀어붙였다고 느끼는 일이 생깁니다. 마음이 아니라 속도의 문제입니다 — 한 박자만 늦추면 대부분 풀립니다.' }; } }
    if(d>=52){ if(d<=68){ return { n:'거드는 각', d:d,
      say:'금성과 화성이 서로 거드는 자리에 있습니다. 좋아하는 쪽으로 자연스럽게 힘이 붙습니다. 크게 애쓰지 않아도 하고 싶은 것과 하는 일이 대체로 같은 쪽을 봅니다.' }; } }
    return { n:'각 없음', d:d,
      say:'금성과 화성이 특별한 각을 이루지 않습니다. 두 자리가 서로 간섭하지 않는다는 뜻이에요. 좋아하는 것은 좋아하는 대로, 밀어붙이는 것은 밀어붙이는 대로 따로 굴러갑니다. 한쪽이 흔들려도 다른 쪽까지 무너지지는 않습니다.' };
  }
