/* ════════════════════════════════════════════════════════════
   아르카나를 주제 명단에 넣습니다   ★ 2026-09-09 · 소희 님께

   소희 님 :
     「카드 세 장 골랐는데 세 장 그림 설명만 있어. 예전엔 세 장에
       대한 설명이 있었는데」
     「연애 결혼 재물 등 주제를 묻기로 했잖아. 그래야 해당 카드를
       뽑아서 줄 수 있다고」

   ── 찾은 것 ────────────────────────────────────────
   책 안에 세 장을 묶어 읽는 표가 이미 있습니다. 그런데 빗장이
   이렇게 걸려 있었습니다.

     var DOOR_TOPICS={
       'door-love':  ['결혼운','연애운','재회운','이별운','짝사랑','궁합'],
       'door-career':['취업운','직장운','이직운','퇴사운','사업운','재물운']
     };
     function topicAllowed(s, t){ ... DOOR_TOPICS[s] 에 t 가 있나 ... }

   **아르카나가 이 명단에 없습니다.** 그래서 topicAllowed 가 늘
   false 이고, 잇는 한 줄(CARD_BRIDGE) · 세 장 묶어 읽기(COMBO_TOPIC) ·
   주제별 맺음말(CLOSE_TOPIC)이 전부 건너뛰어졌습니다.
   표를 새로 쓸 필요가 없습니다. 문을 명단에 넣기만 하면 됩니다.

   ── 이 스니펫이 하는 일 ────────────────────────────
   아르카나 문에 실제로 달려 있는 주제를 그대로 읽어서 명단에 넣습니다.
   제가 이름을 지어내지 않습니다 — 문에 있는 그대로 씁니다.
   그래야 이름이 어긋날 일이 없습니다.

   미리보기에서 주제마다 「표 있음 / 표 없음」을 보여드립니다.
   표가 없는 주제는 지금처럼 원래 글로 갑니다. 없어지지 않습니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   미리보기   https://stellasaju.com/?stella_tarot2=1
   넣기       https://stellasaju.com/?stella_tarot2=apply
   되돌리기   https://stellasaju.com/?stella_tarot2=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_tarot2'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_tarot2'] ) );
	$META = '_stella_tarot2_bak_0909';

	$A = <<<'STELLA_A'
'door-career':['취업운','직장운','이직운','퇴사운','사업운','재물운']
STELLA_A;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="google" content="notranslate">'
	   . '<style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.75;'
	   . 'padding:28px 18px;max-width:960px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.2rem;margin:0 0 10px}'
	   . 'h2{color:#9AD1A5;font-size:1rem;margin:26px 0 6px;'
	   . 'border-top:1px solid rgba(255,255,255,.12);padding-top:14px}'
	   . '.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:12px/1.7 monospace;border-left:3px solid #D4AF6A;'
	   . 'word-break:break-all}'
	   . '</style>';

	$book = get_page_by_path( 'reading-book' );
	$door = get_page_by_path( 'door-tarot' );
	if ( ! $book ) { echo '<p class="no">책 쪽이 없습니다.</p>'; exit; }
	if ( ! $door ) { echo '<p class="no">아르카나 쪽이 없습니다.</p>'; exit; }

	if ( 'undo' === $mode ) {
		echo '<h1>되돌립니다</h1>';
		$bak = get_post_meta( $book->ID, $META, true );
		if ( '' === $bak ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ),
			array( 'ID' => $book->ID ) );
		clean_post_cache( $book->ID );
		delete_post_meta( $book->ID, $META );
		if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		echo '<p class="ok">되돌렸습니다.</p>';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $book->ID ) );
	$d = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $door->ID ) );

	$go = ( 'apply' === $mode );
	echo '<h1>아르카나를 주제 명단에 — '
	   . ( $go ? '넣습니다' : '미리보기 (아직 안 넣음)' ) . '</h1>';

	/* ── 아르카나 문에 달린 주제를 그대로 읽습니다 ───── */
	$topics = array();
	$from   = 0;
	while ( true ) {
		$at = strpos( $d, 'ssg-topic', $from );
		if ( false === $at ) { break; }
		$from = $at + 1;
		$lt = strrpos( substr( $d, 0, $at ), '<' );
		if ( false === $lt ) { continue; }
		$gt = strpos( $d, '>', $at );
		if ( false === $gt ) { continue; }
		$tag = substr( $d, $lt, $gt - $lt );
		$vp  = strpos( $tag, 'value="' );
		if ( false === $vp ) { continue; }
		$ve  = strpos( $tag, '"', $vp + 7 );
		if ( false === $ve ) { continue; }
		$v = substr( $tag, $vp + 7, $ve - $vp - 7 );
		$v = trim( $v );
		if ( '' === $v ) { continue; }
		if ( in_array( $v, $topics, true ) ) { continue; }
		$topics[] = $v;
	}

	echo '<h2>아르카나 문에 달린 주제</h2>';
	if ( ! count( $topics ) ) {
		echo '<p class="no">주제 단추를 못 찾았습니다. '
		   . '아르카나 문에 주제 고르는 자리가 없는 것 같습니다.</p>';
		echo '<p class="dim">그렇다면 먼저 주제를 다는 일부터 해야 합니다.</p>';
		exit;
	}

	$out = '';
	foreach ( $topics as $t ) {
		$bridge = ( false !== strpos( $c, "'" . $t . "':{" ) );
		$out .= str_pad( $t, 20 )
		     . ( $bridge ? '표 있음 — 바로 살아납니다' : '표 없음 — 지금처럼 원래 글로 갑니다' )
		     . "\n";
	}
	echo '<pre>' . esc_html( $out ) . '</pre>';

	/* ── 명단에 넣습니다 ─────────────────────────────── */
	$n = substr_count( $c, $A );
	echo '<h2>고칠 자리</h2><pre>' . esc_html( $A ) . "\n\n"
	   . '찾은 개수  ' . (int) $n . '</pre>';

	if ( 1 !== $n ) {
		echo '<p class="no">자리가 하나가 아닙니다. 고치지 않았습니다.</p>';
		exit;
	}

	if ( false !== strpos( $c, "'door-tarot':[" ) ) {
		echo '<p class="ok">이미 명단에 들어 있습니다.</p>';
		echo '<p class="dim">되돌리시려면 /?stella_tarot2=undo 를 여세요.</p>';
		exit;
	}

	$list = array();
	foreach ( $topics as $t ) { $list[] = "'" . $t . "'"; }
	$B = $A . ",\n    'door-tarot':[" . implode( ',', $list ) . "]";

	echo '<h2>이렇게 바뀝니다</h2><pre>' . esc_html( $B ) . '</pre>';

	if ( false !== strpos( $B, '&' ) ) {
		echo '<p class="no">앰퍼샌드가 들어 있습니다. 고치지 않았습니다.</p>';
		exit;
	}

	if ( ! $go ) {
		echo '<h2>끝</h2><p class="dim">이대로 괜찮으시면 '
		   . '/?stella_tarot2=apply 를 여세요.</p>';
		exit;
	}

	$new = str_replace( $A, $B, $c );

	if ( '' === get_post_meta( $book->ID, $META, true ) ) {
		update_post_meta( $book->ID, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ),
		array( 'ID' => $book->ID ) );
	clean_post_cache( $book->ID );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<h2>끝</h2><p class="ok">넣었습니다. '
	   . number_format( strlen( $c ) ) . ' -> '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p class="dim">아르카나로 다시 뽑아보세요. '
	   . '되돌리시려면 /?stella_tarot2=undo 를 여세요.</p>';
	exit;
} );
