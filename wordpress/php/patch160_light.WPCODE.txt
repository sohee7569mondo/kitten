/* ── 하나 · 날것으로 보기 (시험) ───────────────────────── */
add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_raw'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", (int) $page->ID ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<!doctype html><html lang="ko"><head><meta charset="utf-8">';
	echo '<meta name="viewport" content="width=device-width,initial-scale=1">';
	echo '<title>스텔라사주 — 날것으로 보기</title></head><body>';
	echo $c;
	echo '<script>console.log("stella raw ok");</script>';
	echo '</body></html>';
	exit;
}, 0 );

/* ── 둘 · 책 쪽에서 the_content 손길을 비웁니다 ─────────── */
add_action( 'wp', function () {

	if ( is_admin() ) { return; }
	if ( ! is_singular() ) { return; }
	$p = get_post();
	if ( ! $p ) { return; }
	if ( 'reading-book' !== $p->post_name ) { return; }

	global $wp_filter;
	if ( isset( $wp_filter['the_content'] ) ) {
		unset( $wp_filter['the_content'] );
	}
	/* 발췌문에도 같은 일이 벌어집니다 */
	if ( isset( $wp_filter['the_excerpt'] ) ) {
		unset( $wp_filter['the_excerpt'] );
	}
}, 999 );
