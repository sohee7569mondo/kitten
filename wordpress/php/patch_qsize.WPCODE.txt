/* ════════════════════════════════════════════════════════════
   배너 글씨 키우기   patch_qsize
   ★ 2026-09-15 · 소희 님께

   소희 님 : 「배너에 들어간 타이틀 "이 사람과 이어질 수 있을까요?"
              이 흰글씨들 글자 크기를 240프로 정도 키울수 있어?
              특히 무료 배너는 사진이 배경같은 스타일이라서 더 잘
              어울릴거야」

   지금 크기
     주제 배너 질문 (.ask-card .q)   1.02rem
     무료 배너 제목 (.free-sub)      1.12rem
     무료 배너 윗줄 (.free-kicker)   0.75rem

   240% 로 키우면 2.45rem · 2.69rem 입니다. 카드가 세로로 긴 네모라
   서너 줄이 될 수 있어, 화면에서 눌러보며 맞추시게 했습니다.

        https://stellasaju.com/?stella_qsize=1
          140 · 170 · 200 · 240 · 280 · 원래대로

   ★ 큰 글씨에 맞춰 같이 손봅니다
     · 줄 간격을 조금 좁힙니다 (크면 1.5 가 너무 벌어집니다)
     · 글씨 그림자를 넣습니다 — 사진 위 흰 글씨는 그림자가 있어야
       밝은 자리에서도 읽힙니다
     · 카드 아래 어둠(그라데이션)을 조금 더 올립니다

   ★ 끄면 그 즉시 옛 크기입니다. 쪽을 안 고칩니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ════════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {
	$p = (int) get_option( 'stella_qsize', 240 );
	if ( $p < 100 ) { return; }          /* 100 이면 안 건드립니다 */
	if ( $p > 400 ) { $p = 400; }
	$k = $p / 100;

	$q  = round( 1.02 * $k, 2 );          /* 주제 배너 질문 */
	$fs = round( 1.12 * $k, 2 );          /* 무료 배너 제목 */
	$fk = round( 0.75 * ( 1 + ( $k - 1 ) * 0.45 ), 2 );   /* 윗줄은 덜 키웁니다 */
	/* 폰에서는 조금 작게 — 카드가 좁아 글씨가 넘칩니다 */
	$qm  = round( $q  * 0.78, 2 );
	$fsm = round( $fs * 0.78, 2 );
	?>
<!-- stella 배너 글씨 · 판 2026-09-15 · <?php echo (int) $p; ?>% -->
<style id="stella-qsize-css">
#stellar-home .ask-card .q{
  font-size:<?php echo $q; ?>rem !important;
  line-height:1.28 !important;
  letter-spacing:-.02em;
  text-shadow:0 2px 10px rgba(10,6,26,.75), 0 1px 3px rgba(10,6,26,.6); }
#stellar-home .ask-card .info{ padding:20px 15px 18px !important; }
/* 글씨가 커진 만큼 아래 어둠을 조금 더 올려 읽히게 합니다 */
#stellar-home .ask-card .imgwrap::after{
  background:linear-gradient(to top, rgba(16,11,34,.96) 10%, rgba(16,11,34,.62) 52%,
             rgba(16,11,34,.06) 84%) !important; }
#stellar-home .free-sub{
  font-size:<?php echo $fs; ?>rem !important;
  line-height:1.3 !important;
  letter-spacing:-.02em; }
#stellar-home .free-kicker{ font-size:<?php echo $fk; ?>rem !important; }
/* 무료 배너는 사진이 배경이라 글씨가 사진 위에 놓입니다 — 그림자를 줍니다 */
#stellar-home .free-card .free-sub,
#stellar-home .free-card .free-line{
  text-shadow:0 2px 10px rgba(10,6,26,.6), 0 1px 3px rgba(10,6,26,.45); }
@media(max-width:820px){
  #stellar-home .ask-card .q{ font-size:<?php echo $qm; ?>rem !important; }
  #stellar-home .free-sub{ font-size:<?php echo $fsm; ?>rem !important; }
}
</style>
	<?php
} );

/* ── 크기 고르는 화면 ─────────────────────────────────── */
add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_qsize'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}
	if ( isset( $_GET['set'] ) ) {
		check_admin_referer( 'stella_qsize' );
		update_option( 'stella_qsize', (int) $_GET['set'] );
		wp_safe_redirect( add_query_arg( 'stella_qsize', '1', home_url( '/' ) ) );
		exit;
	}

	$now = (int) get_option( 'stella_qsize', 240 );
	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<!doctype html><meta charset="utf-8">';
	echo '<meta name="viewport" content="width=device-width,initial-scale=1">';
	echo '<title>배너 글씨 크기</title>';
	echo '<style>
	 body{font:15px/1.7 system-ui,sans-serif; margin:0; background:#14112b;
	      color:#F6F1E7; padding:20px;}
	 h1{font-size:19px; margin:0 0 6px;}
	 .dim{opacity:.72; font-size:13px;}
	 .btn{display:inline-block; padding:10px 18px; border-radius:8px; margin:8px 8px 0 0;
	      background:#C79A4A; color:#231a08; text-decoration:none; font-weight:700;}
	 .btn.gray{background:#3a3268; color:#F6F1E7;}
	 .demo{margin:20px 0; padding:0; border-radius:14px; overflow:hidden;
	       background:#100B22; max-width:300px;}
	 .demo .pic{height:190px; background:linear-gradient(160deg,#3b2b6b,#1a1030);
	       position:relative;}
	 .demo .pic::after{content:""; position:absolute; inset:0;
	       background:linear-gradient(to top, rgba(16,11,34,.96) 10%, rgba(16,11,34,.62) 52%,
	                  rgba(16,11,34,.06) 84%);}
	 .demo .info{position:relative; margin-top:-90px; padding:20px 15px 18px; z-index:3;}
	 .demo .q{font-family:"Noto Serif KR",serif; font-weight:700; line-height:1.28;
	       letter-spacing:-.02em; color:#FFFDF9;
	       text-shadow:0 2px 10px rgba(10,6,26,.75), 0 1px 3px rgba(10,6,26,.6);}
	</style>';

	echo '<h1>배너 글씨 크기</h1>';
	echo '<p class="dim">지금 <b>' . $now . '%</b>. 눌러서 바꾸고 홈에서 보세요. '
	   . '카드가 세로로 긴 네모라 너무 키우면 넉 줄이 됩니다.</p>';

	foreach ( array( 100, 140, 170, 200, 240, 280 ) as $k ) {
		$go = wp_nonce_url( add_query_arg(
			array( 'stella_qsize' => '1', 'set' => $k ), home_url( '/' ) ), 'stella_qsize' );
		$on = ( $now === $k );
		echo '<a class="btn' . ( $on ? '' : ' gray' ) . '" href="' . esc_url( $go ) . '">'
		   . ( 100 === $k ? '원래대로' : $k . '%' ) . ( $on ? ' ✓' : '' ) . '</a>';
	}

	$q = round( 1.02 * ( $now / 100 ), 2 );
	echo '<div class="demo"><div class="pic"></div>'
	   . '<div class="info"><div class="q" style="font-size:' . $q . 'rem">'
	   . '이 사람과 이어질 수 있을까요?</div></div></div>';
	echo '<p class="dim">위는 넓은 화면에서 보이는 크기입니다 (' . $q . 'rem). '
	   . '폰에서는 78%로 줄여 넘치지 않게 합니다.</p>';
	echo '<a class="btn gray" href="' . esc_url( home_url( '/' ) ) . '">홈 보기</a>';
	exit;
} );
