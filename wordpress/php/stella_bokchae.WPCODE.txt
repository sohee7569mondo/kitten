/* ── ① 받는 문 · 내주는 문 ───────────────────────────── */
add_action( 'rest_api_init', function () {

	register_rest_route( 'stella/v1', '/bokchae', array(
		'methods'             => 'GET',
		'permission_callback' => '__return_true',
		'callback'            => function ( $req ) {
			$slug = sanitize_key( (string) $req->get_param( 'page' ) );
			if ( '' === $slug ) { return new WP_REST_Response( array( 'ok' => false ), 400 ); }
			$n = (int) get_option( 'stella_bokchae_' . $slug, 0 );
			return new WP_REST_Response( array( 'ok' => true, 'n' => $n ), 200 );
		},
	) );

	register_rest_route( 'stella/v1', '/bokchae', array(
		'methods'             => 'POST',
		'permission_callback' => '__return_true',
		'callback'            => function ( $req ) {

			$b    = $req->get_json_params();
			$slug = isset( $b['page'] ) ? sanitize_key( (string) $b['page'] ) : '';
			if ( '' === $slug ) { return new WP_REST_Response( array( 'ok' => false ), 400 ); }

			$who = md5( ( isset( $_SERVER['REMOTE_ADDR'] ) ? $_SERVER['REMOTE_ADDR'] : '' ) . '|' . $slug );

			if ( get_transient( 'stella_bk_' . $who ) ) {
				$n = (int) get_option( 'stella_bokchae_' . $slug, 0 );
				return new WP_REST_Response( array( 'ok' => true, 'n' => $n, 'again' => true ), 200 );
			}

			$key = 'stella_bokchae_' . $slug;
			$n   = (int) get_option( $key, 0 ) + 1;
			update_option( $key, $n, false );

			set_transient( 'stella_bk_' . $who, 1, DAY_IN_SECONDS );

			return new WP_REST_Response( array( 'ok' => true, 'n' => $n ), 200 );
		},
	) );
} );

/* ── ② 센 것 보기 ────────────────────────────────────── */
add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_bokchae'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	$pages = array(
		'today'          => '오늘의 운세',
		'saju'           => '나의 사주풀이',
		'weekly-summary' => '이번주 사주 총평',
		'zodiac'         => '이번주 별자리 운세',
		'zodiac-year'    => '이번주 띠별운세',
		'lotto'          => '로또 번호 추첨',
		'oracle'         => '나에게 필요한 한마디',
		'stars'          => '이번주 스타운세',
	);

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.9;padding:34px 24px;max-width:620px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}'
	   . '.r{display:flex;justify-content:space-between;padding:10px 0;'
	   . 'border-bottom:1px solid rgba(212,175,106,.16)}'
	   . '.n{font-family:ui-monospace,monospace;color:#F0D9A8}'
	   . '.dim{color:#8B849C;font-size:.86rem}</style>';

	echo '<h1>복채</h1>';
	$sum = 0;
	foreach ( $pages as $slug => $name ) {
		$n = (int) get_option( 'stella_bokchae_' . $slug, 0 );
		$sum += $n;
		echo '<div class="r"><span>' . esc_html( $name ) . '</span>'
		   . '<span class="n">' . number_format( $n ) . '</span></div>';
	}
	echo '<div class="r"><b>모두</b><span class="n">' . number_format( $sum ) . '</span></div>';
	echo '<p class="dim">한 기기에서 한 쪽에 하루 한 번까지 셉니다.</p>';
	exit;
}, 0 );

add_filter( 'the_content', function ( $content ) {

	if ( ! is_page() ) { return $content; }
	if ( ! in_the_loop() ) { return $content; }
	if ( ! is_main_query() ) { return $content; }

	             'zodiac-year', 'lotto', 'oracle', 'stars' );

	$slug = get_post_field( 'post_name', get_the_ID() );
	if ( ! in_array( $slug, $ok, true ) ) { return $content; }

	$s = esc_attr( $slug );

	$box = <<<'BOXX'

}
.bokchae .bk-t{ font-size:1rem; color:#221C33; margin:0 0 16px; }
.bokchae .bk-go{
  display:inline-flex; align-items:center; gap:9px;
  padding:12px 26px; border-radius:40px; cursor:pointer;

  font:inherit; font-size:1.02rem; line-height:1;
  transition:transform .15s ease, background .15s ease, border-color .15s ease;
}

.bokchae .bk-go:disabled{ cursor:default; transform:none; opacity:1; }

@media print{ .bokchae{ display:none !important; } }
</style>
<div class="bokchae">
  <p class="bk-t">잘 보셨으면 복채 한 번</p>
  <button type="button" class="bk-go" id="bkGo"><span class="h">&#9825;</span><span class="bk-n" id="bkN"></span></button>

</div>
<script>
/* ══ 복채 한 번 ══
   ★ 이 안에서 홑 앰퍼샌드를 쓰지 않습니다 (사이트 규칙). */
(function(){
  var SLUG = '%%SLUG%%';
  var go = document.getElementById('bkGo');
  var nn = document.getElementById('bkN');
  var dd = document.getElementById('bkD');
  if(!go){ return; }

  var mine = false;
  try{ if(localStorage.getItem('stella_bokchae_' + SLUG) === '1'){ mine = true; } }catch(e){}

  function paint(n){
    if(n > 9){ nn.textContent = Number(n).toLocaleString('ko-KR'); }
    else { nn.textContent = ''; }
  }
  function thanks(){
    go.classList.add('on');
    go.disabled = true;
    dd.textContent = '고맙습니다. 잘 받았습니다.';
  }

  fetch('/wp-json/stella/v1/bokchae?page=' + SLUG, { credentials:'same-origin' })
    .then(function(r){ return r.json(); })
    .then(function(j){ if(j.ok){ paint(j.n); } })
    .catch(function(){});

  if(mine){ thanks(); return; }

  go.addEventListener('click', function(){
    go.disabled = true;
    fetch('/wp-json/stella/v1/bokchae', {
      method:'POST',
      headers:{ 'Content-Type':'application/json' },
      credentials:'same-origin',
      body: JSON.stringify({ page: SLUG })
    })
    .then(function(r){ return r.json(); })
    .then(function(j){
      if(j.ok){
        paint(j.n);
        try{ localStorage.setItem('stella_bokchae_' + SLUG, '1'); }catch(e){}
        thanks();
        return;
      }
      go.disabled = false;
    })
    .catch(function(){ go.disabled = false; });
  });
})();
</script>
BOXX;

	$box = str_replace( '%%SLUG%%', $s, $box );

	return $content . $box;
}, 20 );
