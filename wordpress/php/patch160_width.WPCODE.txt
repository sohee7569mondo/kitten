/* ══════════════════════════════════════════════════════════
   책의 폭 — 이 조각 하나가 정합니다
   ★ 2026-09-12 · 소희 님께      판 WIDTH-1

   소희 님 : 「그냥 페이지 폭 자체를 지정할 수는 없어? 가지각색이야」

   맞습니다. 지금은 쪽마다 시작하는 자리가 다릅니다. 어떤 쪽은
   왼쪽 끝에서, 어떤 쪽은 한참 안쪽에서 시작합니다.

   ── 이 조각이 하는 일 ─────────────────────────────────
   책의 모든 쪽을 **한 폭으로 묶고 가운데에 놓습니다.**
   딱 그 일만 합니다. 다른 것은 건드리지 않습니다.

       쪽 전체      900px
       좌우 여백    78px (여백 조각이 정합니다)
       글줄         744px — 한 줄에 마흔 자 안팎

   창이 960px 보다 좁으면(휴대폰·좁은 창) 화면을 다 씁니다.

   ── 숫자를 바꾸고 싶으시면 ────────────────────────────
   아래 한 줄의 900 만 고치시면 됩니다. 그 한 줄이 책 전체입니다.

       $wide = 900;

   ── 들어갔는지 확인하는 법 ────────────────────────────
   책 쪽에서 Ctrl+U -> Ctrl+F 로 「폭 판」 을 찾으세요.

       <!-- stella 폭 판 WIDTH-1 · 900px -->

   이 줄이 보이면 들어간 것입니다. 없으면 조각이 안 켜졌거나
   위치(Location)가 「어디서나 실행」이 아닙니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요. 코드에 「함수」「반환합니다」가
     보이면 번역된 것입니다.
   ══════════════════════════════════════════════════════════ */

if ( ! function_exists( 'stella_width_css' ) ) {

add_action( 'wp_head', function () {

	if ( ! is_singular() ) { return; }
	$p = get_post();
	if ( ! $p ) { return; }
	if ( 'reading-book' !== $p->post_name ) { return; }

	echo stella_width_css();
}, 99 );

function stella_width_css() {

	$wide = 900;      /* ★ 이 숫자 하나가 책 전체 폭입니다 */

	$out  = "\n<!-- stella 폭 판 WIDTH-1 · " . $wide . "px -->\n";
	$out .= '<style id="stella-width">' . "\n";

	/* 모든 쪽을 같은 폭으로, 가운데에.
	   테두리와 여백이 폭 안에 들어가도록 box-sizing 도 맞춥니다. */
	$out .= "#ssb .page{\n";
	$out .= '  max-width:' . $wide . "px !important;\n";
	$out .= "  margin-left:auto !important;\n";
	$out .= "  margin-right:auto !important;\n";
	$out .= "  box-sizing:border-box !important;\n";
	$out .= "  width:auto !important;\n";
	$out .= "}\n";

	/* 쪽을 담는 칸이 좁게 묶여 있으면 가운데 정렬이 안 보입니다. */
	$out .= "#ssb{ max-width:none !important; width:auto !important; }\n";

	/* 표·그림처럼 넓은 것이 쪽을 밀고 나가지 않게 */
	$out .= "#ssb .page img, #ssb .page table{ max-width:100% !important; }\n";

	/* 창이 좁으면 화면을 다 씁니다 */
	$out .= "@media(max-width:960px){\n";
	$out .= "  #ssb .page{ max-width:100% !important; }\n";
	$out .= "}\n";

	$out .= "</style>\n";
	return $out;
}

}
