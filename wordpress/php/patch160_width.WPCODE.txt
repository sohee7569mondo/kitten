/* ══════════════════════════════════════════════════════════
   책의 모양 — 이 조각 하나가 정합니다
   ★ 2026-09-12 · 소희 님께      판 WIDTH-6

   소희 님 : 「그냥 페이지 폭 자체를 지정할 수는 없어? 가지각색이야」

   맞습니다. 지금은 쪽마다 시작하는 자리가 다릅니다. 어떤 쪽은
   왼쪽 끝에서, 어떤 쪽은 한참 안쪽에서 시작합니다.

   ── 이 조각이 하는 일 ─────────────────────────────────
   책의 겉모양을 맡습니다. 글은 거의 안 건드립니다.

       하나  모든 쪽을 한 폭으로 묶고 가운데에 놓습니다
       둘    차례 쪽의 목록도 그 폭을 다 쓰게 합니다
       셋    쪽번호를 안 보이게 합니다
       넷    표를 쪽 폭 끝까지 늘리고 글씨를 조금 키웁니다   (2026-09-12)
       다섯  장 속표지의 글을 쪽 가운데로 올립니다
       여섯  겹쳐 있던 가로줄 두 개를 하나로 줄입니다
       일곱  별자리 문으로 보내던 상자를 접습니다
       여덟  맺음말 편지를 문장마다 줄을 바꿔 놓습니다
       아홉  「여기까지가 제가 묻기 전에 읽은 전부입니다」를 감춥니다

       쪽 전체      900px
       좌우 여백    78px (여백 조각이 정합니다)
       글줄         744px — 한 줄에 마흔 자 안팎

   창이 960px 보다 좁으면(휴대폰·좁은 창) 화면을 다 씁니다.

   ── 왜 쪽번호를 빼나 (WIDTH-4) ────────────────────────
   소희 님 : 「목차에서는 43인데 실제 프린트하면 73페이지더라고.
             이 숫자가 우리한테 불리한 것 같아」

   맞습니다. 우리가 세는 「쪽」은 화면의 한 칸입니다. 인쇄하면
   그 한 칸이 두 장, 세 장으로 흘러갑니다. 43과 73은 앞으로도
   절대 안 맞습니다 — 용지 크기와 글씨 크기마다 달라지니까요.

   그런데 목차에 43이 적혀 있으면 손님은 「43쪽짜리 책」으로
   읽습니다. 실제로는 73장인데 스스로 작게 말하는 셈입니다.

   그래서 숫자를 감춥니다. 네 군데입니다.

       차례 쪽 오른쪽 번호        30 · 31 · 32 …
       번호로 이어지던 점선        · · · · · · ·
       목차 단추 머리의 「43쪽」
       맨 위 띠의 「43쪽」

   ★ 지우는 게 아니라 감추는 것입니다. 번호는 그대로 매겨집니다.
     아래 「쪽번호」 토막을 지우시면 도로 보입니다.

   ★ 쪽 구석의 작은 번호(.folio)는 남겨 두었습니다. 그것까지
     빼시려면 아래 맨 끝에 주석으로 달아 둔 한 줄을 살리세요.

   ── 숫자를 바꾸고 싶으시면 ────────────────────────────
   아래 한 줄의 900 만 고치시면 됩니다. 그 한 줄이 책 전체입니다.

       $wide = 900;

   ★ WIDTH-3 (2026-09-12) — 머리와 바닥 양쪽으로 내보냅니다.
     WIDTH-2 는 머리(wp_head)로만 내보냈는데 한 글자도 안 나갔습니다.
     다섯 줄짜리로 재보니 바닥(wp_footer)은 잘 나왔습니다.
     그래서 양쪽에 겁니다. <style> 은 바닥에 있어도 똑같이 듣습니다.
     두 번 나가도 같은 규칙이라 탈이 없습니다.

   ★ WIDTH-2 (2026-09-12) — 조건을 다 뺐습니다.
     처음 판은 「책 쪽인지」 세 가지를 따져보고 내보냈는데,
     그 가운데 하나가 걸려 한 글자도 안 나갔습니다.
     이제 어느 쪽에서나 내보냅니다. 어차피 #ssb 는 책 쪽에만 있어서
     다른 쪽에서는 아무 일도 하지 않습니다.

   ── 들어갔는지 확인하는 법 ────────────────────────────
   아무 쪽에서나 Ctrl+U -> Ctrl+F 로 「폭 판」 을 찾으세요.
   (책 쪽이 아니어도 됩니다. 첫 쪽에서도 보입니다)

       <!-- stella 모양 판 WIDTH-6 · 900px -->

   이 줄이 보이면 들어간 것입니다. 없으면 조각이 안 켜졌거나
   위치(Location)가 「어디서나 실행」이 아닙니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요. 코드에 「함수」「반환합니다」가
     보이면 번역된 것입니다.
   ══════════════════════════════════════════════════════════ */

if ( ! function_exists( 'stella_look6_css' ) ) {

/* 조건을 두지 않습니다. #ssb 는 책 쪽에만 있으니
   다른 쪽에서는 이 규칙이 걸릴 자리가 없습니다.
   ★ 머리와 바닥 양쪽에 겁니다. 한쪽이 막혀도 다른 쪽이 나갑니다. */
add_action( 'wp_head', function () {
	echo stella_look6_css( '머리' );
}, 99 );

add_action( 'wp_footer', function () {
	echo stella_look6_css( '바닥' );
	echo stella_look6_js();
}, 99 );

function stella_look6_css( $where = '' ) {

	$wide = 900;      /* ★ 이 숫자 하나가 책 전체 폭입니다 */

	$out  = "\n<!-- stella 모양 판 WIDTH-6 · " . $wide . "px · " . $where . " -->\n";
	$out .= '<style id="stella-width">' . "\n";

	/* 모든 쪽을 같은 폭으로, 가운데에.
	   테두리와 여백이 폭 안에 들어가도록 box-sizing 도 맞춥니다. */
	$out .= "#ssb .page{\n";
	$out .= '  max-width:' . $wide . "px !important;\n";
	$out .= "  margin-left:auto !important;\n";
	$out .= "  margin-right:auto !important;\n";
	$out .= "  box-sizing:border-box !important;\n";
	$out .= "  width:auto !important;\n";
	$out .= "}\n";

	/* 쪽을 담는 칸이 좁게 묶여 있으면 가운데 정렬이 안 보입니다. */
	$out .= "#ssb{ max-width:none !important; width:auto !important; }\n";

	/* 표·그림처럼 넓은 것이 쪽을 밀고 나가지 않게 */
	$out .= "#ssb .page img, #ssb .page table{ max-width:100% !important; }\n";

	/* ══ 차례 쪽 — 폭 ═══════════════════════════════════
	   소희 님 : 「목차 폭이 좀 작아. 예전엔 넓게 보였거든」
	   목록이 30rem(480px)로 묶여 있어 900px 안에서 가운데만
	   쓰고 있었습니다. 본문과 같은 폭을 쓰게 풉니다. */
	$out .= "#ssb .tocpage .tocp-list{ max-width:none !important; margin:0 !important; }\n";
	$out .= "#ssb .tocpage .tocp-sub{ padding-left:26px !important; }\n";

	/* ══ 쪽번호 — 감춥니다 ══════════════════════════════
	   화면의 「쪽」과 인쇄되는 장수가 다릅니다 (43 대 73).
	   숫자가 적은 쪽이 손님 눈에 얇은 책으로 보입니다.
	   ★ 이 토막만 지우시면 번호가 도로 보입니다. */
	$out .= "#ssb .tocpage .n, #ssb .tocpage .d{ display:none !important; }\n";
	$out .= "#ssb .tocpage .t{ flex:1 1 auto !important; }\n";
	$out .= "#ssb .toclist button .no{ display:none !important; }\n";
	$out .= "#ssb .tochead span{ display:none !important; }\n";
	$out .= "#bkN{ display:none !important; }\n";

	/* 쪽 구석의 작은 번호까지 빼시려면 앞의 빗금 두 개를 지우세요
	$out .= "#ssb .page .folio{ display:none !important; }\n";
	*/

	/* ══ 표 — 쪽 폭 끝까지 ══════════════════════════════
	   소희 님 : 「표 글씨가 좀 작고, 표가 끝까지 있었으면」

	   표는 원래 width:100% 였습니다. 그런데 인쇄 대비로 넣었던
	   한 줄이 표를 글자 덩어리로 바꿔 놓았습니다.

	       #ssb table{ display:block !important; }   ← 이것

	   display:block 이 되면 표는 폭을 채우지 않고 글자 길이만큼만
	   줄어듭니다. 그래서 900px 쪽 안에서 460px 만 쓰고 있었습니다.
	   도로 표로 세우고 폭을 채웁니다. 글씨도 한 단계 키웁니다.

	   ★ WIDTH-6 (2026-09-12) — 처음에는 .tw 상자 안의 표만 고쳤습니다.
	     그런데 책의 표는 열아홉 개이고 그 가운데 열한 개는 상자가 없는
	     <table class="luck"> 입니다. 대운 표도 기둥 표도 그쪽이었습니다.
	     그래서 쪽 안의 표를 전부 고칩니다.
	   좁은 화면에서는 예전처럼 옆으로 미는 표가 됩니다. */
	$out .= "#ssb .page table{ display:table !important; width:100% !important;\n";
	$out .= "  min-width:0 !important; font-size:1rem !important; }\n";
	$out .= "#ssb .page td .mini{ font-size:.92rem !important; }\n";
	$out .= "#ssb .tw{ overflow-x:auto !important; }\n";

	/* ══ 겹친 가로줄 ════════════════════════════════════
	   소희 님 : 「줄이 2개 들어간 거 이상해」
	   맞습니다. 책 원본에 <hr> 이 두 번 나란히 적혀 있습니다.
	   글을 고치지 않고, 둘째 줄만 안 보이게 합니다. */
	$out .= "#ssb .page hr + hr{ display:none !important; }\n";

	/* ══ 별자리 문으로 보내던 상자 ══════════════════════
	   별자리는 문을 접고 사주 안의 코너로 가기로 했습니다.
	   그러니 책 안에서 그 문을 사라고 보내면 안 됩니다.
	   상자째 접습니다 — 안내문과 단추가 같이 들어 있습니다. */
	$out .= "#ssb .more{ display:none !important; }\n";

	/* ══ 장 속표지 ══════════════════════════════════════
	   소희 님 : 「표지의 내용이 중간으로 갈 수 있게. 아래로 쏠려서
	             아랫부분의 여백이 좁아 보여」

	   ★ WIDTH-6 — 속표지는 **이미 가운데로 서 있었습니다.**
	     제가 WIDTH-5 에서 가운데로 세운다고 넣은 줄은 아무 일도
	     안 했습니다. 이미 같은 규칙이 있었으니까요.

	     진짜 문제는 쪽의 키였습니다.

	         #ssb .page{ min-height:clamp(560px, 138vw, 1100px); }

	     모든 쪽이 1,100px 입니다. 속표지에 들어가는 글은 400px 남짓이라
	     700px 이 빈 채로 남습니다. 가운데에 있어도 허공에 뜬 것처럼
	     보입니다. 그래서 속표지만 키를 줄입니다. 620px 이면 글이
	     가운데 앉고 위아래 여백이 비슷해집니다.

	   속표지에는 쪽번호를 안 답니다 — 종이책도 그렇습니다. */
	$out .= "#ssb .page.divider{ min-height:620px !important;\n";
	$out .= "  display:flex !important; flex-direction:column !important;\n";
	$out .= "  justify-content:center !important;\n";
	$out .= "  padding-top:56px !important; padding-bottom:56px !important; }\n";
	$out .= "#ssb .page.divider .folio{ display:none !important; }\n";

	/* 창이 좁으면 화면을 다 씁니다 */
	$out .= "@media(max-width:960px){\n";
	$out .= "  #ssb .page{ max-width:100% !important; }\n";
	$out .= "}\n";
	$out .= "@media(max-width:640px){\n";
	$out .= "  #ssb .page table{ display:block !important; overflow-x:auto !important; }\n";
	$out .= "}\n";

	$out .= "</style>\n";
	return $out;
}


/* 글자를 손대는 곳은 여기 두 군데뿐입니다.
   CSS 로는 글자를 바꿀 수도, 문장을 나눌 수도 없어서 그렇습니다.
   책 원본은 한 글자도 안 고칩니다 — 그려진 뒤에 고쳐 놓습니다. */
function stella_look6_js() {
	return <<<'STELLALOOK6'
<script>
/* ═══════════════════════════════════════════════════════════
   STELLA SAJU — 책 모양 손질 두 가지      판 WIDTH-6
   ① 「여기까지가 제가 묻기 전에 읽은 전부입니다」 한 줄 감추기
   ② 맺음말 편지를 문장마다 줄 바꾸기
   ★ 홑 앰퍼샌드와 두 글자 논리기호를 쓰지 않습니다 (사이트 규칙)
   ═══════════════════════════════════════════════════════════ */
(function(){
  'use strict';

  /* ── ① 그 한 줄은 없앱니다 ──────────────────────────
     소희 님 : 「사주가 말하는 소희님 모습, 이건 그냥 없애도 될듯.
               다른 데는 이런 문구 없어」

     맞습니다. 한 장이 끝날 때마다 「여기까지가 …입니다」를 적는
     쪽은 여기 하나뿐이었습니다. 다른 장은 그냥 끝납니다.
     한 군데만 다르면 손님은 그것을 규칙으로 읽습니다.

     지우는 게 아니라 감춥니다 — 책 원본은 그대로입니다. */
  function hideLine(box){
    var ps=box.getElementsByClassName('mini'), i, hit=0;
    for(i=0;i<ps.length;i++){
      if(ps[i].textContent.indexOf('묻기 전에 읽은')<0){ continue; }
      ps[i].style.display='none';
      hit+=1;
    }
    return hit;
  }

  /* ── ② 편지 — 문장마다 줄을 바꿉니다 ────────────────
     마침표 뒤에 빈칸이 오면 거기서 한 줄 내립니다.
     「2.4도」처럼 숫자 사이의 점은 뒤에 빈칸이 없어 그대로 둡니다.
     뒤돌아보기(lookbehind)를 쓰지 않습니다 — 옛 브라우저에서
     그 하나 때문에 조각 전체가 멈춥니다. */
  function cut(t){
    var parts=[], cur='', i, c;
    for(i=0;i<t.length;i++){
      c=t.charAt(i);
      cur+=c;
      if(c!=='.'){ if(c!=='?'){ if(c!=='!'){ continue; } } }
      if(t.charAt(i+1)!==' '){ continue; }
      parts.push(cur);
      cur='';
      i+=1;
    }
    if(cur!==''){ parts.push(cur); }
    return parts;
  }

  function brk(el){
    var kids=[].slice.call(el.childNodes), i, node, parts, j, frag;
    for(i=0;i<kids.length;i++){
      node=kids[i];
      if(node.nodeType===1){ brk(node); continue; }
      if(node.nodeType!==3){ continue; }
      parts=cut(node.nodeValue);
      if(parts.length<2){ continue; }
      frag=document.createDocumentFragment();
      for(j=0;j<parts.length;j++){
        if(j>0){ frag.appendChild(document.createElement('br')); }
        frag.appendChild(document.createTextNode(parts[j]));
      }
      node.parentNode.replaceChild(frag, node);
    }
  }

  function letter(box){
    var ps=box.querySelectorAll('.page.close .lead'), i, done=0;
    for(i=0;i<ps.length;i++){
      if(ps[i].getAttribute('data-br')==='1'){ continue; }
      ps[i].setAttribute('data-br','1');
      brk(ps[i]);
      done+=1;
    }
    return done;
  }

  var tries=0;
  function tick(){
    tries+=1;
    var box=document.getElementById('ssb');
    var ok=0;
    if(box){
      if(box.getElementsByClassName('page').length>4){
        try{ hideLine(box); }catch(e){}
        try{ letter(box); }catch(e){}
        ok=1;
      }
    }
    if(ok){ return; }
    if(tries>60){ return; }
    setTimeout(tick, 400);
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', function(){ setTimeout(tick, 500); });
  } else { setTimeout(tick, 500); }
})();
</script>
STELLALOOK6;
}
