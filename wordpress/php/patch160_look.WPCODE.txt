/* ════════════════════════════════════════════════════════════
   책 보기 한 판   ★ 2026-09-09 · 소희 님께

   ── 무엇을 고치나 ─────────────────────────────────────
   ① 금빛을 책에서 전부 뺍니다. 제목은 군청색이 됩니다.        (25·27·28·33)
   ② 본문 글씨를 진하게 합니다. 흐린 글씨도 읽히게 올립니다.    (3·17)
   ③ 소제목을 본문보다 크게 키웁니다. 지금은 크기가 같습니다.   (25)
   ④ 본문 속 강조는 검정 볼드로. 금빛 얇은 글씨를 없앱니다.     (25)
   ⑤ 표지의 작은 글씨(쓴 이·생년월일시·태어난 곳·사주 원국)를
      감춥니다. 사진과 제목만 남습니다.                        (15)
   ⑥ 가로로 잘리던 것을 막습니다.                             (16)

   ── 왜 금빛을 빼나 ────────────────────────────────────
   흰 종이 위에서 금빛은 대비가 3:1 밖에 안 나옵니다. 굵게 써도
   눈에는 연하게 보입니다. 종이에 찍으면 겨자색이 됩니다.
   군청색은 진하게 나오고, 인쇄해도 그대로입니다.

   ★ 위 도구막(PDF 로 저장 단추)은 책 밖이라 그대로 둡니다.

   ── 안 건드리는 것 ────────────────────────────────────
   글은 한 글자도 안 고칩니다. 보이는 모양만 바꿉니다.
   덩어리 하나를 쪽 맨 뒤에 붙이고, 표를 달아 둡니다.
   두 번 넣어도 덩어리가 하나만 남습니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   ① 먼저 보기   https://stellasaju.com/?stella_look=1
   ② 정말 고치기 https://stellasaju.com/?stella_look=apply
   ③ 되돌리기    https://stellasaju.com/?stella_look=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_look'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_look_backup_0909';
	$MARK = 'stella-look-0909';
	$mode = sanitize_key( wp_unslash( $_GET['stella_look'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.85;'
	   . 'padding:34px 24px;max-width:900px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.35rem;margin:0 0 18px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.88rem}a{color:#D4AF6A}'
	   . 'li{margin:6px 0}'
	   . 'code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:12px/1.55 monospace;border-left:3px solid #D4AF6A}</style>';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '<h1 class="no">reading-book 쪽을 못 찾았습니다.</h1>'; exit; }
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) { echo '<h1 class="no">내용을 못 읽었습니다.</h1>'; exit; }
	$c = $row->post_content;

	/* ── 붙일 덩어리 ─────────────────────────────────
	   따옴표를 쓰지 않습니다. 앰퍼샌드도 쓰지 않습니다. */
	$css  = "\n<!-- " . $MARK . " 시작 -->\n<style>\n";
	$css .= "/* 책 보기 한 판 · 2026-09-09 */\n";

	$css .= "#ssb{\n";
	$css .= "  --gold:#26346B !important;\n";
	$css .= "  --gold-lite:#26346B !important;\n";
	$css .= "  --star:#1A1713 !important;\n";
	$css .= "  --soft:#26211C !important;\n";
	$css .= "  --dim:#5A5148 !important;\n";
	$css .= "  --rule:rgba(38,52,107,.20) !important;\n";
	$css .= "}\n";

	/* 본문 진하게 */
	$css .= "#ssb p{ color:#26211C !important; }\n";
	$css .= "#ssb .dek{ color:#5A5148 !important; }\n";

	/* 소제목을 본문보다 크게 · 군청 */
	$css .= "#ssb h2{ color:#26346B !important; }\n";
	$css .= "#ssb h3{ color:#26346B !important; font-weight:700 !important;\n";
	$css .= "  font-size:1.3em !important; line-height:1.5 !important;\n";
	$css .= "  margin-top:2.1em !important; }\n";
	$css .= "#ssb .eyebrow{ color:#26346B !important; }\n";
	$css .= "#ssb .sipno{ color:#26346B !important; font-weight:700 !important;\n";
	$css .= "  font-size:1em !important; }\n";

	/* 본문 속 강조는 검정 볼드 */
	$css .= "#ssb em,#ssb strong,#ssb b{ color:#12100D !important;\n";
	$css .= "  font-weight:700 !important; font-style:normal !important; }\n";

	/* 제 N 장 · 장 제목은 검정 크게 */
	$css .= "#ssb .divider .dvno{ color:#1A1713 !important; }\n";
	$css .= "#ssb .divider h2{ color:#12100D !important; }\n";
	$css .= "#ssb .divider .dvwho{ color:#26346B !important; }\n";

	/* 표지 — 작은 글씨를 감추고 제목만 */
	$css .= "#ssb .cover dl{ display:none !important; }\n";
	$css .= "#ssb .cover .rule{ margin:26px auto !important; }\n";
	$css .= "#ssb .cover h1{ color:#12100D !important; }\n";
	$css .= "@media(max-width:640px){\n";
	$css .= "  #ssb .cover{ padding:44px 26px 56px !important; }\n";
	$css .= "  #ssb .cover h1{ font-size:1.86rem !important; line-height:1.42 !important; }\n";
	$css .= "}\n";

	/* 가로로 잘리지 않게 */
	$css .= "#ssb,#ssb .book,#ssb .page{ max-width:100% !important; }\n";
	$css .= "#ssb .chart{ max-width:100% !important; overflow-x:auto !important; }\n";
	$css .= "#ssb pre,#ssb table{ max-width:100% !important;\n";
	$css .= "  overflow-x:auto !important; display:block !important; }\n";
	$css .= "#ssb img{ max-width:100% !important; height:auto; }\n";

	/* 말상자 — 벼리가 말하는 자리 */
	$css .= "#ssb .voice p{ color:#1A1713 !important; }\n";
	$css .= "#ssb .voice .who{ color:#26346B !important; font-weight:700 !important; }\n";

	$css .= "</style>\n<!-- " . $MARK . " 끝 -->\n";

	$s = '<!-- ' . $MARK . ' 시작 -->';
	$e = '<!-- ' . $MARK . ' 끝 -->';
	$has = ( false !== strpos( $c, $s ) );

	echo '<h1>' . ( 'apply' === $mode ? '고쳤습니다' : '이렇게 바꿉니다 (아직 안 고쳤습니다)' ) . '</h1>';

	echo '<ul>'
	   . '<li>금빛을 책에서 전부 뺍니다. 제목은 <b>군청색</b>이 됩니다.</li>'
	   . '<li>본문 글씨를 <b>진하게</b> 합니다.</li>'
	   . '<li>소제목을 본문보다 <b>1.3배</b> 크게 키웁니다.</li>'
	   . '<li>본문 속 강조는 <b>검정 볼드</b>로 바꿉니다.</li>'
	   . '<li>표지의 작은 글씨(쓴 이·생년월일시·태어난 곳·사주 원국)를 <b>감춥니다</b>.</li>'
	   . '<li>가로로 잘리던 것을 막습니다.</li>'
	   . '</ul>';
	echo '<p class="dim">글은 한 글자도 안 고칩니다. 보이는 모양만 바꿉니다. '
	   . ( $has ? '이미 들어가 있는 덩어리를 <b>새것으로 바꿉니다</b>.' : '덩어리를 쪽 맨 뒤에 <b>새로 붙입니다</b>.' )
	   . '</p>';

	if ( 'apply' !== $mode ) {
		echo '<pre>' . esc_html( $css ) . '</pre>';
		echo '<p>정말 고치시려면 <code>/?stella_look=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}

	if ( $has ) {
		$a = strpos( $c, $s );
		$b = strpos( $c, $e );
		if ( false === $b || $b < $a ) {
			echo '<h1 class="no">표는 찾았는데 끝을 못 찾았습니다. 아무것도 안 바꿨습니다.</h1>';
			exit;
		}
		$b   = $b + strlen( $e );
		$new = substr( $c, 0, $a ) . ltrim( $css, "\n" ) . substr( $c, $b );
	} else {
		$new = $c . $css;
	}

	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<p class="ok">고쳤습니다. ' . number_format( strlen( $c ) ) . ' -> '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p><a href="/reading-book/?demo=door-love&v=' . time() . '">책 열어보기</a></p>';
	echo '<p class="dim">폰에서 그대로면 캐시입니다. 주소 뒤에 <code>?v=2</code> 를 붙여 여세요.</p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_look=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
