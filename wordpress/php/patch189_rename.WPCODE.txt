/* ════════════════════════════════════════════════════════════
   /guide/ 이름표 고치기 · 소희 님께

   무엇을 왜 고치나
     189 페이지는 990원 낱권을 설명하고 있는데 제목만
     「나의 인생길잡이」였습니다. 인생길잡이는 여섯 문을 한 권으로
     묶는 25구슬짜리 종합본이라 이름이 겹칩니다.
     이름표를 「나의 사주와 별자리」로 바꿔 둘을 갈라놓습니다.
     「나의 인생길잡이」라는 이름은 나중 종합본을 위해 비워둡니다.

   고치는 것
     1. 페이지 제목        나의 인생길잡이 → 나의 사주와 별자리
     2. 화면 큰 제목(h1)   같이
     3. 「지금 한 권 받아보기 →」 단추 삭제
        (근거 없이 직업운 문으로만 보내고 있었습니다.
         바로 위 타일 여섯 개가 이미 그 역할을 합니다)
     4. 상단 메뉴에서 /guide/ 로 오는 항목의 이름도 같이

   건드리지 않는 것
     목차 미리보기 · 값 990원 · 문 여섯 개 타일 · 샘플 링크
     가격 안내(/price/) 의 인생길잡이 예고 줄 — 그건 종합본 이야기라 그대로 둡니다

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 Run Everywhere →
              첫 줄 PHP 태그 아래에 붙여넣기 → 저장 → 활성

   ① 먼저 미리보기 — 아무것도 바뀌지 않습니다
        https://stellasaju.com/?stella_patch=guide189

   ② 화면 보고 괜찮으면 적용
        https://stellasaju.com/?stella_patch=guide189&go=1

   되돌리기
        https://stellasaju.com/?stella_patch=guide189&undo=1
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_patch'] ) || 'guide189' !== $_GET['stella_patch'] ) {
		return;
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		return;
	}

	global $wpdb;

	$pid     = 189;
	$OLDNAME = '나의 인생길잡이';
	$NEWNAME = '나의 사주와 별자리';

	$go   = isset( $_GET['go'] );
	$undo = isset( $_GET['undo'] );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:15px/1.8 -apple-system,system-ui,sans-serif;background:#111;color:#eee;padding:24px;max-width:1000px">';
	printf(
		'<h1 style="color:#ffd76a">/guide/ 이름표 고치기 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);

	/* ── 되돌리기 ─────────────────────────────────────── */
	if ( $undo ) {
		$bc = get_post_meta( $pid, '_stella_g189_content', true );
		$bt = get_post_meta( $pid, '_stella_g189_title', true );
		if ( ! $bc ) {
			echo '<p style="color:#ff7b7b">되돌릴 사본이 없습니다.</p></body>';
			exit;
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $bc, 'post_title' => $bt ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );

		$menus = get_option( 'stella_g189_menu_backup' );
		if ( is_array( $menus ) ) {
			foreach ( $menus as $mid => $mc ) {
				$wpdb->update( $wpdb->posts, array( 'post_content' => $mc ), array( 'ID' => (int) $mid ) );
				clean_post_cache( (int) $mid );
			}
		}
		if ( function_exists( 'wp_cache_flush' ) ) {
			wp_cache_flush();
		}
		echo '<h2 style="color:#7bff9b">되돌렸습니다.</h2></body>';
		exit;
	}

	/* ── 페이지 189 ───────────────────────────────────── */
	$row = $wpdb->get_row( $wpdb->prepare( "SELECT post_title, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) {
		echo '<p style="color:#ff7b7b">페이지 189 를 못 읽었습니다.</p></body>';
		exit;
	}
	$c = $row->post_content;

	$pairs = array(
		array( '화면 큰 제목', '<h1>' . $OLDNAME . '</h1>', '<h1>' . $NEWNAME . '</h1>' ),
		array( '「지금 한 권 받아보기」 단추 삭제', '<p style="text-align:center"><a class="cta" href="/door-career/">지금 한 권 받아보기 →</a></p>', '' ),
	);

	echo '<h2 style="color:#9fd">1. 페이지 189 본문</h2>';
	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%"><tr style="color:#ffd76a"><td>찾음</td><td>무엇을</td></tr>';
	$ready = true;
	$done  = true;
	foreach ( $pairs as $p ) {
		$n = substr_count( $c, $p[1] );
		$m = ( '' === $p[2] ) ? 0 : substr_count( $c, $p[2] );
		if ( 1 !== $n ) {
			$ready = false;
		}
		if ( 0 !== $n ) {
			$done = false;
		}
		printf(
			'<tr style="border-top:1px solid #333"><td style="color:%s"><b>%d</b></td><td>%s</td></tr>',
			1 === $n ? '#7bff9b' : '#ff7b7b',
			$n,
			esc_html( $p[0] )
		);
	}
	printf(
		'<tr style="border-top:1px solid #333"><td style="color:%s">%s</td><td>페이지 제목 : 지금 「%s」</td></tr>',
		$OLDNAME === $row->post_title ? '#7bff9b' : '#ffd76a',
		$OLDNAME === $row->post_title ? '있음' : '다름',
		esc_html( $row->post_title )
	);
	echo '</table>';

	/* ── 상단 메뉴 ────────────────────────────────────── */
	echo '<h2 style="color:#9fd">2. 상단 메뉴에서 /guide/ 로 오는 항목</h2>';

	$menu_posts = $wpdb->get_results(
		$wpdb->prepare(
			"SELECT ID, post_title, post_type, post_content FROM {$wpdb->posts}
			  WHERE post_type IN ('wp_template_part','wp_navigation','wp_template')
			    AND post_status IN ('publish','draft')
			    AND post_content LIKE %s",
			'%' . $wpdb->esc_like( $OLDNAME ) . '%'
		)
	);

	/* /guide/ 를 가리키는 자리의 이름만 바꿉니다 */
	$q = preg_quote( $OLDNAME, '/' );
	/* 메뉴 정보에 label 이 url 보다 먼저 적힌 경우 */
	$re_block = '/("label"\s*:\s*")' . $q . '("[^}]*?"url"\s*:\s*"[^"]*\/guide\/)/u';
	/* url 이 먼저 적힌 경우 */
	$re_blk2  = '/("url"\s*:\s*"[^"]*\/guide\/[^"]*"[^}]*?"label"\s*:\s*")' . $q . '(")/u';
	/* 보통 링크로 적힌 경우 */
	$re_href  = '/(<a\b[^>]*href="[^"]*\/guide\/[^"]*"[^>]*>\s*)' . $q . '(\s*<\/a>)/u';

	$menu_plan = array();
	if ( ! $menu_posts ) {
		echo '<p style="color:#ffd76a">메뉴에서 「' . esc_html( $OLDNAME ) . '」 을 못 찾았습니다. 메뉴는 손대지 않습니다.</p>';
	} else {
		echo '<table cellpadding="8" style="border-collapse:collapse;width:100%"><tr style="color:#ffd76a"><td>어디</td><td>바꿀 곳</td><td>전체에 몇 번</td></tr>';
		foreach ( $menu_posts as $mp ) {
			$k1 = preg_match_all( $re_block, $mp->post_content ) + preg_match_all( $re_blk2, $mp->post_content );
			$k2 = preg_match_all( $re_href, $mp->post_content );
			$all = substr_count( $mp->post_content, $OLDNAME );
			if ( $k1 + $k2 > 0 ) {
				$menu_plan[ $mp->ID ] = $mp->post_content;
			}
			printf(
				'<tr style="border-top:1px solid #333"><td>%s <span style="color:#888">(%s · %d)</span></td><td style="color:%s"><b>%d</b></td><td>%d</td></tr>',
				esc_html( $mp->post_title ),
				esc_html( $mp->post_type ),
				$mp->ID,
				( $k1 + $k2 ) > 0 ? '#7bff9b' : '#ffd76a',
				$k1 + $k2,
				$all
			);
		}
		echo '</table>';
		echo '<p style="color:#888">「전체에 몇 번」이 더 많아도 괜찮습니다. /guide/ 를 가리키는 자리만 바꿉니다.</p>';
	}

	/* ── 미리보기면 여기까지 ──────────────────────────── */
	if ( ! $go ) {
		if ( $done ) {
			echo '<h2 style="color:#7bff9b">이미 고쳐져 있습니다.</h2>';
		} elseif ( ! $ready ) {
			echo '<h2 style="color:#ff7b7b">찾은 횟수가 1이 아닌 것이 있어 적용할 수 없습니다.</h2>'
				. '<p>글이 그새 바뀌었을 수 있습니다. 이 화면을 저에게 보내주세요.</p>';
		} else {
			echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다. 아무것도 바뀌지 않았습니다.</h2>'
				. '<p style="font-size:17px">괜찮으시면 <a style="color:#ffd76a" href="?stella_patch=guide189&go=1">→ 이대로 적용하기</a></p>';
		}
		echo '</body>';
		exit;
	}

	/* ── 적용 ─────────────────────────────────────────── */
	if ( ! $ready ) {
		echo '<h2 style="color:#ff7b7b">찾은 횟수가 1이 아니라 손대지 않았습니다. 글은 그대로입니다.</h2></body>';
		exit;
	}

	if ( ! get_post_meta( $pid, '_stella_g189_content', true ) ) {
		update_post_meta( $pid, '_stella_g189_content', $c );
		update_post_meta( $pid, '_stella_g189_title', $row->post_title );
	}

	$new = $c;
	foreach ( $pairs as $p ) {
		$new = str_replace( $p[1], $p[2], $new );
	}

	$wpdb->update(
		$wpdb->posts,
		array( 'post_content' => $new, 'post_title' => $NEWNAME ),
		array( 'ID' => $pid )
	);
	clean_post_cache( $pid );

	/* 메뉴 */
	$menu_done = 0;
	if ( $menu_plan ) {
		if ( false === get_option( 'stella_g189_menu_backup' ) ) {
			update_option( 'stella_g189_menu_backup', $menu_plan, false );
		}
		foreach ( $menu_plan as $mid => $mc ) {
			$mn = preg_replace( $re_block, '${1}' . $NEWNAME . '${2}', $mc );
			$mn = preg_replace( $re_blk2, '${1}' . $NEWNAME . '${2}', $mn );
			$mn = preg_replace( $re_href, '${1}' . $NEWNAME . '${2}', $mn );
			if ( $mn !== $mc ) {
				$wpdb->update( $wpdb->posts, array( 'post_content' => $mn ), array( 'ID' => (int) $mid ) );
				clean_post_cache( (int) $mid );
				$menu_done++;
			}
		}
	}
	if ( function_exists( 'wp_cache_flush' ) ) {
		wp_cache_flush();
	}

	$after = $wpdb->get_row( $wpdb->prepare( "SELECT post_title, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );

	echo '<h2 style="color:#7bff9b">바꿨습니다</h2><ul>';
	printf( '<li>페이지 제목 : %s</li>', esc_html( $after->post_title ) );
	printf( '<li>본문 : %d → %d 바이트</li>', strlen( $c ), strlen( $after->post_content ) );
	printf( '<li>큰 제목에 「%s」 남았나 : <b style="color:%s">%d</b> (0 이어야 정상)</li>',
		esc_html( $OLDNAME ),
		substr_count( $after->post_content, '<h1>' . $OLDNAME . '</h1>' ) ? '#ff7b7b' : '#7bff9b',
		substr_count( $after->post_content, '<h1>' . $OLDNAME . '</h1>' ) );
	printf( '<li>「지금 한 권 받아보기」 남았나 : <b style="color:%s">%d</b> (0 이어야 정상)</li>',
		substr_count( $after->post_content, '한 권 받아보기' ) ? '#ff7b7b' : '#7bff9b',
		substr_count( $after->post_content, '한 권 받아보기' ) );
	printf( '<li>메뉴 고친 곳 : <b>%d</b></li>', $menu_done );
	echo '</ul>';

	echo '<p style="margin-top:18px;font-size:17px"><a style="color:#ffd76a" href="/guide/">→ 페이지 열어보기</a></p>';
	echo '<p style="color:#888">되돌리려면 <a style="color:#888" href="?stella_patch=guide189&undo=1">?stella_patch=guide189&undo=1</a></p>';
	echo '</body>';
	exit;
} );
