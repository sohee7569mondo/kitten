<!-- stella 유앤미 · 내 생일 기억 · 판 2026-09-16 -->
<style id="stella-umremember">
#umrm-note{
  margin:10px 22px 0; font-size:.74rem; line-height:1.6; color:#B3A9C9;
}
#umrm-note button{
  border:0; background:none; padding:0; margin:0; cursor:pointer;
  font:inherit; color:#8E7BD0; text-decoration:underline;
}
</style>
<script>
(function(){
  if(window.StellaUmRemember){ return; }
  window.StellaUmRemember = 1;

  var KEY = 'stella_um_me';

  function get(){
    try{
      var raw = localStorage.getItem(KEY);
      if(!raw){ return null; }
      var o = JSON.parse(raw);
      if(!o){ return null; }
      if(!o.y){ return null; }
      return o;
    }catch(e){ return null; }
  }
  function put(o){ try{ localStorage.setItem(KEY, JSON.stringify(o)); }catch(e){} }
  function wipe(){ try{ localStorage.removeItem(KEY); }catch(e){} }

  function el(id){ return document.getElementById(id); }
  function val(id){ var n = el(id); return n ? String(n.value).trim() : ''; }

  /* ═══ 지금 보이는 화면이 「내 것」인가 ═════════════════
     칸 이름에 안 기댑니다 — 화면 맨 위 제목으로 가립니다. */
  function screenIsMine(){
    if(!el('f-y')){ return 0; }
    var hs = document.querySelectorAll('.h1'), i, t;
    var mine = 0, yours = 0;
    for(i = 0; i < hs.length; i++){
      t = String(hs[i].textContent === null ? '' : hs[i].textContent);
      if(t.indexOf('내 것') >= 0){ mine = 1; }
      if(t.indexOf('상대는') >= 0){ yours = 1; }
    }
    if(yours){ return 0; }
    return mine;
  }

  /* ═══ 담기 ═══════════════════════════════════════════ */
  function save(){
    if(!screenIsMine()){ return; }
    var y = val('f-y');
    if(y.length < 4){ return; }
    var o = {
      name: val('f-name'), y: y, m: val('f-m'), d: val('f-d'),
      h: val('f-h'), mi: val('f-mi'),
      ampm: segOn('data-ampm'), sex: segOn('data-sex')
    };
    put(o);
  }

  function segOn(attr){
    try{
      var bs = document.querySelectorAll('[' + attr + ']'), i, c;
      for(i = 0; i < bs.length; i++){
        c = String(bs[i].className === null ? '' : bs[i].className);
        if(c.indexOf('on') >= 0){ return String(bs[i].getAttribute(attr)); }
      }
    }catch(e){}
    return '';
  }
  /* 단추를 눌러 줍니다. 누르면 화면을 다시 그리기도 하므로,
     이미 켜져 있으면 안 누릅니다. 몇 번 이상은 포기합니다. */
  var taps = {};
  function segTap(attr, want){
    if(!want){ return; }
    if(segOn(attr) === want){ return; }
    var n = taps[attr] ? taps[attr] : 0;
    if(n > 3){ return; }
    try{
      var b = document.querySelector('[' + attr + '="' + want + '"]');
      if(!b){ return; }
      taps[attr] = n + 1;
      b.click();
    }catch(e){}
  }

  /* ═══ 채우기 ═════════════════════════════════════════
     비어 있는 칸만 채웁니다. 손님이 고치는 중에는 안 끼어듭니다. */
  function fill(id, v){
    if(!v){ return; }
    var n = el(id);
    if(!n){ return; }
    if(String(n.value).trim()){ return; }
    n.value = v;
    try{
      var ev = document.createEvent('HTMLEvents');
      ev.initEvent('input', true, false);
      n.dispatchEvent(ev);
    }catch(e){}
  }

  function restore(){
    if(!screenIsMine()){ return; }
    var o = get();
    if(!o){ return; }
    fill('f-name', o.name);
    fill('f-y', o.y);
    fill('f-m', o.m);
    fill('f-d', o.d);
    fill('f-h', o.h);
    fill('f-mi', o.mi);
    segTap('data-ampm', o.ampm);
    segTap('data-sex', o.sex);
    note();
  }

  /* ═══ 발치 한 줄 — 담아 둔 것을 지울 수 있게 ══════════ */
  function note(){
    if(!screenIsMine()){ return; }
    if(el('umrm-note')){ return; }
    var y = el('f-y');
    if(!y){ return; }
    var box = y.parentElement;
    if(box){ box = box.parentElement; }
    if(!box){ return; }
    var p = document.createElement('p');
    p.id = 'umrm-note';
    p.appendChild(document.createTextNode('이 기기에 기억해 둡니다 · '));
    var b = document.createElement('button');
    b.setAttribute('type', 'button');
    b.appendChild(document.createTextNode('지우기'));
    b.addEventListener('click', function(ev){
      ev.preventDefault();
      ev.stopPropagation();
      wipe();
      var ids = ['f-name', 'f-y', 'f-m', 'f-d', 'f-h', 'f-mi'], i, n;
      for(i = 0; i < ids.length; i++){
        n = el(ids[i]);
        if(n){ n.value = ''; }
      }
      p.textContent = '지웠습니다.';
    });
    p.appendChild(b);
    if(box.parentNode){ box.parentNode.insertBefore(p, box.nextSibling); }
  }

  /* ═══ 돌리기 ═════════════════════════════════════════ */
  document.addEventListener('input', function(){ save(); }, true);
  document.addEventListener('change', function(){ save(); }, true);
  document.addEventListener('click', function(){
    save();
    setTimeout(function(){ taps = {}; }, 60);
  }, true);

  function tick(){ try{ restore(); }catch(e){} }
  function start(){
    tick();
    setInterval(tick, 400);
  }
  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', start);
  } else { start(); }
})();
</script>
