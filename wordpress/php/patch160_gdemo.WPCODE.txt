       order.partner.day 가 있나        ← ★

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_gdemo'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	$mode = (string) $_GET['stella_gdemo'];
	$on   = ( '0' !== $mode );
	$see  = ( 'see' === $mode );

	echo '<!DOCTYPE html><html lang="ko" translate="no"><head><meta charset="utf-8">'

	   . '</head><body>';

	if ( $see ) {
		echo '<h3>지금 저장된 주문</h3><pre id="out">보는 중…</pre>';
		?>
<script>
(function(){
  var o=document.getElementById('out'), L=[];
  function say(s){ L.push(s); o.textContent=L.join('\n'); }
  var raw=null;
  try{ raw=localStorage.getItem('stella_demo'); }catch(e){
    say('★ 브라우저가 자료를 못 읽습니다 (시크릿 창이거나 쿠키를 막아두셨습니다)');
    return;
  }
  if(!raw){ say('★ 주문이 아예 없습니다. 먼저 주문을 하나 하셔야 합니다.'); return; }
  var d=null;
  try{ d=JSON.parse(raw); }catch(e){ say('★ 주문이 깨져 있습니다.'); return; }

  say('문      ' + (d.guardian_slug||'(없음)'));
  say('주제    ' + (d.topic||'(없음)'));
  say('');
  say('내 정보(profile)');
  if(d.profile){
    ['name','year','month','day','hour','minute','sex','city','country'].forEach(function(k){
      say('   ' + k + (new Array(10-k.length).join(' ')) + ' ' +
          (d.profile[k]===undefined ? '(없음)' : d.profile[k]));
    });
  } else { say('   ★ 없습니다'); }
  say('');
  say('상대 정보(partner)   ← 궁합에 꼭 있어야 합니다');
  if(!d.partner){
    say('   ★★ 아예 없습니다.');
    say('   door-love 가 상대 생년월일을 order.partner 로 저장하지 않습니다.');
    say('   이것이 궁합 책이 안 나오는 까닭입니다.');
  } else {
    ['year','month','day','hour','minute','sex','rel','alias'].forEach(function(k){
      var v=d.partner[k];
      var miss=(v===undefined); if(v===null){ miss=true; } if(v===''){ miss=true; }
      say('   ' + k + (new Array(8-k.length).join(' ')) + ' ' +
          (miss ? '★ 없음' : v));
    });
    var ok=true;
    if(!d.partner.year){ ok=false; }
    if(!d.partner.month){ ok=false; }
    if(!d.partner.day){ ok=false; }
    say('');
    say(ok ? '   → 년·월·일이 다 있습니다. 책이 궁합으로 가야 합니다.'
           : '   → ★ 년·월·일 가운데 빠진 것이 있어 책이 일반으로 흘러갑니다.');
  }
  say('');
  say('셈 조각');
  say('   StellaGunghapCalc  ' + (window.StellaGunghapCalc ? '있음' : '★ 없음'));
  say('   StellaGunghapText  ' + (window.StellaGunghapText ? '있음' : '★ 없음'));
  say('   (이 화면은 책 쪽이 아니라 조각이 안 보일 수 있습니다 — 책에서는 보입니다)');
  say('');
  say('── 주문 전체 (그대로) ──');
  say(raw.length>3000 ? raw.slice(0,3000)+' …' : raw);
})();
</script>
