/* 로그인한 분을 우리 통로에서 알아보게 합니다.        2026-09-03

   무슨 일이 있었나
   ─────────────────
   워드프레스는 /wp-json/... 요청에 「열쇠(nonce)」가 붙어 있지 않으면,
   로그인 쿠키가 멀쩡히 있어도 그 요청을 손님(익명)으로 처리합니다.
   워드프레스 안쪽(rest_cookie_check_errors)에서 wp_set_current_user(0)
   을 부르기 때문입니다.

   그래서 회원가입을 마치고 /mypage/ 에 들어가도
     /wp-json/stella/v1/me  →  member:false
   가 돌아왔고, 화면은 「가입하신 적이 있다면 들어오세요」 상자를 띄웠습니다.
   저장(profile)·결제(pay)·이메일 바꾸기(email)도 같은 이유로 조용히
   손님 취급을 받고 있었습니다.

   그동안은 문 페이지에만 열쇠를 붙여 왔습니다(patch_door_nonce).
   페이지가 늘 때마다 열쇠를 하나씩 붙이는 방식이라, 하나 빠지면
   오늘 같은 일이 또 생깁니다. 이번에는 서버 쪽에서 한 번에 막습니다.

   무엇을 하나
   ─────────────
   stella/v1 통로로 들어온 요청에 한해, 로그인 쿠키를 직접 확인해서
   그 회원으로 되돌려 놓습니다. 워드프레스의 다른 통로는 손대지 않습니다.

   안전한가
   ─────────
   · 읽기(GET)는 그냥 되돌립니다. 아무것도 바꾸지 않으니까요.
   · 쓰기(POST)는 요청이 우리 사이트 화면에서 출발했을 때만 되돌립니다.
     (Origin / Referer 가 stellasaju.com 일 때. 다른 사이트가 몰래
      우리 회원 이름으로 요청을 보내는 일 — CSRF — 을 막습니다.)
   · 열쇠가 제대로 붙어 온 요청은 이미 로그인 상태이므로 그냥 지나갑니다.
   · 쿠키는 wp_validate_auth_cookie 로 워드프레스가 직접 검사합니다.
     위조한 쿠키는 통과하지 못합니다.

   지우면 원래대로 돌아갑니다. 다른 스니펫을 건드리지 않습니다. */

if ( ! function_exists( 'stella_rest_cookie_same_site' ) ) {

	/* 이 요청이 우리 사이트 화면에서 출발했나 */
	function stella_rest_cookie_same_site() {

		$mine = wp_parse_url( home_url(), PHP_URL_HOST );
		if ( ! $mine ) {
			return false;
		}
		$mine = strtolower( $mine );

		$heads = array( 'HTTP_ORIGIN', 'HTTP_REFERER' );

		foreach ( $heads as $h ) {
			if ( empty( $_SERVER[ $h ] ) ) {
				continue;
			}
			$host = wp_parse_url( (string) $_SERVER[ $h ], PHP_URL_HOST );
			if ( ! $host ) {
				continue;
			}
			$host = strtolower( $host );
			if ( $host === $mine ) {
				return true;
			}
			if ( $host === 'www.' . $mine ) {
				return true;
			}
			if ( 'www.' . $host === $mine ) {
				return true;
			}
			/* 주소는 있는데 우리 것이 아니면 — 거기서 끝냅니다 */
			return false;
		}

		return false;
	}
}

add_filter( 'rest_pre_dispatch', function ( $result, $server, $request ) {

	if ( null !== $result ) {
		return $result;
	}
	if ( is_user_logged_in() ) {
		return $result;
	}

	$route = (string) $request->get_route();
	if ( 0 !== strpos( $route, '/stella/v1/' ) ) {
		return $result;
	}

	$method = strtoupper( (string) $request->get_method() );

	/* 바꾸는 요청은 우리 화면에서 온 것만 */
	if ( 'GET' !== $method && 'HEAD' !== $method ) {
		if ( ! stella_rest_cookie_same_site() ) {
			return $result;
		}
	}

	if ( ! defined( 'LOGGED_IN_COOKIE' ) ) {
		return $result;
	}
	if ( empty( $_COOKIE[ LOGGED_IN_COOKIE ] ) ) {
		return $result;
	}

	$uid = wp_validate_auth_cookie( $_COOKIE[ LOGGED_IN_COOKIE ], 'logged_in' );
	if ( ! $uid ) {
		return $result;
	}

	wp_set_current_user( (int) $uid );

	return $result;
}, 1, 3 );
