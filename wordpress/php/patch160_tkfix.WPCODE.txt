/* ══════════════════════════════════════════════════════════
   책이 하얗게 나오는 것을 고칩니다 — topicKey
                                        ★ 2026-09-13 · 소희 님께
                                        판 TKFIX-1

   ★★ 이것은 궁합만의 일이 아닙니다. **모든 책**이 이것 때문에
      하얗게 나오고 있습니다. 먼저 넣어주세요.

   ── 무엇이 잘못돼 있었나 ──────────────────────────────
   진단 띠가 이렇게 알려줬습니다 —

       ★★ 이 쪽에서 처음 난 탈 — topicKey is not defined
          [reading-book 줄 9500]

   WEDSPLIT 이 결혼운을 미혼·기혼으로 가르려고 topicKey 라는 이름을
   만들었습니다. 그런데 그 이름을 **만든 자리와 쓰는 자리가 다른
   울타리 안**에 있습니다. 만든 곳은 책을 그리는 큰 덩어리 안이고,
   쓰는 곳 몇 군데는 따로 떨어진 작은 덩어리 안입니다. 거기서는
   topicKey 가 없는 이름이라 자바스크립트가 그 자리에서 멈춥니다.

   멈추면 화면에는 아무 말도 안 나옵니다. 그냥 하얗게 남습니다.
   소희 님이 보신 것이 그 모습입니다.

   ── 어떻게 고치나 (쪽을 안 건드립니다) ────────────────
   책이 그려지기 전에 **topicKey 를 온 쪽이 아는 이름으로 하나
   만들어 둡니다.** 그러면 어느 덩어리에서 불러도 있습니다.
   값은 WEDSPLIT 이 정한 것과 똑같이 셉니다 —

       주제가 「결혼운」이 아니면        그대로
       「결혼운」이고 답이 「이미…」면   결혼운_기혼
       그 밖 (답이 없는 옛 손님 포함)   결혼운_미혼

   책을 그리는 큰 덩어리는 자기 것을 따로 만들어 쓰므로 거기는
   그대로입니다. 없던 자리에만 채워집니다.

   앵커를 안 잡습니다. 쪽 글을 한 글자도 안 바꿉니다.
   되돌리기도 없습니다 — 스니펫을 끄면 원래대로입니다.

   ── 붙이는 법 ─────────────────────────────────────────
   WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.

   넣으신 뒤 ?gwhy=1 로 보시면 「이 쪽에서 난 탈 없음」이 되어야
   합니다. 그리고 책이 나와야 합니다.

   ── 검사한 것 ─────────────────────────────────────────
   php -l 통과 · 앰퍼샌드 0개 · node --check 통과
   node 로 값 셈을 다섯 가지로 돌려봄
   ══════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {
	?>
<script>
/* ═══ topicKey 를 온 쪽이 아는 이름으로 만들어 둡니다 (TKFIX-1) ═══
   WEDSPLIT 이 만든 이름이 다른 덩어리에서는 없는 이름이라 책이
   그리다가 멈췄습니다. 여기서 미리 하나 만들어 둡니다. */
(function(){

  function read(k){
    try{
      var v=localStorage.getItem(k);
      if(!v){ return null; }
      return JSON.parse(v);
    }catch(e){ return null; }
  }

  function calc(){
    var o=read('stella_demo');
    if(!o){ return ''; }
    var t=String(o.topic===undefined?'':o.topic);
    if(t!=='결혼운'){ return t; }

    /* 결혼운만 미혼·기혼으로 갈립니다.
       답이 없는 옛 손님은 미혼으로 봅니다 — WEDSPLIT 과 같습니다. */
    var a=read('stella_answers');
    var w='';
    if(a){ if(a.wed){ w=String(a.wed); } }
    if(w.indexOf('이미')>=0){ return '결혼운_기혼'; }
    return '결혼운_미혼';
  }

  function set(){
    try{ window.topicKey=calc(); }catch(e){}
  }

  set();
  /* 주문이 늦게 자리잡는 길도 있어, 쪽이 다 그려질 때 한 번 더 봅니다.
     책을 그리는 큰 덩어리는 자기 것을 따로 만들어 쓰므로 거기는
     이 값에 흔들리지 않습니다. */
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', set);
  }

})();
</script>
	<?php
}, 1 );
