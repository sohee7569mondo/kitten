/* ════════════════════════════════════════════════════════════
   가디언을 「벼리」라고 부릅니다 · 여는 글을 새로 씁니다

   지금 사랑 문 책 둘째 쪽은 이렇게 시작합니다.

     연성의 신이 결혼운에 대해
     [연성의 신] 사랑은 마음의 문제 같지만, 저는 다르게 봅니다…

   「연성의 신」은 손님이 처음 보는 말입니다. 그게 무슨 뜻인지
   모르는 채로 책을 펼치게 됩니다. 손님이 산 것은 가디언이
   아니라 결혼운이고요. 그래서 순서를 뒤집고 이름으로 부릅니다.

     결혼운에 대해 — 벼리
     [벼리] 사랑은 마음의 문제 같지만, 사주에서는 조금 다르게 봅니다…

   이름은 한 단 작게 둡니다. 큰 글씨로 나란히 놓으면 이름이
   제목처럼 보여서, 주제가 앞에 온 뜻이 없어집니다.

   「연성의 신」은 사랑 문에서 한 군데도 안 남습니다. 표지의
   가디언 칸, 말풍선 이름표 열셋, 맺음말 서명까지 전부 벼리가
   됩니다. 이름 하나만 갈아 끼우는 자리를 만들어 뒀으니,
   나머지 다섯 문도 이름이 정해지면 한 줄씩만 더하면 됩니다.

     var NICK={ 'door-love':'벼리' };

   지금은 사랑 문만 바뀝니다. 나머지 문은 그대로 칭호로 갑니다.

   여는 글 두 문단은 소희 님이 새로 쓰신 글로 바꿉니다.
   「저는 다르게 봅니다」가 「사주에서는 조금 다르게 봅니다」로
   바뀌면서, 벼리 개인의 의견이 아니라 사주가 그렇게 본다는
   말이 됩니다. 파는 것이 벼리가 아니라 사주라서 이쪽이 맞습니다.

   궁합 표지에 굳어 있던 조사도 같이 풉니다. 「연성의 신이」는
   맞았지만 「벼리이」가 되어버려서, 받침을 보고 고르게 했습니다.

   페이지 내용을 wp_update_post 로 저장하면 워드프레스가 <script> 를
   지워버립니다. 그래서 데이터베이스에 바로 씁니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로 바꾸세요.
     기본값인 「사이트 전체 헤더」로 두면 아래 주소가 그냥 홈으로 넘어갑니다.

   ① 먼저 보기   https://stellasaju.com/?stella_byline=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_byline=apply
   ③ 되돌리기    https://stellasaju.com/?stella_byline=undo

   바꾸고 나서 볼 곳
     https://stellasaju.com/reading-book/?demo=door-love-2
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_byline'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$PAGE = 160;
	$META = '_stella_byline_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_byline'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $PAGE ) );
	if ( ! $row ) { echo '<h1>160쪽을 못 찾았습니다.</h1>'; exit; }

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $PAGE, $META, true );
		echo '<h1>되돌리기</h1>';
		if ( ! $old ) { echo '<p>되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $PAGE ) );
		clean_post_cache( $PAGE );
		delete_post_meta( $PAGE, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		echo '<p><a href="/reading-book/?demo=door-love-2">책 보기</a></p>';
		exit;
	}

	$c = $row->post_content;
	$before = strlen( $c );

	$jobs = array();

	$jobs[] = array(
		'name' => '① 가디언을 이름으로 부르기',
		'find' => '/var\\s+g\\s+=\\s+G\\[slug\\]\\s+\\|\\|\\s+G\\[\'door\\-career\'\\];/u',
		'to'   => "  var g = G[slug] || G['door-career'];\n  /* 2026-09-07 · 가디언을 칭호 대신 이름으로 부릅니다.\n     「연성의 신」은 손님이 처음 보는 말이라 무슨 뜻인지 모릅니다.\n     이름이 정해진 문만 바꾸고, 나머지 문은 그대로 칭호로 둡니다.\n     이름이 나오는 대로 이 표에 한 줄씩 더하면 그 문도 따라 바뀝니다. */\n  var NICK={ 'door-love':'벼리' };\n  if(NICK[slug]){\n    var gk, g2={};\n    for(gk in g){ if(Object.prototype.hasOwnProperty.call(g,gk)){ g2[gk]=g[gk]; } }\n    g2.name=NICK[slug];\n    g=g2;\n  }\n",
	);

	$jobs[] = array(
		'name' => '② 여는 글 제목 — 주제 먼저, 이름 뒤로',
		'find' => '/\'<h2>\'\\+g\\.name\\+\\(R\\.hasBatchim\\(g\\.name\\)\\?\'이\':\'가\'\\)\\+\'\\s+\'\\+esc\\(topic\\)\\+\'에\\s+대해<\/h2>\'\\+/u',
		'to'   => "    '<h2>'+esc(topic)+'에 대해 <span class=\"byline\">— '+g.name+'</span></h2>'+\n",
	);

	$jobs[] = array(
		'name' => '③ 궁합 표지의 굳은 조사 풀기',
		'find' => '/\'<p\\s+class="dek">\'\\+g\\.name\\+\'이\\s+두\\s+분의\\s+여덟\\s+글자를\\s+나란히\\s+놓고\\s+봅니다<\/p>\',\\s+\'cover\'\\);/u',
		'to'   => "          '<p class=\"dek\">'+g.name+(R.hasBatchim(g.name)?'이':'가')+' 두 분의 여덟 글자를 나란히 놓고 봅니다</p>', 'cover');\n",
	);

	$jobs[] = array(
		'name' => '④ 이름 글자 모양 (CSS)',
		'find' => '/<\/style>\\s+<div\\s+id="ssb">/u',
		'to'   => "\n/* ══ 2026-09-07 · 여는 글 제목에 붙는 이름 ═══════════════════\n   「연성의 신이 결혼운에 대해」 → 「결혼운에 대해 — 벼리」\n   주제가 먼저 오고 이름이 뒤에 붙습니다. 손님이 산 것은\n   가디언이 아니라 결혼운이라서요. 이름은 한 단 작게 둡니다. */\n#ssb h2 .byline{\n  font-size:.62em; font-weight:500; letter-spacing:.01em;\n  color:var(--gold-lite); opacity:.9; white-space:nowrap;\n}\n#ssb .page[data-part=\"two\"] h2 .byline{ color:var(--speak-lite); }\n</style>\n<div id=\"ssb\">\n",
	);

	$jobs[] = array(
		'name' => '⑤ 여는 글 두 문단 새로 쓰기',
		'find' => '/open:\\[\\s+\'사랑은\\s+마음의\\s+문제\\s+같지만,\\s+저는\\s+다르게\\s+봅니다\\.\\s+사랑은\\s+습관의\\s+문제예요\\.\\s+어떤\\s+사람에게\\s+끌리는지,\\s+다투면\\s+어떻게\\s+하는지,\\s+그건\\s+태어날\\s+때부터\\s+몸에\\s+새겨져\\s+있습니다\\.\',\\s+\'그래서\\s+저는\\s+"그\\s+사람이\\s+당신을\\s+사랑한다"는\\s+말을\\s+하지\\s+않습니다\\.\\s+그건\\s+제가\\s+알\\s+수\\s+없어요\\.\\s+대신\\s+당신이\\s+사랑을\\s+어떻게\\s+하는\\s+사람인지,\\s+그\\s+방식이\\s+어디서\\s+자꾸\\s+걸리는지를\\s+말합니다\\.\'\\s+\\],/u',
		'to'   => "    open:[\n      '사랑은 마음의 문제 같지만, 사주에서는 조금 다르게 봅니다. 사랑을 하는 방식에도 타고난 기질과 패턴이 있기 때문입니다. 어떤 사람에게 끌리는지, 관계에서 무엇을 원하는지, 갈등이 생겼을 때 어떻게 반응하는지에는 사주의 성향이 드러납니다.',\n      '그래서 저는 “그 사람이 당신을 사랑한다”고 단정하지 않습니다. 상대의 마음까지 사주로 확인할 수는 없으니까요. 대신 당신의 사주를 통해, 당신이 사랑을 어떤 방식으로 하는 사람인지, 어떤 인연에 끌리는지, 그리고 반복해서 관계가 어려워지는 지점이 어디인지를 살펴봅니다.'\n    ],\n",
	);

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$ok = true;
	foreach ( $jobs as $j ) {
		$n = preg_match_all( $j['find'], $c );
		echo '<h2>' . esc_html( $j['name'] ) . '</h2>';
		if ( 1 === $n ) {
			echo '<p class="ok">찾았습니다 — 한 군데.</p>';
		} elseif ( 0 === $n ) {
			echo '<p class="no">못 찾았습니다. 이미 바꿨거나 코드가 달라졌습니다.</p>';
			$ok = false;
		} else {
			echo '<p class="no">' . intval( $n ) . '군데가 걸립니다. 한 군데여야 안전합니다.</p>';
			$ok = false;
		}
	}

	if ( ! $ok ) {
		echo '<p class="no">한 자리가 한 군데씩 걸릴 때만 바꿉니다. 아무것도 건드리지 않았습니다.</p>';
		exit;
	}

	$new = $c;
	foreach ( $jobs as $j ) {
		$new = preg_replace( $j['find'], str_replace( '\\', '\\\\', $j['to'] ), $new, 1 );
		if ( null === $new ) { echo '<p class="no">바꾸다가 멈췄습니다. 아무것도 저장하지 않았습니다.</p>'; exit; }
	}

	echo '<p>' . number_format( $before ) . ' → ' . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_byline=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $PAGE, $META, true ) ) {
		update_post_meta( $PAGE, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $PAGE ) );
	clean_post_cache( $PAGE );

	echo '<p class="ok">바꿨습니다.</p>';
	echo '<p>되돌리려면 <code>/?stella_byline=undo</code></p>';
	echo '<p><a href="/reading-book/?demo=door-love-2">책 보기</a></p>';
	exit;
} );
