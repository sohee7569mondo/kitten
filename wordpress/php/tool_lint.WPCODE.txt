/* ══════════════════════════════════════════════════════════
   스니펫 문법 검사기 — 어느 스니펫이 깨졌는지 찾아줍니다
   ★ 2026-09-10 · 소희 님께

   「스니펫을 활성화하는 중에 오류가 발생했습니다」가 뜨면
   이걸 켜고 주소창에 쳐보세요. 켜져 있는 스니펫을 전부 훑어서
   문법이 깨진 것을 찾아 드립니다.

     https://stellasaju.com/?stella_lint=1

   깨진 스니펫이 있으면 이름과 몇 줄째가 문제인지, 그리고 그
   앞뒤 세 줄을 그대로 보여줍니다.

   ★ 이 검사기는 아무것도 안 고칩니다. 보기만 합니다.
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_lint'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$rows = $wpdb->get_results(
		"SELECT ID, post_title, post_status FROM {$wpdb->posts}
		 WHERE post_type = 'wpcode' ORDER BY ID DESC LIMIT 400" );

	if ( ! $rows ) { echo '스니펫을 못 찾았습니다.'; exit; }

	echo "═══ 스니펫 문법 검사 ═══\n";
	echo count( $rows ) . "개를 훑습니다\n\n";

	$bad = 0;
	$php = 0;
	foreach ( $rows as $r ) {
		$code = get_post_meta( $r->ID, '_wpcode_code', true );
		if ( ! is_string( $code ) ) { continue; }
		if ( '' === trim( $code ) ) { continue; }
		$type = get_post_meta( $r->ID, '_wpcode_code_type', true );
		if ( 'php' !== $type ) { continue; }
		$php++;

		$src = $code;
		if ( strpos( $src, '<?php' ) === false ) { $src = "<?php\n" . $src; }

		$err = '';
		$line = 0;
		try {
			/* 실행하지 않고 문법만 봅니다 */
			$tmp = @token_get_all( $src, TOKEN_PARSE );
		} catch ( ParseError $e ) {
			$err  = $e->getMessage();
			$line = $e->getLine();
		} catch ( Error $e ) {
			$err  = $e->getMessage();
			$line = $e->getLine();
		}

		if ( '' === $err ) { continue; }

		$bad++;
		echo "★ " . $r->post_title . "  (번호 " . $r->ID . " - " . $r->post_status . ")\n";
		echo "   " . $err . "\n";
		if ( $line > 0 ) {
			$ls = explode( "\n", $src );
			$from = max( 0, $line - 3 );
			$to   = min( count( $ls ), $line + 2 );
			echo "   ── " . $line . "줄 언저리 ──\n";
			for ( $i = $from; $i < $to; $i++ ) {
				$mark = ( $i + 1 === $line ) ? ' >>' : '   ';
				echo $mark . ' ' . str_pad( (string) ( $i + 1 ), 5, ' ', STR_PAD_LEFT ) . ' | ' . $ls[ $i ] . "\n";
			}
		}
		echo "\n";
	}

	echo "───────────────\n";
	echo 'PHP 스니펫 ' . $php . "개 가운데\n";
	if ( $bad ) {
		echo '깨진 것 ' . $bad . "개.\n";
		echo "위에 표시된 줄을 보시고, 따옴표가 빠졌는지 확인해 주세요.";
	} else {
		echo "깨진 것 없습니다.\n";
		echo "그렇다면 오류는 저장이 안 된 스니펫에서 났을 수 있습니다.\n";
		echo "방금 붙여넣으신 스니펫을 다시 열어 처음부터 끝까지 다시 붙여넣어 보세요.";
	}
	exit;
}, 1 );
