    if(c.spd){ return c.spd; }
    var i = TAROT.indexOf(c);
    if(i < 0){ return 0; }
    if(i <= 21){ return i; }
    return 0;
  }

    var i = TAROT.indexOf(c);
    if(i < 0){ return false; }
    if(i <= 21){ return true; }
    return false;
  }

  function comboHtml(got){

    var fKey = 'flat';

    if(flow <= -4){ fKey = 'down'; }

    var rKey = 'rs';
    if(revN === 0){ rKey = 'r0'; }
    if(revN === 3){ rKey = 'r3'; }
