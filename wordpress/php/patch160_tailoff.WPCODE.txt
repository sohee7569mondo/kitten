/* ══════════════════════════════════════════════════════════
   본문에 박힌 옛 TAIL 블록을 떼어냅니다
   ★ 2026-09-12 · 소희 님께

   ── 왜 필요한가 ───────────────────────────────────────
   화면에 나오는 덤 두 쪽이 옛 판입니다.
       눈썹줄이 「띠로 보면」   ← 새 판은 「태어난 띠로 다시 보면」
       물병자리 글이 한 문장   ← 새 원고는 네 문장 + 맺음
       그림이 없음

   ?stella_tail9=undo 로 본문이 TAIL-9 붙이기 직전으로
   돌아갔는데, 거기에는 더 옛날 TAIL 블록이 그대로 박혀
   있었습니다. 그게 지금 보이는 것입니다.

   ★ 새 판(TAIL-RUN)은 본문을 안 건드리고 화면에 바로
     내보냅니다. 그러니 본문에 남은 옛 블록을 먼저
     떼어내야 합니다. 안 그러면 두 번 나옵니다.

   ── 이 조각이 하는 일 ─────────────────────────────────
   본문에서 「BUILD: TAIL-」 이 든 덩어리만 찾아 들어냅니다.
   글은 한 글자도 안 건드립니다. 그 덩어리만입니다.

   ★ 덤으로, 본문이 100KB 안팎 줄어듭니다.
     메모리가 터진 그 자리에 숨통이 트입니다.

   ── 지키는 것 ─────────────────────────────────────────
   · 먼저 보기에서는 몇 개가 걸리는지만 셉니다
   · 되돌림 꾸러미를 wp_slash 로 감쌉니다
   · 두 번 눌러도 안전합니다. 없으면 그냥 물러납니다

   먼저 보기 https://stellasaju.com/?stella_tailoff=look
   떼어내기  https://stellasaju.com/?stella_tailoff=apply
   되돌리기  https://stellasaju.com/?stella_tailoff=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_tailoff'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_tailoff'] ) );
	$META = '_stella_tailoff_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	/* 덩어리를 찾습니다. 정규식을 쓰지 않고 자리만 짚습니다. */
	$found = array();
	$scan  = $c;
	$base  = 0;
	while ( true ) {
		$at = strpos( $scan, 'BUILD: TAIL-' );
		if ( false === $at ) { break; }
		$from = strrpos( substr( $scan, 0, $at ), '<!-- wp:html -->' );
		if ( false === $from ) { break; }
		$to = strpos( $scan, '<!-- /wp:html -->', $at );
		if ( false === $to ) { break; }
		$to = $to + strlen( '<!-- /wp:html -->' );
		$name = trim( substr( $scan, $at, 20 ) );
		$found[] = array( $base + $from, $base + $to, $to - $from, $name );
		$base = $base + $to;
		$scan = substr( $scan, $to );
	}

	echo "═══ 본문에 박힌 TAIL 덩어리 ═══════════════\n";
	echo '쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "자\n\n";

	if ( ! count( $found ) ) {
		echo "하나도 없습니다. 본문은 이미 깨끗합니다.\n";
		echo "덤 두 쪽이 아직 옛 판으로 보인다면 옛 TAIL 조각이\n";
		echo "켜져 있는지 봐주세요.\n";
		exit;
	}

	$sum = 0;
	foreach ( $found as $f ) {
		echo '  ' . $f[3] . '   ' . number_format( $f[2] ) . "자\n";
		$sum = $sum + $f[2];
	}
	echo "\n  모두 " . count( $found ) . '개 · ' . number_format( $sum ) . "자\n";
	echo '  떼어내면  ' . number_format( strlen( $c ) ) . ' → '
		. number_format( strlen( $c ) - $sum ) . "자\n\n";

	if ( 'look' === $mode ) {
		echo "떼어내시려면 ?stella_tailoff=apply\n";
		exit;
	}
	if ( 'apply' !== $mode ) { echo '?stella_tailoff=look 부터 보세요.'; exit; }

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		update_post_meta( $id, $META, wp_slash( $c ) );
	}

	/* 뒤에서부터 잘라내야 앞 자리가 안 밀립니다 */
	$new = $c;
	$rev = array_reverse( $found );
	foreach ( $rev as $f ) {
		$new = substr( $new, 0, $f[0] ) . substr( $new, $f[1] );
	}
	$new = rtrim( $new ) . "\n";

	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "떼어냈습니다 — " . count( $found ) . "개\n";
	echo '  ' . number_format( strlen( $new ) ) . "자로 줄었습니다\n\n";
	echo "이제 patch160_tailrun 조각을 켜시면\n";
	echo "새 판이 화면에 바로 나옵니다.\n\n";
	echo "되돌리려면 ?stella_tailoff=undo\n";
	exit;
}, 1 );
