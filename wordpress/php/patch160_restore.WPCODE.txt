/* ── 답과 카드를 맡아 두는 자리 ───────────────────────── */
add_action( 'rest_api_init', 'stella_keep_routes' );

function stella_keep_routes() {

	register_rest_route( 'stella/v1', '/keep', array(
		array(
			'methods'             => 'GET',
			'callback'            => 'stella_keep_get',
			'permission_callback' => function () { return is_user_logged_in(); },
		),
		array(
			'methods'             => 'POST',
			'callback'            => 'stella_keep_put',
			'permission_callback' => function () { return is_user_logged_in(); },
		),
	) );
}

function stella_keep_get() {
	$uid = get_current_user_id();
	if ( ! $uid ) { return rest_ensure_response( array( 'ok' => false ) ); }

	$raw = get_user_meta( $uid, 'stella_keep', true );
	$got = $raw ? json_decode( $raw, true ) : array();
	if ( ! is_array( $got ) ) { $got = array(); }

	return rest_ensure_response( array(
		'ok'      => true,
		'answers' => isset( $got['answers'] ) ? $got['answers'] : null,
		'cards'   => isset( $got['cards'] ) ? $got['cards'] : null,
		'at'      => isset( $got['at'] ) ? (int) $got['at'] : 0,
	) );
}

function stella_keep_put( WP_REST_Request $req ) {
	$uid = get_current_user_id();
	if ( ! $uid ) { return rest_ensure_response( array( 'ok' => false ) ); }

	$body = $req->get_json_params();
	if ( ! is_array( $body ) ) { $body = array(); }

	$keep = array(
		'answers' => isset( $body['answers'] ) ? $body['answers'] : null,
		'cards'   => isset( $body['cards'] ) ? $body['cards'] : null,
		'at'      => time(),
	);

	$json = wp_json_encode( $keep );

	/* 너무 큰 것은 받지 않습니다 — 답과 카드는 몇 KB 면 넉넉합니다 */
	if ( strlen( $json ) > 61440 ) {
		return rest_ensure_response( array( 'ok' => false, 'why' => '너무 큽니다' ) );
	}

	update_user_meta( $uid, 'stella_keep', $json );
	return rest_ensure_response( array( 'ok' => true, 'size' => strlen( $json ) ) );
}

/* ── 책 쪽에서만 도는 조각 ────────────────────────────── */
add_action( 'wp_footer', function () {

	if ( ! is_singular() ) { return; }
	$p = get_post();
	if ( ! $p ) { return; }
	if ( 'reading-book' !== $p->post_name ) { return; }
	if ( ! is_user_logged_in() ) { return; }

	/* ── 시험용 · 잊게 하기 ───────────────────────────
	   ?stella_forget=1 이면 이 브라우저의 사본을 지우고 다시 엽니다.
	   폰을 바꾼 손님과 똑같은 처지를 만들어 보는 것입니다.
	   ★ 관리자일 때만 듣습니다. 손님 것은 건드릴 수 없습니다. */
	if ( isset( $_GET['stella_forget'] ) ) {
		if ( current_user_can( 'manage_options' ) ) {
			echo stella_forget_js();
			return;
		}
	}

	echo stella_restore_js();
}, 98 );

function stella_forget_js() {
	return <<<'STELLAFORGET'
<script>
(function(){
  'use strict';
  try{
    localStorage.removeItem('stella_demo');
    localStorage.removeItem('stella_answers');
    localStorage.removeItem('stella_cards');
    sessionStorage.removeItem('stella_restored');
    sessionStorage.removeItem('stella_kept');
  }catch(e){}
  var m=document.getElementById('ssLockMsg');
  if(m){ m.textContent='이 브라우저의 사본을 잊었습니다. 서버에서 다시 불러옵니다…'; }
  setTimeout(function(){ location.replace('/reading-book/'); }, 700);
})();
</script>
STELLAFORGET;
}

function stella_restore_js() {
	return <<<'STELLARESTORE'
<script>
/* ═══════════════════════════════════════════════════════════
   STELLA SAJU — 다른 기기에서도 책이 열립니다   판 RESTORE-2
   ★ 홑 앰퍼샌드와 두 글자 논리기호를 쓰지 않습니다 (사이트 규칙)
   ═══════════════════════════════════════════════════════════ */
(function(){
  'use strict';

  function read(k){
    try{
      var v=localStorage.getItem(k);
      if(!v){ return null; }
      return JSON.parse(v);
    }catch(e){ return null; }
  }
  function write(k, v){
    try{ localStorage.setItem(k, JSON.stringify(v)); return true; }
    catch(e){ return false; }
  }
  function said(t){
    var m=document.getElementById('ssLockMsg');
    if(m){ m.textContent=t; }
  }

  var order=read('stella_demo');

  /* ── 흔한 경우 : 주문이 멀쩡히 있습니다 ───────────────
     이때는 서버에 말도 안 겁니다. 답과 카드만 맡겨 둡니다. */
  var has=false;
  if(order){ if(order.profile){ has=true; } }

  if(has){ return keep(); }

  /* ── 한 번만 시도합니다 (돌고 도는 것을 막습니다) ───── */
  var once=null;
  try{ once=sessionStorage.getItem('stella_restored'); }catch(e){ once='1'; }
  if(once){ return; }

  said('받으신 책을 불러오는 중이에요…');

  var head={'Accept':'application/json'};
  if(window.STELLA_NONCE){ head['X-WP-Nonce']=window.STELLA_NONCE; }

  fetch('/wp-json/stella/v1/me',{credentials:'same-origin',headers:head})
    .then(function(r){ return r.json(); })
    .then(function(me){ got(me); })
    .catch(function(){ });

  function got(me){
    if(!me){ return; }
    if(me.member!==true){ return; }          /* 잠금 화면이 알아서 안내합니다 */

    var pf=me.profile;
    if(!pf){
      said('생년월일시를 아직 안 넣으셨어요. 먼저 넣어주시면 바로 책이 만들어집니다.');
      return;
    }

    var list=me.orders;
    if(!list){ return; }
    if(!list.length){ return; }               /* 주문이 없으면 원래 안내문 */

    /* 치르신 것 가운데 가장 나중 것을 폅니다 */
    var pick=null, i;
    for(i=0;i<list.length;i++){
      if(list[i].status==='paid'){ pick=list[i]; }
    }
    if(!pick){ pick=list[list.length-1]; }

    var made={
      guardian:      pick.guardian || '',
      guardian_slug: pick.slug || '',
      topic:         pick.topic || '',
      oneline:       pick.oneline || '',
      memo:          pick.memo || '',
      deeps:         pick.deeps || [],
      profile:       pf,
      restored:      1
    };
    if(!write('stella_demo', made)){ return; }

    /* 답과 카드도 맡겨둔 것이 있으면 같이 되살립니다 */
    fetch('/wp-json/stella/v1/keep',{credentials:'same-origin',headers:head})
      .then(function(r){ return r.json(); })
      .then(function(k){
        if(k){
          if(k.answers){ write('stella_answers', k.answers); }
          if(k.cards){ write('stella_cards', k.cards); }
        }
        open();
      })
      .catch(function(){ open(); });
  }

  function open(){
    try{ sessionStorage.setItem('stella_restored','1'); }catch(e){}
    said('책을 펴는 중이에요…');
    location.reload();
  }

  /* ── 답과 카드를 서버에 맡겨 둡니다 ───────────────────
     다음에 다른 기기로 여셔도 2부가 그대로 나옵니다.
     한 창에서 한 번만 보냅니다. */
  function keep(){
    var sent=null;
    try{ sent=sessionStorage.getItem('stella_kept'); }catch(e){ sent='1'; }
    if(sent){ return; }

    var a=read('stella_answers'), c=read('stella_cards');
    if(!a){ if(!c){ return; } }

    var h={'Content-Type':'application/json','Accept':'application/json'};
    if(window.STELLA_NONCE){ h['X-WP-Nonce']=window.STELLA_NONCE; }

    try{
      fetch('/wp-json/stella/v1/keep',{
        method:'POST', credentials:'same-origin', headers:h,
        body:JSON.stringify({answers:a, cards:c})
      }).then(function(){
        try{ sessionStorage.setItem('stella_kept','1'); }catch(e){}
      }).catch(function(){});
    }catch(e){}
  }
})();
</script>
STELLARESTORE;
}
