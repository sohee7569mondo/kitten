                                               2026-09-16

   소희 님 : 유앤미 쪽이 통째로 비어 있습니다.

   까닭은 쪽이 아닙니다 — 쪽(번호 760)은 살아 있고, 안에는
   딱 한 줄 <div id="um"></div> 만 들어 있습니다.
   화면은 **조각 아홉 개**가 머리말로 그립니다.
   그 가운데 **하나만 꺼져도 화면이 통째로 빕니다.**

   ?stella_check=1 에서 켜진 조각 133개를 세어 보니 유앤미
   것은 여섯 개뿐이었습니다. 어느 셋이 빠졌는지 이 도구가
   이름을 그대로 찍어 드립니다.

   ── 보는 법 ───────────────────────────────────────────
       https://stellasaju.com/?stella_umon=1

   ★ 아무것도 안 건드립니다. 세기만 합니다.
   ★ 조각 **이름**만 찍습니다. 안에 든 글은 한 자도 안 찍습니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_umon'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	echo "═══ 유앤미 조각 아홉 · 켜짐 꺼짐   판 UMON-1 ═══\n\n";

	/* 조각마다 그 조각에만 있는 표지말.
	   표지말로 찾으므로 소희 님이 이름을 어떻게 붙이셨든 찾아냅니다. */
	$parts = array(
		array( '①', '계산기',        'UANDME_TIERS',            '점수를 셉니다',                 '꼭' ),
		array( '②', '화면 밑감',      'um-chars.js',             '그림·설정·연예인·옷',           '꼭' ),
		array( '③', '쪽 만들기',      'stella_um3',              '쪽을 한 번 만드는 것',           '아님' ),
		array( '④', '카톡 카드 그림',  'stella_um_bg',            '카톡에 갈 점수 그림',            '아님' ),
		array( '⑤', '카톡 그림 올리기','um_kimg',                 '카카오로 그림 보내기',           '아님' ),
		array( '⑥', '화면 그리기',    'uandme-screens.js',       '화면 다섯을 실제로 그립니다',     '꼭' ),
		array( '⑦', '에스엔에스 카드','jetpack_open_graph_tags', '스레드·엑스·페북 미리보기',       '아님' ),
		array( '⑧', '짧은 주소',      "REQUEST_URI'] = '/uandme/'",              '물음표 없는 주소 열어주기',       '아님' ),
		array( '⑨', '이상형',        'UM_IDEAL_ON',             '이상형 화면',                   '아님' ),
	);

	$rows = $wpdb->get_results(
		"SELECT ID, post_title, post_status, post_content FROM {$wpdb->posts}
		  WHERE post_type = 'wpcode' ORDER BY post_title ASC" );

	$missing = array();
	$offList = array();

	foreach ( $parts as $p ) {

		$found = array();
		foreach ( (array) $rows as $r ) {
			if ( false === strpos( (string) $r->post_content, $p[2] ) ) { continue; }
			$found[] = $r;
		}

		$anyOn = false;
		foreach ( $found as $r ) {
			if ( 'publish' === $r->post_status ) { $anyOn = true; }
		}

		$mark = '✔ 켜짐';
		if ( ! $anyOn ) {
			if ( count( $found ) ) { $mark = '✖ 꺼짐'; } else { $mark = '✖ 아예 없음'; }
		}

		printf( "%s %s %s · %s — %s\n",
			$p[0], ( '꼭' === $p[4] ? '★ 꼭 있어야' : '        있으면' ), $mark, $p[1], $p[3] );

		foreach ( $found as $r ) {
			printf( "      · %s   [%s · 번호 %d]\n",
				$r->post_title,
				( 'publish' === $r->post_status ? '켜짐' : '꺼짐' ),
				$r->ID );
			if ( 'publish' !== $r->post_status ) { $offList[] = $r->post_title; }
		}
		if ( ! count( $found ) ) {
			echo "      · WPCode 어디에도 없습니다 — 새로 붙이셔야 합니다\n";
		}

		if ( '꼭' === $p[4] ) {
			if ( ! $anyOn ) { $missing[] = $p[0] . ' ' . $p[1]; }
		}
		echo "\n";
	}

	/* ── 판정 ──────────────────────────────────────────── */
	echo "── 그래서 ─────────────────────────────────────\n";
	if ( count( $missing ) ) {
		echo "  ★ 유앤미가 비는 까닭 — 꼭 있어야 할 것이 꺼져 있습니다\n";
		foreach ( $missing as $m ) { echo '      ' . $m . "\n"; }
		echo "\n  WPCode 목록에서 위에 적힌 **꺼짐** 이름을 찾아\n";
		echo "  Active 로 켜시고 /uandme/ 를 새로고침해 보세요.\n";
	} else {
		echo "  꼭 있어야 할 것 셋(①②⑥)은 다 켜져 있습니다.\n";
		echo "  그래도 화면이 비면 자바스크립트가 멈춘 것입니다 —\n";
		echo "  /uandme/ 에서 F12 를 눌러 빨간 줄을 보여주세요.\n";
	}

	/* ── 이름에 유앤미가 든 조각 모두 ───────────────────── */
	echo "\n── 이름에 유앤미가 든 조각 모두 ───────────────\n";
	$seen = 0;
	foreach ( (array) $rows as $r ) {
		$t = (string) $r->post_title;
		$hit = false;
		if ( false !== strpos( $t, '유앤미' ) ) { $hit = true; }
		if ( false !== stripos( $t, 'uandme' ) ) { $hit = true; }
		if ( false !== stripos( $t, 'um_' ) ) { $hit = true; }
		if ( ! $hit ) { continue; }
		$seen++;
		printf( "  %s · %s   [번호 %d]\n",
			( 'publish' === $r->post_status ? '켜짐' : '꺼짐' ), $t, $r->ID );
	}
	if ( ! $seen ) { echo "  없습니다 (이름을 다르게 붙이신 듯합니다)\n"; }

	echo "\n★ 조각 이름만 찍었습니다. 안에 든 글은 한 자도 안 찍었습니다.\n";
	exit;
}, 1 );
