/* ════════════════════════════════════════════════════════════
   유앤미 · 카톡 그림 진단 (보기만 합니다)

   무엇을 보는가
     ① 카톡에 보낼 그림 주소가 설정에 들어 있는지
     ② 그 그림을 서버가 실제로 열 수 있는지
     ③ 이 서버가 그림을 그릴 줄 아는지 (GD)
     ④ 한글 글꼴이 있는지 — 점수 카드를 그리려면 필요합니다

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Auto Insert → Run Everywhere → 저장 → Active

   보기 : https://stellasaju.com/?um_diag=1
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['um_diag'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:14px/1.75 system-ui;background:#111;color:#eee;padding:24px;max-width:820px">';
	echo '<h1 style="color:#ffd76a">유앤미 · 카톡 그림 진단</h1>';

	/* ── ① 그림을 그릴 줄 아는가 ───────────────────────── */
	echo '<h2 style="color:#9fd">① 그림 그리기(GD)</h2>';
	if ( ! function_exists( 'gd_info' ) ) {
		echo '<p style="color:#f88">GD 가 없습니다 — 서버가 그림을 못 그립니다.</p>';
	} else {
		$gd = gd_info();
		echo '<p style="color:#9f9">GD 있습니다 · ' . esc_html( $gd['GD Version'] ) . '</p>';
		echo '<p>PNG ' . ( ! empty( $gd['PNG Support'] ) ? '예' : '아니오' ) .
		     ' · JPEG ' . ( ! empty( $gd['JPEG Support'] ) ? '예' : '아니오' ) .
		     ' · FreeType(글꼴) ' . ( ! empty( $gd['FreeType Support'] ) ? '<b style="color:#9f9">예</b>' : '<b style="color:#f88">아니오</b>' ) . '</p>';
	}

	/* ── ② 한글 글꼴이 있는가 ──────────────────────────── */
	echo '<h2 style="color:#9fd">② 한글 글꼴</h2>';
	$dirs = array(
		'/usr/share/fonts', '/usr/local/share/fonts', '/usr/share/fonts/truetype',
		WP_CONTENT_DIR . '/uploads', ABSPATH . 'wp-includes/fonts',
	);
	$found = array();
	foreach ( $dirs as $d ) {
		if ( ! is_dir( $d ) ) { continue; }
		$it = @new RecursiveIteratorIterator(
			new RecursiveDirectoryIterator( $d, FilesystemIterator::SKIP_DOTS ),
			RecursiveIteratorIterator::SELF_FIRST );
		if ( ! $it ) { continue; }
		$n = 0;
		foreach ( $it as $f ) {
			$n++;
			if ( $n > 4000 ) { break; }
			$p = $f->getPathname();
			if ( ! preg_match( '/\.(ttf|otf|ttc)$/i', $p ) ) { continue; }
			$b = strtolower( basename( $p ) );
			$korean = ( strpos( $b, 'nanum' ) !== false )
			       || ( strpos( $b, 'noto' ) !== false )
			       || ( strpos( $b, 'malgun' ) !== false )
			       || ( strpos( $b, 'gothic' ) !== false )
			       || ( strpos( $b, 'cjk' ) !== false )
			       || ( strpos( $b, 'jua' ) !== false );
			$found[] = array( $p, $korean );
		}
	}
	if ( ! $found ) {
		echo '<p style="color:#fa8">글꼴 파일을 못 찾았습니다. 그래도 괜찮습니다 — 글꼴은 미디어에 올려서 쓸 수 있습니다.</p>';
	} else {
		$kor = 0;
		echo '<div style="max-height:260px;overflow:auto;background:#000;padding:10px;border-radius:8px">';
		foreach ( $found as $f ) {
			if ( $f[1] ) { $kor++; echo '<div style="color:#9f9">' . esc_html( $f[0] ) . '</div>'; }
			else { echo '<div style="color:#777">' . esc_html( $f[0] ) . '</div>'; }
		}
		echo '</div>';
		echo '<p>모두 ' . count( $found ) . '개 · 한글로 보이는 것 <b style="color:' .
		     ( $kor ? '#9f9' : '#f88' ) . '">' . $kor . '개</b> (초록색)</p>';
	}

	/* ── ③ 올릴 곳이 쓰기 가능한가 ─────────────────────── */
	echo '<h2 style="color:#9fd">③ 파일 쓰기</h2>';
	$up = wp_upload_dir();
	echo '<p>' . esc_html( $up['basedir'] ) . ' → ' .
	     ( is_writable( $up['basedir'] ) ? '<b style="color:#9f9">쓸 수 있음</b>' : '<b style="color:#f88">못 씀</b>' ) . '</p>';

	/* ── ④ 카톡에 보낼 그림을 서버가 열 수 있는가 ──────── */
	echo '<h2 style="color:#9fd">④ 카톡 그림</h2>';
	$pic = 'https://stellasaju.com/wp-content/uploads/2026/09/uandme_cha-02-300px.png';
	echo '<p>' . esc_html( $pic ) . '</p>';
	$r = wp_remote_head( $pic, array( 'timeout' => 12, 'redirection' => 3 ) );
	if ( is_wp_error( $r ) ) {
		echo '<p style="color:#f88">못 열었습니다 — ' . esc_html( $r->get_error_message() ) . '</p>';
	} else {
		$code = wp_remote_retrieve_response_code( $r );
		echo '<p>응답 <b style="color:' . ( 200 === $code ? '#9f9' : '#f88' ) . '">' . intval( $code ) . '</b>' .
		     ' · 형식 ' . esc_html( wp_remote_retrieve_header( $r, 'content-type' ) ) .
		     ' · 크기 ' . esc_html( wp_remote_retrieve_header( $r, 'content-length' ) ) . ' 바이트</p>';
	}
	$path = str_replace( 'https://stellasaju.com/wp-content/uploads', $up['basedir'], $pic );
	echo '<p>서버 안 파일 : ' . esc_html( $path ) . ' → ' .
	     ( file_exists( $path ) ? '<b style="color:#9f9">있음</b>' : '<b style="color:#f88">없음</b>' ) . '</p>';

	/* ── ⑤ 바깥으로 나갈 수 있는가 (글꼴 받아오기용) ───── */
	echo '<h2 style="color:#9fd">⑤ 바깥 인터넷</h2>';
	$g = wp_remote_head( 'https://fonts.gstatic.com/', array( 'timeout' => 10 ) );
	echo '<p>fonts.gstatic.com → ' . ( is_wp_error( $g )
		? '<b style="color:#f88">막힘 (' . esc_html( $g->get_error_message() ) . ')</b>'
		: '<b style="color:#9f9">열림 (' . intval( wp_remote_retrieve_response_code( $g ) ) . ')</b>' ) . '</p>';

	echo '<hr style="margin:26px 0;border-color:#333">';
	echo '<p style="color:#888">이 화면을 통째로 찍어서 보내주세요.</p>';
	exit;
} );
