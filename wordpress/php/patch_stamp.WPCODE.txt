/* ════════════════════════════════════════════════════════════
   홈 아래 흰 띠 없애기 · 소희 님께

   무엇이었나
     마지막 어두운 칸 바로 밑에 개발용 도장이 하나 깔려 있습니다.

       <div class="build-stamp">빌드 v16c · 2026년 9월 8일</div>

     여기에 종이색 바탕과 위아래 여백 48px 이 붙어 있어서,
     어두운 칸과 푸터 사이에 흰 줄이 그어진 것처럼 보입니다.
     글자는 거의 투명해서 아무것도 안 적힌 흰 띠로만 보입니다.

   무엇을 하나
     그 도장의 겉모습 규칙 하나를 감춤으로 바꿉니다.
     글자는 코드에 그대로 남아 있어서, 필요하면 되돌리기로
     다시 보이게 할 수 있습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 「어디서나 실행」 → 저장 → 활성

   ① 미리보기
        https://stellasaju.com/?stella_stamp=1
   ② 적용
        https://stellasaju.com/?stella_stamp=1&go=1
   되돌리기
        https://stellasaju.com/?stella_stamp=1&undo=1
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_stamp'] ) ) {
		return;
	}
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다. 관리자로 로그인한 창에서 열어주세요.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	$PID  = 62;
	$META = '_stella_stamp_backup';
	$OPEN = '#stellar-home .build-stamp{';
	$NEW  = "\n  display:none;}";
	$go   = isset( $_GET['go'] );
	$undo = isset( $_GET['undo'] );

	$safe = function ( $s ) {
		if ( function_exists( 'mb_convert_encoding' ) ) {
			$s = mb_convert_encoding( $s, 'UTF-8', 'UTF-8' );
		}
		return esc_html( $s );
	};

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:15px/1.8 -apple-system,system-ui,sans-serif;'
		. 'background:#111;color:#eee;padding:24px;max-width:900px">';
	printf(
		'<h1 style="color:#ffd76a">홈 아래 흰 띠 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);

	if ( $undo ) {
		$bak = get_post_meta( $PID, $META, true );
		if ( $bak ) {
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $PID ) );
			delete_post_meta( $PID, $META );
			clean_post_cache( $PID );
			if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
			echo '<h2 style="color:#7bff9b">되돌렸습니다.</h2></body>';
		} else {
			echo '<h2 style="color:#888">사본이 없습니다.</h2></body>';
		}
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare( "SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $PID ) );
	if ( ! $row ) {
		echo '<h2 style="color:#ff7b7b">62번 페이지를 못 읽었습니다.</h2></body>';
		exit;
	}
	$c = $row->post_content;

	$p = strpos( $c, $OPEN );
	if ( false === $p ) {
		echo '<h2 style="color:#ff7b7b">도장 규칙을 못 찾았습니다. 아무것도 바꾸지 않았습니다.</h2></body>';
		exit;
	}
	$q = strpos( $c, '}', $p + strlen( $OPEN ) );
	if ( false === $q ) {
		echo '<h2 style="color:#ff7b7b">규칙의 끝을 못 찾았습니다. 아무것도 바꾸지 않았습니다.</h2></body>';
		exit;
	}

	$old_rule = substr( $c, $p, $q - $p + 1 );
	if ( false !== strpos( $old_rule, 'display:none' ) ) {
		echo '<h2 style="color:#7bff9b">이미 감춰져 있습니다.</h2></body>';
		exit;
	}

	$new_c = substr( $c, 0, $p ) . $OPEN . $NEW . substr( $c, $q + 1 );

	printf(
		'<p style="margin:16px 0 4px;color:#ff9b9b">지금</p>'
		. '<pre style="white-space:pre-wrap;background:#000;padding:12px;'
		. 'font:12.5px/1.6 monospace">%s</pre>',
		$safe( $old_rule )
	);
	printf(
		'<p style="margin:16px 0 4px;color:#7bff9b">바뀔 모습</p>'
		. '<pre style="white-space:pre-wrap;background:#000;padding:12px;'
		. 'font:12.5px/1.6 monospace">%s</pre>',
		$safe( $OPEN . $NEW )
	);

	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다. 아무것도 바뀌지 않았습니다.</h2>';
		echo '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_stamp=1&go=1">'
			. '→ 이대로 바꾸기</a></p></body>';
		exit;
	}

	if ( ! get_post_meta( $PID, $META, true ) ) {
		update_post_meta( $PID, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new_c ), array( 'ID' => $PID ) );
	clean_post_cache( $PID );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<h2 style="color:#7bff9b">바꿨습니다</h2>';
	echo '<p style="font-size:17px">휴대폰으로 <a style="color:#ffd76a" href="/">홈</a> 맨 아래를 봐주세요.</p>';
	echo '<p style="color:#888">되돌리려면 ?stella_stamp=1&undo=1</p></body>';
	exit;
} );
