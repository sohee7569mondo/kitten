/* ══════════════════════════════════════════════════════════
   「두 부분으로 나뉩니다」를 장 안내로  ★ 2026-09-13 · 소희 님께
                                        판 INTRO-2

   소희 님 : 「바로 아래 「이 책은 두 부분으로 나뉩니다」 이 부분은
             다시 써야 할 거 같아」

   맞습니다. 책은 이미 장으로 짜여 있는데 1부·2부로 설명하고 있어요.
   patch160_open 의 세 번째 자리가 그것이었는데, 글자가 안 맞아
   건너뛰었고 지금은 「이미 들어갔다」에 막혀 다시 못 넣습니다.

   그래서 그 자리 하나만 따로 고칩니다.

   ── 어떻게 찾나 ───────────────────────────────────────
   긴 글자를 맞추지 않습니다. 앞뒤 표시로 구간을 잡습니다.

       앞   「이 책은 두 부분으로 나뉩니다」가 있는 줄의 머리
       뒤   그 뒤 처음 나오는  ');

   그 사이를  stellaIntro());  한 줄로 바꿉니다.
   stellaIntro() 는 patch160_open 이 이미 넣어둔 함수입니다 —
   주제마다 다른 장 안내를 만들어 줍니다.

   ★ preview 가 **지워질 글을 그대로 보여드립니다.** 눈으로
     확인하시고 apply 하세요. 이상하면 저에게 그 글을 주세요.

   ── 판 INTRO-2 (2026-09-13) · 이름을 바꿨습니다 ────────
   소희 님 : 「?stella_intro=preview 이거 홈으로 감」

   조각이 안 돌고 홈으로 넘어갔습니다. 이름을 stella_bookintro 로
   바꿨습니다 — intro 라는 말은 테마나 다른 플러그인이 가로채는
   일이 있습니다.

   ★ 그래도 홈으로 가면 조각이 안 켜진 것입니다. 이렇게 보세요.

       1  WPCode -> Code Snippets 목록에서 이 조각을 찾습니다
       2  Active(초록) 인지 봅니다. 회색이면 켜세요.
       3  조각을 열어 Location(위치)이
          「Run Everywhere / 어디서나 실행」인지 봅니다.
          Auto Insert 의 다른 값이면 안 돕니다.
       4  저장하고 다시 주소를 엽니다.

   ★ 다른 조각이 도는지로 가려낼 수 있습니다.
       https://stellasaju.com/?stella_audit=1
     이것이 잘 나오면 WPCode 는 멀쩡하고 이 조각만 문제입니다.
     이것도 홈으로 가면 WPCode 쪽을 보셔야 합니다.

   ── 하는 법 ───────────────────────────────────────────
   ① 먼저 보기   https://stellasaju.com/?stella_bookintro=preview
   ② 정말 바꾸기 https://stellasaju.com/?stella_bookintro=apply
   ③ 되돌리기    https://stellasaju.com/?stella_bookintro=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_bookintro'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_bookintro'] ) );
	$META = '_stella_bookintro_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽을 못 찾았습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$back = get_post_meta( $id, $META, true );
		if ( '' === $back ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $back ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다. ' . number_format( strlen( $back ) ) . "바이트\n";
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	if ( '' === $c ) { echo '내용을 못 읽었습니다.'; exit; }

	echo "「두 부분으로 나뉩니다」를 장 안내로   판 INTRO-2\n";
	echo '책 쪽 ' . $id . ' · ' . number_format( strlen( $c ) ) . "바이트\n\n";

	/* stellaIntro 가 있어야 합니다 */
	if ( false === strpos( $c, 'function stellaIntro' ) ) {
		echo "★ stellaIntro() 가 책에 없습니다.\n";
		echo "  patch160_open 을 먼저 넣으셔야 합니다.\n";
		exit;
	}
	echo "stellaIntro() 있음 — 장 안내를 만들 준비가 되어 있습니다.\n";

	if ( false !== strpos( $c, 'stellaIntro())' ) ) {
		echo "\n이미 바뀌어 있습니다. 손댈 것이 없습니다.\n";
		exit;
	}

	$KEY = '이 책은 두 부분으로 나뉩니다';
	$n   = substr_count( $c, $KEY );
	if ( 1 !== $n ) {
		echo "\n★ 「" . $KEY . "」가 " . $n . "군데입니다. 한 군데여야 합니다.\n";
		if ( 0 === $n ) {
			echo "  이미 누가 고쳐 두었을 수 있습니다. 책을 한 번 보세요.\n";
		}
		exit;
	}
	$at = strpos( $c, $KEY );

	/* 그 줄의 머리 */
	$ls = strrpos( substr( $c, 0, $at ), "\n" );
	if ( false === $ls ) { $ls = 0; } else { $ls = $ls + 1; }

	/* 그 뒤 처음 나오는  ');  */
	$ce = strpos( $c, "');", $at );
	if ( false === $ce ) {
		echo "\n★ 문단의 끝( '); )을 못 찾았습니다. 그만둡니다.\n";
		exit;
	}
	$ce = $ce + 3;

	$old = substr( $c, $ls, $ce - $ls );
	echo '바꿀 크기 : ' . number_format( strlen( $old ) ) . "바이트\n";

	if ( strlen( $old ) > 4000 ) {
		echo "\n★ 너무 큽니다(4,000바이트 넘음). 구간을 잘못 잡은 듯합니다.\n";
		echo "  손대지 않고 그만둡니다. 아래를 저에게 주세요.\n";
		echo "\n──── 잡은 구간의 앞머리 ────\n";
		echo substr( $old, 0, 700 ) . " …\n";
		exit;
	}

	echo "\n──── 지워질 글 (그대로입니다) ─────────────────────\n";
	echo $old;
	echo "\n──── 여기까지 ─────────────────────────────────────\n";
	echo "\n──── 그 자리에 들어갈 글 ──────────────────────────\n";
	echo "    stellaIntro());\n";
	echo "──── 끝 ───────────────────────────────────────────\n";
	echo "\n  stellaIntro() 는 주제마다 다른 장 안내를 만듭니다.\n";
	echo "  보기 — 이혼운이면 「이 결혼에서 무엇이 오래 쌓였고, 이제\n";
	echo "         무엇을 정리해야 하는지」가 가운데 한 줄로 들어갑니다.\n";

	$made = substr( $c, 0, $ls ) . "    stellaIntro());" . substr( $c, $ce );

	echo "\n" . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $made ) )
	   . '바이트 (' . number_format( strlen( $made ) - strlen( $c ) ) . ")\n";

	if ( 'apply' !== $mode ) {
		echo "\n아직 안 바꿨습니다. 정말 바꾸시려면 ?stella_bookintro=apply\n";
		exit;
	}

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		update_post_meta( $id, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $made ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "\n바꿨습니다. 주제를 바꿔가며 여는 쪽을 보세요.\n";
	echo "되돌리기 : ?stella_bookintro=undo\n";
	exit;
}, 1 );
