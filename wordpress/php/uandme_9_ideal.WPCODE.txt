  function ideal(y, m, d) {
    var S = window.StellaSaju;
    if (!S) { return null; }

    for (i = 0; i < c.five.length; i++) {

    }

    /* 숫자만 받고, 다 채우면 다음 칸으로 넘어갑니다 */
    var ids = ['#iy', '#im', '#id'], caps = [4, 2, 2], q;
    for (q = 0; q < ids.length; q++) {
      (function (n, cap, nextSel) {
        if (!n) { return; }
        n.addEventListener('input', function () {
          var v = this.value.replace(/[^0-9]/g, '');
          if (v !== this.value) { this.value = v; }
          if (v.length >= cap) {
            if (nextSel) {
              var nx = root.querySelector(nextSel);
              if (nx) { nx.focus(); }
            }
          }
        });
      }(root.querySelector(ids[q]), caps[q], ids[q + 1]));
    }
    var b = root.querySelector('#igo');
    if (!b) { return; }
    b.addEventListener('click', function () {
      function num(sel) {
        var n = root.querySelector(sel);
        if (!n) { return 0; }
        return parseInt(String(n.value).replace(/[^0-9]/g, ''), 10) || 0;
      }
      var y = num('#iy'), m = num('#im'), d = num('#id');
      var ok = true;
      if (!y) { ok = false; }
      if (!m) { ok = false; }
      if (!d) { ok = false; }
      if (ok) { if (y < 1900) { ok = false; } }
      if (ok) { if (y > 2025) { ok = false; } }
      if (ok) { if (m < 1) { ok = false; } }
      if (ok) { if (m > 12) { ok = false; } }
      if (ok) { if (d < 1) { ok = false; } }
      if (ok) { if (d > 31) { ok = false; } }
      if (!ok) { this.textContent = '날짜를 다시 봐주세요'; return; }
      st.y = y; st.m = m; st.d = d;
      show(y, m, d);
    });
  }
