/* ══════════════════════════════════════════════════════════
   스타 사주(406) — 별자리 자리를 제대로
   ★ 2026-09-09 · 소희 님께

   소희 님 : 「염소자리의 내용보다는 해설 이유가 더 길어」

   맞습니다. 지금은 이렇습니다.

     태어난 날의 해 — 염소자리
     서양 별자리로는 쌓아 올리는 사람입니다. 태양 별자리는
     날짜만으로 정해져서, 태어난 시각을 몰라도 세울 수 있는
     유일한 자리예요.

   알맹이가 「쌓아 올리는 사람」 여섯 글자고, 왜 그런지 하는
   변명이 두 줄입니다. 앞뒤가 바뀌었습니다.

   고치는 것 둘
     1. 열두 별자리마다 세 문장씩 씁니다.
     2. 「사주는 이렇게 말하고 별자리는 이렇게 말합니다」를
        한 줄 붙입니다. 두 셈법이 같은 쪽을 가리키는지 아닌지가
        이 자리에서 제일 재미있는 대목입니다.
     3. 「날짜만으로 정해져서…」는 작은 글씨로 내립니다.

   해보기   https://stellasaju.com/?stella_sun406=preview
   붙이기   https://stellasaju.com/?stella_sun406=apply
   되돌리기 https://stellasaju.com/?stella_sun406=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_sun406'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_sun406'] ) );
	$META = '_stella_sun406_bak';

	$page = get_page_by_path( 'star-saju' );
	if ( ! $page ) { echo '스타 사주 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	if ( false !== strpos( $c, 'SR_SUN2' ) ) {
		echo "이미 붙어 있습니다.\n먼저 ?stella_sun406=undo 로 되돌리세요.";
		exit;
	}

	$OLD = "'<p>서양 별자리로는 <em>'+SUN[p.s]+'</em>입니다. 태양 별자리는 날짜만으로 정해져서, '+";
	$FN  = '  function show(p, known){';

	$anchors = array( '별자리 글' => $OLD, '함수 앞' => $FN );
	$bad = false;
	foreach ( $anchors as $nm => $a ) {
		$n = substr_count( $c, $a );
		echo sprintf( "%-12s %d군데\n", $nm, $n );
		if ( 1 !== $n ) { $bad = true; }
	}
	if ( $bad ) {
		echo "\n한 군데가 아니어서 아무것도 안 바꿨습니다.";
		exit;
	}

	$js = <<<'STELLA_SUN_JS'

  /* ══ 별자리를 제대로 ─ 2026-09-09 ═══════════════════════════ */

  var SR_SUN2 = {
    '양자리':'먼저 움직이고 나중에 생각합니다. 시작하는 일에서 제일 빛나고, 오래 끄는 일에서 지칩니다. 화가 나도 오래 안 갑니다.',
    '황소자리':'좋아하는 것을 오래 좋아합니다. 서두르는 법이 없고, 한번 정한 것은 잘 안 바꿉니다. 머리로 따지기 전에 몸으로 느껴지는 것을 믿습니다.',
    '쌍둥이자리':'말이 먼저 나옵니다. 궁금한 것이 많아서 한자리에 오래 못 있습니다. 지루한 것을 제일 못 견딥니다.',
    '게자리':'안과 밖을 나눕니다. 내 사람이라고 정하면 끝까지 갑니다. 대신 상처를 오래 기억합니다.',
    '사자자리':'보는 눈이 있어야 힘이 납니다. 인정받으면 몇 배로 해냅니다. 무시당하는 것을 제일 못 참습니다.',
    '처녀자리':'어긋난 것이 먼저 보입니다. 남보다 자기한테 더 엄격합니다. 도와주는 방식이 잔소리로 보일 때가 있습니다.',
    '천칭자리':'한쪽으로 안 기울려고 합니다. 사람 사이의 공기를 읽습니다. 그래서 무엇 하나 고르는 데 오래 걸립니다.',
    '전갈자리':'깊이 들어갑니다. 겉으로 하는 말과 속이 다릅니다. 한번 등을 돌리면 잘 되돌아오지 않습니다.',
    '사수자리':'멀리 봅니다. 답답한 것을 못 견디고 묶이는 것을 싫어합니다. 솔직해서 가끔 자기가 다칩니다.',
    '염소자리':'쌓아 올립니다. 지금 참는 것이 나중에 값이 된다는 걸 압니다. 놀 때도 계획이 있습니다.',
    '물병자리':'남들과 같은 것이 싫습니다. 사람을 좋아하는데 거리도 필요합니다. 늘 한 발 떨어져서 봅니다.',
    '물고기자리':'남의 기분이 그대로 들어옵니다. 경계가 흐려서 잘 스며듭니다. 현실보다 마음이 먼저입니다.'
  };

  /* 별자리 네 갈래와 오행 — 대강 겹치는 자리만 */
  var SR_SUN_EL = {
    '양자리':'화','사자자리':'화','사수자리':'화',
    '황소자리':'토','처녀자리':'토','염소자리':'토',
    '쌍둥이자리':'목','천칭자리':'목','물병자리':'목',
    '게자리':'수','전갈자리':'수','물고기자리':'수'
  };
  var SR_SUN_GROUP = { '화':'불','토':'흙','목':'바람','수':'물' };

  function srSun(p){
    var body = SR_SUN2[p.s] ? SR_SUN2[p.s] : '';
    var se = SR_SUN_EL[p.s];
    if(!se){ return body; }

    var line;
    if(se === p.hi){
      line = '이 별자리는 <b>' + SR_SUN_GROUP[se] + '</b> 갈래입니다. ' +
             '사주에서 가장 많은 것도 <b>' + p.hi + '</b>입니다. ' +
             '동양과 서양이 같은 쪽을 가리킵니다.';
    } else if(p.hi === '금'){
      line = '이 별자리는 <b>' + SR_SUN_GROUP[se] + '</b> 갈래입니다. ' +
             '사주에서 가장 많은 것은 <b>금</b>인데, 서양 별자리에는 금에 맞는 갈래가 아예 없습니다. ' +
             '두 셈법이 세상을 나누는 방식이 다릅니다.';
    } else {
      line = '이 별자리는 <b>' + SR_SUN_GROUP[se] + '</b> 갈래입니다. ' +
             '그런데 사주에서 가장 많은 것은 <b>' + p.hi + '</b>입니다. ' +
             '두 셈법이 서로 다른 쪽을 가리킵니다 — 사람은 원래 한마디로 안 끝납니다.';
    }
    return body + ' ' + line;
  }


STELLA_SUN_JS;

	$NEWP = "'<p><em>'+SUN[p.s]+'</em>입니다. '+srSun(p)+'</p>'+\n"
	      . "      '<p class=\"mini\">태양 별자리는 날짜만으로 정해져서, '+";

	$new = $c;
	$new = str_replace( $OLD, $NEWP, $new );
	$new = str_replace( $FN, $js . $FN, $new );

	if ( 'preview' === $mode ) {
		echo "\n자리를 찾았습니다. 글자 수 "
		   . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $new ) ) . "\n";
		echo "붙이시려면 ?stella_sun406=apply\n";
		exit;
	}

	update_post_meta( $id, $META, $c );
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "\n붙였습니다.\n되돌리려면 ?stella_sun406=undo\n";
	exit;
}, 1 );
