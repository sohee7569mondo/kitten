       140ms  wpautop  ← 게다가 16,473글자를 멋대로 집어넣습니다

   그래서 책 쪽에서만 이 손질들을 빼냅니다.

add_action( 'template_redirect', function () {

	if ( ! is_page( 'reading-book' ) ) { return; }

	global $wp_filter;
	if ( ! isset( $wp_filter['the_content'] ) ) { return; }

	/* 이름에 이 조각이 들어 있으면 뺍니다 */
	$KILL = array(
		'sharing_display',
		'Subscriptions\\',
		'apply_block_hooks_to_content',
		'wpautop',
		'wptexturize',
		'convert_smilies',
		'capital_P_dangit',
		'crowdsignal_link',
		'WP_Embed::autoembed',
		'wpcomsh_make_content_clickable',
		'Image_CDN',
		'Jetpack_RelatedPosts',
		'photon_subsizes',
		'jetpack_spotify',
		'jetpack_fix_youtube',
		'jetpack_videopress',
		'latex_markup',
		'Jetpack_Carousel',
		'MailPoet',
		'Newspack_Blocks',
		'Paypal_Payments',
	);

	$gone = 0;
	foreach ( $wp_filter['the_content']->callbacks as $prio => $cbs ) {
		foreach ( $cbs as $idn => $cb ) {
			$fn = $cb['function'];

			if ( is_string( $fn ) ) { $nm = $fn; }
			elseif ( is_array( $fn ) ) {
				$cls = is_object( $fn[0] ) ? get_class( $fn[0] ) : (string) $fn[0];
				$nm  = $cls . '::' . (string) $fn[1];
			} else { continue; }

			foreach ( $KILL as $k ) {
				if ( false === strpos( $nm, $k ) ) { continue; }
				remove_filter( 'the_content', $fn, $prio );
				$gone++;
				break;
			}
		}
	}

	if ( current_user_can( 'manage_options' ) ) {
		add_action( 'wp_footer', function () use ( $gone ) {
			echo '<script>console.log("STELLA FAST-1 : 손질 ' . (int) $gone . '가지 뺐습니다");</script>';
		}, 999 );
	}
}, 5 );
