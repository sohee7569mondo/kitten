	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;
	}

  };
  function normTopic(t){
    if(!t){ return ''; }
    if(TMAP[t]){ return TMAP[t]; }
    if(ZO[t]){ return t; }
    if(DD[t]){ return t; }
    /* 「연애운세」처럼 뒤에 '세'가 붙어 오면 떼고 다시 찾습니다 */
    if(t.length>2){ if(t.slice(-2)==='운세'){
      var s=t.slice(0,-1);
      if(TMAP[s]){ return TMAP[s]; }
      if(ZO[s]){ return s; }
      if(DD[s]){ return s; }
    } }
    /* 「연애운」처럼 왔는데 원고가 「연애운세」인 경우 */
    if(t.slice(-1)==='운'){
      var w=t+'세';
      if(ZO[w]){ return w; }
      if(DD[w]){ return w; }
    }
    return t;
  }

    var at=html.indexOf(word);
    if(at<0){ return html; }
    var from=at, to=at;
    while(from>0){
      var ch=html.charAt(from-1);
      if(ch==='>'){ break; }
      if(ch===' '){ if(html.charAt(from-2)==='.'){ break; } }
      from-=1;
    }
    to=html.indexOf('.', at);
    if(to<0){ to=html.length; } else { to+=1; }
    var mid=html.slice(from,to);
    if(mid.indexOf('<')>=0){ return html; }
    return html.slice(0,from)+'<strong class="mine">'+mid+'</strong>'+html.slice(to);
  }

  function sheet(html){
    return '<div class="page" data-part="tail">'+html+'<div class="folio"></div></div>';
  }

  function build(topic, sign, animal){
    var made=[], zo=ZO[topic], dd=DD[topic];
    var zc=null, dc=null, zlong=0;
    if(zo){ if(zo.c){ zc=zo.c[sign]; } if(zo.long){ zlong=1; } }
    if(dd){ dc=dd[animal]; }

    var badge='<p class="bonus">덤으로 드리는 두 쪽</p>';

    /* 하나 · 별자리 */
    if(zc){
      var zbody;
      if(zlong){ zbody='<h3>'+sign+' · '+zc[0]+'</h3>'+zc[1]; }
      else {
        zbody='<h3>'+sign+'</h3><p>'+zc+'</p>';
        if(zo.z){ zbody+=zo.z; }
      }
      made.push(sheet(
        badge+
        '<p class="eyebrow">별자리로 다시 보면</p>'+
        '<h2>'+zo.q+'</h2>'+
        '<p class="dek">여덟 글자로 보신 것을 하늘로 한 번 더 겹쳐 봅니다. '+
        '여기서는 누구나 아는 태양 자리로만 봅니다</p>'+
        zbody));
    }

    /* 둘 · 띠 */
    if(dc){
      made.push(sheet(
        badge+
        '<p class="eyebrow">띠로 보면</p>'+
        '<h2>'+animal+' · '+dc[0]+'</h2>'+
        '<p class="dek">여덟 글자의 맨 앞 글자가 곧 띠입니다. '+
        '어릴 때부터 알고 계셨던 그 한 글자가 사실은 사주의 첫 칸이에요</p>'+
        dc[1]+
        '<p class="mini">앞에서 본 여덟 글자가 세밀한 결이라면, 띠는 그 아래 깔린 바탕색입니다. '+
        '겹치는 데가 있다면 그 자리가 그만큼 분명하다는 뜻입니다.</p>'));
    }
    return made;
  }

  function start(){
    var box=document.getElementById('ssb');
    if(!box){ return false; }
    if(box.getAttribute('data-tail')==='1'){ return true; }

    var order=null;
    try{ order=JSON.parse(localStorage.getItem('stella_demo')); }catch(e){ order=null; }
    if(!order){ return false; }

    var S=window.StellaSaju;
    if(!S){ return false; }
    if(!S.lastChart){ return false; }
    var chart=S.lastChart();
    if(!chart){ return false; }

    var topic=order.topic;
    if(!topic){ return true; }
    topic=normTopic(topic);
    if(!ZO[topic]){ if(!DD[topic]){ return true; } }

    var sign='';
    if(chart.stars){ if(chart.stars.sun){ sign=chart.stars.sun.name; } }
    var animal='';
    if(chart.pillars){ if(chart.pillars.year){
      var jz=chart.pillars.year.ji;
      if(typeof jz==='number'){ animal=ANIMAL[jz]+'띠'; }
    } }

    var made=build(topic, sign, animal);
    if(!made.length){ return true; }

    /* 맺음말 앞에 끼웁니다. 아직 맺음말이 안 그려졌으면 다음에 다시 옵니다. */
    var pages=box.getElementsByClassName('page');
    var anchor=null, i, eb;
    for(i=0;i<pages.length;i++){
      eb=pages[i].getElementsByClassName('eyebrow')[0];
      if(!eb){ continue; }
      if(eb.textContent==='맺음말'){ anchor=pages[i]; break; }
    }
    if(!anchor){ return false; }

    var hold=document.createElement('div');
    hold.innerHTML=made.join('');
    while(hold.firstChild){ anchor.parentNode.insertBefore(hold.firstChild, anchor); }

    box.setAttribute('data-tail','1');
    renumber();
    return true;
  }

  /* 다른 조각들이 쪽을 다 넣을 때까지 기다립니다. 최대 20초. */
  var tries=0;
  function tick(){
    tries+=1;
    var done=false;
    try{ done=start(); }catch(e){ done=true; }
    if(done){ return; }
    if(tries>60){ return; }
    setTimeout(tick, 330);
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', function(){ setTimeout(tick, 400); });
  } else { setTimeout(tick, 400); }
})();
</script>
