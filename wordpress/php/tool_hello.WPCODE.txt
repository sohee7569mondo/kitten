/* ══════════════════════════════════════════════════════════
   조각이 도는지만 봅니다 — 다섯 줄짜리
   ★ 2026-09-12 · 소희 님께      판 HELLO-1

   patch160_width 가 켜져 있고 위치도 「어디서나 실행」인데
   글이 한 글자도 안 나옵니다. 그래서 가장 단순한 것으로 재봅니다.

   이 조각은 쪽 머리에 한 줄만 적습니다. 그것 말고는 아무 일도 안 합니다.

   보는 법 : 아무 쪽이나 열고 Ctrl+U -> Ctrl+F 로 「여보세요」

       <!-- stella 여보세요 · 머리 · 2026-09-12 12:34:56 -->
       <!-- stella 여보세요 · 바닥 · 2026-09-12 12:34:56 -->

   ── 무엇을 알 수 있나 ─────────────────────────────────
   둘 다 보이면   → WPCode 는 멀쩡합니다. patch160_width 쪽 문제입니다
   바닥만 보이면  → wp_head 가 막혀 있습니다 (그럼 폭도 바닥으로 옮기면 됩니다)
   둘 다 없으면   → WPCode 가 새 조각을 안 돌리고 있습니다
                    (담아둔 화면을 보고 계신 것일 수도 있습니다 —
                     주소 뒤에 ?zzz=9 를 붙여 다시 보세요)

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> ★ 활성
   ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {
	echo "\n<!-- stella 여보세요 · 머리 · " . gmdate( 'Y-m-d H:i:s' ) . " -->\n";
}, 99 );

add_action( 'wp_footer', function () {
	echo "\n<!-- stella 여보세요 · 바닥 · " . gmdate( 'Y-m-d H:i:s' ) . " -->\n";
}, 99 );
