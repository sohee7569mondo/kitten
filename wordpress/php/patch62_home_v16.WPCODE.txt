    return { y:t.getUTCFullYear(), m:t.getUTCMonth() + 1, d:t.getUTCDate() };
  }

  function stemOf(y, m, d){
    var diff = Math.round((Date.UTC(y, m - 1, d) - LBASE) / 86400000);
    return ((diff % 10) + 10) % 10;
  }

  function showBack(pf){
    if(!pf){ return; }
    if(!pf.year){ return; }

    var sy = pf.year, sm = pf.month, sd = pf.day;
    if(pf.calendar === 'lunar'){
      var conv = lunarToSolar(pf.year, pf.month, pf.day, pf.leap ? true : false);
      if(!conv){ return; }
      sy = conv.y; sm = conv.m; sd = conv.d;
    }

    var mine = GAN_EL[stemOf(sy, sm, sd)];
    var now = new Date();
    var todayStem = stemOf(now.getFullYear(), now.getMonth() + 1, now.getDate());
    var dayEl = GAN_EL[todayStem];

    /* 이번주 월요일 */
    var back = (now.getDay() + 6) % 7;
    var mon = new Date(now.getFullYear(), now.getMonth(), now.getDate() - back);
    var weekEl = GAN_EL[stemOf(mon.getFullYear(), mon.getMonth() + 1, mon.getDate())];

    var rToday = relOf(mine, dayEl);
    var rWeek  = relOf(mine, weekEl);

    var nm = pf.name || '';
    $h('bDate2').textContent = (now.getMonth() + 1) + '월 ' + now.getDate() + '일 '
                             + DOW[now.getDay()] + '요일 · ' + GAN_L[todayStem] + '의 날';
    $h('bHi').innerHTML = (nm ? (nm + '님, ') : '') + '<em>' + TODAY_T[rToday] + '</em>';
    $h('bToday').textContent = TODAY_P[rToday];
    $h('bWeek').textContent = WEEK_P[rWeek]
      + '  (' + nm + '님은 ' + mine + ' ' + EL_NAME[mine] + '의 사람이에요)';

    $h('bNew').classList.remove('on');
    $h('bBack').classList.add('on');
  }
