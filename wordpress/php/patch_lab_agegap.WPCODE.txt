/* ════════════════════════════════════════════════════════════
   스텔라 랩 · 나이 차이 검사 2번 문항 고치기
   ★ 2026-09-09 · 소희 님께

   소희 님 : 「질문이 이상하지 않아? 상대가 아픈데 죽을 사오라 시킨다?」

   맞습니다. 말이 안 됩니다.

   이 검사의 축은 「기대고 싶다 ↔ 챙겨주고 싶다」입니다.
   그런데 질문 주어가 「상대가」로 되어 있어서, 기대는 쪽 선택지를
   쓸 자리가 없습니다. 그래서 「상대가 아픈데 그 사람더러 죽을 사
   오라고 시킨다」는 이상한 문장이 나왔습니다.

     지금   상대가 아플 때 나는
              죽을 사 오라고 시킨다        (기대는 쪽)   ← 말이 안 됨
              죽을 사서 문 앞에 둔다       (챙기는 쪽)

     고침   내가 아플 때 나는
              아프다고 바로 말하고 죽 사다 달라고 한다   (기대는 쪽)
              티 안 내고 혼자 약 사 먹는다              (챙기는 쪽)

   주어를 뒤집으니 두 선택지가 다 말이 되고 축도 선명해집니다.

   ★ 나머지 열한 문항은 다시 읽어봤고 축이 잘 맞습니다. 안 건드립니다.
   ★ 검사 열 개 120문항 전체는 새 스타일로 다시 지을 때 같이 봅니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 이 스니펫은 랩 본체와 「검사 더 넣기」 스니펫이 켜져 있어야 돕니다.

   확인   https://stellasaju.com/?stella_labfix=1
   ════════════════════════════════════════════════════════════ */

/* 옛 문항이 지나가면 새 문항으로 바꿔치웁니다.
   맨 나중(99)에 걸어야 「검사 더 넣기」가 다 얹힌 뒤에 손댑니다. */
add_filter( 'stella_lab_more', function ( $js ) {

	$old = "'상대가 아플 때 나는','죽을 사 오라고 시킨다','죽을 사서 문 앞에 둔다'";
	$new = "'내가 아플 때 나는','아프다고 바로 말하고 죽 사다 달라고 한다','티 안 내고 혼자 약 사 먹는다'";

	if ( false === strpos( $js, $old ) ) { return $js; }
	return str_replace( $old, $new, $js );
}, 99 );

/* 잘 바뀌었는지 눈으로 보는 자 */
add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_labfix'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="google" content="notranslate">'
	   . '<style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.8;'
	   . 'padding:28px 18px;max-width:860px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.2rem;margin:0 0 12px}'
	   . '.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}'
	   . 'pre{background:#000;padding:14px;white-space:pre-wrap;'
	   . 'font:13px/1.9 monospace;border-left:3px solid #D4AF6A}'
	   . '</style>';

	echo '<h1>나이 차이 검사 2번 문항</h1>';

	$js = apply_filters( 'stella_lab_more', '' );

	if ( '' === $js ) {
		echo '<p class="no">검사 자료가 비어 있습니다. '
		   . '랩 본체와 「검사 더 넣기」 스니펫이 켜져 있는지 보세요.</p>';
		exit;
	}

	$oldQ = '상대가 아플 때 나는';
	$newQ = '내가 아플 때 나는';

	echo '<pre>'
	   . '옛 문항 남아 있나   ' . ( false !== strpos( $js, $oldQ ) ? '예 — 안 바뀌었습니다' : '아니오' ) . "\n"
	   . '새 문항 들어갔나    ' . ( false !== strpos( $js, $newQ ) ? '예' : '아니오' ) . "\n"
	   . '자료 크기          ' . number_format( strlen( $js ) ) . ' 바이트'
	   . '</pre>';

	/* 그 자리 앞뒤를 보여드립니다 */
	$at = strpos( $js, $newQ );
	if ( false === $at ) { $at = strpos( $js, $oldQ ); }
	if ( false !== $at ) {
		$from = max( 0, $at - 120 );
		echo '<pre>' . esc_html( substr( $js, $from, 360 ) ) . '</pre>';
	}

	echo '<p class="dim">이 자는 읽기만 합니다. '
	   . '검사 화면에서도 확인해 보세요 — /lab/agegap/</p>';
	exit;
} );
