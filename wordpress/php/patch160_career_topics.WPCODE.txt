add_action( 'init', function () {

	if ( ! isset( $_GET['stella_patch'] ) || 'career1' !== $_GET['stella_patch'] ) {
		return;
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		return;
	}

	global $wpdb;
	$pid  = 160;
	$go   = isset( $_GET['go'] );
	$undo = isset( $_GET['undo'] );
	$WANT = '53da826c90ce4c1269834b2efa6c0876fa521362';
	$NEED = 12576;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:15px/1.8 -apple-system,system-ui,sans-serif;background:#111;color:#eee;padding:24px;max-width:1000px">';
	printf(
		'<h1 style="color:#ffd76a">직성의 신 · 주제별로 갈라 쓰기 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);

	if ( $undo ) {
		$bak = get_post_meta( $pid, '_stella_career1_backup', true );
		if ( ! $bak ) {
			echo '<p style="color:#ff7b7b">되돌릴 사본이 없습니다.</p></body>';
			exit;
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		if ( function_exists( 'wp_cache_flush' ) ) {
			wp_cache_flush();
		}
		printf( '<h2 style="color:#7bff9b">되돌렸습니다 — %d 바이트</h2></body>', strlen( $bak ) );
		exit;
	}

	$B64 = stella_career1_payload();
	if ( strlen( $B64 ) !== $NEED ) {
		printf(
			'<h2 style="color:#ff7b7b">스니펫이 잘렸습니다</h2><p>글 뭉치가 %d자여야 하는데 %d자입니다. 다시 붙여넣어 주세요.</p></body>',
			$NEED,
			strlen( $B64 )
		);
		exit;
	}
	$data = json_decode( gzdecode( base64_decode( $B64 ) ), true );
	if ( ! is_array( $data ) || empty( $data['pairs'] ) ) {
		echo '<h2 style="color:#ff7b7b">글 뭉치를 풀지 못했습니다.</h2></body>';
		exit;
	}

	$cur = $wpdb->get_var( $wpdb->prepare( "SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $cur ) {
		echo '<p style="color:#ff7b7b">페이지 160 을 못 읽었습니다.</p></body>';
		exit;
	}
	printf( '<p>지금 글 : %d 바이트 · sha1 <b>%s</b></p>', strlen( $cur ), substr( sha1( $cur ), 0, 12 ) );

	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%"><tr style="color:#ffd76a"><td>찾음</td><td>붙일 자리</td></tr>';
	$ready = true;
	$done  = true;
	foreach ( $data['pairs'] as $p ) {
		$n = substr_count( $cur, $p[1] );
		$m = substr_count( $cur, $p[2] );
		if ( 1 !== $n ) {
			$ready = false;
		}
		if ( 0 === $m ) {
			$done = false;
		}
		printf(
			'<tr style="border-top:1px solid #333"><td style="color:%s"><b>%d</b></td><td>%s%s</td></tr>',
			1 === $n ? '#7bff9b' : '#ff7b7b',
			$n,
			esc_html( $p[0] ),
			$m ? ' <span style="color:#9fd">(이미 들어 있음)</span>' : ''
		);
	}
	$hasTail = ( false !== strpos( $cur, '문장이 끝나면 줄을 바꿉니다' ) );
	printf(
		'<tr style="border-top:1px solid #333"><td style="color:%s">%s</td><td>책 전체 · 문장이 끝나면 줄바꿈</td></tr></table>',
		$hasTail ? '#9fd' : '#7bff9b',
		$hasTail ? '있음' : '넣음'
	);

	if ( $done && $hasTail ) {
		echo '<h2 style="color:#7bff9b">이미 다 들어가 있습니다.</h2>'
			. '<p><a style="color:#ffd76a" href="/reading-book/">전자책 열어보기</a></p></body>';
		exit;
	}
	if ( ! $ready ) {
		echo '<h2 style="color:#ff7b7b">붙일 자리를 정확히 못 찾아 손대지 않았습니다.</h2>'
			. '<p>이 화면을 저에게 보내주세요.</p></body>';
		exit;
	}
	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다. 아무것도 바뀌지 않았습니다.</h2>'
			. '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_patch=career1&go=1">→ 이대로 적용하기</a></p></body>';
		exit;
	}

	if ( ! get_post_meta( $pid, '_stella_career1_backup', true ) ) {
		update_post_meta( $pid, '_stella_career1_backup', $cur );
	}

	$new = $cur;
	foreach ( $data['pairs'] as $p ) {
		if ( substr_count( $new, $p[2] ) ) {
			continue;
		}
		$new = str_replace( $p[1], $p[1] . $p[2], $new );
	}
	if ( ! $hasTail ) {
		$new = rtrim( $new, "\n" ) . $data['tail'];
	}

	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) {
		wp_cache_flush();
	}
	$after = $wpdb->get_var( $wpdb->prepare( "SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );

	printf( '<h2 style="color:#7bff9b">넣었습니다 · %d → %d 바이트 (%+d)</h2>', strlen( $cur ), strlen( $after ), strlen( $after ) - strlen( $cur ) );
	printf(
		'<p>맞아야 하는 sha1 <b>%s</b><br>지금 글의 sha1 <b>%s</b> → %s</p>',
		substr( $WANT, 0, 16 ),
		substr( sha1( $after ), 0, 16 ),
		sha1( $after ) === $WANT ? '<b style="color:#7bff9b">똑같습니다</b>' : '<b style="color:#ffd76a">다릅니다 — 그래도 다섯 자리 모두 들어갔습니다</b>'
	);

	echo '<p style="margin-top:18px;font-size:17px"><a style="color:#ffd76a" href="/reading-book/">→ 전자책 열어보기</a></p>';
	echo '<p style="color:#888">되돌리려면 ?stella_patch=career1&undo=1</p>';
	echo '</body>';
	exit;
} );
