/* ══════════════════════════════════════════════════════════
   살아 있는 사이트를 들여다봅니다   ★ 2026-09-13 · 소희 님께
                                    판 CHECK-1

   소희 님 : 「궁합을 선택하면 정보 입력칸 아래에 상대 생년월일
             묻게 설계된 걸로 아는데」

   제가 잘못 봤습니다. 제 door-love 사본은 **9월 6일 것**이고
   43,061바이트인데 살아 있는 쪽은 59,306바이트입니다.
   16,245바이트가 모자랐습니다. 그 안에 상대 입력이 있을 겁니다.

   그래서 짐작하지 않고 **살아 있는 쪽을 직접 셉니다.**

   ── 보는 법 ───────────────────────────────────────────
       https://stellasaju.com/?stella_check=1

   세 가지를 보여드립니다.

       ①  쪽마다 크기와 제 사본과의 차이
       ②  궁합에 필요한 것 셋이 살아 있나
       ③  WPCode 조각 — 켜진 것과 꺼진 것

   ★ 아무것도 안 건드립니다. 세기만 합니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_check'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	echo "═══ 살아 있는 사이트를 들여다봅니다   판 CHECK-1 ═══\n\n";

	/* ── ① 쪽마다 크기 ──────────────────────────────────── */
	echo "── ① 쪽마다 크기 ──────────────────────────────\n";
	$mine = array(
		'reading-book'  => 1115096,
		'reading-start' => 22241,
		'door-love'     => 43061,
		'door-career'   => 0,
		'door-health'   => 0,
		'door-fortune'  => 0,
	);
	$pages = array();
	foreach ( $mine as $slug => $was ) {
		$pg = get_page_by_path( $slug );
		if ( ! $pg ) { printf( "  %-15s 없음\n", $slug ); continue; }
		$c = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", (int) $pg->ID ) );
		$pages[ $slug ] = $c;
		$now = strlen( $c );
		if ( $was ) {
			printf( "  %-15s %9s바이트   제 사본과 차이 %+s\n",
				$slug, number_format( $now ), number_format( $now - $was ) );
		} else {
			printf( "  %-15s %9s바이트\n", $slug, number_format( $now ) );
		}
	}

	/* ── ② 궁합에 필요한 것 셋 ──────────────────────────── */
	echo "\n── ② 궁합에 필요한 것 셋 ──────────────────────\n";

	$dl = isset( $pages['door-love'] ) ? $pages['door-love'] : '';
	$bk = isset( $pages['reading-book'] ) ? $pages['reading-book'] : '';

	echo "  가. 상대 생년월일을 묻는 자리 (door-love)\n";
	$words = array( 'partner', 'alias', '상대', '생년월일', 'couple', 'lover', 'pY', 'pMonth' );
	foreach ( $words as $w ) {
		printf( "      %-12s %d번\n", $w, substr_count( $dl, $w ) );
	}

	echo "\n  나. 책의 궁합 갈래 (reading-book)\n";
	printf( "      %-22s %s\n", "if(topic==='궁합')",
		( false !== strpos( $bk, "if(topic==='궁합')" ) ) ? '있음' : '없음' );
	printf( "      %-22s %s\n", 'StellaGunghapCalc',
		( false !== strpos( $bk, 'StellaGunghapCalc' ) ) ? '이름을 부름' : '안 부름' );
	printf( "      %-22s %s\n", 'StellaGunghapText',
		( false !== strpos( $bk, 'StellaGunghapText' ) ) ? '이름을 부름' : '안 부름' );

	echo "\n  다. 그 이름을 만드는 조각이 있나 (WPCode 전체에서)\n";
	$rows = $wpdb->get_results(
		"SELECT ID, post_title, post_status, post_content FROM {$wpdb->posts}
		  WHERE post_type = 'wpcode' ORDER BY post_title ASC" );
	$calcIn = array(); $textIn = array();
	foreach ( (array) $rows as $r ) {
		if ( false !== strpos( (string) $r->post_content, 'StellaGunghapCalc' ) ) {
			$calcIn[] = $r->post_title . ' [' . $r->post_status . ']';
		}
		if ( false !== strpos( (string) $r->post_content, 'StellaGunghapText' ) ) {
			$textIn[] = $r->post_title . ' [' . $r->post_status . ']';
		}
	}
	echo '      Calc 를 만드는 조각 : ' . ( count( $calcIn ) ? implode( ' / ', $calcIn ) : '없음' ) . "\n";
	echo '      Text 를 만드는 조각 : ' . ( count( $textIn ) ? implode( ' / ', $textIn ) : '없음' ) . "\n";

	/* ── ③ WPCode 조각 ─────────────────────────────────── */
	echo "\n── ③ WPCode 조각 ──────────────────────────────\n";
	$on = 0; $off = 0; $onList = array();
	foreach ( (array) $rows as $r ) {
		if ( 'publish' === $r->post_status ) { $on++; $onList[] = $r->post_title; }
		else { $off++; }
	}
	echo '  켜진 것 ' . $on . '개 · 꺼진 것 ' . $off . "개 · 모두 " . ( $on + $off ) . "개\n\n";
	echo "  켜진 것 이름\n";
	foreach ( $onList as $t ) { echo '    · ' . $t . "\n"; }

	exit;
}, 1 );
