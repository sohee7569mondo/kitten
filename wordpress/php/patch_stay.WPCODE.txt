/* ── 하나. 들어온 상태를 90일로 ── */
add_filter( 'auth_cookie_expiration', function ( $length, $uid, $remember ) {
	if ( ! $remember ) { return $length; }
	if ( user_can( $uid, 'edit_posts' ) ) { return $length; }   /* 관리자는 그대로 둡니다 */
	return 90 * DAY_IN_SECONDS;
}, 10, 3 );
