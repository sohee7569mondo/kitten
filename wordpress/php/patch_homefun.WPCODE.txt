// 홈 아래에 「친구와 나누는 것」 구역 — 유앤미 · 스텔라 랩
// 2026-09-27 · 자세한 것은 notes/2026-09-27-홈-친구배너.md
//
//   소희 님 「그건 좀 다른 사이트라서 아래에 따로 배너 만들었으면 좋겠어」
//   무료 다섯 장 줄에 끼워넣던 것을 치우고, 홈 맨 아래 인사말 앞에
//   가로로 넓은 카드 두 장을 따로 놓습니다.
//
//   patch_homeviral 은 꺼 주셔도 됩니다. 켜 두셔도 이 조각이 그 카드를
//   치웁니다. 쪽 원본은 한 글자도 안 고칩니다 — 끄면 그대로 돌아갑니다.
//
//   [로고]
//   유앤미 로고는 미디어에 이미 있습니다 (uandme_logo-01.png).
//   스텔라 랩 로고는 파일 이름에 lab 과 logo 가 들어가게 올려주세요 —
//   이를테면  stellalab-logo.png  입니다. 조각이 미디어에서 찾아 씁니다.
//   못 찾으면 그 카드만 로고 없이 나옵니다. 깨지지 않습니다.
//
// 위치(Location)는 「어디서나 실행 / Run Everywhere」로.
// 크롬 번역을 끄고 붙여넣으세요.
// 맨 아랫줄이  } );  인지 확인하세요. 아니면 붙여넣기가 잘린 것입니다.

add_action( 'wp_head', function () {

	if ( ! is_front_page() ) { return; }

	global $wpdb;

	// 로고 — 미디어에서 파일 이름으로 찾습니다. 못 찾으면 그 카드만 로고가 빕니다
	$logo = array( 'um' => '', 'lab' => '' );

	$um = $wpdb->get_var(
		"SELECT meta_value FROM {$wpdb->postmeta} WHERE meta_key = '_wp_attached_file'"
		. " AND meta_value LIKE '%uandme%logo%' ORDER BY post_id DESC LIMIT 1" );
	if ( $um ) { $logo['um'] = content_url( 'uploads/' . $um ); }

	$lab = $wpdb->get_var(
		"SELECT meta_value FROM {$wpdb->postmeta} WHERE meta_key = '_wp_attached_file'"
		. " AND meta_value LIKE '%lab%logo%' ORDER BY post_id DESC LIMIT 1" );
	if ( $lab ) { $logo['lab'] = content_url( 'uploads/' . $lab ); }

	// 큰 그림을 그대로 쓰면 화면이 느립니다 — CDN 으로 줄여 씁니다
	foreach ( $logo as $k => $v ) {
		if ( '' === $v ) { continue; }
		$logo[ $k ] = 'https://i0.wp.com/'
			. str_replace( array( 'https://', 'http://' ), '', $v ) . '?w=160';
	}

	$b  = '';
	$b .= "<style id='sfun-css'>\n";
	$b .= "#sfun .sfhead{ margin-bottom:20px; }\n";
	$b .= "#sfun .sfkick{ font-size:.72rem; letter-spacing:.16em;\n";
	$b .= "  text-transform:uppercase; color:var(--gold); margin-bottom:7px; }\n";
	$b .= "#sfun .sfbig{ font-size:clamp(1.24rem,3vw,1.62rem); font-weight:700;\n";
	$b .= "  line-height:1.45; color:var(--ink); }\n";
	$b .= "#sfun .sfdek{ margin-top:8px; font-size:.9rem; color:var(--ink-dim); }\n";
	$b .= "#sfun .sfgrid{ display:grid; grid-template-columns:repeat(2,1fr); gap:16px; }\n";
	$b .= "@media(max-width:760px){ #sfun .sfgrid{ grid-template-columns:1fr; } }\n";
	$b .= "#sfun a.sfc{ display:block; position:relative; overflow:hidden;\n";
	$b .= "  border-radius:14px; padding:26px 24px 22px; min-height:172px;\n";
	$b .= "  text-decoration:none; color:#FFFDF9; isolation:isolate;\n";
	$b .= "  display:flex; flex-direction:column; align-items:flex-start;\n";
	$b .= "  box-shadow:var(--shadow); transition:transform .25s ease, box-shadow .25s ease; }\n";
	$b .= "#sfun a.sfc:hover{ transform:translateY(-6px); box-shadow:var(--shadow-up); }\n";
	$b .= "#sfun a.sfc:focus-visible{ outline:2px solid var(--plum); outline-offset:3px; }\n";
	$b .= "#sfun #sfc1{ background:linear-gradient(135deg,#E0607F 0%,#A83457 100%); }\n";
	$b .= "#sfun #sfc2{ background:linear-gradient(135deg,#4A87BA 0%,#22506F 100%); }\n";
	$b .= "#sfun a.sfc::after{ content:''; position:absolute; z-index:-1;\n";
	$b .= "  right:-60px; top:-70px; width:210px; height:210px; border-radius:50%;\n";
	$b .= "  background:rgba(255,253,249,.12); }\n";
	$b .= "#sfun .sftop{ display:flex; align-items:center; gap:11px; }\n";
	$b .= "#sfun .sfbadge{ flex:0 0 auto; width:58px; height:58px; border-radius:50%;\n";
	$b .= "  background:#FFFDF9; display:flex; align-items:center; justify-content:center;\n";
	$b .= "  overflow:hidden; box-shadow:0 3px 10px -4px rgba(20,16,49,.45); }\n";
	$b .= "#sfun .sfbadge img{ width:100%; height:100%; object-fit:contain; display:block; }\n";
	$b .= "#sfun .sfbrand{ font-size:.79rem; letter-spacing:.06em;\n";
	$b .= "  color:rgba(255,253,249,.82); }\n";
	$b .= "#sfun .sfhook{ margin-top:17px; font-weight:700; line-height:1.36;\n";
	$b .= "  font-size:clamp(1.22rem,4.2vw,1.46rem); letter-spacing:-.01em;\n";
	$b .= "  text-wrap:balance; }\n";
	$b .= "#sfun .sfsay{ margin-top:8px; font-size:.91rem; line-height:1.7;\n";
	$b .= "  color:rgba(255,253,249,.86); max-width:23em; }\n";
	$b .= "#sfun .sfgo{ margin-top:auto; padding-top:16px; font-size:.86rem; font-weight:700;\n";
	$b .= "  color:rgba(255,253,249,.95); }\n";
	$b .= "#sfun .sffree{ margin-top:16px; font-size:.82rem; color:var(--ink-dim);\n";
	$b .= "  text-align:center; }\n";
	$b .= "</style>\n";

	$b .= "<script>\n";
	$b .= "(function(){\n";
	$b .= "  if(window.StellaHomeFun){ return; }\n";
	$b .= "  window.StellaHomeFun = 1;\n";

	$b .= "  var LOGO_UM = '" . str_replace( "'", '', $logo['um'] ) . "';\n";
	$b .= "  var LOGO_LAB = '" . str_replace( "'", '', $logo['lab'] ) . "';\n";
	// 카피는 소희 님 판입니다 (2026-09-27). 이름보다 후킹 문구가 큽니다.
	$b .= "  var CARDS = [\n";
	$b .= "    ['sfc1','/uandme/','유앤미',\n";
	$b .= "     '그 사람과 나, 정말 잘 맞을까?',\n";
	$b .= "     '이름과 생일만 넣으면 바로 나옵니다',\n";
	$b .= "     '연인 · 가족 · 직장 · 스타 궁합보기', LOGO_UM],\n";
	$b .= "    ['sfc2','/lab/','스텔라 랩',\n";
	$b .= "     '내가 이런 성격이라고?',\n";
	$b .= "     '행동으로 알아보는 진짜 나',\n";
	$b .= "     '성격 · 취향 · 여행 테스트', LOGO_LAB]\n";
	$b .= "  ];\n";

	$b .= "  function esc(s){\n";
	$b .= "    var d = document.createElement('div');\n";
	$b .= "    d.textContent = s;\n";
	$b .= "    return d.innerHTML;\n";
	$b .= "  }\n";

	$b .= "  function build(){\n";
	$b .= "    var home = document.getElementById('stellar-home');\n";
	$b .= "    if(!home){ return false; }\n";
	$b .= "    if(document.getElementById('sfun')){ return true; }\n";
	$b .= "    var i, c, h = '';\n";
	$b .= "    for(i = 0; i < CARDS.length; i++){\n";
	$b .= "      c = CARDS[i];\n";
	$b .= "      var badge = '';\n";
	$b .= "      if(c[6]){\n";
	$b .= "        badge = '<span class=sfbadge><img role=presentation src='\n";
	$b .= "              + c[6] + '></span>';\n";
	$b .= "      }\n";
	$b .= "      h += '<a class=sfc id=' + c[0] + ' href=' + c[1] + '>'\n";
	$b .= "         + '<div class=sftop>' + badge\n";
	$b .= "         + '<span class=sfbrand>' + esc(c[2]) + '</span></div>'\n";
	$b .= "         + '<div class=sfhook>' + esc(c[3]) + '</div>'\n";
	$b .= "         + '<div class=sfsay>' + esc(c[4]) + '</div>'\n";
	$b .= "         + '<div class=sfgo>' + esc(c[5]) + ' →</div></a>';\n";
	$b .= "    }\n";
	$b .= "    var sec = document.createElement('section');\n";
	$b .= "    sec.id = 'sfun';\n";
	$b .= "    sec.className = 'sh-section';\n";
	$b .= "    sec.innerHTML = '<div class=wrap>'\n";
	$b .= "      + '<div class=sfhead>'\n";
	$b .= "      + '<div class=sfkick>ALSO FROM STELLA</div>'\n";
	$b .= "      + '<div class=sfbig>친구와 나누는 것</div>'\n";
	$b .= "      + '<div class=sfdek>사주와는 조금 다른 자리입니다. '\n";
	$b .= "      + '가볍게 보고 결과를 그대로 보내보세요.</div>'\n";
	$b .= "      + '</div>'\n";
	$b .= "      + '<div class=sfgrid>' + h + '</div>'\n";
	$b .= "      + '<div class=sffree>둘 다 값 없이 보실 수 있습니다.</div>'\n";
	$b .= "      + '</div>';\n";
	$b .= "    var fin = home.querySelector('section.final');\n";
	$b .= "    if(fin){ home.insertBefore(sec, fin); }\n";
	$b .= "    if(!fin){ home.appendChild(sec); }\n";
	$b .= "    return true;\n";
	$b .= "  }\n";

	// 옛 조각(patch_homeviral)이 무료 줄에 넣은 카드를 치웁니다
	$b .= "  function sweep(){\n";
	$b .= "    var old = document.querySelectorAll('a.free-card[data-viral=on]');\n";
	$b .= "    var i;\n";
	$b .= "    for(i = 0; i < old.length; i++){\n";
	$b .= "      if(old[i].parentNode){ old[i].parentNode.removeChild(old[i]); }\n";
	$b .= "    }\n";
	$b .= "  }\n";

	$b .= "  var tries = 0;\n";
	$b .= "  function tick(){\n";
	$b .= "    tries++;\n";
	$b .= "    build();\n";
	$b .= "    sweep();\n";
	$b .= "    if(tries < 12){ setTimeout(tick, 400); }\n";
	$b .= "  }\n";

	$b .= "  if(document.readyState === 'loading'){\n";
	$b .= "    document.addEventListener('DOMContentLoaded', tick);\n";
	$b .= "  }\n";
	$b .= "  if(document.readyState !== 'loading'){ tick(); }\n";
	$b .= "})();\n";
	$b .= "</script>\n";

	echo $b;
} );
