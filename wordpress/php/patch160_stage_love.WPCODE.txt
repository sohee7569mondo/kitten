			echo '<h2 style="color:#888">사본이 없습니다.</h2>';
		}
		exit;
	}

	$has = substr_count( $now, '__stgLove' );
	$n   = preg_match_all( $re, $now );

	echo '<table cellpadding="8" style="border-collapse:collapse">';
	printf( '<tr><td>페이지 크기</td><td><b>%d</b> 바이트</td></tr>', strlen( $now ) );
	printf( '<tr><td>새 글 크기</td><td>%d 바이트 · 18 문단</td></tr>', strlen( $ins ) );
	printf( '<tr><td>이미 들어감</td><td>%d</td></tr>', $has );
	printf( '<tr><td>넣을 자리 (holdTxt)</td><td style="color:%s"><b>%d</b></td></tr>', 1 === $n ? '#7bff9b' : '#ff7b7b', $n );
	echo '</table>';

	if ( $has > 0 ) {
		echo '<h2 style="color:#7bff9b">이미 들어가 있습니다.</h2>'
			. '<p style="color:#888">되돌리려면 ?stella_agel=1&undo=1</p>';
		exit;
	}
	if ( 1 !== $n ) {
		echo '<h2 style="color:#ff7b7b">자리가 딱 한 번이 아니라 아무것도 바꾸지 않았습니다.</h2>'
			. '<p>이 화면을 저에게 보내주세요.</p>';
		exit;
	}
	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다.</h2>'
			. '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_agel=1&go=1">→ 이대로 넣기</a></p>';
		exit;
	}

	if ( ! get_post_meta( $pid, '___stgLove_backup', true ) ) {
		update_post_meta( $pid, '___stgLove_backup', $now );
	}

	$out = preg_replace_callback(
		$re,
		function ( $m ) use ( $ins ) { return str_replace( 'holdTxt+', 'holdTxt+' . $ins . '+', $m[0] ); },
		$now,
		1
	);

	$wpdb->update( $wpdb->posts, array( 'post_content' => $out ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) {
		wp_cache_flush();
	}

	printf( '<h2 style="color:#7bff9b">넣었습니다 — %d → %d 바이트</h2>', strlen( $now ), strlen( $out ) );
	echo '<p style="color:#888">되돌리려면 ?stella_agel=1&undo=1</p>';
	exit;
} );
