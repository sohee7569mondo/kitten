/* ════════════════════════════════════════════════════════════
   문 다섯 곳의 버튼에서 「신」을 「가디언」으로

   소희 님 —
     「별을 읽는 신에게 물어보기 → 별을 읽는 가디언에게 물어보기」

   찾아보니 다섯 문이 다 같은 꼴이라 하나만 바꾸면 짝이 안 맞습니다.

     별을 읽는 신에게 물어보기   →  별을 읽는 가디언에게 물어보기
     일을 짓는 신에게 물어보기   →  일을 짓는 가디언에게 물어보기
     때를 아는 신에게 물어보기   →  때를 아는 가디언에게 물어보기
     몸을 지키는 신에게 물어보기 →  몸을 지키는 가디언에게 물어보기
     인연을 잇는 신에게 물어보기 →  인연을 잇는 가디언에게 물어보기

   타로 문은 원래 「아르카나에게 물어보기」라 「신」이 없습니다. 그대로 둡니다.

   ── 이 패치가 다른 것과 다른 점 ──────────────────
   페이지 160 이 아니라 문 다섯 쪽을 건드립니다. 그런데 그 쪽들의
   번호를 모릅니다. 그래서 번호 대신 주소(door-astro 같은 것)로
   찾습니다 — get_page_by_path.

   그리고 이 쪽들에도 <script> 가 들어 있어서, wp_update_post 로
   저장하면 워드프레스가 지워버립니다. 160 때와 같이 데이터베이스에
   바로 씁니다.

   ── 한 군데도 못 찾으면 아무것도 안 바꿉니다 ─────────
   저장소에 있는 사본이 사이트와 다를 수 있어서, 먼저 보기로
   몇 군데 걸리는지부터 보여드립니다. 한 쪽이라도 못 찾으면
   그 쪽만 건너뛰고 나머지는 그대로 바꿉니다. 무엇을 건너뛰었는지
   화면에 적습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로 바꾸세요.

   ① 먼저 보기   https://stellasaju.com/?stella_door=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_door=apply
   ③ 되돌리기    https://stellasaju.com/?stella_door=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_door'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_door_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_door'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$doors = array(
		'door-astro'   => '별을 읽는',
		'door-career'  => '일을 짓는',
		'door-fortune' => '때를 아는',
		'door-health'  => '몸을 지키는',
		'door-love'    => '인연을 잇는',
	);

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$n = 0;
		foreach ( $doors as $slug => $lead ) {
			$p = get_page_by_path( $slug );
			if ( ! $p ) { continue; }
			$old = get_post_meta( $p->ID, $META, true );
			if ( ! $old ) { echo '<p>' . esc_html( $slug ) . ' — 되돌릴 것이 없습니다.</p>'; continue; }
			$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $p->ID ) );
			clean_post_cache( $p->ID );
			delete_post_meta( $p->ID, $META );
			echo '<p class="ok">' . esc_html( $slug ) . ' — ' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
			$n++;
		}
		echo '<p>' . intval( $n ) . '쪽을 되돌렸습니다.</p>';
		exit;
	}

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$done = 0; $skip = 0;

	foreach ( $doors as $slug => $lead ) {
		$from = $lead . ' 신에게 물어보기';
		$to   = $lead . ' 가디언에게 물어보기';

		echo '<h2>' . esc_html( $slug ) . '</h2>';

		$p = get_page_by_path( $slug );
		if ( ! $p ) { echo '<p class="no">쪽을 못 찾았습니다. 건너뜁니다.</p>'; $skip++; continue; }

		$row = $wpdb->get_row( $wpdb->prepare(
			"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
		if ( ! $row ) { echo '<p class="no">내용을 못 읽었습니다. 건너뜁니다.</p>'; $skip++; continue; }

		$c = $row->post_content;
		$n = substr_count( $c, $from );

		if ( 0 === $n ) {
			echo '<p class="no">「' . esc_html( $from ) . '」를 못 찾았습니다. '
			   . '이미 바꿨거나 글이 달라졌습니다. 건너뜁니다.</p>';
			$skip++;
			continue;
		}

		echo '<p class="ok">' . intval( $n ) . '군데 — 「' . esc_html( $from ) . '」 → <b>「'
		   . esc_html( $to ) . '」</b></p>';

		if ( ! $apply ) { $done++; continue; }

		$new = str_replace( $from, $to, $c );

		if ( ! get_post_meta( $p->ID, $META, true ) ) {
			update_post_meta( $p->ID, $META, $c );
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $p->ID ) );
		clean_post_cache( $p->ID );
		$done++;
	}

	echo '<h2>정리</h2>';
	echo '<p>' . intval( $done ) . '쪽 ' . ( $apply ? '바꿨습니다' : '바꿀 수 있습니다' )
	   . ' · ' . intval( $skip ) . '쪽 건너뜀</p>';

	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_door=apply</code> 를 여세요.</p>';
	} else {
		echo '<p class="ok">되돌리시려면 <code>/?stella_door=undo</code> 를 여세요.</p>';
	}
	exit;
}, 0 );
