/* ════════════════════════════════════════════════════════════
   문마다 제 미리보기를 보여줍니다   patch_doorpreview
   ★ 2026-09-15 · 소희 님께

   ?stella_preview=1 이 찾아낸 것 —

       door-health   → demo=door-career · door-career-2
       door-fortune  → demo=door-career · door-career-2
       door-astro    → demo=door-career · door-career-2

   건강운 문에서 「미리보기」를 누르면 **마루의 직업운 책**이 나옵니다.
   시기운 문도, 별자리 문도 그렇습니다.

   까닭 : 원래 여섯 문이 모두 같은 견본 둘(/reading-sample/ 직업운 ·
   /reading-wealth/ 재물운)을 가리키고 있었습니다. patch_demolink 가
   그것을 살아 있는 책으로 돌려주긴 했지만, **주제는 그대로** 직업운
   입니다. door-love 와 door-career 만 나중에 제 주제로 고쳐졌습니다.

   ── 하는 일 ──
     그 문에 서 있을 때, 미리보기 링크의 주소만 제 문으로 갈아 끼웁니다.

       첫째 링크  ?demo=door-health
       둘째 링크  ?demo=door-health-2

     데모 사람이 열둘이고 문이 여섯이라 문마다 둘입니다 —
       직업운 서준 · 재물운 명주 · 연애운 하은 · 결혼운 도윤
       건강운 민재 · 가족운 수아 · 시기운 지우 · 대운 현우
       별자리 은설 · 별자리와 사주 라온 · 오늘의 선택 태윤 · 지훈

   ★ 쪽 글을 안 고칩니다. 주소만 갈아 끼웁니다. 제 사본이 낡아도
     안 놓칩니다. 스위치를 끄면 그 즉시 원래대로입니다.
   ★ 그 문 쪽에서만 실립니다. 다른 쪽에는 한 자도 안 나갑니다.
   ★ 둘째 사람이 없는 문이면 그 링크는 감춥니다 — 눌러도 빈 책이
     나오느니 안 보이는 편이 낫습니다. 확인하시고 말씀 주시면
     아래 TWO 표에서 켜 드립니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ════════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {

	$doors = array( 'door-love', 'door-career', 'door-health',
	                'door-fortune', 'door-astro', 'door-tarot' );

	$here = '';
	foreach ( $doors as $d ) {
		if ( is_page( $d ) ) { $here = $d; break; }
	}
	if ( ! $here ) { return; }

	/* 둘째 사람이 있는 문 — 확인된 것만 켭니다.
	   door-career 는 이미 잘 돌고 있어 그대로 둡니다. */
	$TWO = array(
		'door-career'  => 1,   /* 재물운 명주 — 지금도 잘 나옵니다 */
		'door-love'    => 0,   /* 결혼운 도윤 — 확인 뒤 1 로 */
		'door-health'  => 0,   /* 가족운 수아 */
		'door-fortune' => 0,   /* 대운 현우 */
		'door-astro'   => 0,   /* 별자리와 사주 라온 */
		'door-tarot'   => 0,
	);
	$two = isset( $TWO[ $here ] ) ? (int) $TWO[ $here ] : 0;
	?>
<!-- stella 문 미리보기 · 판 2026-09-15 · <?php echo esc_html( $here ); ?> -->
<script>
(function(){
  if(window.StellaDoorPreview){ return; }
  window.StellaDoorPreview = 1;

  var HERE = '<?php echo esc_js( $here ); ?>';
  var TWO  = <?php echo $two ? '1' : '0'; ?>;

  /* 주소에서 demo= 뒤의 값만 꺼냅니다 */
  function demoOf(h){
    var at = String(h).indexOf('demo=');
    if(at < 0){ return ''; }
    var v = String(h).slice(at + 5);
    var amp = String.fromCharCode(38);
    var cut = v.length, k, c;
    for(k = 0; k < v.length; k++){
      c = v.charAt(k);
      if(c === amp){ cut = k; break; }
      if(c === '#'){ cut = k; break; }
    }
    return v.slice(0, cut);
  }

  function go(){
    var as = document.querySelectorAll('a[href*="demo="]');
    var i, a, h, n = 0;
    for(i = 0; i < as.length; i++){
      a = as[i];
      if(a.getAttribute('data-dpv') === '1'){ continue; }
      h = String(a.getAttribute('href') || '');
      if(h.indexOf('/reading-book/') < 0){ continue; }

      /* 둘째 사람을 가리키던 링크인가 (…-2)
         ★ 정규식을 안 씁니다 — 앰퍼샌드도 역빗금도 안 들어가야 합니다.
           집 규칙이고, 역빗금은 편집기 저장에 벗겨진 적이 있습니다. */
      var isTwo = (demoOf(h).slice(-2) === '-2');

      if(isTwo){
        if(!TWO){
          /* 그 문에 둘째 사람이 없습니다 — 링크를 감춥니다 */
          a.style.display = 'none';
          a.setAttribute('data-dpv', '1');
          n++;
          continue;
        }
        a.setAttribute('href', '/reading-book/?demo=' + HERE + '-2');
      } else {
        a.setAttribute('href', '/reading-book/?demo=' + HERE);
      }
      a.setAttribute('data-dpv', '1');
      n++;
    }
    return n;
  }

  var tries = 0;
  function tick(){
    tries++;
    var n = 0;
    try{ n = go(); }catch(e){ return; }
    if(n > 0){ return; }
    if(tries < 20){ setTimeout(tick, 300); }
  }
  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', tick);
  } else { tick(); }
})();
</script>
	<?php
} );
