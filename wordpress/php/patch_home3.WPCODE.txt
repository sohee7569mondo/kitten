/* ════════════════════════════════════════════════════════════
   홈 · 문 소개를 세 줄로 · 배너 여백 · 문구 여섯 · 소희 님께

   ① 문구 여섯 줄을 바꿉니다
        연애 스타일과 되풀이되는 걸림을 찾아드립니다.
          → 연애에 관한 궁금증을 풀어드립니다.
        여섯 문 모두 같은 꼴로 맞춥니다.
        「모든」은 넣지 않았습니다 — 표시광고법에서 걸리는 단정입니다.

   ② 휴대폰에서 세 줄로 앉힙니다
        1줄  인연을 잇는 가디언 · 벼리        모두 보기 ›
        2줄  연애에 관한 궁금증을 풀어드립니다.
        3줄  정환이의 연애운 미리보기
        「모두 보기」를 첫 줄 오른쪽 끝으로 올려서, 없애지 않고도
        세 줄이 됩니다.

   ③ 글씨를 키웁니다 (휴대폰에서 한 줄 설명 · 미리보기 단추)

   ④ 배너와 헤더 사이 여백을 넓힙니다
        지금 휴대폰에서 위쪽 여백이 12px 뿐이라 헤더에 붙습니다.
        24px 로 넓힙니다.

   컴퓨터 화면은 하나도 안 바뀝니다. 700px 아래에서만 달라집니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 「어디서나 실행」 → 저장 → 활성

   ① 미리보기 — 아무것도 바뀌지 않습니다
        https://stellasaju.com/?stella_home3=1
   ② 적용
        https://stellasaju.com/?stella_home3=1&go=1
   되돌리기
        https://stellasaju.com/?stella_home3=1&undo=1
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_home3'] ) ) {
		return;
	}
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다. 관리자로 로그인한 창에서 열어주세요.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	$PID  = 62;
	$META = '_stella_home3_backup';
	$MARK = 'stella-home3-css';
	$go   = isset( $_GET['go'] );
	$undo = isset( $_GET['undo'] );

	$safe = function ( $s ) {
		if ( function_exists( 'mb_convert_encoding' ) ) {
			$s = mb_convert_encoding( $s, 'UTF-8', 'UTF-8' );
		}
		return esc_html( $s );
	};

	$LINES = array(
		'door-love'    => '연애에 관한 궁금증을 풀어드립니다.',
		'door-career'  => '직장/사업/재물에 관한 궁금증을 풀어드립니다.',
		'door-health'  => '건강/가족에 관한 궁금증을 풀어드립니다.',
		'door-fortune' => '대운/올해 흐름에 관한 궁금증을 풀어드립니다.',
		'door-astro'   => '별자리에 관한 궁금증을 풀어드립니다.',
		'door-tarot'   => '타로 한 장에 관한 궁금증을 풀어드립니다.',
	);

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:15px/1.8 -apple-system,system-ui,sans-serif;'
		. 'background:#111;color:#eee;padding:24px;max-width:980px">';
	printf(
		'<h1 style="color:#ffd76a">홈 손보기 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);

	/* ── 되돌리기 ── */
	if ( $undo ) {
		$bak = get_post_meta( $PID, $META, true );
		if ( $bak ) {
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $PID ) );
			delete_post_meta( $PID, $META );
			clean_post_cache( $PID );
			if ( function_exists( 'wp_cache_flush' ) ) {
				wp_cache_flush();
			}
			echo '<h2 style="color:#7bff9b">되돌렸습니다.</h2></body>';
		} else {
			echo '<h2 style="color:#888">사본이 없습니다.</h2></body>';
		}
		exit;
	}

	$row = $wpdb->get_row( $wpdb->prepare( "SELECT post_title, post_content FROM {$wpdb->posts} WHERE ID = %d", $PID ) );
	if ( ! $row ) {
		echo '<h2 style="color:#ff7b7b">62번 페이지를 못 읽었습니다.</h2></body>';
		exit;
	}
	$c = $row->post_content;

	/* ── ① 문구 여섯 ── */
	$TAG   = '<h2 class="rh-d">';
	$edits = array();
	$show  = array();
	$pos   = 0;

	while ( ( $h = strpos( $c, $TAG, $pos ) ) !== false ) {
		$start = $h + strlen( $TAG );
		$end   = strpos( $c, '</h2>', $start );
		if ( false === $end ) {
			break;
		}
		$inner = substr( $c, $start, $end - $start );

		/* 문구는 주석이나 링크가 시작되기 전까지입니다 */
		$cut = strlen( $inner );
		$p1  = strpos( $inner, '<!--' );
		$p2  = strpos( $inner, '<a ' );
		if ( false !== $p1 ) {
			$cut = min( $cut, $p1 );
		}
		if ( false !== $p2 ) {
			$cut = min( $cut, $p2 );
		}
		$old = substr( $inner, 0, $cut );

		/* 어느 문인지 — 바로 뒤의 「모두 보기」 주소로 알아냅니다 */
		$slug = '';
		$gp   = strpos( $c, '<a class="rh-go" href="/', $end );
		if ( false !== $gp ) {
			$s1 = $gp + strlen( '<a class="rh-go" href="/' );
			$s2 = strpos( $c, '/', $s1 );
			if ( false !== $s2 ) {
				$slug = substr( $c, $s1, $s2 - $s1 );
			}
		}

		$new = isset( $LINES[ $slug ] ) ? $LINES[ $slug ] : '';
		$show[] = array( $slug, trim( $old ), $new );
		if ( '' !== $new && trim( $old ) !== $new ) {
			$edits[] = array( $start, $cut, $new );
		}

		$pos = $end + 5;
	}

	/* 뒤에서부터 갈아끼웁니다 — 앞의 자리가 안 밀립니다 */
	$new_c = $c;
	for ( $i = count( $edits ) - 1; $i >= 0; $i-- ) {
		list( $s, $len, $txt ) = $edits[ $i ];
		$new_c = substr( $new_c, 0, $s ) . $txt . substr( $new_c, $s + $len );
	}

	/* ── ②③④ 겉모습 — 맨 끝에 한 덩어리 ── */
	$css = "\n\n<!-- wp:html -->\n<style id=\"" . $MARK . "\">\n"
		. "/* 2026-09-08 · 휴대폰에서만. 컴퓨터 화면은 그대로입니다.\n"
		. "   1줄 가디언 이름 + 모두 보기 / 2줄 한 줄 설명 / 3줄 미리보기 */\n"
		. "@media(max-width:700px){\n"
		. "  #stellar-home .banner-wrap{padding:24px 14px 0 !important;}\n"
		. "  #stellar-home .rowhead{\n"
		. "    display:grid !important; grid-template-columns:1fr auto;\n"
		. "    align-items:start; column-gap:12px;}\n"
		. "  #stellar-home .rowhead .rh-l{min-width:0;}\n"
		. "  #stellar-home .rowhead .rh-go{\n"
		. "    white-space:nowrap; align-self:start; margin-top:2px; font-size:.82rem;}\n"
		. "  #stellar-home .rowhead .kick{font-size:.82rem; margin-bottom:6px;}\n"
		. "  #stellar-home .rowhead .rh-d{\n"
		. "    font-size:1.06rem !important; line-height:1.62; margin:0;\n"
		. "    word-break:keep-all; text-wrap:balance;}\n"
		. "  #stellar-home .sbtn{\n"
		. "    display:block !important; width:-moz-fit-content; width:fit-content;\n"
		. "    margin-top:11px; font-size:.9rem;}\n"
		. "}\n</style>\n<!-- /wp:html -->\n";

	$has_css = ( false !== strpos( $c, $MARK ) );
	if ( ! $has_css ) {
		$new_c .= $css;
	}

	/* ── 보여주기 ── */
	echo '<h2 style="color:#9fd">① 문구 여섯</h2>';
	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
	echo '<tr style="color:#ffd76a"><td>문</td><td>지금</td><td>바꿀 말</td></tr>';
	foreach ( $show as $s ) {
		printf(
			'<tr style="border-top:1px solid #333"><td><code>%s</code></td>'
			. '<td style="color:#888">%s</td><td style="color:%s">%s</td></tr>',
			$safe( $s[0] ),
			$safe( $s[1] ),
			'' === $s[2] ? '#ff7b7b' : '#7bff9b',
			'' === $s[2] ? '모르는 문 — 손대지 않습니다' : $safe( $s[2] )
		);
	}
	echo '</table>';

	printf(
		'<h2 style="color:#9fd">②③④ 겉모습</h2><p>%s</p>',
		$has_css ? '<span style="color:#888">이미 들어가 있습니다. 다시 넣지 않습니다.</span>'
			: '<span style="color:#7bff9b">맨 끝에 한 덩어리를 넣습니다 (휴대폰 전용).</span>'
	);
	printf(
		'<pre style="white-space:pre-wrap;background:#000;padding:12px;'
		. 'font:12.5px/1.6 monospace">%s</pre>',
		$safe( $css )
	);

	printf(
		'<p>바꿀 문구 <b style="color:#ffd76a">%d</b>군데 · 글자 수 %s → %s</p>',
		count( $edits ),
		number_format( strlen( $c ) ),
		number_format( strlen( $new_c ) )
	);

	if ( 0 === count( $edits ) && $has_css ) {
		echo '<h2 style="color:#7bff9b">이미 다 되어 있습니다.</h2></body>';
		exit;
	}

	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다. 아무것도 바뀌지 않았습니다.</h2>';
		echo '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_home3=1&go=1">'
			. '→ 이대로 바꾸기</a></p></body>';
		exit;
	}

	if ( ! get_post_meta( $PID, $META, true ) ) {
		update_post_meta( $PID, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new_c ), array( 'ID' => $PID ) );
	clean_post_cache( $PID );
	if ( function_exists( 'wp_cache_flush' ) ) {
		wp_cache_flush();
	}

	echo '<h2 style="color:#7bff9b">바꿨습니다</h2>';
	echo '<p style="font-size:17px">휴대폰으로 <a style="color:#ffd76a" href="/">홈</a>을 열어 '
		. '세 줄로 앉았는지 봐주세요.</p>';
	echo '<p style="color:#888">되돌리려면 ?stella_home3=1&undo=1</p></body>';
	exit;
} );
