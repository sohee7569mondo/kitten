/* ════════════════════════════════════════════════════════════
   지금 페이지에 뭐가 들어 있나 (읽기만)   ★ 2026-09-09 · 소희 님께

   아무것도 안 바꿉니다.

   그림만 봐서는 어느 판이 들어갔는지 알 수가 없어서,
   페이지 안을 그대로 세어봅니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.

   보기   https://stellasaju.com/?stella_state=1
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_state'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="google" content="notranslate">'
	   . '<style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.75;'
	   . 'padding:28px 18px;max-width:900px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.25rem;margin:0 0 12px}'
	   . 'h2{color:#9AD1A5;font-size:1rem;margin:24px 0 6px;'
	   . 'border-top:1px solid rgba(255,255,255,.12);padding-top:14px}'
	   . '.ok{color:#9AD1A5}.no{color:#E5A0A0}.dim{color:#8B849C;font-size:.86rem}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:12px/1.75 monospace;border-left:3px solid #D4AF6A}'
	   . '</style>';

	$look = function ( $slug, $rows ) use ( $wpdb ) {
		$p = get_page_by_path( $slug );
		echo '<h2>' . esc_html( $slug ) . '</h2>';
		if ( ! $p ) { echo '<p class="no">쪽 없음</p>'; return; }
		$c = (string) $wpdb->get_var( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
		$out = '';
		foreach ( $rows as $label => $needle ) {
			$n = substr_count( $c, $needle );
			$out .= str_pad( $label, 30 ) . ( $n ? '있음  ' . $n : '없음' ) . "\n";
		}
		$out .= str_pad( '쪽 크기', 30 ) . number_format( strlen( $c ) ) . " 바이트\n";
		echo '<pre>' . esc_html( $out ) . '</pre>';
	};

	echo '<h1>지금 페이지에 뭐가 들어 있나</h1>';

	$look( 'door-love', array(
		'벼리 카드'              => 'class="g-card"',
		'사진 칸'                => 'class="g-portrait"',
		'벼리 사진 주소'         => 'stellardoorlovers',
		'문 다섯 패치'           => 'stella-doors1-0909',
		'그 패치가 지금 판(v5)'  => 'stella-doors1-v5',
		'단추 겉모습(못 박은 판)'=> 'text-indent:0 !important',
		'접기 규칙'              => 'stella-fold-0909',
		'주제 단추'              => 'name="ssg-topic"',
	) );

	$look( 'door-career', array(
		'마루 카드'              => 'class="g-card"',
		'문 다섯 패치'           => 'stella-doors1-0909',
		'그 패치가 지금 판(v5)'  => 'stella-doors1-v5',
		'단추 겉모습(못 박은 판)'=> 'text-indent:0 !important',
	) );

	echo '<h2>스니펫 쪽</h2>';
	$snips = $wpdb->get_results(
		"SELECT ID, post_title, post_status, post_content FROM {$wpdb->posts}
		  WHERE post_type = 'wpcode' AND ID > 1050 ORDER BY ID DESC LIMIT 30" );
	$out = '';
	foreach ( $snips as $r ) {
		$mark = '';
		if ( false !== strpos( $r->post_content, 'stella-doors1-v5' ) ) { $mark = '  <- doors1 v5'; }
		if ( false !== strpos( $r->post_content, 'stellardoorlovers' ) ) { $mark = '  <- 벼리 사진 있는 판'; }
		$out .= str_pad( (string) $r->ID, 6 ) . str_pad( mb_substr( $r->post_title, 0, 22 ), 24 )
		     . ( 'publish' === $r->post_status ? '켜짐' : '꺼짐' ) . $mark . "\n";
	}
	echo '<pre>' . esc_html( $out ) . '</pre>';
	echo '<p class="dim">스니펫 쪽에 「v5」나 「벼리 사진 있는 판」 표시가 없으면 '
	   . '붙여넣기가 안 된 것입니다.</p>';

	exit;
}, 0 );
