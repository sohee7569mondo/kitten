/* ════════════════════════════════════════════════════════════
   「보는 순서」 04 를 「책 한 권으로 받기」로 · 소희 님께

   지금        04  책으로 받기
   바뀔 말     04  책 한 권으로 받기

   왜 「단행본」이 아닌가
     뜻은 정확하지만 출판하는 사람들이 쓰는 말이라 손님 머릿속에
     그림이 안 그려집니다. 「한 권」 두 글자가 같은 일을 하면서
     훨씬 쉽게 읽힙니다. 홈 위쪽의 「책 한 권으로 받으세요」와도
     말투가 이어집니다.

   ★ 「책으로 받기」라고 적힌 자리만 바꿉니다. 다른 「책」은
     하나도 안 건드립니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 「어디서나 실행」 → 저장 → 활성

   ① 미리보기
        https://stellasaju.com/?stella_step4=1
   ② 적용
        https://stellasaju.com/?stella_step4=1&go=1
   되돌리기
        https://stellasaju.com/?stella_step4=1&undo=1
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_step4'] ) ) {
		return;
	}
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다. 관리자로 로그인한 창에서 열어주세요.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	$META = '_stella_step4_backup';
	$FROM = '책으로 받기';
	$TO   = '책 한 권으로 받기';
	$go   = isset( $_GET['go'] );
	$undo = isset( $_GET['undo'] );

	$safe = function ( $s ) {
		if ( function_exists( 'mb_convert_encoding' ) ) {
			$s = mb_convert_encoding( $s, 'UTF-8', 'UTF-8' );
		}
		return esc_html( $s );
	};

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:15px/1.8 -apple-system,system-ui,sans-serif;'
		. 'background:#111;color:#eee;padding:24px;max-width:900px">';
	printf(
		'<h1 style="color:#ffd76a">「책으로 받기」 → 「책 한 권으로 받기」 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);

	/* 그 말이 적힌 쪽을 찾습니다 */
	$rows = $wpdb->get_results(
		$wpdb->prepare(
			"SELECT ID, post_title, post_content FROM {$wpdb->posts}
			  WHERE post_status IN ('publish','draft','private')
			    AND post_type IN ('page','post','wp_template','wp_template_part','wp_block')
			    AND post_content LIKE %s",
			'%' . $wpdb->esc_like( $undo ? $TO : $FROM ) . '%'
		)
	);

	if ( $undo ) {
		$n = 0;
		$all = $wpdb->get_col(
			$wpdb->prepare( "SELECT post_id FROM {$wpdb->postmeta} WHERE meta_key = %s", $META )
		);
		foreach ( $all as $pid ) {
			$bak = get_post_meta( $pid, $META, true );
			if ( ! $bak ) {
				continue;
			}
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $pid ) );
			delete_post_meta( $pid, $META );
			clean_post_cache( $pid );
			$n++;
		}
		if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		printf( '<h2 style="color:#7bff9b">%d 쪽을 되돌렸습니다.</h2></body>', $n );
		exit;
	}

	if ( ! $rows ) {
		echo '<h2 style="color:#7bff9b">「책으로 받기」라고 적힌 자리가 없습니다. 이미 바뀌었거나 원래 없습니다.</h2></body>';
		exit;
	}

	$total = 0;
	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
	echo '<tr style="color:#ffd76a"><td>쪽</td><td>제목</td><td>몇 번</td></tr>';
	foreach ( $rows as $r ) {
		$n = substr_count( $r->post_content, $FROM );
		$total += $n;
		printf(
			'<tr style="border-top:1px solid #333"><td><b>%d</b></td><td>%s</td>'
			. '<td style="color:#7bff9b"><b>%d</b></td></tr>',
			$r->ID, esc_html( $r->post_title ), $n
		);

		$p = strpos( $r->post_content, $FROM );
		if ( false !== $p ) {
			printf(
				'<tr><td colspan="3"><pre style="white-space:pre-wrap;background:#000;padding:10px;'
				. 'border-left:3px solid #ffd76a;font:12px/1.55 monospace">%s</pre></td></tr>',
				$safe( substr( $r->post_content, max( 0, $p - 180 ), 420 ) )
			);
		}

		if ( $go ) {
			if ( ! get_post_meta( $r->ID, $META, true ) ) {
				update_post_meta( $r->ID, $META, $r->post_content );
			}
			$wpdb->update(
				$wpdb->posts,
				array( 'post_content' => str_replace( $FROM, $TO, $r->post_content ) ),
				array( 'ID' => $r->ID )
			);
			clean_post_cache( $r->ID );
		}
	}
	echo '</table>';

	printf( '<p>모두 <b style="color:#ffd76a">%d</b>군데.</p>', $total );

	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다. 아무것도 바뀌지 않았습니다.</h2>';
		echo '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_step4=1&go=1">'
			. '→ 이대로 바꾸기</a></p></body>';
		exit;
	}

	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
	echo '<h2 style="color:#7bff9b">바꿨습니다</h2>';
	echo '<p style="color:#888">되돌리려면 ?stella_step4=1&undo=1</p></body>';
	exit;
} );
