  function markMine(html, word){
    if(!word){ return html; }
    var at=html.indexOf(word);
    if(at<0){ return html; }
    var from=at, to=at;
    while(from>0){
      var ch=html.charAt(from-1);
      if(ch==='>'){ break; }
      if(ch===' '){ if(html.charAt(from-2)==='.'){ break; } }
      from-=1;
    }
    to=html.indexOf('.', at);
    if(to<0){ to=html.length; } else { to+=1; }
    var mid=html.slice(from,to);
    if(mid.indexOf('<')>=0){ return html; }
    return html.slice(0,from)+'<strong class="mine">'+mid+'</strong>'+html.slice(to);
  }

  function start(){
    var box=document.getElementById('ssb');
    if(!box){ return false; }
    if(box.getAttribute('data-tail')==='1'){ return true; }

    var order=null;
    try{ order=JSON.parse(localStorage.getItem('stella_demo')); }catch(e){ order=null; }
    if(!order){ return false; }

    var S=window.StellaSaju;
    if(!S){ return false; }
    if(!S.lastChart){ return false; }
    var chart=S.lastChart();
    if(!chart){ return false; }

    var topic=order.topic;
    if(!topic){ return true; }
    topic=normTopic(topic);
    if(!ZO[topic]){ if(!DD[topic]){ return true; } }

    var sign='';
    if(chart.stars){ if(chart.stars.sun){ sign=chart.stars.sun.name; } }
    var animal='';
    if(chart.pillars){ if(chart.pillars.year){
      var jz=chart.pillars.year.ji;
      if(typeof jz==='number'){ animal=ANIMAL[jz]+'띠'; }
    } }

    var made=build(topic, sign, animal);
    if(!made.length){ return true; }

    var pages=box.getElementsByClassName('page');
    if(!pages.length){ return false; }

    /* ── 맺음말 쪽을 찾습니다. 아직 없으면 다음에 다시 옵니다. ── */
    var anchor=null, i, eb;
    for(i=0;i<pages.length;i++){
      eb=pages[i].getElementsByClassName('eyebrow')[0];
      if(!eb){ continue; }
      if(eb.textContent==='맺음말'){ anchor=pages[i]; break; }
    }
    if(!anchor){ return false; }

    /* ── 근거 묶음을 맨 뒤로 옮깁니다 ──────────────────────
       장 속표지 패치가 「뒤에 붙이는 것」 속표지와 근거 다섯 쪽을
       맺음말 앞에 두고 있습니다. 그러면 계산 자료가 맺음말과 편지
       사이를 막습니다. 이름 그대로 진짜 뒤로 보냅니다.
       ── 순서 ──
         … 마지막에 놓는 세 장
         별자리로 다시 보면 · 띠로 보면
         맺음말 · ○○님께
         뒤에 붙이는 것 · 근거 하나~여섯             */
    var all=[].slice.call(pages);
    var backs=[], k, p, cls, ebt;
    for(k=0;k<all.length;k++){
      p=all[k];
      cls=p.className;
      if(cls.indexOf('divider')>=0){
        if(p.innerHTML.indexOf('뒤에 붙이는 것')>=0){ backs.push(p); continue; }
      }
      ebt='';
      var e2=p.getElementsByClassName('eyebrow')[0];
      if(e2){ ebt=e2.textContent; }
      if(ebt){ if(ebt.indexOf('근거')===0){ backs.push(p); } }
    }

    if(backs.length>0){
      /* 편지(farewell) 쪽을 찾아 그 뒤에 붙입니다. 없으면 맨 끝. */
      var tail=null;
      for(k=all.length-1;k>=0;k--){
        if(all[k].className.indexOf('farewell')>=0){ tail=all[k]; break; }
      }
      if(!tail){
        for(k=all.length-1;k>=0;k--){
          if(backs.indexOf(all[k])>=0){ continue; }
          tail=all[k]; break;
        }
      }
      if(tail){
        var host=tail.parentNode;
        for(k=0;k<backs.length;k++){
          if(tail.nextSibling){ host.insertBefore(backs[k], tail.nextSibling); }
          else { host.appendChild(backs[k]); }
          tail=backs[k];
        }
      }
    }

    /* ── 덤 두 쪽을 맺음말 앞에 ── */
    var hold=document.createElement('div');
    hold.innerHTML=made.join('');
    while(hold.firstChild){ anchor.parentNode.insertBefore(hold.firstChild, anchor); }

    box.setAttribute('data-tail','1');
    renumber();
    return true;
  }

  /* 다른 조각들이 쪽을 다 넣을 때까지 기다립니다. 최대 20초. */
  var tries=0;
  function tick(){
    tries+=1;
    var done=false;
    try{ done=start(); }catch(e){ done=true; }
    if(done){ return; }
    if(tries>60){ return; }
    setTimeout(tick, 330);
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', function(){ setTimeout(tick, 400); });
  } else { setTimeout(tick, 400); }
})();
