/* ══════════════════════════════════════════════════════════
   전자책까지 가는 길 줄이기
   ★ 2026-09-10 · 소희 님께

   소희 님 : 「전자책까지 가는 길이 너무 멀어」

   지금 일곱 걸음입니다.

     문 → 정보 입력 → 가디언 인사 → 질문 일곱 → 카드 석 장
        → 자료 확인 → 전자책

   두 걸음을 덜어냅니다.

   ── 하나. 자료 확인(156)을 길에서 뺍니다 ────────────────
   질문 쪽에서 발행을 마치면 자료 확인을 건너뛰고 곧바로
   전자책으로 갑니다.

   ★ 쪽을 지우지 않습니다. 주소를 눌러 들어가면 그대로 보입니다.
     질문 쪽에서 넘어올 때만 건너뜁니다. 그래서 나중에 책 안에서
     「내 자료 확인하기」로 이어 붙이기만 하면 됩니다.

   ── 둘. 카드를 안 골라도 되게 합니다 ────────────────────
   카드 화면 아래에 한 줄이 생깁니다.

     고르기 어려우시면 — 아무 세 장이나 뽑아드릴게요

   누르면 석 장이 저절로 날아가 앉고, 곧바로 발행됩니다.
   카드는 여전히 석 장이라 책 내용은 하나도 안 달라집니다.

   ── 되돌리기 ────────────────────────────────────────────
   ★ 이 스니펫은 쪽을 고치지 않습니다. 스위치를 끄면 그 즉시
     원래대로 돌아옵니다. undo 가 따로 없습니다.

   살펴보기 https://stellasaju.com/?stella_short=1

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

/* ── 하나. 질문 쪽에서 넘어왔으면 자료 확인을 건너뜁니다 ── */
add_action( 'template_redirect', function () {

	if ( ! is_page( 'reading-ready' ) ) { return; }

	$ref = isset( $_SERVER['HTTP_REFERER'] ) ? (string) $_SERVER['HTTP_REFERER'] : '';
	if ( '' === $ref ) { return; }
	if ( false === strpos( $ref, '/reading-start' ) ) { return; }

	wp_safe_redirect( home_url( '/reading-book/' ), 302 );
	exit;
}, 9 );


/* ── 둘. 카드 화면에 「아무 세 장이나」 한 줄 ── */
add_action( 'wp_footer', function () {

	if ( ! is_page( 'reading-start' ) ) { return; }

	echo "<script>\n";
	echo <<<'STELLA_SHORT_JS'

(function(){
  var done = document.getElementById('qDone');
  if(!done){ return; }
  if(document.getElementById('stSkip')){ return; }

  var css = document.createElement('style');
  css.textContent =
    '#ssq .stskip{ display:block; margin:14px auto 0; background:none; border:0;' +
    '  color:rgba(255,255,255,.58); font-size:.88rem; text-decoration:underline;' +
    '  cursor:pointer; padding:6px 10px; font-family:inherit; }' +
    '#ssq .stskip:disabled{ opacity:.45; text-decoration:none; cursor:default; }';
  document.head.appendChild(css);

  var skip = document.createElement('button');
  skip.type = 'button';
  skip.id = 'stSkip';
  skip.className = 'stskip';
  skip.textContent = '고르기 어려우시면 — 아무 세 장이나 뽑아드릴게요';
  done.parentNode.insertBefore(skip, done.nextSibling);

  skip.addEventListener('click', function(){
    skip.disabled = true;
    skip.textContent = '뽑는 중…';

    var all  = document.querySelectorAll('#qDeck .back');
    var left = [];
    var i;
    for(i = 0; i < all.length; i++){
      if(all[i].classList.contains('gone')){ continue; }
      left.push(all[i]);
    }
    var need = 3 - (all.length - left.length);
    var wait = 0;

    while(need > 0){
      if(!left.length){ break; }
      var k = Math.floor(Math.random() * left.length);
      (function(el, ms){
        setTimeout(function(){ el.click(); }, ms);
      })(left[k], wait);
      left.splice(k, 1);
      need = need - 1;
      wait = wait + 300;
    }

    setTimeout(function(){
      var d = document.getElementById('qDone');
      if(!d){ return; }
      if(d.disabled){ skip.disabled = false; skip.textContent = '아무 세 장이나 뽑아드릴게요'; return; }
      d.click();
    }, wait + 520);
  });
})();

STELLA_SHORT_JS;
	echo "\n</script>\n";
}, 99 );


/* ── 살펴보기 ── */
add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_short'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/plain; charset=utf-8' );

	echo "═══ 길 줄이기 ═══\n\n";
	echo "이 스니펫이 켜져 있으면\n\n";
	echo "  · 질문 쪽에서 발행을 마치면 자료 확인을 건너뛰고\n";
	echo "    곧바로 전자책으로 갑니다\n";
	echo "  · 카드 화면에 「아무 세 장이나 뽑아드릴게요」가 생깁니다\n\n";

	foreach ( array( 'reading-start', 'reading-ready', 'reading-book' ) as $slug ) {
		$p = get_page_by_path( $slug );
		echo sprintf( "  %-16s %s\n", $slug, $p ? ( '쪽 ' . $p->ID ) : '★ 없습니다' );
	}

	echo "\n자료 확인 쪽은 지우지 않았습니다.\n";
	echo '  ' . home_url( '/reading-ready/' ) . "  ← 눌러 들어가면 그대로 보입니다\n";
	echo "  질문 쪽에서 넘어올 때만 건너뜁니다.\n";
	echo "\n되돌리려면 이 스니펫 스위치를 끄시면 됩니다. 그 즉시입니다.\n";
	exit;
}, 1 );
