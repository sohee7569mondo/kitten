/* ════════════════════════════════════════════════════════════
   책 표지에 주제 사진 넣기 · 소희 님께

   지금 표지는 글자만 있습니다 — 딱지 · 이름 · 주제 · 쓴 이.
   그림이 한 장도 없습니다.

   이 화면에서 주제마다 사진을 고르면, 그 주제로 나온 책의
   표지에 그 사진이 깔립니다. 글자는 사진 위에 그대로 얹히고,
   읽히도록 옅은 막을 한 겹 씌웁니다.

   ★ 파일 이름은 아무래도 됩니다. 목록에서 고르는 방식입니다.
   ★ 저장하면 그 자리에서 책에 들어갑니다. 패치를 따로 안 넣으셔도 됩니다.
   ★ 나중에 사진을 바꾸실 때도 이 화면에서 다시 고르시면 됩니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 「어디서나 실행」 → 저장 → 활성

        https://stellasaju.com/?stella_coverpic=1

   ★ 되돌리기 (표지에서 사진을 통째로 걷어냅니다)
        https://stellasaju.com/?stella_coverpic=1&undo=1
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_coverpic'] ) ) { return; }
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	$OPT   = 'stella_cover_pics';
	$META  = '_stella_coverpic_backup';
	$MARK  = '2026-09-08 · 표지 주제 사진';
	$page  = get_page_by_path( 'reading-book' );
	if ( ! $page ) { wp_die( 'reading-book 쪽을 못 찾았습니다.', '스텔라', array( 'response' => 200 ) ); }
	$PID = $page->ID;

	/* 주제 — 문 여섯이 내놓는 것들 */
	$TOPICS = array(
		'벼리 · 연애와 결혼' => array( '연애운', '재회운', '결혼운', '이별운', '짝사랑', '궁합' ),
		'마루 · 일과 돈'     => array( '직업운', '재물운', '이직운', '퇴사운', '직장운', '사업운' ),
		'아람 · 건강과 가족' => array( '건강운', '가족운' ),
		'미르 · 흐름과 시기' => array( '시기운', '대운', '인생총운 · 말년' ),
		'아라 · 점성술'      => array( '별자리로 보는 나', '별자리와 사주' ),
		'아르카나 · 타로'    => array( '오늘의 선택' ),
	);

	$pics = get_option( $OPT );
	if ( ! is_array( $pics ) ) { $pics = array(); }

	/* ── 되돌리기 ── */
	if ( isset( $_GET['undo'] ) ) {
		$bak = get_post_meta( $PID, $META, true );
		header( 'Content-Type: text/html; charset=utf-8' );
		echo '<meta charset="utf-8"><body style="background:#111;color:#eee;'
			. 'font:15px/1.8 system-ui,sans-serif;padding:24px">';
		if ( $bak ) {
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $PID ) );
			delete_post_meta( $PID, $META );
			clean_post_cache( $PID );
			if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
			echo '<h1 style="color:#7bff9b">표지에서 사진을 걷어냈습니다.</h1>';
		} else {
			echo '<h1 style="color:#888">사본이 없습니다.</h1>';
		}
		echo '<p style="color:#888">고르신 사진 목록은 그대로 두었습니다. '
			. '다시 저장하시면 되살아납니다.</p></body>';
		exit;
	}

	$saved = 0;

	/* ── 저장 ── */
	if ( isset( $_POST['stella_coverpic_save'] ) ) {
		check_admin_referer( 'stella_coverpic' );

		$pick = isset( $_POST['pic'] ) ? (array) $_POST['pic'] : array();
		$next = array();
		foreach ( $pick as $topic => $aid ) {
			$topic = sanitize_text_field( wp_unslash( $topic ) );
			$aid   = (int) $aid;
			if ( $aid < 1 ) { continue; }
			$url = wp_get_attachment_url( $aid );
			if ( ! $url ) { continue; }
			/* 주소에 앰퍼샌드가 있으면 넣지 않습니다 (사이트 규칙) */
			if ( false !== strpos( $url, '&' ) ) { continue; }
			$next[ $topic ] = $url;
			$saved++;
		}
		update_option( $OPT, $next, false );
		$pics = $next;

		/* 책에 심기 */
		$row = $wpdb->get_row( $wpdb->prepare(
			"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $PID ) );
		if ( $row ) {
			$c = $row->post_content;
			if ( ! get_post_meta( $PID, $META, true ) ) {
				update_post_meta( $PID, $META, $c );
			}
			/* 전에 심은 덩어리가 있으면 걷어냅니다 */
			$s = strpos( $c, '<!-- ' . $MARK . ' -->' );
			if ( false !== $s ) {
				$e = strpos( $c, '<!-- /' . $MARK . ' -->', $s );
				if ( false !== $e ) {
					$e = $e + strlen( '<!-- /' . $MARK . ' -->' );
					$c = substr( $c, 0, $s ) . substr( $c, $e );
				}
			}

			$json = wp_json_encode( $pics, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES );
			if ( ! is_string( $json ) ) { $json = '{}'; }

			$block  = "\n<!-- " . $MARK . " -->\n";
			$block .= "<style>\n";
			$block .= "#ssb .page.cover.haspic{position:relative; overflow:hidden;\n";
			$block .= "  min-height:74vh; background-size:cover; background-position:center;\n";
			$block .= "  display:flex; flex-direction:column; justify-content:center;}\n";
			$block .= "#ssb .page.cover.haspic:before{content:''; position:absolute; inset:0;\n";
			$block .= "  background:linear-gradient(180deg, rgba(247,243,235,.90) 0%,\n";
			$block .= "    rgba(247,243,235,.74) 42%, rgba(247,243,235,.95) 100%);}\n";
			$block .= "#ssb .page.cover.haspic > *{position:relative; z-index:1;}\n";
			$block .= "@media print{#ssb .page.cover.haspic{min-height:auto;}}\n";
			$block .= "</style>\n";
			$block .= "<script>\n";
			$block .= "/* 표지 주제 사진 — 이 안에 홑 앰퍼샌드를 쓰지 않습니다 */\n";
			$block .= "window.STELLA_COVER_PICS = " . $json . ";\n";
			$block .= "(function(){\n";
			$block .= "  var tries = 0;\n";
			$block .= "  function put(){\n";
			$block .= "    var box = document.getElementById('ssb');\n";
			$block .= "    if(!box){ return false; }\n";
			$block .= "    var cv = box.querySelector('.page.cover');\n";
			$block .= "    if(!cv){ return false; }\n";
			$block .= "    if(cv.classList.contains('haspic')){ return true; }\n";
			$block .= "    var h1 = cv.querySelector('h1');\n";
			$block .= "    if(!h1){ return false; }\n";
			$block .= "    var t = h1.textContent.replace(/\\s+/g, '');\n";
			$block .= "    var map = window.STELLA_COVER_PICS;\n";
			$block .= "    var keys = Object.keys(map);\n";
			$block .= "    keys.sort(function(a, b){ return b.length - a.length; });\n";
			$block .= "    var i, k, kk;\n";
			$block .= "    for(i = 0; i < keys.length; i++){\n";
			$block .= "      k  = keys[i];\n";
			$block .= "      kk = k.replace(/\\s+/g, '');\n";
			$block .= "      if(t.slice(-kk.length) === kk){\n";
			$block .= "        cv.style.backgroundImage = 'url(' + map[k] + ')';\n";
			$block .= "        cv.classList.add('haspic');\n";
			$block .= "        return true;\n";
			$block .= "      }\n";
			$block .= "    }\n";
			$block .= "    return true;\n";
			$block .= "  }\n";
			$block .= "  var tm = setInterval(function(){\n";
			$block .= "    tries = tries + 1;\n";
			$block .= "    if(put()){ clearInterval(tm); return; }\n";
			$block .= "    if(tries > 90){ clearInterval(tm); }\n";
			$block .= "  }, 300);\n";
			$block .= "})();\n";
			$block .= "</script>\n";
			$block .= "<!-- /" . $MARK . " -->\n";

			$wpdb->update( $wpdb->posts, array( 'post_content' => $c . $block ), array( 'ID' => $PID ) );
			clean_post_cache( $PID );
			if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		}
	}

	/* ── 미디어 서랍 ── */
	$atts = get_posts( array(
		'post_type'      => 'attachment',
		'post_mime_type' => 'image',
		'post_status'    => 'inherit',
		'posts_per_page' => 300,
		'orderby'        => 'date',
		'order'          => 'DESC',
	) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">';
	echo '<style>body{background:#12101B;color:#F2ECE0;'
		. 'font:15px/1.7 -apple-system,"Apple SD Gothic Neo",system-ui,sans-serif;'
		. 'padding:22px 16px;max-width:1000px;margin:0 auto}'
		. 'h1{color:#D4AF6A;font-size:1.3rem;margin:0 0 6px}'
		. 'h2{color:#9AD1E5;font-size:1rem;margin:30px 0 6px;'
		. 'border-bottom:1px solid #2A2438;padding-bottom:6px}'
		. '.dim{color:#8B849C;font-size:.88rem}'
		. '.row{display:flex;gap:14px;align-items:center;'
		. 'border-top:1px solid #2A2438;padding:13px 0}'
		. '.thumb{width:66px;height:92px;object-fit:cover;border-radius:7px;'
		. 'background:#221D33;flex:0 0 auto}'
		. '.mid{flex:1 1 auto;min-width:0}'
		. '.tp{font-weight:700;margin-bottom:6px}'
		. 'select{width:100%;max-width:420px;padding:9px;border-radius:7px;'
		. 'background:#1B172A;color:#F2ECE0;border:1px solid #3A3350;font:inherit}'
		. 'button{background:#D4AF6A;color:#1A1230;border:0;border-radius:8px;'
		. 'padding:14px 30px;font:inherit;font-weight:700;cursor:pointer}'
		. '.ok{color:#9AD1A5}a{color:#D4AF6A}</style>';
	echo '<body><h1>책 표지에 넣을 주제 사진</h1>';
	echo '<p class="dim">주제마다 사진을 고르면 그 주제로 나온 책 표지에 깔립니다. '
		. '글자는 사진 위에 얹히고, 읽히도록 옅은 막을 한 겹 씌웁니다. '
		. '안 고른 주제는 지금처럼 글자만 나옵니다.</p>';

	if ( $saved ) {
		printf( '<p class="ok">%d 개를 저장하고 책에 넣었습니다. '
			. '<a href="/reading-book/?demo=door-health">책에서 보기</a></p>', $saved );
	}

	echo '<form method="post">';
	wp_nonce_field( 'stella_coverpic' );

	foreach ( $TOPICS as $door => $list ) {
		printf( '<h2>%s</h2>', esc_html( $door ) );
		foreach ( $list as $tp ) {
			$cur = isset( $pics[ $tp ] ) ? $pics[ $tp ] : '';
			printf(
				'<div class="row"><img class="thumb" id="pv%s" src="%s" alt=""><div class="mid">'
				. '<div class="tp">%s</div>'
				. '<select name="pic[%s]" data-pv="pv%s"><option value="">없음 (글자만)</option>',
				esc_attr( md5( $tp ) ), esc_url( $cur ), esc_html( $tp ),
				esc_attr( $tp ), esc_attr( md5( $tp ) )
			);
			foreach ( $atts as $a ) {
				$u = wp_get_attachment_url( $a->ID );
				printf(
					'<option value="%d" data-u="%s"%s>%s</option>',
					$a->ID, esc_url( $u ),
					( $u === $cur ) ? ' selected' : '',
					esc_html( basename( (string) $u ) )
				);
			}
			echo '</select></div></div>';
		}
	}

	echo '<p style="margin:28px 0 44px"><button type="submit" name="stella_coverpic_save" value="1">'
		. '저장하고 책에 넣기</button></p></form>';

	echo '<script>'
		. 'document.querySelectorAll("select").forEach(function(s){'
		. ' s.addEventListener("change",function(){'
		. '  var o=s.options[s.selectedIndex];'
		. '  var u=o.getAttribute("data-u");'
		. '  var img=document.getElementById(s.getAttribute("data-pv"));'
		. '  if(!img){ return; }'
		. '  img.src = u ? u : "";'
		. ' });'
		. '});</script>';

	echo '<p class="dim">표지에서 사진을 통째로 걷어내려면 '
		. '<code>?stella_coverpic=1&undo=1</code></p></body>';
	exit;
}, 0 );
