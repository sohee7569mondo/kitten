//
// 그동안 띠는 자바스크립트가 만들어 끼워넣었습니다. 그래서 여백이
// 덮이고 자리가 흔들렸습니다. 이제 쪽 글에 박아 CSS 한 줄이 여백을 정합니다.
//
//   살펴보기   https://stellasaju.com/?stella_homeband=1
//   넣기       https://stellasaju.com/?stella_homeband=apply
//   빼기       https://stellasaju.com/?stella_homeband=undo
//
// 넣으신 뒤 patch_homeprice 를 꺼 주세요 - 켜 두면 띠가 둘이 됩니다.
// 위치는 어디서나 실행. 크롬 번역을 끄고 붙여넣으세요.

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_homeband'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_homeband'] ) );
	$pid  = 62;

	$p = get_post( $pid );
	if ( ! $p ) { echo '홈 쪽 62 를 못 찾았습니다'; exit; }

	$c = (string) $p->post_content;

	// 넣을 자리 — 히어로 바로 다음, 폼 앞
	$AN = '  <!-- ══ 폼 ═';
	$MK = 'id=' . chr(34) . 'hp-open' . chr(34);

	// 넣을 글
	$S = '  <!-- STELLA-BAND-START -->';
	$E = '  <!-- STELLA-BAND-END -->';
	$H = '';
	$H .= $S . chr(10);
	$H .= '  <!-- ══ 오픈기념 배너 ════════════════════' . chr(10);
	$H .= '       2026-09-29 · 조각이 자바스크립트로 끼워넣던 것을 쪽 글로 옮겼습니다.' . chr(10);
	$H .= '       여백은 #hp-open 의 padding 한 줄이 정합니다.' . chr(10);
	$H .= '       ★ patch_homeprice 는 꺼야 합니다 — 켜 두면 띄가 둘이 됩니다. -->' . chr(10);
	$H .= '  <style>' . chr(10);
	$H .= '  #stellar-home #hp-open{ padding:56px 0 76px; }' . chr(10);
	$H .= '  @media(max-width:700px){ #stellar-home #hp-open{ padding:38px 0 52px; } }' . chr(10);
	$H .= '  #hp-band{ position:relative; overflow:hidden;' . chr(10);
	$H .= '    padding:32px 36px; border-radius:20px;' . chr(10);
	$H .= '    display:flex; flex-wrap:wrap; align-items:center; gap:22px 34px;' . chr(10);
	$H .= '    background:linear-gradient(120deg,#F3EEFC 0%,#FFFFFF 62%);' . chr(10);
	$H .= '    border:1px solid #D9CFF2;' . chr(10);
	$H .= '    box-shadow:0 18px 44px -26px rgba(59,44,120,.45); }' . chr(10);
	$H .= '  #hp-band .hp-kick{ display:inline-block; padding:5px 13px; border-radius:30px;' . chr(10);
	$H .= '    background:#6B4FB0; color:#FFFDF9;' . chr(10);
	$H .= '    font-size:.72rem; font-weight:700; letter-spacing:.12em; }' . chr(10);
	$H .= '  #hp-band .hp-main{ flex:1 1 400px; min-width:0; }' . chr(10);
	$H .= '  #hp-band .hp-row{ display:flex; flex-wrap:wrap; align-items:flex-end;' . chr(10);
	$H .= '    gap:8px 18px; margin-top:16px; }' . chr(10);
	$H .= '  #hp-band .hp-big{ font-family:Noto Serif KR,serif; font-weight:700;' . chr(10);
	$H .= '    font-size:clamp(1.5rem,4vw,2.1rem); line-height:1.3; color:var(--ink,#221C33); }' . chr(10);
	$H .= '  #hp-band .hp-price{ display:flex; align-items:baseline; gap:8px; padding-bottom:5px; }' . chr(10);
	$H .= '  #hp-band .hp-was{ color:var(--ink-dim,#8B849C); text-decoration:line-through;' . chr(10);
	$H .= '    font-size:.98rem; }' . chr(10);
	$H .= '  #hp-band .hp-arrow{ color:var(--ink-dim,#8B849C); font-size:.9rem; }' . chr(10);
	$H .= '  #hp-band .hp-now{ color:#C4453A; font-weight:800; font-size:1.34rem; }' . chr(10);
	$H .= '  #hp-band .hp-say{ margin-top:14px; max-width:660px;' . chr(10);
	$H .= '    color:var(--ink-soft,#4E4763); font-size:.94rem; line-height:1.85; }' . chr(10);
	$H .= '  #hp-band .hp-go{ flex:0 0 auto; text-decoration:none;' . chr(10);
	$H .= '    padding:13px 26px; border-radius:30px; font-weight:700; font-size:.95rem;' . chr(10);
	$H .= '    color:#FFFDF9; background:linear-gradient(120deg,#3A2E77,#241C4E);' . chr(10);
	$H .= '    box-shadow:0 16px 34px -16px rgba(36,28,78,.7); }' . chr(10);
	$H .= '  #hp-band .hp-go:hover{ background:#2A2158; }' . chr(10);
	$H .= '  @media(max-width:700px){' . chr(10);
	$H .= '    #hp-band{ border-radius:16px; }' . chr(10);
	$H .= '    #hp-band .hp-now{ font-size:1.2rem; }' . chr(10);
	$H .= '    #hp-band{ padding:26px 20px; gap:18px; }' . chr(10);
	$H .= '    #hp-band .hp-main{ flex:1 1 100%; }' . chr(10);
	$H .= '    #hp-band .hp-go{ flex:1 1 100%; text-align:center; } }' . chr(10);
	$H .= '  </style>' . chr(10);
	$H .= '  <section class="sh-section" id="hp-open">' . chr(10);
	$H .= '    <div class="wrap">' . chr(10);
	$H .= '      <div id="hp-band">' . chr(10);
	$H .= '        <div class="hp-main">' . chr(10);
	$H .= '        <span class="hp-kick">오픈기념</span>' . chr(10);
	$H .= '        <div class="hp-row">' . chr(10);
	$H .= '          <div class="hp-big">여섯 문이 모두 무료입니다</div>' . chr(10);
	$H .= '          <div class="hp-price">' . chr(10);
	$H .= '            <span class="hp-was">9,000원</span>' . chr(10);
	$H .= '            <span class="hp-arrow">›</span>' . chr(10);
	$H .= '            <span class="hp-now">0원</span>' . chr(10);
	$H .= '          </div>' . chr(10);
	$H .= '        </div>' . chr(10);
	$H .= '        <div class="hp-say">오픈 기간에는 값을 받지 않습니다. 가입도, 카드도 필요 없습니다. 글은 한 줄도 줄이지 않아서, 풀이 한 편이 책 한 권 그대로입니다.</div>' . chr(10);
	$H .= '        </div>' . chr(10);
	$H .= '        <a class="hp-go" href="/door-fortune/?topic=2026%EB%85%84%20%EC%9A%B4%EC%84%B8">2026년 운세부터 보기 ›</a>' . chr(10);
	$H .= '      </div>' . chr(10);
	$H .= '    </div>' . chr(10);
	$H .= '  </section>' . chr(10);
	$H .= $E . chr(10);

	echo "오픈기념 띠 - 홈 쪽 글에 넣기\n\n";
	echo '지금 쪽 ' . number_format( strlen( $c ) ) . "자\n";
	echo '넣을 자리 ' . substr_count( $c, $AN ) . "군데\n";
	echo '이미 들어감 ' . substr_count( $c, $MK ) . "군데\n\n";

	if ( 'undo' === $mode ) {
		$a = strpos( $c, $S );
		if ( false === $a ) { echo '뺄 것이 없습니다'; exit; }
		$b = strpos( $c, $E, $a );
		if ( false === $b ) { echo '끝 표지를 못 찾았습니다'; exit; }
		$new = substr( $c, 0, $a ) . substr( $c, $b + strlen( $E ) + 1 );
	} else {
		if ( substr_count( $c, $MK ) > 0 ) { echo '이미 들어가 있습니다.'; exit; }
		if ( 1 !== substr_count( $c, $AN ) ) { echo '넣을 자리를 못 찾았거나 여럿입니다.'; exit; }
		if ( 'apply' !== $mode ) {
			echo "넣으려면  ?stella_homeband=apply\n";
			echo "빼려면    ?stella_homeband=undo\n";
			exit;
		}
		$new = str_replace( $AN, $H . $AN, $c );
	}

	$b1 = substr_count( $c, chr(92) );
	wp_update_post( array( 'ID' => $pid, 'post_content' => wp_slash( $new ) ) );
	clean_post_cache( $pid );
	$saved = (string) get_post( $pid )->post_content;
	$b2 = substr_count( $saved, chr(92) );

	if ( $b2 !== $b1 ) {
		wp_update_post( array( 'ID' => $pid, 'post_content' => wp_slash( $c ) ) );
		clean_post_cache( $pid );
		echo '되돌렸습니다 - 역빗금 ' . $b1 . ' 에서 ' . $b2;
		exit;
	}

	echo '됐습니다 · 지금 ' . number_format( strlen( $saved ) ) . '자 · 띠 '
		. substr_count( $saved, $MK ) . '군데 · 역빗금 ' . $b2 . "개 그대로\n\n";
	echo "이제 patch_homeprice 를 꺼 주세요.\n";
	exit;
} );
