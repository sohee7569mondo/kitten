		array( '④ 미루는 편이 나은 자리', '/동업 · 보증 · 큰 지출은 이 구간을 피하시는 게 낫습니다\\.\'\\s*\\};/u', $i4, $k4, 'after' ),
		array( '⑤ 지금 판과 다음 판', '/nowTxt\\+\\s*\'<div class=\\"voice\\">/u', $i5, $k5, 'call' ),
	);

	printf(
		'<h1 style="color:#ffd76a">세성의 신 주제별 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);

	$live = $wpdb->get_var( $wpdb->prepare( "SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	/* 첫 판이 남아 있으면 그것을 걷어낸 자리에서 시작합니다 */
	$first = get_post_meta( $pid, '_stella_fortune_backup', true );
	$now   = $first ? $first : $live;
	if ( null === $now ) {
		echo '<h2 style="color:#ff7b7b">160 을 못 읽었습니다.</h2>';
		exit;
	}

	if ( $undo ) {
		$bak = get_post_meta( $pid, '_stella_fortune2_backup', true );
		if ( $bak ) {
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $pid ) );
			delete_post_meta( $pid, '_stella_fortune2_backup' );
			clean_post_cache( $pid );
			if ( function_exists( 'wp_cache_flush' ) ) {
				wp_cache_flush();
			}
			echo '<h2 style="color:#7bff9b">되돌렸습니다.</h2>';
		} else {
			echo '<h2 style="color:#888">사본이 없습니다.</h2>';
		}
		exit;
	}

	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
	echo '<tr style="color:#ffd76a"><td>할 일</td><td>자리</td><td>이미 됨</td><td>글자 수</td></tr>';

	$ready = true;
	$done  = true;

	foreach ( $steps as $s ) {
		$n = preg_match_all( $s[1], $now );
		$m = substr_count( $now, $s[3] );

		if ( $m > 0 ) {
			printf(
				'<tr style="border-top:1px solid #333"><td>%s</td><td style="color:#888">%d</td><td style="color:#9fd">이미 됨</td><td>%d</td></tr>',
				esc_html( $s[0] ), $n, strlen( $s[2] )
			);
			continue;
		}
		$done = false;
		if ( 1 !== $n ) {
			$ready = false;
		}
		printf(
			'<tr style="border-top:1px solid #333"><td>%s</td><td style="color:%s"><b>%d</b></td><td></td><td>%d</td></tr>',
			esc_html( $s[0] ), 1 === $n ? '#7bff9b' : '#ff7b7b', $n, strlen( $s[2] )
		);
	}
	echo '</table>';

	if ( $done ) {
		echo '<h2 style="color:#7bff9b">이미 다 들어가 있습니다.</h2>'
			. '<p style="color:#888">되돌리려면 ?stella_fortune2=1&undo=1</p>';
		exit;
	}
	if ( ! $ready ) {
		echo '<h2 style="color:#ff7b7b">자리가 딱 한 번씩이 아니라 아무것도 바꾸지 않았습니다.</h2>'
			. '<p>이 화면을 저에게 보내주세요.</p>';
		exit;
	}
	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다.</h2>'
			. '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_fortune2=1&go=1">→ 이대로 넣기</a></p>';
		exit;
	}

	if ( ! get_post_meta( $pid, '_stella_fortune2_backup', true ) ) {
		update_post_meta( $pid, '_stella_fortune2_backup', $now );
	}

	$out = $now;
	foreach ( $steps as $s ) {
		if ( substr_count( $out, $s[3] ) > 0 ) {
			continue;
		}
		$ins  = $s[2];
		$mode = $s[4];
		$out  = preg_replace_callback(
			$s[1],
			function ( $mm ) use ( $ins, $mode ) {
				if ( 'after' === $mode ) {
					return $mm[0] . $ins;
				}
				return str_replace( 'nowTxt+', 'nowTxt+__fNow+', $mm[0] );
			},
			$out,
			1
		);
	}

	$wpdb->update( $wpdb->posts, array( 'post_content' => $out ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) {
		wp_cache_flush();
	}

	printf( '<h2 style="color:#7bff9b">넣었습니다 — %d → %d 바이트</h2>', strlen( $now ), strlen( $out ) );
	echo '<p style="color:#888">되돌리려면 ?stella_fortune2=1&undo=1</p>';
	exit;
} );
