/* ════════════════════════════════════════════════════════════
   메뉴 이름 바꾸기 · 그려질 때   ★ 2026-09-09 · 소희 님께

   소희 님 : 「메뉴에 있는 타로를 스텔라 타로로 바꾸는 거,
             데스크탑은 됐는데 모바일이...」

   ── 왜 이렇게 하나 ─────────────────────────────
   메뉴 자료를 고치는 대신, 화면에 그려지는 순간에 이름을 바꿉니다.
   그래서 메뉴가 어디에 있든 상관이 없습니다 —
   위쪽 메뉴 · 모바일 메뉴 · 바닥글, 다 같이 바뀝니다.
   테마 파일 안에 있어서 제 자로 못 찾던 메뉴에도 걸립니다.

   ── 무엇을 바꾸나 ──────────────────────────────
   아래 표에 적은 것만 바꿉니다. 메뉴 안에서만요.
   본문 글은 한 글자도 안 건드립니다.

   ── 먼저 볼 것 ─────────────────────────────────
   휴대폰으로 이 주소를 열고 메뉴를 펼쳐보세요.

     https://stellasaju.com/?stella_navlabels=1

   화면 맨 아래에 「지금 이 화면에 그려진 메뉴 이름」이 전부 나옵니다.
   모바일이 정말 다른 메뉴를 쓰는지 여기서 판가름납니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 끄면 그대로 돌아갑니다. 자료를 안 건드리기 때문입니다.
   ════════════════════════════════════════════════════════════ */

/* ── 여기만 고치면 됩니다 ───────────────────────── */
function stella_nav_renames() {
	return array(
		'아르카나 · 타로' => '아르카나 · 스텔라 타로',
		/* 더 바꾸실 것이 있으면 이렇게 한 줄씩 :
		   '옛 이름' => '새 이름', */
	);
}

/* 그려진 메뉴 이름을 모아두는 곳 (아래 진단용) */
global $stella_nav_seen;
$stella_nav_seen = array();

add_filter( 'render_block', function ( $html, $block ) {

	if ( empty( $block['blockName'] ) ) { return $html; }
	$bn = $block['blockName'];

	$isNav = false;
	if ( 0 === strpos( $bn, 'core/navigation' ) ) { $isNav = true; }
	if ( 'core/page-list' === $bn ) { $isNav = true; }
	if ( ! $isNav ) { return $html; }

	/* 그려진 이름을 모읍니다 — 무엇이 있는지 눈으로 보려고 */
	if ( isset( $_GET['stella_navlabels'] ) ) {
		global $stella_nav_seen;
		$from = 0;
		while ( true ) {
			$at = strpos( $html, '__label">', $from );
			if ( false === $at ) { break; }
			$s = $at + 9;
			$e = strpos( $html, '<', $s );
			if ( false === $e ) { break; }
			$from = $e;
			$t = trim( wp_strip_all_tags( substr( $html, $s, $e - $s ) ) );
			if ( '' !== $t ) { $stella_nav_seen[] = $bn . '  |  ' . $t; }
		}
	}

	foreach ( stella_nav_renames() as $old => $new ) {
		if ( '' === $old ) { continue; }
		/* 이름표 안의 글자만 바꿉니다 */
		$html = str_replace( '>' . $old . '<', '>' . $new . '<', $html );
		/* 제목 속성(title)에도 옛 이름이 남아 있을 수 있습니다 */
		$html = str_replace( '"' . $old . '"', '"' . $new . '"', $html );
	}

	return $html;
}, 10, 2 );

/* ── 지금 화면에 그려진 메뉴 이름 보여주기 ───────── */
add_action( 'wp_footer', function () {

	if ( ! isset( $_GET['stella_navlabels'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $stella_nav_seen;
	$seen = is_array( $stella_nav_seen ) ? $stella_nav_seen : array();

	echo '<div style="position:relative;z-index:99999;background:#12101B;'
	   . 'color:#F2ECE0;padding:20px 16px;font:13px/1.7 monospace;'
	   . 'white-space:pre-wrap;border-top:3px solid #D4AF6A">';
	echo "지금 이 화면에 그려진 메뉴 이름  " . count( $seen ) . "개\n";
	echo "(휴대폰이면 메뉴를 한 번 펼치고 다시 보세요)\n\n";
	if ( ! count( $seen ) ) {
		echo "하나도 없습니다 — 이 쪽에는 메뉴 블록이 없거나,\n";
		echo "메뉴가 블록이 아닌 다른 방식으로 그려지고 있습니다.\n";
	} else {
		$i = 0;
		foreach ( $seen as $s ) { $i++; echo str_pad( (string) $i, 4 ) . esc_html( $s ) . "\n"; }
	}
	echo "\n바꾸기로 적어둔 것\n";
	foreach ( stella_nav_renames() as $o => $n ) {
		echo '   ' . esc_html( $o ) . '  ->  ' . esc_html( $n ) . "\n";
	}
	echo '</div>';
}, 999 );
