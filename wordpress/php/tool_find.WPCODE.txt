/* ══════════════════════════════════════════════════════════
   글 찾기 — 책에서 이 문장이 어디에 적혀 있는지
   ★ 2026-09-10 · 소희 님께

   책에서 고치고 싶은 문장을 보시면, 그 문장을 그대로 여기에
   넣어보세요. 책 안 어디에 적혀 있는지 찾아 드립니다.

     https://stellasaju.com/?stella_find=여기에 문장

   책의 글은 코드 안에서 여러 조각으로 잘려 있을 때가 많습니다.
   화면에는 한 문장으로 보여도 코드에는

       '올해는 벌여둔 것을 '+
       '내보내는 해입니다.'

   처럼 나뉘어 있는 식이에요. 그래서 통째로 못 찾으면 앞에서부터
   한 글자씩 줄여가며 찾히는 가장 긴 조각을 알려드립니다.
   그 조각이 제가 고칠 때 쓰는 손잡이가 됩니다.

   쓰는 법
     ?stella_find=올해는 벌여둔 것을        ← 그냥 문장
     ?stella_find=올해는&n=5                ← 앞뒤로 더 넓게 (기본 3)
     ?stella_find=올해는&all=1              ← 나온 자리를 전부

   ★ 사람마다 달라지는 값(이름 · 날짜 · 별자리)은 빼고 넣으세요.
     「소희님은 물의 기운이」 대신 「의 기운이」처럼요.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_find'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$q = (string) wp_unslash( $_GET['stella_find'] );
	$q = trim( $q );
	if ( '' === $q ) { echo '찾을 말을 넣어주세요.'; exit; }

	$wide = 3;
	if ( isset( $_GET['n'] ) ) { $wide = max( 1, min( 12, (int) $_GET['n'] ) ); }
	$all = isset( $_GET['all'] );

	$slug = 'reading-book';
	if ( isset( $_GET['slug'] ) ) { $slug = sanitize_title( wp_unslash( $_GET['slug'] ) ); }
	$page = get_page_by_path( $slug );
	if ( ! $page ) { echo $slug . ' 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	echo "═══ 글 찾기 ═══\n";
	echo '쪽 ' . $id . ' (' . $slug . ') - ' . number_format( strlen( $c ) ) . "글자\n";
	echo '찾는 말 : ' . $q . "\n\n";

	$use = $q;
	$n   = substr_count( $c, $use );

	if ( 0 === $n ) {
		/* 통째로 없으면 뒤에서 한 글자씩 잘라가며 가장 긴 조각을 찾습니다 */
		$len = mb_strlen( $q, 'UTF-8' );
		$found = '';
		for ( $i = $len; $i >= 4; $i-- ) {
			$try = mb_substr( $q, 0, $i, 'UTF-8' );
			if ( substr_count( $c, $try ) > 0 ) { $found = $try; break; }
		}
		if ( '' === $found ) {
			/* 앞에서도 잘라봅니다 */
			for ( $i = 1; $i < $len - 3; $i++ ) {
				$try = mb_substr( $q, $i, $len - $i, 'UTF-8' );
				if ( substr_count( $c, $try ) > 0 ) { $found = $try; break; }
			}
		}
		if ( '' === $found ) {
			echo "이 말은 책 어디에도 없습니다.\n";
			echo "- 화면에서 보신 그대로 넣으셨는지 확인해 주세요.\n";
			echo "- 이름이나 날짜처럼 사람마다 달라지는 값은 빼고 넣어주세요.\n";
			echo "- 가운뎃점(-)이나 따옴표(「」)는 코드에 다르게 적혀 있을 수 있습니다.";
			exit;
		}
		echo "통째로는 없습니다. 코드에서 잘려 있어요.\n";
		echo '찾은 조각 : ' . $found . "\n";
		echo '이 조각이 ' . substr_count( $c, $found ) . "군데 있습니다.\n\n";
		$use = $found;
		$n   = substr_count( $c, $use );
	} else {
		echo $n . "군데 있습니다.\n\n";
	}

	$max = $all ? 8 : 3;
	$pos = -1;
	$k   = 0;
	while ( $k < $max ) {
		$pos = strpos( $c, $use, $pos + 1 );
		if ( false === $pos ) { break; }
		$k++;

		/* 앞뒤로 줄 단위로 넓혀서 보여줍니다 */
		$s = $pos;
		$e = $pos + strlen( $use );
		$back = $wide;
		while ( $back > 0 ) {
			$p = strrpos( substr( $c, 0, $s ), "\n" );
			if ( false === $p ) { $s = 0; break; }
			$s = $p;
			$back--;
		}
		$fwd = $wide;
		while ( $fwd > 0 ) {
			$p = strpos( $c, "\n", $e );
			if ( false === $p ) { $e = strlen( $c ); break; }
			$e = $p + 1;
			$fwd--;
		}
		$line = substr_count( substr( $c, 0, $pos ), "\n" ) + 1;
		echo "───── " . $k . "번째 - " . $line . "줄쯤 ─────\n";
		echo substr( $c, $s, $e - $s ) . "\n";
	}
	if ( $k < $n ) {
		echo "\n( 나머지 " . ( $n - $k ) . "군데는 안 보여드렸습니다. 전부 보시려면 뒤에 " . '&all=1' . " 을 붙이세요 )";
	}
	exit;
}, 1 );
