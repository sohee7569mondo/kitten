  /* 맨 위에 붙박인 것들의 아랫변 — 가장 낮은 것 */
  function headBottom(){
    var all = document.querySelectorAll('header, nav, div, section');
    var i, el, st, r, max = 0, pos;
    for(i = 0; i < all.length; i++){
      el = all[i];
      st = window.getComputedStyle(el);
      if(!st){ continue; }
      pos = String(st.position);
      if(pos !== 'fixed'){ if(pos !== 'sticky'){ continue; } }
      r = el.getBoundingClientRect();
      if(r.top > 4){ continue; }        /* 맨 위에 붙은 것만 */
      if(r.height < 20){ continue; }
      if(r.height > 220){ continue; }   /* 쪽 전체를 덮는 것은 머리글이 아닙니다 */
      if(r.bottom > max){ max = r.bottom; }
    }

  /* ★★ 오행 막대 다섯 줄을 한 줄로 — 2026-09-16
     소희 님 「밑에 토 금 수 화 이거 분량이 너무 많아 위에 6개 표에
              있는 박스 아래에 금0개 화0개 이런식으로 한줄로 쓰면」

         목 ────────────── 1자
         화 ────────────── 0자      ← 다섯 줄이 자리를 많이 먹습니다
         토 ────────────── 2자
         금 ────────────── 1자
         수 ────────────── 2자

              ↓

         목 1 · 화 0 · 토 2 · 금 1 · 수 2      ← 사주표 바로 아래 한 줄

     ★ 값은 화면에서 그대로 읽습니다. 다시 셈하지 않습니다 —
       셈을 두 군데서 하면 언젠가 어긋납니다.
     ★ 막대는 지우지 않고 숨깁니다. 조각을 끄면 돌아옵니다. */
  var OHS = ['목', '화', '토', '금', '수'];

  /* 「목 1자」 처럼 오행 하나와 글자 수만 든 줄을 찾습니다 */
  function ohRow(o){
    var all = document.querySelectorAll('div, li, p, tr, span, td'), i, el, t, num;
    for(i = 0; i < all.length; i++){
      el = all[i];
      if(el.getAttribute('data-ssoh') === '1'){ continue; }
      t = squash(el.textContent);
      if(t.length < 3){ continue; }
      if(t.length > 6){ continue; }          /* 다섯을 싼 상자는 거릅니다 */
      if(t.indexOf(o) !== 0){ continue; }
      if(t.charAt(t.length - 1) !== '자'){ continue; }
      num = t.slice(o.length, t.length - 1);
      if(!num){ continue; }
      if(num !== String(parseInt(num, 10))){ continue; }
      return { el: el, n: num };
    }
    return null;
  }

  /* 다섯을 다 품는 가장 가까운 상자 */
  function commonBox(els){
    var n = els[0], i, ok, lv = 0;
    while(n){
      ok = 1;
      for(i = 0; i < els.length; i++){
        if(!n.contains(els[i])){ ok = 0; }
      }
      if(ok){ if(n !== els[0]){ return n; } }
      n = n.parentElement;
      lv++;
      if(lv > 8){ break; }
      if(!n){ break; }
      if(n === document.body){ break; }
    }
    return null;
  }

  /* 사주표 — 「연주」와 「일주」가 둘 다 든 가장 안쪽 상자 */
  function pillarBox(){
    var all = document.querySelectorAll('table, div, section'), i, el, t, best = null;
    for(i = 0; i < all.length; i++){
      el = all[i];
      t = String(el.textContent === null ? '' : el.textContent);
      if(t.indexOf('연주') < 0){ continue; }
      if(t.indexOf('일주') < 0){ continue; }
      if(t.length > 400){ continue; }
      if(!best){ best = el; }
      else if(best.contains(el)){ best = el; }
    }
    return best;
  }

  var OHDONE = 0;

  function ohOneLine(){
    if(OHDONE){ return; }
    try{
      var rows = [], i, r;
      for(i = 0; i < OHS.length; i++){
        r = ohRow(OHS[i]);
        if(!r){ return; }                 /* 다섯이 다 있어야 손댑니다 */
        rows.push(r);
      }
      var els = [];
      for(i = 0; i < rows.length; i++){ els.push(rows[i].el); }
      var box = commonBox(els);
      if(!box){ return; }

      var p = document.createElement('p');
      p.className = 'stella-oh';
      var txt = '';
      for(i = 0; i < rows.length; i++){
        if(i){ txt += '   ·   '; }
        txt += OHS[i] + ' ' + rows[i].n;
      }
      p.appendChild(document.createTextNode(txt));

      var tb = pillarBox();
      if(tb){ if(tb.parentNode){ tb.parentNode.insertBefore(p, tb.nextSibling); } }
      if(!p.parentNode){ if(box.parentNode){ box.parentNode.insertBefore(p, box); } }
      if(!p.parentNode){ return; }

      for(i = 0; i < els.length; i++){ els[i].setAttribute('data-ssoh', '1'); }
      box.setAttribute('data-ssoh', '1');
      box.style.display = 'none';
      OHDONE = 1;
    }catch(e){}
  }

  /* ★ 「내 사주로 같은 것을 보기」 단추는 뺍니다
