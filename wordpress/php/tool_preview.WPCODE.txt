/* ════════════════════════════════════════════════════════════
   미리보기가 어디로 가나 — 보기만 하는 도구   tool_preview
   ★ 2026-09-15 · 소희 님께

   소희 님 : 「사이트의 모든 미리보기는 최신형으로 업데이트 해야해」

   맞습니다. 다만 무엇이 낡았는지부터 정확히 봐야 합니다.

   ── 두 가지 미리보기가 섞여 있습니다 ──
     옛 견본 쪽   /reading-love/ 처럼 8월에 한 번 뽑아 붙여둔 붙박이 글.
                  그 뒤에 책을 고친 것이 하나도 안 들어갑니다.
     살아 있는 책 /reading-book/?demo=door-love 처럼 **그 자리에서**
                  지금 엔진으로 그립니다. 다시는 안 낡습니다.

   patch_demolink 가 여섯 개를 살아 있는 책으로 돌려 두었습니다.
   이 도구는 **그 뒤에 새로 생긴 주제**까지 다 훑어, 아직 옛 쪽으로
   가는 링크가 남았는지 · 미리보기가 아예 없는 주제가 있는지 찍습니다.

   아무것도 안 바꿉니다.

   보기   https://stellasaju.com/?stella_preview=1

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_preview'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$OLD = array( 'reading-sample', 'reading-wealth', 'reading-love',
	              'reading-health', 'reading-fortune', 'reading-astro' );

	echo "═══ 미리보기가 어디로 가나 ═══\n\n";

	/* ── 1. 옛 견본 쪽은 언제 굳었나 ──────────────────── */
	echo "── 1. 옛 견본 쪽 (붙박이 글) ──\n";
	foreach ( $OLD as $slug ) {
		$p = get_page_by_path( $slug );
		if ( ! $p ) { echo sprintf( "  %-18s (쪽 없음)\n", $slug ); continue; }
		echo sprintf( "  %-18s 쪽 %-4d %s  마지막 손질 %s\n",
			$slug, $p->ID, mb_substr( $p->post_title, 0, 18 ),
			mb_substr( $p->post_modified, 0, 10 ) );
	}
	$bk = get_page_by_path( 'reading-book' );
	if ( $bk ) {
		echo "\n  책 자신 (reading-book)  마지막 손질 "
		   . mb_substr( $bk->post_modified, 0, 10 ) . "\n";
		echo "  ★ 그 뒤로도 조각이 계속 고칩니다. 붙박이 견본은 안 따라옵니다.\n";
	}
	echo "\n";

	/* ── 2. 어느 쪽이 옛 견본으로 가고 있나 ──────────── */
	echo "── 2. 아직 옛 견본으로 가는 링크 ──\n";
	$rows = $wpdb->get_results(
		"SELECT ID, post_title, post_name, post_content
		   FROM {$wpdb->posts}
		  WHERE post_type = 'page' AND post_status = 'publish'" );
	$bad = 0;
	foreach ( (array) $rows as $one ) {
		if ( in_array( $one->post_name, $OLD, true ) ) { continue; }
		$c = (string) $one->post_content;
		foreach ( $OLD as $slug ) {
			$n = substr_count( $c, '/' . $slug . '/' );
			if ( ! $n ) { continue; }
			echo sprintf( "  %-16s → /%s/  %d군데\n",
				$one->post_name, $slug, $n );
			$bad += $n;
		}
	}
	if ( ! $bad ) {
		echo "  없습니다. 다 살아 있는 책으로 갑니다.\n";
	} else {
		echo "  ★ " . $bad . "군데 남았습니다 — ?stella_demolink=apply 로 돌립니다.\n";
	}
	echo "\n";

	/* ── 3. 살아 있는 책으로 가는 링크 ───────────────── */
	echo "── 3. 살아 있는 책으로 가는 미리보기 ──\n";
	$seen = array();
	foreach ( (array) $rows as $one ) {
		$c = (string) $one->post_content;
		if ( preg_match_all( '#/reading-book/\?demo=([a-z0-9_-]+)#i', $c, $m ) ) {
			foreach ( $m[1] as $d ) {
				$k = $one->post_name . ' → ' . $d;
				if ( isset( $seen[ $k ] ) ) { $seen[ $k ]++; continue; }
				$seen[ $k ] = 1;
			}
		}
	}
	if ( ! $seen ) {
		echo "  하나도 없습니다.\n";
	} else {
		foreach ( $seen as $k => $n ) {
			echo '  ' . $k . '  ' . $n . "군데\n";
		}
	}
	echo "\n";

	/* ── 4. 새로 생긴 주제에 미리보기가 있나 ─────────── */
	echo "── 4. 새로 생긴 주제 ──\n";
	$NEW = array(
		'신년운세' => '2026 · 2027 두 권',
		'삼재'     => '2026-09-15 에 지음',
		'가족운'   => '원고를 새로 씀',
		'건강운'   => '두 층 원고',
		'체질'     => '',
		'궁합'     => '',
		'평생운'   => '',
		'전환기'   => '',
	);
	foreach ( $NEW as $t => $memo ) {
		$has = 0;
		foreach ( (array) $rows as $one ) {
			if ( false !== strpos( (string) $one->post_content, 'demo=' . $t ) ) { $has = 1; break; }
			if ( false !== strpos( (string) $one->post_content, rawurlencode( $t ) ) ) {
				if ( false !== strpos( (string) $one->post_content, 'demo=' ) ) { $has = 1; break; }
			}
		}
		echo sprintf( "  %-10s %s   %s\n", $t,
			$has ? '미리보기 있음' : '★ 미리보기 없음', $memo );
	}
	echo "\n";

	echo "── 읽는 법 ──\n";
	echo "  2번이 비어 있으면 옛 견본으로 가는 링크는 없습니다.\n";
	echo "  4번의 ★ 는 그 주제를 미리 볼 길이 아예 없다는 뜻입니다.\n";
	echo "  새 주제 미리보기를 만들려면 책 안의 데모 표에 사람을 더해야\n";
	echo "  합니다 — 벼리에게 이 글을 그대로 붙여넣어 주세요.\n";
	echo "\n═══ 끝 ═══\n";
	exit;
}, 1 );
