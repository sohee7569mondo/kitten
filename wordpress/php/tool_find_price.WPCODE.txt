/* ════════════════════════════════════════════════════════════
   990원이 적힌 자리를 전부 찾습니다 · 소희 님께

   왜
     KG이니시스가 1,000원 미만을 받지 않아 풀이 한 편 값을
     990원에서 1,000원으로 올립니다. 그런데 엔진의 숫자만 고치면
     화면에는 990원이라 적혀 있는데 실제로는 1,000원이 빠지는
     상태가 됩니다. 심사에서 바로 걸립니다.
     그래서 먼저 어디에 적혀 있는지 전부 세어봅니다.

   아무것도 고치지 않습니다. 찾아서 보여주기만 합니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet →
              Location 은 Run Everywhere →
              첫 줄 PHP 태그 아래에 붙여넣기 → 저장 → 활성

   ★ 저장 직후 먼저 확인
        https://stellasaju.com/wp-json/stella/v1/me
     {"ok":false,...} 면 정상, rest_no_route 면 즉시 꺼주세요.

   그 다음
        https://stellasaju.com/?stella_price=1
   ════════════════════════════════════════════════════════════ */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_price'] ) ) {
		return;
	}
	if ( ! is_user_logged_in() ) {
		wp_die( '로그인이 안 되어 있습니다. 관리자로 로그인한 창에서 열어주세요.', '스텔라', array( 'response' => 200 ) );
	}
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_die( '관리자만 볼 수 있습니다.', '스텔라', array( 'response' => 200 ) );
	}

	global $wpdb;

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><body style="font:14px/1.8 -apple-system,system-ui,sans-serif;background:#111;color:#eee;padding:24px;max-width:1100px">';
	echo '<h1 style="color:#ffd76a">값이 적힌 자리 찾기</h1>';

	/* 찾을 말들 */
	$words = array( '990원', '990 원', '24,750', '24750', '구슬 1개', '구슬 한 개' );

	/* 1. 페이지와 글 */
	echo '<h2 style="color:#9fd">1. 페이지 · 글</h2>';
	$rows = $wpdb->get_results(
		"SELECT ID, post_title, post_name, post_type, post_status, post_content
		   FROM {$wpdb->posts}
		  WHERE post_status IN ('publish','draft','private')
		    AND post_type IN ('page','post','wp_template','wp_template_part','wp_navigation','wp_block')
		    AND ( post_content LIKE '%990%' OR post_content LIKE '%24,750%' OR post_content LIKE '%24750%' )
		  ORDER BY post_type, ID"
	);
	if ( ! $rows ) {
		echo '<p>없습니다.</p>';
	} else {
		echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
		echo '<tr style="color:#ffd76a"><td>ID</td><td>제목</td><td>주소</td><td>종류</td><td>몇 번</td></tr>';
		foreach ( $rows as $r ) {
			$hits = array();
			foreach ( $words as $w ) {
				$n = substr_count( $r->post_content, $w );
				if ( $n ) {
					$hits[] = esc_html( $w ) . ' ' . $n;
				}
			}
			printf(
				'<tr style="border-top:1px solid #333"><td><b>%d</b></td><td>%s</td><td>%s</td><td style="color:#888">%s</td><td style="color:#9fd">%s</td></tr>',
				$r->ID,
				esc_html( $r->post_title ),
				$r->post_name ? '/' . esc_html( $r->post_name ) . '/' : '',
				esc_html( $r->post_type ),
				implode( ' · ', $hits )
			);
		}
		echo '</table>';

		/* 각 자리의 앞뒤를 보여줍니다 */
		echo '<h2 style="color:#9fd">2. 적힌 모습 그대로</h2>';
		foreach ( $rows as $r ) {
			$shown = 0;
			foreach ( $words as $w ) {
				$pos = 0;
				while ( ( $pos = strpos( $r->post_content, $w, $pos ) ) !== false ) {
					$ctx = substr( $r->post_content, max( 0, $pos - 90 ), 220 );
					printf(
						'<p style="margin:14px 0 4px;color:#ffd76a">%d · %s — <b>%s</b></p>'
						. '<pre style="white-space:pre-wrap;background:#000;padding:10px;border-left:3px solid #ffd76a;font:12.5px/1.6 monospace">%s</pre>',
						$r->ID,
						esc_html( $r->post_title ),
						esc_html( $w ),
						esc_html( $ctx )
					);
					$pos += strlen( $w );
					$shown++;
					if ( $shown >= 6 ) {
						break 2;
					}
				}
			}
		}
	}

	/* 3. 스니펫 */
	echo '<h2 style="color:#9fd">3. 스니펫</h2>';
	$snips = $wpdb->get_results(
		"SELECT ID, post_title, post_status, post_content FROM {$wpdb->posts}
		  WHERE post_type = 'wpcode' AND ( post_content LIKE '%990%' OR post_content LIKE '%24750%' )
		  ORDER BY post_title"
	);
	if ( ! $snips ) {
		echo '<p>없습니다.</p>';
	} else {
		echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
		echo '<tr style="color:#ffd76a"><td>ID</td><td>이름</td><td>상태</td><td>줄</td></tr>';
		foreach ( $snips as $s ) {
			$lines = preg_split( "/\r\n|\n|\r/", $s->post_content );
			$found = array();
			foreach ( $lines as $i => $line ) {
				if ( false !== strpos( $line, '990' ) || false !== strpos( $line, '24750' ) ) {
					$found[] = ( $i + 1 ) . '행 : ' . trim( $line );
				}
			}
			printf(
				'<tr style="border-top:1px solid #333"><td>%d</td><td>%s</td><td style="color:%s">%s</td><td><pre style="white-space:pre-wrap;margin:0;font:12.5px/1.6 monospace">%s</pre></td></tr>',
				$s->ID,
				esc_html( $s->post_title ),
				'publish' === $s->post_status ? '#7bff9b' : '#888',
				'publish' === $s->post_status ? '켜짐' : '꺼짐',
				esc_html( implode( "\n", array_slice( $found, 0, 8 ) ) )
			);
		}
		echo '</table>';
	}

	/* 4. 지금 살아 있는 값 */
	echo '<h2 style="color:#9fd">4. 지금 엔진이 쓰는 값</h2><table cellpadding="8" style="border-collapse:collapse">';
	foreach ( array( 'STELLA_ORB_KRW', 'STELLA_PRICE_BASE', 'STELLA_PRICE_FULL' ) as $k ) {
		printf(
			'<tr style="border-top:1px solid #333"><td>%s</td><td><b style="color:#7bff9b">%s</b></td></tr>',
			esc_html( $k ),
			defined( $k ) ? esc_html( (string) constant( $k ) ) : '<span style="color:#ff7b7b">없음</span>'
		);
	}
	if ( defined( 'STELLA_ORB_KRW' ) && defined( 'STELLA_PRICE_BASE' ) ) {
		printf(
			'<tr style="border-top:1px solid #333"><td>풀이 한 편</td><td><b>%s원</b></td></tr>',
			number_format( (int) STELLA_PRICE_BASE * (int) STELLA_ORB_KRW )
		);
	}
	if ( defined( 'STELLA_ORB_KRW' ) && defined( 'STELLA_PRICE_FULL' ) ) {
		printf(
			'<tr style="border-top:1px solid #333"><td>나의 인생길잡이</td><td><b>%s원</b></td></tr>',
			number_format( (int) STELLA_PRICE_FULL * (int) STELLA_ORB_KRW )
		);
	}
	echo '</table>';

	echo '<p style="margin-top:20px;color:#888">아무것도 바꾸지 않았습니다.</p></body>';
	exit;
} );
