add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_peek'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$arg = sanitize_text_field( wp_unslash( $_GET['stella_peek'] ) );
	$from = 12470; $to = 12620;
	if ( preg_match( '/^(\d+)\s*-\s*(\d+)$/', $arg, $m ) ) {
		$from = (int) $m[1];
		$to   = (int) $m[2];
	}
	if ( $to < $from ) { $to = $from; }
	if ( $to - $from > 400 ) { $to = $from + 400; }

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", 160 ) );

	$lines = explode( "\n", $c );
	$total = count( $lines );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<!DOCTYPE html><html lang="ko" translate="no">';
	echo '<head><meta charset="utf-8">';
	echo '<meta name="google" content="notranslate">';
	echo '<meta name="viewport" content="width=device-width,initial-scale=1">';
	echo '<title>PEEK</title>';
	echo '<style>body{background:#12101B;color:#F2ECE0;margin:0;padding:16px;'
	   . 'font:12px/1.65 ui-monospace,Menlo,Consolas,monospace}'
	   . 'h1{color:#D4AF6A;font-size:14px;margin:0 0 12px}'
	   . '.n{color:#6F6A80;user-select:none}'
	   . 'pre{white-space:pre-wrap;word-break:break-all;margin:0}'
	   . '</style></head><body class="notranslate" translate="no">';

	echo '<h1>PEEK &middot; 쪽 160 &middot; 줄 ' . (int) $from . ' ~ ' . (int) $to
	   . ' (모두 ' . number_format( $total ) . '줄)</h1>';

	echo '<pre>';
	$i = $from;
	while ( $i <= $to && $i <= $total ) {
		$txt = isset( $lines[ $i - 1 ] ) ? $lines[ $i - 1 ] : '';
		echo '<span class="n">' . sprintf( '%6d', $i ) . '</span>  '
		   . esc_html( $txt ) . "\n";
		$i++;
	}
	echo '</pre>';

	echo '</body></html>';
	exit;
} );
