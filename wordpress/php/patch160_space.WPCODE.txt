/* ══════════════════════════════════════════════════════════
   읽기 좋게 — 여백 · 실선 정리
   ★ 2026-09-12 · 소희 님께

   ── 소희 님이 짚으신 것 ───────────────────────────────
   1. 「여백이 전체적으로 너무 없어… 단락과 단락도」
   2. 「실선은 없애자. 어떨 땐 단락선과 너무 붙어서 보기 안 좋아」
   3. 「장 사이에만 실선 주는데 아래 위로 여백을 좀 주면…」

   ── 재보니 이랬습니다 ─────────────────────────────────
       문단 사이   14px    ← 한 줄이 32px 인데 14px 이면 안 나뉩니다
       제목 아래    8px
       소제목 아래  2px    ← 소제목과 본문이 붙어 있습니다
       설명 줄 아래 4px

   ── 책의 층이 셋입니다 ────────────────────────────────
       h2  제목      쪽 전체의 주제
       h3  소제목    **주제가 바뀌는 자리**
       p   문단      같은 이야기 안의 숨
     층이 셋이면 여백도 셋이어야 합니다.

   ── 이렇게 바꿉니다 ───────────────────────────────────
       문단 사이   30px   ← 같은 이야기 안의 숨입니다. 한 줄이 32px 입니다.
       제목 아래   12px   ← 설명 줄과는 붙여 둡니다. 한 덩어리니까요.
       설명 줄 아래 30px   ← 여기가 벌어져야 합니다. 다음은 본문이거든요.
       소제목 아래 12px  (위는 48px)

   ── 실선 ──────────────────────────────────────────────
   쪽과 쪽 사이 실선을 없애고, **장이 바뀔 때만** 실선과 넉넉한 여백을
   둡니다. 장이 바뀌는 곳은 eyebrow(「맺음말」 「더 깊이 · 둘」 …)가
   달라지는 지점입니다. 장 속표지가 있는 쪽은 속표지가 알려주므로
   실선을 겹쳐 넣지 않습니다.

   ── 지키는 것 ─────────────────────────────────────────
   · 글은 한 글자도 건드리지 않습니다. 보이는 모양만 바꿉니다.
   · 두 글자짜리 논리기호를 쓰지 않고 if 를 겹쳐 썼습니다.
   · 두 번 눌러도 한 번만 들어갑니다.

   해보기   https://stellasaju.com/?stella_space=preview
   붙이기   https://stellasaju.com/?stella_space=apply
   되돌리기 https://stellasaju.com/?stella_space=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_space'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_space'] ) );
	$META = '_stella_space_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;
	}

	$c    = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	$MARK = 'BUILD: SPACE-1';

	$BLOCK = <<<'STELLAFLOW'
<!-- wp:html -->
<style>
/* ═══ 읽기 좋게   BUILD: SPACE-1   2026-09-12 ═══════════════════
   여백을 넓히고, 쪽 사이 실선을 없애고, 장이 바뀔 때만 실선을 둡니다.
   ═══════════════════════════════════════════════════════════ */

/* ── 하나 · 여백 ─────────────────────────────────────── */
#ssb p{ margin:30px 0 0 !important; }
#ssb h2{ margin:0 0 12px !important; }       /* 제목 → 설명 줄은 붙여 둡니다 */
#ssb .dek{ margin:0 0 30px !important; }     /* 설명 줄 → 본문은 넉넉히 */
#ssb h3{ margin:48px 0 12px !important; }
#ssb h3:first-of-type{ margin-top:34px !important; }
#ssb .voice{ margin-top:38px !important; padding:28px 30px !important; }
#ssb .voice p + p{ margin-top:18px !important; }
#ssb .alert{ margin-top:32px !important; }

/* ── 둘 · 쪽과 쪽 사이 실선을 없앱니다 ────────────────── */
#ssb .page + .page{ border-top:0 !important; }
#ssb .page{ padding-top:30px !important; padding-bottom:14px !important; }

/* ── 셋 · 장이 바뀌는 쪽에만 실선과 넉넉한 여백 ───────── */
#ssb .page[data-chap="1"]{
  border-top:1px solid var(--rule) !important;
  margin-top:80px !important;
  padding-top:80px !important;
}
/* 장 속표지가 있는 쪽은 속표지가 알려주므로 실선을 겹치지 않습니다 */
#ssb .page.divider{ border-top:0 !important; margin-top:80px !important; padding-top:56px !important; }
#ssb .page.divider + .page{ margin-top:0 !important; padding-top:30px !important; border-top:0 !important; }
#ssb .page[data-part="cover"]{ border-top:0 !important; margin-top:0 !important; }
/* 덤 두 쪽은 청록 실선으로 */
#ssb .page[data-chap="1"][data-part="tail"]{ border-top-color:rgba(31,122,110,.34) !important; }

@media(max-width:640px){
  #ssb p{ margin:26px 0 0 !important; }
  #ssb h3{ margin:38px 0 9px !important; }
  #ssb .page{ padding-top:24px !important; }
  #ssb .page[data-chap="1"]{ margin-top:58px !important; padding-top:58px !important; }
  #ssb .page.divider{ margin-top:58px !important; padding-top:42px !important; }
}
@media print{ #ssFlow{ display:none !important; } }
</style>
<script>
/* ═══════════════════════════════════════════════════════════════
   STELLA SAJU — 읽기 좋게      BUILD: SPACE-1      2026-09-12
   글은 건드리지 않습니다. 장이 바뀌는 쪽에 표시만 붙입니다.
   두 글자짜리 논리기호를 쓰지 않고 if 를 겹쳐 썼습니다.
   ═══════════════════════════════════════════════════════════════ */
(function(){
  'use strict';

  function ebOf(el){
    var e=el.getElementsByClassName('eyebrow')[0];
    if(!e){ return ''; }
    var t=e.textContent;
    if(!t){ return ''; }
    return t.replace(/^\s+/,'').replace(/\s+$/,'');
  }

  function markChapters(){
    var box=document.getElementById('ssb');
    if(!box){ return 0; }
    var pages=box.getElementsByClassName('page');
    if(!pages.length){ return 0; }
    var last='', i, t, first=1, cls;
    for(i=0;i<pages.length;i++){
      pages[i].removeAttribute('data-chap');
      cls=pages[i].className;
      if(pages[i].getAttribute('data-part')==='cover'){ last=''; first=0; continue; }
      /* 장 속표지 자체는 표시하지 않습니다. 속표지가 이미 알려줍니다. */
      if(cls.indexOf('divider')>=0){ last=''; first=0; continue; }
      t=ebOf(pages[i]);
      if(!t){ continue; }
      if(t!==last){
        if(first===0){ pages[i].setAttribute('data-chap','1'); }
        first=0;
        last=t;
      }
    }
    return pages.length;
  }

  var tries=0;
  function tick(){
    tries+=1;
    markChapters();
    if(tries<40){ setTimeout(tick, 500); }
  }
  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', function(){ setTimeout(tick, 500); });
  } else { setTimeout(tick, 500); }
})();
</script>
<!-- /wp:html -->
STELLAFLOW;

	if ( 'preview' === $mode ) {
		$has = ( strpos( $c, $MARK ) !== false );
		echo "═══ 이어 읽기 (FLOW-1) ═══\n";
		echo '쪽 160 · ' . number_format( mb_strlen( $c ) ) . "글자\n\n";
		if ( $has ) { echo "이미 붙어 있습니다.\n되돌리려면 ?stella_space=undo\n"; exit; }
		echo '넣을 조각 ' . number_format( mb_strlen( $BLOCK ) ) . "글자\n";
		echo "앰퍼샌드 " . substr_count( $BLOCK, '&' ) . "개\n\n";
		echo "하는 일\n";
		echo "  · 문단 사이 14px → 30px  (한 줄이 32px 인데 너무 좁았습니다)\n";
		echo "  · 설명 줄 아래 4px → 30px  (설명 줄과 본문이 붙어 있었습니다)\n";
		echo "  · 소제목 아래 2px → 12px · 소제목 위 38px → 48px\n";
		echo "  · 쪽과 쪽 사이 실선을 없앱니다\n";
		echo "  · 장이 바뀔 때만 실선과 여백을 둡니다\n\n";
		echo "글은 한 글자도 건드리지 않습니다. 보이는 모양만 바뀝니다.\n\n";
		echo "붙이려면 ?stella_space=apply\n";
		exit;
	}

	if ( 'apply' !== $mode ) { echo '?stella_space=preview 부터 보세요.'; exit; }
	if ( strpos( $c, $MARK ) !== false ) { echo '이미 붙어 있습니다.'; exit; }

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		update_post_meta( $id, $META, $c );
	}

	$new = $c . "\n\n" . $BLOCK;
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "붙였습니다.\n";
	echo '쪽 160 · ' . number_format( mb_strlen( $new ) ) . "글자\n\n";
	echo "책을 열어 쭉 내려보세요. 위쪽에 막대가 뜨고,\n";
	echo "장이 바뀔 때만 실선이 나옵니다.\n되돌리려면 ?stella_space=undo\n";
	exit;
}, 1 );
