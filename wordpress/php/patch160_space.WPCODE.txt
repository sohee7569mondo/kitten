/* ══════════════════════════════════════════════════════════
   읽기 좋게 — 여백
   ★ 2026-09-12 · 소희 님께

   ── 소희 님이 짚으신 것 ───────────────────────────────
   1. 「여백이 전체적으로 너무 없어… 단락과 단락도」
   2. 「실선은 없애자. 어떨 땐 단락선과 너무 붙어서 보기 안 좋아」
   3. 목차를 보시고 —
      「장은 표지 들어가니까 괜찮고, 그 밑에 소제목끼리는
       80 정도는 줘야 될 거 같은데」

   ── 재보니 이랬습니다 ─────────────────────────────────
       문단 사이     14px    ← 한 줄이 32px 인데 14px 이면 안 나뉩니다
       설명 줄 아래   4px    ← 설명 줄과 본문이 붙어 있습니다
       소제목 아래    2px
       쪽과 쪽 사이  실선 한 줄만

   ── 책의 층이 넷입니다 ────────────────────────────────
       p        문단      같은 이야기 안의 숨
       h3       소제목    주제가 바뀌는 자리
       .page    쪽        한 꼭지가 끝나는 자리
       divider  장 속표지  장이 바뀌는 자리   ← 이건 이미 있습니다
     층이 넷이면 여백도 넷이어야 합니다.

   ── 이렇게 바꿉니다 ───────────────────────────────────
       문단 사이     20px   짧은 문단이 많아 30px 은 성기게 보였습니다
       제목 아래     12px   설명 줄과는 붙여 둡니다. 한 덩어리니까요.
       설명 줄 아래   26px   여기가 벌어져야 합니다. 다음은 본문이거든요.
       소제목 위     48px   주제가 바뀌는 자리입니다.
       쪽과 쪽 사이  200px   ← 소희 님이 말씀하신 그것. 쪽이 끝난 것이 분명히 보입니다.
       장            속표지가 알려줍니다. 실선을 겹치지 않습니다.

   ★ 쪽 사이 실선을 없앴습니다. 장은 속표지가 알려주므로
     실선 없이 여백만으로 나눕니다. 그래서 이 조각에는
     자바스크립트가 없습니다. 스타일만 바꿉니다.

   ── 지키는 것 ─────────────────────────────────────────
   · 글은 한 글자도 건드리지 않습니다. 보이는 모양만 바꿉니다.
   · 두 번 눌러도 한 번만 들어갑니다.

   해보기   https://stellasaju.com/?stella_space4=preview
   붙이기   https://stellasaju.com/?stella_space4=apply
   되돌리기 https://stellasaju.com/?stella_space4=undo

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_space4'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_space4'] ) );
	$META = '_stella_space4_bak';

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $id, $META, true );
		if ( '' === $old ) { echo '되돌릴 것이 없습니다.'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		delete_post_meta( $id, $META );
		echo '되돌렸습니다.';
		exit;
	}

	$c    = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );
	$MARK = 'BUILD: SPACE-4';

	/* 먼저 판(SPACE-1)이 붙어 있으면 apply 가 스스로 떼냅니다 */

	$BLOCK = <<<'STELLAFLOW'
<!-- wp:html -->
<style>
/* ═══ 읽기 좋게   BUILD: SPACE-4   2026-09-12 ═══════════════════
   여백만 바꿉니다. 자바스크립트가 없습니다.
   ═══════════════════════════════════════════════════════════ */

/* ── 하나 · 글 안의 여백 ─────────────────────────────── */
#ssb p{ margin:20px 0 0 !important; }        /* 같은 이야기 안의 숨 */
#ssb h2{ margin:0 0 12px !important; }       /* 제목 → 설명 줄은 붙여 둡니다 */
#ssb .dek{ margin:0 0 26px !important; }     /* 설명 줄 → 본문은 넉넉히 */
#ssb h3{ margin:48px 0 12px !important; }    /* 주제가 바뀌는 자리 */
#ssb h3:first-of-type{ margin-top:34px !important; }
#ssb .voice{ margin-top:42px !important; padding:28px 30px !important; }
#ssb .voice p + p{ margin-top:18px !important; }
#ssb .alert{ margin-top:36px !important; }

/* ── 하나의 반 · 아르카나 말상자 ─────────────────────
   소희 님이 캡처로 보여주신 자리입니다. 여섯 줄이 서로
   너무 벌어져 있어서 한 덩어리로 안 읽혔습니다.
   줄 사이를 좁히고 문단 사이만 벌립니다. */
#ssb .arcv-say{ line-height:1.85 !important; }
#ssb .arcv-say p{ margin:0 0 15px !important; }
#ssb .arcv-say p:last-child{ margin-bottom:0 !important; }

/* ── 하나의 셋 · 카드 세 장의 자리 ───────────────────
   「I · 지금 서 있는 자리」 위가 아래와 비슷해서 층이
   안 보였습니다. 자리 이름 위를 크게 벌리고, 카드 이름
   줄과 그 아래 풀이는 한 덩어리로 붙입니다. */
#ssb h3 + p{ margin-top:12px !important; }

/* ── 둘 · 쪽과 쪽 사이 ───────────────────────────────
   실선을 없애고 여백으로만 나눕니다.
   장은 속표지가 알려주므로 실선을 겹치지 않습니다. */
#ssb .page + .page{ border-top:0 !important; }
#ssb .page{ padding-top:200px !important; padding-bottom:10px !important; }
#ssb .page[data-part="cover"]{ padding-top:6px !important; }
#ssb .page.divider{ padding-top:200px !important; padding-bottom:70px !important; }
#ssb .page.divider + .page{ padding-top:56px !important; }

/* ── 둘의 반 · 쪽 사이 여백을 흐리게 ──────────────────
   소희 님 — 「여백을 흐리게도 해주고」.
   실선을 없애니 여백이 그냥 빈 종이라 쪽이 끝난 줄을 몰랐습니다.
   그 200px 자리에만 옅은 그늘을 깝니다. 위가 제일 짙고
   쪽이 시작되는 곳으로 갈수록 맑아집니다.
   선이 아니라 그늘이라 단락선과 부딪히지 않습니다.

   ★ ::after 를 씁니다. ::before 는 쪽 왼쪽의 금색 세로선이
     이미 쓰고 있어서, 거기에 그늘을 얹으면 세로선이 사라집니다. */
#ssb .page{ position:relative !important; }
#ssb .page + .page::after{
  content:'' !important;
  position:absolute !important;
  left:0 !important; right:0 !important; top:0 !important;
  height:200px !important;
  border-radius:inherit !important;
  background:linear-gradient(to bottom,
    rgba(60,44,18,.070) 0%,
    rgba(60,44,18,.030) 45%,
    rgba(60,44,18,0)   100%) !important;
  pointer-events:none !important;
}
/* 장 속표지 다음 쪽은 56px 만 띄우므로 그늘도 그만큼만 */
#ssb .page.divider + .page::after{ height:56px !important; }
/* 표지 다음 쪽에는 안 깝니다 */
#ssb .page[data-part="cover"] + .page::after{ display:none !important; }

/* ── 둘의 셋 · 좌우 여백 ─────────────────────────────
   소희 님 — 「좌우 여백이 좀 있어야 하지 않을까? 답답해 보여」.
   맞습니다. 글이 쪽 가장자리에 바짝 붙어 있었습니다.
   쪽 안쪽 좌우를 62px 에서 78px 로 넓히고, 덤 상자는
   쪽 안에서 한 번 더 들여 씁니다. 두 겹으로 숨을 줍니다.
   휴대폰에서는 화면이 좁으니 22px 로 둡니다. */
#ssb .page{ padding-left:78px !important; padding-right:78px !important; }
#ssb .page[data-part="cover"]{ padding-left:56px !important; padding-right:56px !important; }
#ssb .tailbox{ padding:30px 34px 28px !important; }
@media(max-width:640px){
  #ssb .page{ padding-left:22px !important; padding-right:22px !important; }
  #ssb .page[data-part="cover"]{ padding-left:20px !important; padding-right:20px !important; }
  #ssb .tailbox{ padding:22px 20px !important; }
}

/* ── 셋 · 「세 가지만 기억하세요」의 넷째 덩어리 ────────
   「이 셋을 다 기억하지 않으셔도 됩니다」에 번호 4가 붙어 있었습니다.
   셋을 기억하라 해놓고 넷이 되어 있으니 이상합니다.
   넷째는 번호를 빼고 육성 상자 모양으로 바꿉니다. 가디언이 하는 말이니까요. */
#ssb .keep .k:nth-of-type(4){
  counter-increment:none !important;
  padding:24px 26px !important;
  margin-top:30px !important;
  background:rgba(212,175,106,.075) !important;
  border-left:2px solid var(--gold) !important;
  border-radius:0 !important;
}
#ssb .keep .k:nth-of-type(4)::before{ display:none !important; }
#ssb .keep .k:nth-of-type(4) .kt{ font-size:1.12rem !important; }

@media(max-width:640px){
  #ssb p{ margin:18px 0 0 !important; }
  #ssb h3{ margin:40px 0 11px !important; }
  #ssb .dek{ margin:0 0 22px !important; }
  #ssb .page{ padding-top:140px !important; }
  #ssb .page.divider{ padding-top:140px !important; padding-bottom:52px !important; }
  #ssb .page.divider + .page{ padding-top:44px !important; }
}
</style>
<!-- /wp:html -->
STELLAFLOW;

	if ( 'preview' === $mode ) {
		$has = ( strpos( $c, $MARK ) !== false );
		echo "═══ 이어 읽기 (FLOW-1) ═══\n";
		echo '쪽 160 · ' . number_format( mb_strlen( $c ) ) . "글자\n\n";
		if ( $has ) { echo "이미 붙어 있습니다.\n되돌리려면 ?stella_space4=undo\n"; exit; }
		echo '넣을 조각 ' . number_format( mb_strlen( $BLOCK ) ) . "글자\n";
		echo "앰퍼샌드 " . substr_count( $BLOCK, '&' ) . "개\n\n";
		echo "하는 일 — 스타일만 바꿉니다. 자바스크립트가 없습니다.\n";
		echo "  · 문단 사이 14px → 20px\n";
		echo "  · 설명 줄 아래 4px → 30px  (설명 줄과 본문이 붙어 있었습니다)\n";
		echo "  · 소제목 아래 2px → 12px · 소제목 위 38px → 48px\n";
		echo "  · 쪽과 쪽 사이 실선을 없애고 여백 200px 으로 나눕니다\n";
		echo "  · 「이 셋을 다 기억하지 않으셔도 됩니다」의 번호 4를 없애고\n";
		echo "    가디언이 하는 말처럼 육성 상자 모양으로 바꿉니다\n";
		echo "  · 장은 속표지가 알려주므로 실선을 겹치지 않습니다\n\n";
		echo "글은 한 글자도 건드리지 않습니다. 보이는 모양만 바뀝니다.\n\n";
		echo "붙이려면 ?stella_space4=apply\n";
		exit;
	}

	if ( 'apply' !== $mode ) { echo '?stella_space4=preview 부터 보세요.'; exit; }
	if ( strpos( $c, $MARK ) !== false ) { echo '이미 붙어 있습니다.'; exit; }

	if ( '' === (string) get_post_meta( $id, $META, true ) ) {
		/* ★ 2026-09-12 · wp_slash 를 반드시 거칩니다.
		   update_post_meta 는 담기는 값에서 역슬래시를 한 겹 벗겨냅니다.
		   책 본문의 /\s+/ · '<\/p>' · \' 가 그때 전부 사라졌고,
		   되돌리기가 그 판을 덮어서 책이 안 보이게 됐습니다. */
		update_post_meta( $id, $META, wp_slash( $c ) );
	}

	/* 먼저 판을 떼냅니다. 정규식을 쓰지 않고 자리만 찾아 자릅니다. */
	$took = 0;
	while ( true ) {
		$at = strpos( $c, 'BUILD: SPACE-' );
		if ( false === $at ) { break; }
		$from = strrpos( substr( $c, 0, $at ), '<!-- wp:html -->' );
		if ( false === $from ) { break; }
		$to = strpos( $c, '<!-- /wp:html -->', $at );
		if ( false === $to ) { break; }
		$to = $to + strlen( '<!-- /wp:html -->' );
		$c = substr( $c, 0, $from ) . substr( $c, $to );
		$took += 1;
	}
	$c = rtrim( $c ) . "\n";
	if ( $took > 0 ) { echo '먼저 판 ' . $took . "개를 떼냈습니다.\n"; }

	$new = $c . "\n\n" . $BLOCK;
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "붙였습니다.\n";
	echo '쪽 160 · ' . number_format( mb_strlen( $new ) ) . "글자\n\n";
	echo "책을 열어 쭉 내려보세요. 위쪽에 막대가 뜨고,\n";
	echo "장이 바뀔 때만 실선이 나옵니다.\n되돌리려면 ?stella_space4=undo\n";
	exit;
}, 1 );
