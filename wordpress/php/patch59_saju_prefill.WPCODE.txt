/* ════════════════════════════════════════════════════════════
   무료 사주 쪽이 홈에서 넣고 오신 값을 받게 합니다  (페이지 59)

   소희 님 —
     「사주보기가 나을까 회원가입이 나을까? 사주보기 눌렀을때
       어디로 갈지 모르겟네」

   홈 첫 화면 폼의 단추는 「무료로 내 사주 펴보기」입니다.
   누르면 무료 사주 쪽(/saju/)으로 가는데, 지금은 값이 안 넘어가서
   손님이 생년월일을 두 번 적게 됩니다. 그걸 막는 패치입니다.

   홈은 이렇게 보냅니다 —
     /saju/?b=1993-4-6-20-10-s
      해-달-날-시-분-달력    달력은 s 양력 · l 음력 · ll 윤달
      태어난 시간을 모르시면 시와 분 자리가 x 입니다.

   물음표 한 칸에 값을 몰아 담은 이유는 하나입니다. 주소에서 값을
   나눌 때 쓰는 앰퍼샌드를 스크립트 안에 쓰면 워드프레스가
   실체참조로 바꿔서 스크립트가 통째로 멈추기 때문입니다.

   ── 하는 일 ────────────────────────────────────
   ① 주소에 b= 가 있으면 칸을 채웁니다.
   ② 「나의 사주 보기」를 대신 한 번 눌러 결과까지 펴 둡니다.
   ③ b= 가 없으면 아무 일도 하지 않습니다. 그냥 오신 분에게는
      지금과 똑같습니다.

   이 쪽은 시를 시진(두 시간짜리) 열둘로 받습니다. 그래서 분은
   여기서 쓰지 않고 흘려보냅니다. 분까지 보는 것은 문 쪽입니다.

   ── 한 군데도 못 찾으면 아무것도 안 바꿉니다 ─────────
   붙일 자리를 못 찾으면 그냥 멈춥니다. 반쯤 바뀌는 일은 없습니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로 바꾸세요.

   ① 먼저 보기   https://stellasaju.com/?stella_pre=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_pre=apply
   ③ 되돌리기    https://stellasaju.com/?stella_pre=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_pre'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$META = '_stella_pre_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_pre'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$page = get_page_by_path( 'saju' );
	if ( ! $page ) {
		echo '<h1 class="no">/saju/ 쪽을 못 찾았습니다.</h1>';
		exit;
	}
	$pid = $page->ID;

	if ( 'undo' === $mode ) {
		echo '<h1>되돌리기</h1>';
		$old = get_post_meta( $pid, $META, true );
		if ( ! $old ) {
			echo '<p class="no">되돌릴 것이 없습니다.</p>';
			exit;
		}
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $pid ) );
		clean_post_cache( $pid );
		delete_post_meta( $pid, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		exit;
	}

	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( ! $row ) {
		echo '<p class="no">내용을 못 읽었습니다.</p>';
		exit;
	}
	$c = $row->post_content;

	$find = "\n})();\n</script>";

	$add = <<<'JS'


  /* ── 2026-09-08 · 홈에서 넣고 오신 값 받기 ─────────────────
     홈 첫 화면 폼이 이렇게 보냅니다 —
       /saju/?b=1993-4-6-20-10-s
       해-달-날-시-분-달력    달력은 s 양력 · l 음력 · ll 윤달
       태어난 시간을 모르시면 시와 분 자리가 x 입니다.

     값이 오면 칸을 채우고 「나의 사주 보기」를 대신 한 번 눌러
     결과까지 펴 둡니다. 손님은 홈에서 한 번만 적으면 됩니다.

     이 쪽은 시를 시진(두 시간짜리) 열둘로 받습니다. 그래서 분은
     여기서 쓰지 않고 흘려보냅니다. 분까지 보는 것은 문 쪽입니다.

     ★ 스크립트 안에 앰퍼샌드를 쓰지 않습니다. 주소를 나눌 때만
       필요해서 글자 번호(38)로 만들어 씁니다. */
  try{
    var qs = location.search;
    var at = qs.indexOf('b=');
    if(at >= 0){
      var raw = qs.slice(at + 2);
      var amp = raw.indexOf(String.fromCharCode(38));
      if(amp >= 0){ raw = raw.slice(0, amp); }
      var pp = raw.split('-');
      if(pp.length >= 6){
        var py = parseInt(pp[0], 10);
        var pm = parseInt(pp[1], 10);
        var pd = parseInt(pp[2], 10);
        var okPre = true;
        if(!py){ okPre = false; }
        if(!pm){ okPre = false; }
        if(!pd){ okPre = false; }
        if(okPre){
          $('sYear').value = py;
          monthSel.value = String(pm);
          daySel.value = String(pd);

          if(pp[5].charAt(0) === 'l'){
            $('calLunar').click();
            if(pp[5] === 'll'){
              if(leapWrap.className.indexOf('hide') < 0){ leapBox.checked = true; }
            }
          }

          if(pp[3] === 'x'){
            unknown.checked = true;
            hourSel.disabled = true;
            hourSel.value = '';
          } else {
            var ph = parseInt(pp[3], 10);
            if(!isNaN(ph)){
              hourSel.value = String(Math.floor(((ph + 1) % 24) / 2));
            }
          }

          $('sajuBtn').click();
        }
      }
    }
  }catch(ePre){}
JS;

	$n = substr_count( $c, $find );

	echo '<h2>붙일 자리</h2>';
	if ( 1 !== $n ) {
		echo '<p class="no">자리를 ' . intval( $n ) . '군데 찾았습니다. 딱 한 군데여야 합니다. '
		   . '아무것도 바꾸지 않았습니다.</p>';
		exit;
	}
	if ( false !== strpos( $c, '홈에서 넣고 오신 값 받기' ) ) {
		echo '<p class="no">이미 붙어 있습니다. 아무것도 바꾸지 않았습니다.</p>';
		exit;
	}
	echo '<p class="ok">한 군데 찾았습니다 — 스크립트 맨 끝에 '
	   . number_format( strlen( $add ) ) . ' 바이트를 붙입니다.</p>';

	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_pre=apply</code> 를 여세요.</p>';
		exit;
	}

	$new = str_replace( $find, $add . $find, $c );

	if ( ! get_post_meta( $pid, $META, true ) ) {
		update_post_meta( $pid, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );

	echo '<p class="ok">' . number_format( strlen( $c ) ) . ' → '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p>이렇게 열어서 확인해 보세요 — '
	   . '<a href="/saju/?b=1993-4-6-20-10-s">/saju/?b=1993-4-6-20-10-s</a></p>';
	echo '<p class="ok">되돌리시려면 <code>/?stella_pre=undo</code> 를 여세요.</p>';
	exit;
}, 0 );
