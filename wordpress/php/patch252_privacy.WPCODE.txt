/* 개인정보처리방침(252 /privacy/)에 휴대폰 번호를 넣습니다.

   왜 — 가입할 때 이메일과 휴대폰을 필수로 받는데 방침에는 휴대폰이
        한 글자도 없고, 이메일은 「선택」으로 적혀 있었습니다.
        04번 위탁 표의 포트원 항목도 「주문번호, 금액, 이메일, 이름」이라
        실제로 넘기는 것(휴대폰 포함)과 달랐습니다.

   ① 01 무엇을 받는지 : 선택·이메일 주소 → 필수·이메일 주소, 휴대폰 번호
      (줄바꿈이 제각각일 수 있어 정규식으로 표의 한 줄을 통째로 갈아끼웁니다)
   ② 02 왜 받는지     : 휴대폰 번호 한 줄 추가
   ③ 04 맡기는 곳     : 포트원에 넘기는 항목에 휴대폰 번호 추가
   ④ 시행일 갱신

   미리보기  ?stella_privfix=1      적용 ?stella_privfix=1&go=1
   되돌리기  ?stella_privfix=1&undo=1                              2026-09-01 */

add_action( 'init', function () {

	if ( ! isset( $_GET['stella_privfix'] ) ) {
		return;
	}

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="google" content="notranslate">';
	echo '<body class="notranslate" translate="no" style="font:15px/1.9 -apple-system,system-ui,sans-serif;background:#111;color:#eee;padding:24px;max-width:960px">';

	if ( ! current_user_can( 'manage_options' ) ) {
		echo '<h1 style="color:#ff7b7b">관리자만 볼 수 있습니다.</h1>';
		exit;
	}

	global $wpdb;

	$pid  = 252;
	$go   = isset( $_GET['go'] );
	$undo = isset( $_GET['undo'] );

	$re_row  = '/<td>\s*선택\s*<\/td>\s*<td>\s*이메일 주소\s*<\/td>\s*<td>[^<]*<\/td>/u';
	$row_new = '<td>필수</td>' . "\n" .
		'          <td>이메일 주소, 휴대폰 번호</td>' . "\n" .
		'          <td>문을 골라 풀이를 청하실 때 — 로그인과 결제, 안내에 씁니다</td>';

	$pairs = array(
		array(
			'name' => '② 왜 받는지에 휴대폰 넣기',
			'old'  => '<li><b>이메일</b> — 로그인 링크를 보내드리고, 구슬 안내나 공지를 전하는 데 씁니다.</li>',
			'new'  => '<li><b>이메일</b> — 로그인 링크를 보내드리고, 구슬 안내나 공지를 전하는 데 씁니다.</li>' . "\n"
				. '        <li><b>휴대폰 번호</b> — 결제할 때 결제대행사에 구매자 확인용으로 보내고, 결제와 주문에 관한 안내를 드리는 데 씁니다.</li>',
		),
		array(
			'name' => '③ 포트원에 넘기는 항목에 휴대폰 넣기',
			'old'  => '<td>주문번호, 금액, 이메일, 이름</td>',
			'new'  => '<td>주문번호, 금액, 이메일, 이름, 휴대폰 번호</td>',
		),
		array(
			'name' => '④ 시행일 고치기',
			'old'  => '<p class="when">시행일 2026년 8월 21일</p>',
			'new'  => '<p class="when">시행일 2026년 9월 1일 · 휴대폰 번호를 받는 항목이 더해졌습니다</p>',
		),
	);

	printf(
		'<h1 style="color:#ffd76a">개인정보처리방침에 휴대폰 넣기 — %s</h1>',
		$undo ? '되돌리기' : ( $go ? '적용' : '<span style="color:#9fd">미리보기 (아무것도 안 바뀝니다)</span>' )
	);

	$c = $wpdb->get_var( $wpdb->prepare( "SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $pid ) );
	if ( null === $c ) {
		echo '<h2 style="color:#ff7b7b">252 를 못 읽었습니다.</h2>';
		exit;
	}

	if ( $undo ) {
		$bak = get_post_meta( $pid, '_stella_privfix_backup', true );
		if ( $bak ) {
			$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ), array( 'ID' => $pid ) );
			delete_post_meta( $pid, '_stella_privfix_backup' );
			clean_post_cache( $pid );
			if ( function_exists( 'wp_cache_flush' ) ) {
				wp_cache_flush();
			}
			echo '<h2 style="color:#7bff9b">되돌렸습니다.</h2>';
		} else {
			echo '<h2 style="color:#888">사본이 없습니다.</h2>';
		}
		exit;
	}

	$ready = true;
	$done  = true;

	echo '<table cellpadding="8" style="border-collapse:collapse;width:100%">';
	echo '<tr style="color:#ffd76a"><td>바꿀 것</td><td>찾음</td><td>이미 됨</td></tr>';

	$n_row = preg_match_all( $re_row, $c );
	$m_row = substr_count( $c, '<td>이메일 주소, 휴대폰 번호</td>' );
	if ( $m_row > 0 ) {
		echo '<tr style="border-top:1px solid #333"><td>① 이메일·휴대폰을 필수로</td><td style="color:#888">-</td><td style="color:#9fd">이미 됨</td></tr>';
	} else {
		$done = false;
		if ( 1 !== $n_row ) {
			$ready = false;
		}
		printf(
			'<tr style="border-top:1px solid #333"><td>① 이메일·휴대폰을 필수로</td><td style="color:%s"><b>%d</b></td><td></td></tr>',
			1 === $n_row ? '#7bff9b' : '#ff7b7b',
			$n_row
		);
	}

	foreach ( $pairs as $p ) {
		$n = substr_count( $c, $p['old'] );
		$m = substr_count( $c, $p['new'] );
		if ( $m > 0 ) {
			printf(
				'<tr style="border-top:1px solid #333"><td>%s</td><td style="color:#888">%d</td><td style="color:#9fd">이미 됨</td></tr>',
				esc_html( $p['name'] ),
				$n
			);
			continue;
		}
		$done = false;
		if ( 1 !== $n ) {
			$ready = false;
		}
		printf(
			'<tr style="border-top:1px solid #333"><td>%s</td><td style="color:%s"><b>%d</b></td><td></td></tr>',
			esc_html( $p['name'] ),
			1 === $n ? '#7bff9b' : '#ff7b7b',
			$n
		);
	}
	echo '</table>';

	if ( $done ) {
		echo '<h2 style="color:#7bff9b">이미 다 되어 있습니다.</h2>';
		exit;
	}
	if ( ! $ready ) {
		echo '<h2 style="color:#ff7b7b">딱 한 번씩이 아닌 것이 있어 아무것도 바꾸지 않았습니다.</h2>'
			. '<p>이 화면을 저에게 보내주세요.</p>';
		exit;
	}
	if ( ! $go ) {
		echo '<h2 style="color:#ffd76a">여기까지가 미리보기입니다.</h2>'
			. '<p style="font-size:17px"><a style="color:#ffd76a" href="?stella_privfix=1&go=1">→ 이대로 바꾸기</a></p>';
		exit;
	}

	if ( ! get_post_meta( $pid, '_stella_privfix_backup', true ) ) {
		update_post_meta( $pid, '_stella_privfix_backup', $c );
	}

	$out = $c;
	if ( substr_count( $out, '<td>이메일 주소, 휴대폰 번호</td>' ) === 0 ) {
		$out = preg_replace( $re_row, $row_new, $out, 1 );
	}
	foreach ( $pairs as $p ) {
		if ( substr_count( $out, $p['new'] ) > 0 ) {
			continue;
		}
		$out = str_replace( $p['old'], $p['new'], $out );
	}

	$wpdb->update( $wpdb->posts, array( 'post_content' => $out ), array( 'ID' => $pid ) );
	clean_post_cache( $pid );
	if ( function_exists( 'wp_cache_flush' ) ) {
		wp_cache_flush();
	}

	printf( '<h2 style="color:#7bff9b">바꿨습니다 — %d → %d 바이트</h2>', strlen( $c ), strlen( $out ) );
	echo '<p><a style="color:#ffd76a" href="/privacy/">→ 개인정보처리방침에서 확인</a></p>';
	echo '<p style="color:#888">되돌리려면 ?stella_privfix=1&undo=1</p>';
	exit;
} );
