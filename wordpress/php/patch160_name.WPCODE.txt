/* ════════════════════════════════════════════════════════════
   「당신」이라고 부르지 않고 손님 이름으로 부릅니다

   책 안에 「당신」이 405곳 있습니다. 한 손님이 실제로 읽는 것은
   문에 따라 27~56곳이에요. 문장 종류로는 113가지입니다.

   「당신」도 「소희님」도 받침으로 끝납니다. 그래서 뒤에 붙는 조사가
   똑같아요 — 당신이/소희님이, 당신은/소희님은, 당신에게/소희님에게,
   당신을/소희님을, 당신의/소희님의. 그대로 갈아 끼워도 말이 안 어긋납니다.

   그래서 쪽을 만드는 자리에서 한 번에 갈아 끼웁니다. 113가지 문장을
   손으로 고치지 않아도 되고, 앞으로 새로 쓰는 글에 「당신」이 들어가도
   저절로 이름으로 바뀝니다.

   빽빽하지 않은지 재봤습니다 — 한 쪽 평균 1.2번, 제일 많은 쪽이 여섯 번.
   편지처럼 읽히는 정도라고 봅니다.

   다만 한 문장 안에 이름이 두 번 들어가는 자리 셋은 따로 고칩니다.

     ② 여는 글
        지금 : 제가 별을 읽은 것인지 당신 말을 옮겨 적은 것인지,
               당신이 직접 판단하실 수 있으니까요.
        고침 : … 소희님 말을 옮겨 적은 것인지, 직접 판단하실 수 있으니까요.

     ③ 짝사랑 · 지금 뗄 한 걸음
        지금 : 그리고 당신 인생의 주인공은 당신입니다.
        고침 : 그리고 그 인생의 주인공은 소희님입니다.
        여기 있던 「닿더라고요」도 「닿습니다」로 고칩니다.
        「~더라고요」는 안 쓰기로 한 말인데 책 전체에 이 한 곳만 남아 있었습니다.

     ④ 별자리
        지금 : 처음 본 사람이 아는 당신과 오래 본 사람이 아는 당신이 다릅니다.
        고침 : 처음 본 사람이 아는 모습과 오래 본 사람이 아는 모습이 다릅니다.
        여기는 이름을 넣기보다 「모습」이 더 정확합니다.

   페이지 내용을 wp_update_post 로 저장하면 워드프레스가 <script> 를
   지워버립니다. 그래서 데이터베이스에 바로 씁니다.

   붙여넣기 : WPCode → 새 스니펫 → PHP Snippet → 저장 → Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로 바꾸세요.

   ① 먼저 보기   https://stellasaju.com/?stella_name=1
   ② 정말 바꾸기 https://stellasaju.com/?stella_name=apply
   ③ 되돌리기    https://stellasaju.com/?stella_name=undo
   ════════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_name'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$PAGE = 160;
	$META = '_stella_name_backup';
	$mode = sanitize_key( wp_unslash( $_GET['stella_name'] ) );

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,"Apple SD Gothic Neo","Malgun Gothic",sans-serif;'
	   . 'line-height:1.8;padding:34px 24px;max-width:820px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.4rem;margin:0 0 18px}h2{color:#D4AF6A;font-size:1rem;margin:22px 0 6px}'
	   . 'b{color:#F0D9A8}.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . 'a{color:#D4AF6A}code{background:rgba(212,175,106,.12);padding:2px 6px;border-radius:3px}</style>';

	$row = $wpdb->get_row( $wpdb->prepare(
		"SELECT ID, post_content FROM {$wpdb->posts} WHERE ID = %d", $PAGE ) );
	if ( ! $row ) { echo '<h1>160쪽을 못 찾았습니다.</h1>'; exit; }

	if ( 'undo' === $mode ) {
		$old = get_post_meta( $PAGE, $META, true );
		echo '<h1>되돌리기</h1>';
		if ( ! $old ) { echo '<p>되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $old ), array( 'ID' => $PAGE ) );
		clean_post_cache( $PAGE );
		delete_post_meta( $PAGE, $META );
		echo '<p class="ok">' . number_format( strlen( $old ) ) . ' 바이트로 되돌렸습니다.</p>';
		echo '<p><a href="/reading-book/">책 보기</a></p>';
		exit;
	}

	$c = $row->post_content;
	$before = strlen( $c );

	$jobs = array();
	$jobs[] = array(
		'name' => '① 쪽을 만드는 자리에서 「당신」을 이름으로',
		'find' => '/\\s+function\\s+page\\(html,\\s+kind,\\s+part\\)\\{\\s+n\\+\\+;/u',
		'to'   => "  function page(html, kind, part){\n    n++;\n    /* 2026-09-07 · 「당신」이라고 부르지 않고 손님 이름으로 부릅니다.\n       「당신」도 「이름님」도 받침으로 끝나서 뒤에 붙는 조사가 같습니다.\n       그래서 그대로 갈아 끼워도 「~이/가」 「~은/는」이 어긋나지 않습니다. */\n    html = String(html).split('당신').join(esc(name)+'님');",
	);

	$jobs[] = array(
		'name' => '② 여는 글 · 한 문장에 두 번 들어가는 자리',
		'find' => '/\\s+\'<p>그렇게\\s+나눈\\s+이유가\\s+있습니다\\.\\s+그래야\\s+제가\\s+별을\\s+읽은\\s+것인지\\s+당신\\s+말을\\s+옮겨\\s+적은\\s+것인지,\\s+\'\\+\\s+\'당신이\\s+직접\\s+판단하실\\s+수\\s+있으니까요\\.\\s+그리고\\s+두\\s+쪽이\\s+어긋나는\\s+자리가\\s+있다면\\s+—\\s+그\\s+자리도\\s+숨기지\\s+않고\\s+적었습니다\\.<\\/p>\'\\);/u',
		'to'   => "    '<p>그렇게 나눈 이유가 있습니다. 그래야 제가 별을 읽은 것인지 당신 말을 옮겨 적은 것인지, '+\n    '직접 판단하실 수 있으니까요. 그리고 두 쪽이 어긋나는 자리가 있다면 — 그 자리도 숨기지 않고 적었습니다.</p>');",
	);

	$jobs[] = array(
		'name' => '③ 짝사랑 · 주인공 문장 + 「더라고요」',
		'find' => '/\\s+\\+\\s+\'<p>그리고\\s+<em>당신\\s+인생의\\s+주인공은\\s+당신입니다\\.<\\/em>\\s+\'\\s+\\+\\s+\'주인공은\\s+몇\\s+회를\\s+돌아가더라도\\s+결국\\s+인연이\\s+닿더라고요\\.\\s+\'/u',
		'to'   => "        + '<p>그리고 <em>그 인생의 주인공은 당신입니다.</em> '\n        + '주인공은 몇 회를 돌아가더라도 결국 인연이 닿습니다. '",
	);

	$jobs[] = array(
		'name' => '④ 별자리 · 이름 대신 「모습」으로',
		'find' => '/\\s+:\\s+\'\\s+태양은\\s+\'\\+st\\.sun\\.name\\+\'인데\\s+상승궁은\\s+\'\\+st\\.ascendant\\.name\\+\'라,\\s+처음\\s+본\\s+사람이\\s+아는\\s+당신과\\s+\'\\+\\s+\'오래\\s+본\\s+사람이\\s+아는\\s+당신이\\s+다릅니다\\.\\s+설명을\\s+두\\s+번\\s+해야\\s+하는\\s+자리가\\s+자주\\s+생기고,\\s+\'\\+/u',
		'to'   => "      : ' 태양은 '+st.sun.name+'인데 상승궁은 '+st.ascendant.name+'라, 처음 본 사람이 아는 모습과 '+\n        '오래 본 사람이 아는 모습이 다릅니다. 설명을 두 번 해야 하는 자리가 자주 생기고, '+",
	);


	$apply = ( 'apply' === $mode );
	echo '<h1>' . ( $apply ? '바꿨습니다' : '바꾸면 이렇게 됩니다 (아직 안 바꿨습니다)' ) . '</h1>';

	$ok = true;
	foreach ( $jobs as $j ) {
		$n = preg_match_all( $j['find'], $c );
		echo '<h2>' . esc_html( $j['name'] ) . '</h2>';
		if ( 1 === $n ) {
			echo '<p class="ok">찾았습니다 — 한 군데.</p>';
		} elseif ( 0 === $n ) {
			echo '<p class="no">못 찾았습니다. 이미 바꿨거나 코드가 달라졌습니다.</p>';
			$ok = false;
		} else {
			echo '<p class="no">' . intval( $n ) . '군데가 걸립니다. 한 군데여야 안전합니다.</p>';
			$ok = false;
		}
	}

	if ( ! $ok ) {
		echo '<p class="no">한 자리가 한 군데씩 걸릴 때만 바꿉니다. 아무것도 건드리지 않았습니다.</p>';
		exit;
	}

	$new = $c;
	foreach ( $jobs as $j ) {
		$new = preg_replace( $j['find'], $j['to'], $new, 1 );
		if ( null === $new ) { echo '<p class="no">바꾸다가 멈췄습니다. 아무것도 저장하지 않았습니다.</p>'; exit; }
	}

	echo '<p>' . number_format( $before ) . ' → ' . number_format( strlen( $new ) ) . ' 바이트</p>';

	if ( ! $apply ) {
		echo '<p>정말 바꾸시려면 <code>/?stella_name=apply</code> 를 여세요.</p>';
		exit;
	}

	if ( ! get_post_meta( $PAGE, $META, true ) ) {
		update_post_meta( $PAGE, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $PAGE ) );
	clean_post_cache( $PAGE );

	echo '<p class="ok">바꿨습니다.</p>';
	echo '<p>되돌리려면 <code>/?stella_name=undo</code></p>';
	echo '<p><a href="/reading-book/">책 보기</a></p>';
	exit;
} );
