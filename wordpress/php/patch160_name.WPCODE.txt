/* ══════════════════════════════════════════════════════════
   ① 가디언 이름 통일 · 소희 님께                    2026-09-11

   무엇을 고치나
     책 안의 가디언 이름이 옛 이름으로 남아 있습니다.
       직성의 신 → 일을 짓는 마루      세성의 신 → 때를 아는 미르
       연성의 신 → 인연을 잇는 벼리    성좌의 신 → 별을 읽는 아라
       강성의 신 → 몸을 지키는 아람    아르카나  → 그대로

   왜 여섯 줄만 고치면 되나
     책은 이름을 한 군데(GUIDE 표)에 적어두고 나머지는 다 거기서
     가져다 씁니다. 그래서 그 여섯 줄만 고치면 책 전체가 따라옵니다.

   짧은 이름도 같이 넣습니다
     말풍선에서 말하는 사람 이름은 짧아야 읽힙니다.
       말풍선   벼리
       제목     인연을 잇는 벼리가 연애운에 대해
     그래서 표에 short 를 하나 더 둡니다.

   주소
     해보기   https://stellasaju.com/?stella_b1_name=preview
     붙이기   https://stellasaju.com/?stella_b1_name=apply
     되돌리기 https://stellasaju.com/?stella_b1_name=undo
     쌓인 것  https://stellasaju.com/?stella_b1_name=list

   ★ 스니펫 이름 「STELLA - 가디언 이름」
   ★ 위치(Location)는 「어디서나 실행 / Run Everywhere」
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_b1_name'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode  = sanitize_text_field( wp_unslash( $_GET['stella_b1_name'] ) );
	$STACK = '_stella_b1_name_stack';
	$ROUND = '가디언 이름 통일 2026-09-11';

	/* ══ 여기부터 · 고칠 목록 ══════════════════════════════
	   array( '이름', '지금 글', '바꿀 글' )
	   ★ 지금 글은 화면에 보이는 그대로가 아니라, ?stella_find= 으로
	     찾은 조각을 씁니다. 코드에서 문장이 잘려 있을 때가 많아서요. */
	$FIX = array(

		/* ── GUIDE 표 여섯 줄 — 여기가 책 전체의 이름입니다 ── */
		array( '마루', 'name:\'직성의 신\', en:\'THE ARCHITECT\',',
		               'name:\'일을 짓는 마루\', short:\'마루\', en:\'THE ARCHITECT\',' ),
		array( '벼리', 'name:\'연성의 신\', en:\'THE LOVERS\',',
		               'name:\'인연을 잇는 벼리\', short:\'벼리\', en:\'THE LOVERS\',' ),
		array( '아람', 'name:\'강성의 신\', en:\'THE NURTURER\',',
		               'name:\'몸을 지키는 아람\', short:\'아람\', en:\'THE NURTURER\',' ),
		array( '미르', 'name:\'세성의 신\', en:\'THE SEASONS\',',
		               'name:\'때를 아는 미르\', short:\'미르\', en:\'THE SEASONS\',' ),
		array( '아라', 'name:\'성좌의 신\', en:\'ASTRAIOS\',',
		               'name:\'별을 읽는 아라\', short:\'아라\', en:\'ASTRAIOS\',' ),
		array( '아르카나', 'name:\'아르카나\', en:\'ARCANA\',',
		                   'name:\'아르카나\', short:\'아르카나\', en:\'ARCANA\',' ),

		/* ── 말풍선 화자는 짧은 이름으로 (열여섯 군데가 한꺼번에) ── */
		array( '말풍선 화자', '\'<div class="voice"><div class="who">\'+g.name+\'</div>\'',
		                      '\'<div class="voice"><div class="who">\'+(g.short||g.name)+\'</div>\'' ),

		/* ── 예시 주문에 남은 옛 이름 ── */
		array( '예시주문 마루', 'guardian:\'직성의 신\'', 'guardian:\'마루\'' ),
		array( '예시주문 벼리', 'guardian:\'연성의 신\'', 'guardian:\'벼리\'' ),
		array( '예시주문 아람', 'guardian:\'강성의 신\'', 'guardian:\'아람\'' ),
		array( '예시주문 미르', 'guardian:\'세성의 신\'', 'guardian:\'미르\'' ),

		/* ── 책 끝의 문 링크 ── */
		array( '문 링크', '직성의 신에게 가기', '마루에게 가기' ),

	);
	/* ══ 여기까지 ═══════════════════════════════════════════ */

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	$stack = get_post_meta( $id, $STACK, true );
	if ( ! is_array( $stack ) ) { $stack = array(); }

	if ( 'list' === $mode ) {
		echo "═══ 되돌릴 수 있는 판 ═══\n";
		if ( ! count( $stack ) ) { echo '쌓인 것이 없습니다.'; exit; }
		$i = count( $stack );
		foreach ( array_reverse( $stack ) as $s ) {
			echo '  ' . $i . '. ' . $s['when'] . ' - ' . $s['name'] . ' - '
			   . number_format( strlen( $s['body'] ) ) . "글자\n";
			$i--;
		}
		echo "\n?stella_b1_name=undo 를 치면 맨 위 것부터 하나씩 돌아갑니다.";
		exit;
	}

	if ( 'undo' === $mode ) {
		if ( ! count( $stack ) ) { echo '되돌릴 것이 없습니다.'; exit; }
		$last = array_pop( $stack );
		$wpdb->update( $wpdb->posts, array( 'post_content' => $last['body'] ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		update_post_meta( $id, $STACK, $stack );
		echo '되돌렸습니다 — ' . $last['when'] . ' - ' . $last['name'] . "\n";
		echo '남은 판 ' . count( $stack ) . '개';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	echo "═══ 가디언 이름 통일 ═══\n";
	echo '쪽 ' . $id . ' - ' . number_format( strlen( $c ) ) . "글자\n";
	echo '고칠 자리 ' . count( $FIX ) . "개\n\n";

	if ( ! count( $FIX ) ) {
		echo "목록이 비어 있습니다.\n";
		echo "고치실 데를 알려주시면 채워 드리겠습니다.";
		exit;
	}

	$todo = array();
	$done = 0;
	$miss = 0;
	foreach ( $FIX as $r ) {
		list( $nm, $old, $new ) = $r;
		$got = substr_count( $c, $old );
		if ( $got > 0 ) {
			echo sprintf( "  %-22s 찾음 %d군데\n", $nm, $got );
			$todo[] = $r;
			continue;
		}
		if ( substr_count( $c, $new ) > 0 ) {
			echo sprintf( "  %-22s 이미 고쳐져 있습니다\n", $nm );
			$done++;
			continue;
		}
		echo sprintf( "  %-22s ★ 못 찾았습니다 — 건너뜁니다\n", $nm );
		$miss++;
	}

	if ( ! count( $todo ) ) {
		if ( $done ) { echo "\n이미 다 고쳐져 있습니다."; }
		else { echo "\n하나도 못 찾았습니다. 아무것도 안 바꿨습니다.\n?stella_find= 으로 지금 글을 다시 확인해 주세요."; }
		exit;
	}

	$new = $c;
	foreach ( $todo as $r ) { $new = str_replace( $r[1], $r[2], $new ); }

	if ( 'preview' === $mode ) {
		echo "\n" . count( $todo ) . "자리를 고칩니다";
		if ( $miss ) { echo ' (못 찾은 ' . $miss . '자리는 건너뜁니다)'; }
		echo ".\n" . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $new ) ) . "글자\n";
		echo "\n괜찮으시면 ?stella_b1_name=apply\n";
		exit;
	}

	if ( 'apply' !== $mode ) { echo 'preview - apply - undo - list 중에 골라주세요.'; exit; }

	$stack[] = array( 'when' => current_time( 'Y-m-d H:i' ), 'name' => $ROUND, 'body' => $c );
	while ( count( $stack ) > 5 ) { array_shift( $stack ); }
	update_post_meta( $id, $STACK, $stack );

	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "\n고쳤습니다. " . count( $todo ) . "자리.\n";
	echo "되돌리려면 ?stella_b1_name=undo\n";
	exit;
}, 1 );
