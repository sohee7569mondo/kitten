/* ══════════════════════════════════════════════════════════
   둘째 배치 · 소희 님께                              2026-09-11

   지금 책을 직접 물어보고 만든 패치입니다. 짐작으로 만들지 않았습니다.

   ① 접기 열쇠가 어긋나 있습니다 — 이게 제일 급합니다
        제목      여덟 글자에서 제일 많은 것
        접기 열쇠  여덟 글자에서 제일 센 것
      책이 제목 글자로 장을 찾는데 「많은」과 「센」이 달라서
      지금 그 장을 못 찾고 있습니다. 열쇠를 제목에 맞춥니다.

   ② 가디언 긴 이름
      짧은 이름(벼리·마루…)은 이미 들어가 있습니다.
      머리말 옆에 붙는 이름만 긴 이름으로 바꿉니다.
        지금  연애운에 대해 — 벼리
        바꿈  연애운에 대해 — 인연을 잇는 벼리
      말풍선 화자는 짧은 이름 그대로 둡니다. 표에 full 을 하나 더 둡니다.

   ③ 13장 제목
      지금 제목에 (사전질문)이 들어 있어서 지난번에 못 찾았습니다.
        지금  타고난 성향과 적어주신 현재의 생활(사전질문)을 정리해보았습니다
        바꿈  타고난 나와 지금의 내가 어긋난 자리
              (부제) 다르다고 틀린 것이 아닙니다. 그만큼 힘이 더 든다는 뜻이에요

   주소
     해보기   https://stellasaju.com/?stella_b2=preview
     붙이기   https://stellasaju.com/?stella_b2=apply
     되돌리기 https://stellasaju.com/?stella_b2=undo
     쌓인 것  https://stellasaju.com/?stella_b2=list

   ★ 스니펫 이름 「STELLA - 둘째 배치」
   ★ 위치(Location)는 「어디서나 실행 / Run Everywhere」
   ══════════════════════════════════════════════════════════ */

add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_b2'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;
	header( 'Content-Type: text/plain; charset=utf-8' );

	$mode  = sanitize_text_field( wp_unslash( $_GET['stella_b2'] ) );
	$STACK = '_stella_b2_stack';
	$ROUND = '둘째 배치 2026-09-11';

	/* ══ 여기부터 · 고칠 목록 ══════════════════════════════
	   array( '이름', '지금 글', '바꿀 글' )
	   ★ 지금 글은 화면에 보이는 그대로가 아니라, ?stella_find= 으로
	     찾은 조각을 씁니다. 코드에서 문장이 잘려 있을 때가 많아서요. */
	$FIX = array(
		array( '접기 열쇠',
		  'var SIP=\'여덟 글자에서 제일 센 것\'',
		  'var SIP=\'여덟 글자에서 제일 많은 것\'' ),
		array( '긴이름 마루',
		  'name:\'마루\', en:\'THE ARCHITECT\'',
		  'name:\'마루\', full:\'일을 짓는 마루\', en:\'THE ARCHITECT\'' ),
		array( '긴이름 벼리',
		  'name:\'벼리\', en:\'THE LOVERS\'',
		  'name:\'벼리\', full:\'인연을 잇는 벼리\', en:\'THE LOVERS\'' ),
		array( '긴이름 아람',
		  'name:\'아람\', en:\'THE NURTURER\'',
		  'name:\'아람\', full:\'몸을 지키는 아람\', en:\'THE NURTURER\'' ),
		array( '긴이름 미르',
		  'name:\'미르\', en:\'THE SEASONS\'',
		  'name:\'미르\', full:\'때를 아는 미르\', en:\'THE SEASONS\'' ),
		array( '긴이름 아라',
		  'name:\'아라\', en:\'ASTRAIOS\'',
		  'name:\'아라\', full:\'별을 읽는 아라\', en:\'ASTRAIOS\'' ),
		array( '머리말 옆 이름',
		  '<span class="byline">— \'+g.name+\'</span>',
		  '<span class="byline">— \'+(g.full||g.name)+\'</span>' ),
		array( '13 제목',
		  '<h2>타고난 성향과 적어주신 현재의 생활(사전질문)을 정리해보았습니다</h2>\'+',
		  '<h2>타고난 나와 지금의 내가 어긋난 자리</h2>\'+\'<p class="dek">다르다고 틀린 것이 아닙니다. 그만큼 힘이 더 든다는 뜻이에요</p>\'+' ),

	);
	/* ══ 여기까지 ═══════════════════════════════════════════ */

	$page = get_page_by_path( 'reading-book' );
	if ( ! $page ) { echo '책 쪽이 없습니다.'; exit; }
	$id = (int) $page->ID;

	$stack = get_post_meta( $id, $STACK, true );
	if ( ! is_array( $stack ) ) { $stack = array(); }

	if ( 'list' === $mode ) {
		echo "═══ 되돌릴 수 있는 판 ═══\n";
		if ( ! count( $stack ) ) { echo '쌓인 것이 없습니다.'; exit; }
		$i = count( $stack );
		foreach ( array_reverse( $stack ) as $s ) {
			echo '  ' . $i . '. ' . $s['when'] . ' - ' . $s['name'] . ' - '
			   . number_format( strlen( $s['body'] ) ) . "글자\n";
			$i--;
		}
		echo "\n?stella_b2=undo 를 치면 맨 위 것부터 하나씩 돌아갑니다.";
		exit;
	}

	if ( 'undo' === $mode ) {
		if ( ! count( $stack ) ) { echo '되돌릴 것이 없습니다.'; exit; }
		$last = array_pop( $stack );
		$wpdb->update( $wpdb->posts, array( 'post_content' => $last['body'] ), array( 'ID' => $id ) );
		clean_post_cache( $id );
		update_post_meta( $id, $STACK, $stack );
		echo '되돌렸습니다 — ' . $last['when'] . ' - ' . $last['name'] . "\n";
		echo '남은 판 ' . count( $stack ) . '개';
		exit;
	}

	$c = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $id ) );

	echo "═══ 둘째 배치 ═══\n";
	echo '쪽 ' . $id . ' - ' . number_format( strlen( $c ) ) . "글자\n";
	echo '고칠 자리 ' . count( $FIX ) . "개\n\n";

	if ( ! count( $FIX ) ) {
		echo "목록이 비어 있습니다.\n";
		echo "고치실 데를 알려주시면 채워 드리겠습니다.";
		exit;
	}

	$todo = array();
	$done = 0;
	$miss = 0;
	foreach ( $FIX as $r ) {
		list( $nm, $old, $new ) = $r;
		$got = substr_count( $c, $old );
		if ( $got > 0 ) {
			echo sprintf( "  %-22s 찾음 %d군데\n", $nm, $got );
			$todo[] = $r;
			continue;
		}
		if ( substr_count( $c, $new ) > 0 ) {
			echo sprintf( "  %-22s 이미 고쳐져 있습니다\n", $nm );
			$done++;
			continue;
		}
		echo sprintf( "  %-22s ★ 못 찾았습니다 — 건너뜁니다\n", $nm );
		$miss++;
	}

	if ( ! count( $todo ) ) {
		if ( $done ) { echo "\n이미 다 고쳐져 있습니다."; }
		else { echo "\n하나도 못 찾았습니다. 아무것도 안 바꿨습니다.\n?stella_find= 으로 지금 글을 다시 확인해 주세요."; }
		exit;
	}

	$new = $c;
	foreach ( $todo as $r ) { $new = str_replace( $r[1], $r[2], $new ); }

	if ( 'preview' === $mode ) {
		echo "\n" . count( $todo ) . "자리를 고칩니다";
		if ( $miss ) { echo ' (못 찾은 ' . $miss . '자리는 건너뜁니다)'; }
		echo ".\n" . number_format( strlen( $c ) ) . ' → ' . number_format( strlen( $new ) ) . "글자\n";
		echo "\n괜찮으시면 ?stella_b2=apply\n";
		exit;
	}

	if ( 'apply' !== $mode ) { echo 'preview - apply - undo - list 중에 골라주세요.'; exit; }

	$stack[] = array( 'when' => current_time( 'Y-m-d H:i' ), 'name' => $ROUND, 'body' => $c );
	while ( count( $stack ) > 5 ) { array_shift( $stack ); }
	update_post_meta( $id, $STACK, $stack );

	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ), array( 'ID' => $id ) );
	clean_post_cache( $id );

	echo "\n고쳤습니다. " . count( $todo ) . "자리.\n";
	echo "되돌리려면 ?stella_b2=undo\n";
	exit;
}, 1 );
