/* ════════════════════════════════════════════════════════════
   문 여섯 쪽의 「충전일」 한 줄을 고칩니다
   ★ 2026-09-15 · 소희 님께

   왜 — 결제대행(이니시스) 심사에 「포인트는 이벤트로 드리는 것이고
        손님이 충전하는 것이 아니다」라고 적어 보내는데, 화면에
        「구슬은 충전일로부터 1년간 유효합니다」가 남아 있으면
        심사관이 충전형 포인트로 읽습니다. 말이 어긋납니다.

   바뀌기 전  정식 발행 뒤에는 구슬을 쓰며, 구슬은 충전일로부터
              1년간 유효합니다.
   바뀐 뒤    정식 발행 뒤에는 구슬을 쓰며, 구슬은 이벤트로 드리는
              것입니다. 미리 충전해 두는 것은 없고, 받으신 날로부터
              1년간 쓰실 수 있습니다.

   ★ 쪽 글을 고치지 않고 화면에서 그 줄만 갈아 끼웁니다. 제 사본이
     낡아도 안 놓칩니다 — 「충전일로부터」라는 낱말만 보고 찾습니다.
   ★ 끄면 원래대로 돌아옵니다. 되돌리기가 따로 없습니다.

   붙여넣기 : WPCode -> 새 스니펫 -> PHP Snippet -> 저장 -> Active
   ★ 위치(Location)를 반드시 「어디서나 실행 / Run Everywhere」로.
   ★ 크롬 번역을 끄고 붙여넣으세요.
   ════════════════════════════════════════════════════════════ */

add_action( 'wp_head', function () {

	/* 울타리 — 문 여섯 쪽에서만 실립니다 */
	if ( ! is_page( array(
		'door-love', 'door-career', 'door-health',
		'door-fortune', 'door-astro', 'door-tarot'
	) ) ) { return; }

	?>
<!-- stella 구슬 문구 · 판 2026-09-15 -->
<script>
(function(){
  if(window.StellaOrbWord){ return; }
  window.StellaOrbWord = 1;

  var OLD = '충전일로부터';
  var NEW = '정식 발행 뒤에는 구슬을 쓰며, 구슬은 이벤트로 드리는 것입니다. '
          + '미리 충전해 두는 것은 없고, 받으신 날로부터 1년간 쓰실 수 있습니다.';

  /* 글자만 갈아끼웁니다 — 태그 구조를 전혀 안 건드립니다.
     ★ 처음에 innerHTML 을 통째로 바꿨다가 윈에 있는 상자까지
       지워졌습니다 (jsdom 으로 돌려보고 알았습니다). */
  function go(){
    var walk = document.createTreeWalker(document.body, 4, null);
    var hits = [], nd, tg;
    while(walk.nextNode()){
      nd = walk.currentNode;
      if(String(nd.nodeValue).indexOf(OLD) < 0){ continue; }
      tg = '';
      if(nd.parentNode){ tg = String(nd.parentNode.nodeName).toLowerCase(); }
      if(tg === 'script'){ continue; }
      if(tg === 'style'){ continue; }
      if(tg === 'textarea'){ continue; }
      hits.push(nd);
    }
    var i;
    for(i = 0; i < hits.length; i++){ hits[i].nodeValue = NEW; }
    return hits.length;
  }

  /* 찾아 바꿔도 몇 번 더 살핍니다 — 쪽의 다른 스크립트가
     나중에 그 칸을 다시 그리면 옛 글이 돌아오기 때문입니다. */
  var tries = 0;
  function tick(){
    tries++;
    try{ go(); }catch(e){ return; }
    if(tries < 24){ setTimeout(tick, 400); }
  }

  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', tick);
  } else {
    tick();
  }
})();
</script>
	<?php
} );
