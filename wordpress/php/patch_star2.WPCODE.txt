add_action( 'template_redirect', function () {

	if ( ! isset( $_GET['stella_starmerge'] ) ) { return; }
	if ( ! current_user_can( 'manage_options' ) ) { return; }

	global $wpdb;

	$mode = sanitize_text_field( wp_unslash( $_GET['stella_starmerge'] ) );
	$META = '_stella_starmerge_bak_0909';

	header( 'Content-Type: text/html; charset=utf-8' );
	echo '<meta charset="utf-8"><meta name="google" content="notranslate">'
	   . '<style>body{background:#12101B;color:#F2ECE0;'
	   . 'font-family:-apple-system,sans-serif;line-height:1.7;'
	   . 'padding:28px 18px;max-width:1000px;margin:0 auto}'
	   . 'h1{color:#D4AF6A;font-size:1.2rem;margin:0 0 10px}'
	   . 'h2{color:#9AD1A5;font-size:1rem;margin:26px 0 6px;'
	   . 'border-top:1px solid rgba(255,255,255,.12);padding-top:14px}'
	   . '.ok{color:#9AD1A5}.no{color:#E5A0A0}'
	   . '.dim{color:#8B849C;font-size:.86rem}'
	   . 'pre{background:#000;padding:12px;white-space:pre-wrap;'
	   . 'font:12px/1.7 monospace;border-left:3px solid #D4AF6A}'
	   . '</style>';

	$p = get_page_by_path( 'star-saju' );
	if ( ! $p ) { echo '<p class="no">스타 사주 쪽이 없습니다.</p>'; exit; }

	if ( 'undo' === $mode ) {
		echo '<h1>되돌립니다</h1>';
		$bak = get_post_meta( $p->ID, $META, true );
		if ( '' === $bak ) { echo '<p class="no">되돌릴 것이 없습니다.</p>'; exit; }
		$wpdb->update( $wpdb->posts, array( 'post_content' => $bak ),
			array( 'ID' => $p->ID ) );
		clean_post_cache( $p->ID );
		delete_post_meta( $p->ID, $META );
		if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }
		echo '<p class="ok">되돌렸습니다.</p>';
		exit;
	}

	$c  = (string) $wpdb->get_var( $wpdb->prepare(
		"SELECT post_content FROM {$wpdb->posts} WHERE ID = %d", $p->ID ) );
	$go = ( 'apply' === $mode );

	echo '<h1>스타 사주에 사람 더하기 — '
	   . ( $go ? '붙입니다' : '미리보기 (아직 안 붙임)' ) . '</h1>';

	/* ── 지금 있는 사람들 ────────────────────────────── */
	$have = array();
	$last = -1;
	$from = 0;
	while ( true ) {
		$at = strpos( $c, '{"n":"', $from );
		if ( false === $at ) { break; }
		$depth = 0; $end = $at; $len = strlen( $c ); $inStr = false;
		for ( $i = $at; $i < $len; $i++ ) {
			$ch = $c[ $i ];
			if ( '"' === $ch ) {
				if ( $i > 0 ) { if ( '\\' !== $c[ $i - 1 ] ) { $inStr = ! $inStr; } }
				continue;
			}
			if ( $inStr ) { continue; }
			if ( '{' === $ch ) { $depth++; }
			if ( '}' === $ch ) { $depth--; if ( 0 === $depth ) { $end = $i; break; } }
		}
		if ( $end <= $at ) { break; }
		$from = $end + 1;
		$last = $end;
		$ns = $at + 6;
		$ne = strpos( $c, '"', $ns );
		if ( false !== $ne ) { $have[ substr( $c, $ns, $ne - $ns ) ] = 1; }
	}

	echo '<p class="dim">쪽 번호 ' . (int) $p->ID . ' &middot; 지금 '
	   . count( $have ) . '명 &middot; ' . number_format( strlen( $c ) )
	   . ' 바이트</p>';

	if ( $last < 0 ) {
		echo '<p class="no">사람 자료를 못 찾았습니다. 아무것도 안 했습니다.</p>';
		exit;
	}

	/* ── 넣을 사람들 ────────────────────────────────── */
	$raw = apply_filters( 'stella_star_rows', '' );
	$raw = trim( $raw );
	if ( '' === $raw ) {
		echo '<p class="no">자료 스니펫(①②)이 안 켜져 있습니다.</p>';
		exit;
	}

	$EL = array( '목', '화', '토', '금', '수' );
	$json = ''; $add = 0; $dup = array(); $bad = 0; $show = ''; $split = array();

	foreach ( explode( "\n", $raw ) as $line ) {
		$line = trim( $line );
		if ( '' === $line ) { continue; }
		$f = explode( '|', $line );
		if ( count( $f ) < 15 ) { $bad++; continue; }
		$grp = ( isset( $f[15] ) ) ? trim( $f[15] ) : '';

		/* 이름이 겹치면 그룹 이름을 붙여 갈라줍니다.
		   같은 활동명을 쓰는 다른 사람이 있어서, 그냥 건너뛰면
		   진짜 사람이 빠집니다. */
		$nm = $f[0];
		if ( isset( $have[ $nm ] ) ) {
			if ( '' === $grp ) { $dup[] = $nm; continue; }
			$nm2 = $nm . '(' . $grp . ')';
			if ( isset( $have[ $nm2 ] ) ) { $dup[] = $nm; continue; }
			$split[] = $nm . ' -> ' . $nm2;
			$nm = $nm2;
		}

		$nums = explode( ',', $f[10] );
		if ( count( $nums ) !== 5 ) { $bad++; continue; }
		$el = array();
		for ( $k = 0; $k < 5; $k++ ) { $el[ $EL[ $k ] ] = (int) $nums[ $k ]; }

		$no  = ( '' === $f[12] ) ? array() : explode( '.', $f[12] );
		$sip = explode( '.', $f[13] );

		$rec = array(
			'n' => $nm, 'c' => $f[1], 'b' => $f[2], 'k' => '',
			'j' => $f[3], 'a' => $f[4], 'y' => $f[5], 'm' => $f[6],
			'dd' => $f[7], 'g' => $f[8], 'gk' => $f[9],
			'el' => $el, 'hi' => $f[11], 'no' => $no,
			'sip' => $sip, 's' => $f[14],
		);
		$one = wp_json_encode( $rec, JSON_UNESCAPED_UNICODE );
		if ( ! $one ) { $bad++; continue; }
		$json .= ',' . $one;
		$add++;
		if ( $add <= 3 ) { $show .= $one . "\n\n"; }
		$have[ $nm ] = 1;
	}

	echo '<h2>세어보니</h2><pre>'
	   . '지금 있는 사람        ' . count( $have ) - $add . "명\n"
	   . '새로 넣을 사람        ' . $add . "명\n"
	   . '이름이 겹쳐 건너뜀    ' . count( $dup ) . "명\n"
	   . '그룹 이름을 붙임     ' . count( $split ) . "명\n"
	   . '모양이 이상해 건너뜀  ' . $bad . "명\n"
	   . '넣고 나면             ' . ( count( $have ) ) . '명'
	   . '</pre>';

	if ( count( $split ) ) {
		echo '<h2>같은 이름이 있어 그룹 이름을 붙인 사람 ' . count( $split ) . '명</h2><pre>'
		   . esc_html( implode( "\n", $split ) ) . '</pre>';
	}

	if ( count( $dup ) ) {
		echo '<h2>겹친 이름</h2><pre>'
		   . esc_html( implode( ' · ', array_slice( $dup, 0, 60 ) ) )
		   . ( count( $dup ) > 60 ? ' …' : '' ) . '</pre>';
	}

	echo '<h2>넣을 사람 처음 세 명</h2><pre>' . esc_html( $show ) . '</pre>';

	if ( ! $add ) {
		echo '<p class="no">넣을 사람이 없습니다.</p>'; exit;
	}
	if ( $bad ) {
		echo '<p class="no">모양이 이상한 줄이 ' . $bad . '개 있습니다. 붙이지 않았습니다.</p>';
		exit;
	}

	if ( ! $go ) {
		echo '<h2>끝</h2><p class="dim">이대로 괜찮으시면 '
		   . '/?stella_starmerge=apply 를 여세요.</p>';
		exit;
	}

	$new = substr( $c, 0, $last + 1 ) . $json . substr( $c, $last + 1 );

	if ( '' === get_post_meta( $p->ID, $META, true ) ) {
		update_post_meta( $p->ID, $META, $c );
	}
	$wpdb->update( $wpdb->posts, array( 'post_content' => $new ),
		array( 'ID' => $p->ID ) );
	clean_post_cache( $p->ID );
	if ( function_exists( 'wp_cache_flush' ) ) { wp_cache_flush(); }

	echo '<h2>끝</h2><p class="ok">' . $add . '명을 넣었습니다. '
	   . number_format( strlen( $c ) ) . ' -> '
	   . number_format( strlen( $new ) ) . ' 바이트</p>';
	echo '<p class="dim">되돌리시려면 /?stella_starmerge=undo 를 여세요.</p>';
	exit;
} );
