  var TESTS = [T1, T2, T3, T4];

  /* ── 화면을 그리는 부분 ──
     앰퍼샌드와 여는 꺾쇠를 쓰지 않기 위해 문자열이 아니라
     createElement 로만 만듭니다. */
  function el(id){ return document.getElementById(id); }
  function clear(n){ while(n.firstChild){ n.removeChild(n.firstChild); } }
  function mk(tag, cls, text){
    var n = document.createElement(tag);
    if(cls){ n.className = cls; }
    if(text){ n.textContent = text; }
    return n;
  }
  function show(which){
    el('lab-home').hidden = (which !== 'home');
    el('lab-stage').hidden = (which !== 'stage');
    el('lab-result').hidden = (which !== 'result');
  }
  function toTop(){
    var box = el('slab');
    if(!box){ return; }
    var r = box.getBoundingClientRect();
    window.scrollTo(0, r.top + window.pageYOffset - 16);
  }

  var cur = null, step = 0, picks = [];

  function buildHome(){
    var box = el('lab-cards');
    clear(box);
    TESTS.forEach(function(t){
      var b = mk('button','card');
      b.type = 'button';
      b.appendChild(mk('span','no', t.no));
      b.appendChild(mk('div','nm', t.name));
      b.appendChild(mk('div','sb', t.sub));
      b.appendChild(mk('div','mt', t.meta));
      b.addEventListener('click', function(){ start(t); });
      box.appendChild(b);
    });
  }

  function start(t){
    cur = t; step = 0; picks = [];
    show('stage');
    render();
    toTop();
  }

  function render(){
    var total = cur.q.length;
    var q = cur.q[step];
    el('lab-cnt').textContent = cur.no + '  ·  ' + (step + 1) + ' / ' + total;
    el('lab-fill').style.width = Math.round(step / total * 100) + '%';
    el('lab-q').textContent = q.t;
    var box = el('lab-opts');
    clear(box);
    q.o.forEach(function(op, i){
      var b = mk('button','opt', op.x);
      b.type = 'button';
      b.addEventListener('click', function(){ pick(i); });
      box.appendChild(b);
    });
    el('lab-back').style.visibility = (step === 0) ? 'hidden' : 'visible';
  }

  function pick(i){
    picks[step] = i;
    step = step + 1;
    if(step === cur.q.length){ finish(); return; }
    render();
    toTop();
  }

  function finish(){
    var tally = {};
    cur.keys.forEach(function(k){ tally[k] = 0; });
    picks.forEach(function(oi, qi){
      var k = cur.q[qi].o[oi].k;
      tally[k] = tally[k] + 1;
    });
    var best = cur.keys[0];
    cur.keys.forEach(function(k){
      if(tally[k] > tally[best]){ best = k; }
    });
    el('lab-fill').style.width = '100%';
    draw(cur.r[best]);
  }

  function draw(r){
    el('lab-rtag').textContent = r.tag;
    el('lab-rname').textContent = r.name;
    el('lab-rsub').textContent = r.sub;

    var body = el('lab-rbody');
    clear(body);
    r.body.forEach(function(p){ body.appendChild(mk('p', '', p)); });

    var care = el('lab-rcare');
    clear(care);
    r.care.forEach(function(c){ care.appendChild(mk('li', '', c)); });

    var go = el('lab-rgo');
    clear(go);
    var list = [];
    if(r.du){ list.push({ t: r.name + ' 문 열기', d: '이 가디언에게 직접 묻습니다', u: r.du }); }
    cur.go.forEach(function(g){ list.push(g); });
    list.forEach(function(g){
      var a = document.createElement('a');
      a.href = g.u;
      a.appendChild(mk('b', '', g.t));
      a.appendChild(mk('span', '', g.d));
      a.appendChild(mk('i', '', '→'));
      go.appendChild(a);
    });

    show('result');
    toTop();
  }

  el('lab-back').addEventListener('click', function(){
    if(step === 0){ return; }
    step = step - 1;
    render();
    toTop();
  });
  el('lab-quit').addEventListener('click', function(){ show('home'); toTop(); });
  el('lab-again').addEventListener('click', function(){ start(cur); });
  el('lab-other').addEventListener('click', function(){ show('home'); toTop(); });

  buildHome();
  show('home');
})();
